38/**
39 * Applies a mutant *warm* and returns its verdict. The parent process
40 * boots once; each mutant runs in a `pcntl_fork`ed child that loads the
41 * mutated bytes on first reference — through a prepended autoloader that,
42 * while a mutant is active, serves it from a unique temp path (a fresh
43 * path means opcache compiles it cleanly, so no invalidate is ever
44 * needed). The child reports its verdict to a file and dies by SIGKILL,
45 * skipping every shutdown handler the parent registered.
46 *
47 * The precondition is redeclare-safety: a class already loaded cannot be
48 * re-mutated in a forked child (its declaration is inherited), so callers
49 * check {@see canApply} first and cold-fallback when it is false. This is
50 * the mechanism proven before the engine was built; the covering-test
51 * ordering and catalog live above it.
52 *
53 * Warm application needs `pcntl` and `posix` — the fork and the signal.
54 * They are absent on Windows and on minimal PHP builds, so a runner asks
55 * {@see isSupported} first and cold-forks each mutant through `proc_open`
56 * (the portable path the rest of Crucible already uses) when it is false.
57 * The rest of the test runner never touches these extensions.
58 */
59final readonly class MutantApplier
60{
61    private const int POLL_MICROSECONDS = 1_000;
62
63    /**
64     * The memory a single mutant may use before it is cut off.
65     *
66     * ✓ Measured 2026-09-20, and it is why this exists: a mutant that
67     * turns a bounded loop into an unbounded one grows ~130 MiB/s when
68     * the loop writes to anything that accumulates — a FakeTerminal
69     * collecting frames does exactly that. The wall-clock timeout is no
70     * defence, because 10s of that is 1.3 GB and the cold path's 30s is
71     * 3.9 GB, so the OOM killer reaches the whole process tree first. It
72     * did: three runs died with SIGKILL around mutant 2444, and the
73     * journal named the mutant each time.
74     *
75     * A limit turns that into an ordinary errored verdict, because a
76     * child that dies without writing one is already read as "the mutant
77     * crashed the run" rather than as an escape.
78     *
79     * Generous against a worker running a handful of covering tests —
80     * the mutate parent itself peaks near 245 MiB — and still an order
81     * of magnitude under the runaway.
82     *
83     * Shared with ColdMutantExecutor on purpose: one budget for a mutant,
84     * whichever way it runs, or the two paths disagree about what is
85     * survivable.
86     */
87    public const string MEMORY_LIMIT = '512M';
88
89    private MutationAutoloader $autoloader;
90
91    public function __construct()
92    {
93        // Dormant in the parent; each fork arms it for its own mutant. The
94        // same loader serves the cold worker from the environment.
95        $this->autoloader = new MutationAutoloader();
96        $this->autoloader->register();
97    }
98
99    /**
100     * Whether this platform can run the warm path at all: the fork and the
101     * signal it terminates the child with. False on Windows and minimal
102     * builds — the runner cold-falls-back through `proc_open` instead.
103     */
104    public static function isSupported(): bool
105    {
106        return function_exists('pcntl_fork')
107            && function_exists('pcntl_waitpid')
108            && function_exists('posix_kill')
109            && function_exists('posix_getpid');
110    }
111
112    /**
113     * Whether the mutant's class is still unloaded in this process — the
114     * warm precondition. Once a class is declared it cannot be replaced
115     * without a redeclare fatal, so a false here means cold-fallback.
116     */
117    public function canApply(Mutant $mutant): bool
118    {
119        return !class_exists($mutant->class, false);
120    }
121
122    /**
123     * Run the covering tests against the mutant in an isolated child.
124     * $covering runs the ordered covering tests and returns the id of the
125     * first that failed — the kill — or null when every one passed.
126     *
127     * @param callable(): ?non-empty-string $covering
128     */
129    public function run(Mutant $mutant, callable $covering, float $timeoutSeconds = 10.0): MutationVerdict
130    {
131        // A safety net, not the routing point: a runner checks isSupported()
132        // up front and cold-falls-back, so this only fires on direct misuse
133        // — and reports it plainly rather than fataling on an undefined fork.
134        if (!self::isSupported()) {
135            return MutationVerdict::errored($mutant, 'Warm mutation requires the pcntl and posix extensions.', 0.0);
136        }
137
138        $started    = microtime(true);
139        $mutatedTmp = tempnam(sys_get_temp_dir(), 'crucible-mutant-');
140        $resultTmp  = tempnam(sys_get_temp_dir(), 'crucible-verdict-');
141
142        if ($mutatedTmp === false || $resultTmp === false) {
143            return MutationVerdict::errored($mutant, 'Could not allocate a temp file for the mutant.', 0.0);
144        }
145
146        file_put_contents($mutatedTmp, $mutant->mutatedSource);
147
148        // Remembered BEFORE the fork, so the child can prove it is not
149        // the parent. ✓ Measured 2026-09-20: mutating the `$pid === 0`
150        // branch below makes the parent take the child's path, and the
151        // child's last act is SIGKILL on its own pid — uncatchable, so
152        // the whole run dies with no verdict and no log line. Three
153        // mutation runs ended exactly there.
154        //
155        // The guard is worth more than that one mutant: any fork bug
156        // reaching child() in the parent takes the run with it, and this
157        // is the one invariant that cannot be true by accident.
158        $parent = posix_getpid();
159        $pid    = pcntl_fork();
160
161        if ($pid === -1) {
162            @unlink($mutatedTmp);
163            @unlink($resultTmp);
164
165            return MutationVerdict::errored($mutant, 'Could not fork to isolate the mutant.', microtime(true) - $started);
166        }
167
168        if ($pid === 0) {
169            $this->child($mutant, $mutatedTmp, $resultTmp, $covering, $parent);
170        }
171
172        // Guarded in BOTH directions, because either branch being wrong is
173        // unrecoverable in a different way. A parent that reaches child()
174        // must not SIGKILL itself — that is the guard inside child(). A
175        // CHILD that skips child() must not reach here and resume the
176        // parent's work: it would finish the parent's run, fork again from
177        // inside the loop, and multiply. Its own pid is the proof.
178        if (posix_getpid() !== $parent) {
179            posix_kill(posix_getpid(), SIGKILL);
180        }
181
182        $verdict = $this->await($pid, $mutant, $resultTmp, $timeoutSeconds, $started);
183
184        @unlink($mutatedTmp);
185        @unlink($resultTmp);
186
187        return $verdict;
188    }
189
190    /**
191     * The child: activate the mutant, run the covering tests, write the
192     * verdict, and terminate hard.
193     *
194     * @param non-empty-string              $mutatedTmp
195     * @param non-empty-string              $resultTmp
196     * @param callable(): ?non-empty-string $covering
197     * @param int                           $parentPid the pid that forked this one
198     */
199    private function child(Mutant $mutant, string $mutatedTmp, string $resultTmp, callable $covering, int $parentPid): never
200    {
201        // Only the child is capped. The parent is bookkeeping and must
202        // outlive every mutant it runs.
203        ini_set('memory_limit', self::MEMORY_LIMIT);
204
205        $this->autoloader->activate($mutant->class, $mutatedTmp);
206
207        try {
208            $killedBy = $covering();
209            $payload  = $killedBy === null
210                ? ['outcome' => 'escaped']
211                : ['outcome' => 'killed', 'killedBy' => $killedBy];
212        } catch (Throwable $throwable) {
213            $payload = ['outcome' => 'errored', 'reason' => $throwable->getMessage()];
214        }
215
216        file_put_contents($resultTmp, (string) json_encode($payload));
217
218        // The one check that must hold before an uncatchable signal: this
219        // is a child, not the process that forked it. Throwing satisfies
220        // `never` and leaves a run that can still report; SIGKILL here in
221        // the parent would end it with no verdict and no message.
222        if (posix_getpid() === $parentPid) {
223            throw new RuntimeException('child() ran in the process that forked it; refusing to signal the parent.');
224        }
225
226        // Hard exit: the verdict file is the only thing this child leaves
227        // behind — no shutdown handler, no flushed output buffer.
228        posix_kill(posix_getpid(), SIGKILL);
229
230        exit(1); // unreachable once the signal lands; the `never` guarantee.
231    }
232
233    /**
234     * Wait for the child, enforcing the timeout, then read its verdict.
235     *
236     * @param non-empty-string $resultTmp
237     */
238    private function await(int $pid, Mutant $mutant, string $resultTmp, float $timeoutSeconds, float $started): MutationVerdict
239    {
240        $deadline = $started + $timeoutSeconds;
241        $status   = 0;
242
243        while (true) {
244            if (pcntl_waitpid($pid, $status, WNOHANG) === $pid) {
245                break;
246            }
247
248            if (microtime(true) >= $deadline) {
249                posix_kill($pid, SIGKILL);
250                pcntl_waitpid($pid, $status);
251
252                return MutationVerdict::timedOut($mutant, microtime(true) - $started);
253            }
254
255            usleep(self::POLL_MICROSECONDS);
256        }
257
258        return $this->verdictFrom($mutant, $resultTmp, microtime(true) - $started);
259    }
260
261    /**
262     * @param non-empty-string $resultTmp
263     */
264    private function verdictFrom(Mutant $mutant, string $resultTmp, float $duration): MutationVerdict
265    {
266        $raw     = file_get_contents($resultTmp);
267        $decoded = $raw === false || $raw === '' ? null : json_decode($raw, true);
268
269        if (!is_array($decoded) || !is_string($decoded['outcome'] ?? null)) {
270            // No verdict written — the child died before finishing, i.e. a
271            // fatal in the mutated code. That is an error, never a silent
272            // escape.
273            return MutationVerdict::errored($mutant, 'The mutant crashed the run before a verdict was recorded.', $duration);
274        }
275
276        $killedBy = is_string($decoded['killedBy'] ?? null) && $decoded['killedBy'] !== '' ? $decoded['killedBy'] : 'unknown';
277        $reason   = is_string($decoded['reason'] ?? null) && $decoded['reason'] !== '' ? $decoded['reason'] : 'The mutant errored.';
278
279        return match ($decoded['outcome']) {
280            'killed'  => MutationVerdict::killed($mutant, $killedBy, $duration),
281            'escaped' => MutationVerdict::escaped($mutant, $duration),
282            default   => MutationVerdict::errored($mutant, $reason, $duration),
283        };
284    }
