35/**
36 * The ambient browser runtime behind `visit()` — the Snapshots
37 * pattern (D-042): one static context the runner opens and closes
38 * around every test attempt, because a test body cannot be handed
39 * parameters. One driver session and one launched browser per
40 * process (launching is the expensive part); one fresh context per
41 * visit (the isolation unit); every context closed when its test
42 * ends. The default-off gate stays where it always was —
43 * Session::start — this class only adds the dialect-facing states:
44 * explicitly-disabled means SKIP here, not error.
45 */
46final class Browsing
47{
48    private static ?BrowserConfiguration $configuration = null;
49
50    private static ?WorkingDirectory $workingDirectory = null;
51    /**
52     * The configured directory, or the '/' the property used to default
53     * to. A static property cannot hold a `new`, so the default moved
54     * here rather than changing what an unconfigured read answers.
55     */
56    private static function workingDirectory(): WorkingDirectory
57    {
58        return self::$workingDirectory ??= new WorkingDirectory('/');
59    }

95    /** Closes every context the test opened. */
96    public static function end(): void
97    {
98        foreach (self::$testContexts as $context) {
99            try {
100                $context->close();
101            } catch (Throwable) {
102                // the browser may already be gone; teardown stays quiet
103            }
104        }
105
106        self::$testContexts = [];
107        self::$lastPage     = null;
108    }

110    /**
111     * The dialect entry point: a navigated page in a fresh context —
112     * or, for a URL list, one page per URL behind the fan-out surface
113     * (D-064).
114     *
115     * @param list<non-empty-string>|string $url
116     *
117     * @return ($url is string ? Page : PageCollection)
118     */
119    public static function visit(array|string $url, ?ContextOptions $options = null): Page|PageCollection
120    {
121        if (is_array($url)) {
122            $pages = [];
123
124            foreach ($url as $one) {
125                $pages[] = self::visit($one, $options);
126            }
127
128            if ($pages === []) {
129                throw new BrowserNotEnabledException('visit([]): the URL list is empty.');
130            }
131
132            return new PageCollection($pages);
133        }
134
135        $configuration = self::$configuration ?? new BrowserConfiguration();
136
137        if ($configuration->enabled === false) {
138            throw new SkippedTestError('The browser tier is explicitly disabled (->browser(enabled: false)).');
139        }
140
141        $relative = !str_starts_with($url, 'http://') && !str_starts_with($url, 'https://') && !str_starts_with($url, 'data:');
142
143        if ($relative && $configuration->requestHandler === null) {
144            throw new BrowserNotEnabledException(sprintf(
145                "visit('%s'): relative URLs resolve against the in-process application server — configure ->browser(requestHandler: YourHandler::class) (the Laravel bridge ships one). Until then, visit absolute URLs.",
146                $url,
147            ));
148        }
149
150        // The server ideally exists BEFORE the session so the driver's
151        // select loop watches it from the first frame (D-057); when a
152        // session already runs (an earlier absolute visit, another
153        // test), the transport late-watches it instead.
154        if ($configuration->requestHandler !== null && !self::$server instanceof InProcessServer) {
155            self::$server = new InProcessServer(self::handler($configuration->requestHandler));
156            self::$session?->watch(self::$server);
157        }
158
159        self::$session ??= Session::start($configuration, self::workingDirectory(), self::$server);
160        self::$browser ??= self::$session->launch();
161
162        if ($relative && self::$server instanceof InProcessServer) {
163            $url = rtrim(self::$server->baseUrl(), '/') . (str_starts_with($url, '/') ? $url : '/' . $url);
164        }
165
166        $context              = self::$browser->newContext($options);
167        self::$testContexts[] = $context;
168
169        $page           = $context->newPage()->navigate($url);
170        self::$lastPage = $page;
171
172        return $page;
173    }
174
175    /**
176     * Resolves the configured handler class — every misconfiguration
177     * is a named error at first use, never a silent 404.
178     *
179     * @param non-empty-string $class
180     */
181    private static function handler(string $class): RequestHandler
182    {
183        if (!class_exists($class)) {
184            throw new BrowserNotEnabledException(sprintf(
185                '->browser(requestHandler: %s) names a class that does not exist.',
186                $class,
187            ));
188        }
189
190        $handler = new $class();
191
192        if (!$handler instanceof RequestHandler) {
193            throw new BrowserNotEnabledException(sprintf(
194                '->browser(requestHandler: %s): the handler must implement %s.',
195                $class,
196                RequestHandler::class,
197            ));
198        }
199
200        return $handler;
201    }
202
203    /**
204     * A failure artifact from the test's most recent page, saved
205     * under tests/Browser/Screenshots — the incumbent's location.
206     * Null when the test never visited anything.
207     */
208    public static function failureScreenshot(TestId $test): ?string
209    {
210        self::$lastScreenshotFailure = null;
211        $page                        = self::$lastPage;
212
213        if (!$page instanceof Page) {
214            self::$lastScreenshotFailure = new RuntimeException('no page was visited by this test');
215
216            return null;
217        }
218
219        $directory = self::workingDirectory()->path . '/tests/Browser/Screenshots';
220
221        if (!is_dir($directory) && !@mkdir($directory, 0o755, true)) {
222            self::$lastScreenshotFailure = new RuntimeException('could not create ' . $directory);
223
224            return null;
225        }
226
227        $slug = trim((string) preg_replace('/[^\w-]+/', '_', $test->name), '_');
228
229        try {
230            return $page->screenshotTo(sprintf('%s/%s.png', $directory, $slug === '' ? 'test' : $slug));
231        } catch (Throwable $dead) {
232            // A dead page must not mask the real failure — but the
233            // reason must not vanish with it either. Returning a bare
234            // null left a missing screenshot with nothing to diagnose:
235            // a driver that died under load and a screenshot directory
236            // that could not be created looked identical from here.
237            self::$lastScreenshotFailure = $dead;
238
239            return null;
240        }
241    }
242
243    /**
244     * Why the last failure screenshot was not written, or null when one
245     * was.
246     *
247     * Kept as the Throwable rather than a message, because the class is
248     * the part that separates "the browser session died" — a
249     * BrowserProtocolException, and a precondition rather than a defect
250     * — from a screenshot path this engine genuinely got wrong.
251     */
252    public static function lastScreenshotFailure(): ?Throwable
253    {
254        return self::$lastScreenshotFailure;
255    }

257    /** Ends the process-wide session (shutdown / between runs). */
258    public static function shutdown(): void
259    {
260        try {
261            self::$browser?->close();
262            self::$session?->close();
263            self::$server?->close();
264        } catch (Throwable) {
265            // shutdown stays quiet
266        }
267
268        self::$browser = null;
269        self::$session = null;
270        self::$server  = null;
271    }
