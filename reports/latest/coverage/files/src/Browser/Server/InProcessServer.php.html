40/**
41 * The dependency-free HTTP/1.1 server that lives INSIDE the test
42 * process. Bound to 127.0.0.1 on an ephemeral port; pumped from the
43 * driver transport's select loop, so the browser's page loads are
44 * served while PHP waits for the navigation reply — one cooperative
45 * loop, no second process, and therefore one shared world: whatever
46 * the test faked, the served request sees.
47 *
48 * Deliberately simple where simple is correct: one request per
49 * connection (Connection: close), bodies via Content-Length only.
50 * Honest limit: a handler that blocks forever starves the loop —
51 * the handler runs app code in-process by design.
52 *
53 * Nothing here is browser-specific beyond where it was first needed:
54 * the watch loop's retrigger endpoint (D-083) is its second consumer,
55 * pumping the same cooperative loop from `stream_select`.
56 */
57final class InProcessServer implements Pumpable
58{
59    /** @var resource */
60    private $listener;
61
62    private readonly int $port;
63
64    /** @var array<int, array{stream: resource, buffer: string}> */
65    private array $connections = [];
66
67    /**
68     * @param int $port 0 = an ephemeral port; a fixed one only where the
69     *                  environment demands it (a container, a firewall rule)
70     */
71    public function __construct(
72        private readonly RequestHandler $handler,
73        int $port = 0,
74    ) {
75        $listener = @stream_socket_server('tcp://127.0.0.1:' . $port, $errorCode, $errorMessage);
76
77        if ($listener === false) {
78            throw new BrowserProtocolException(sprintf('Could not bind the in-process server: %s (%d).', $errorMessage, $errorCode));
79        }
80
81        $this->listener = $listener;
82        stream_set_blocking($this->listener, false);
83
84        $name       = stream_socket_get_name($this->listener, false);
85        $port       = $name === false ? null : parse_url('tcp://' . $name, PHP_URL_PORT);
86        $this->port = is_int($port) ? $port : throw new BrowserProtocolException('The in-process server has no port.');
87    }
88
89    public function port(): int
90    {
91        return $this->port;
92    }
93
94    public function baseUrl(): string
95    {
96        return sprintf('http://127.0.0.1:%d', $this->port);
97    }
98
99    /**
100     * @return list<resource>
101     */
102    #[Override]
103    public function watchStreams(): array
104    {
105        $streams = [$this->listener];
106
107        foreach ($this->connections as $connection) {
108            $streams[] = $connection['stream'];
109        }
110
111        return $streams;
112    }
113
114    #[Override]
115    public function pump(): void
116    {
117        // accept whatever is queued
118        while (($client = @stream_socket_accept($this->listener, 0)) !== false) {
119            stream_set_blocking($client, false);
120            $this->connections[(int) $client] = ['stream' => $client, 'buffer' => ''];
121        }
122
123        if ($this->connections === []) {
124            return;
125        }
126
127        // read what arrived, serve every completed request
128        $read = [];
129
130        foreach ($this->connections as $connection) {
131            $read[] = $connection['stream'];
132        }
133
134        $write  = null;
135        $except = null;
136
137        if (@stream_select($read, $write, $except, 0) === false) {
138            return;
139        }
140
141        foreach ($read as $stream) {
142            $key   = (int) $stream;
143            $chunk = fread($stream, 65_536);
144
145            if ($chunk === false || $chunk === '') {
146                // speculative connection closed by the browser
147                unset($this->connections[$key]);
148                fclose($stream);
149
150                continue;
151            }
152
153            $buffer                  = $this->connections[$key]['buffer'] . $chunk;
154            $this->connections[$key] = ['stream' => $stream, 'buffer' => $buffer];
155            $request                 = $this->completedRequest($buffer);
156
157            if ($request instanceof HttpRequest) {
158                $response = $this->handler->handle($request);
159                fwrite($stream, $response->render());
160                unset($this->connections[$key]);
161                fclose($stream);
162            }
163        }
164    }
165
166    public function close(): void
167    {
168        foreach ($this->connections as $connection) {
169            fclose($connection['stream']);
170        }
171
172        $this->connections = [];
173        fclose($this->listener);
174    }
175
176    /** Parses the buffer once headers (and any declared body) are whole. */
177    private function completedRequest(string $buffer): ?HttpRequest
178    {
179        $headerEnd = strpos($buffer, "\r\n\r\n");
180
181        if ($headerEnd === false) {
182            return null;
183        }
184
185        $head        = substr($buffer, 0, $headerEnd);
186        $lines       = explode("\r\n", $head);
187        $requestLine = explode(' ', $lines[0]);
188
189        if (count($requestLine) < 2) {
190            return null;
191        }
192
193        $headers = [];
194
195        foreach (array_slice($lines, 1) as $line) {
196            if (str_contains($line, ':')) {
197                [$name, $value]                   = explode(':', $line, 2);
198                $headers[strtolower(trim($name))] = trim($value);
199            }
200        }
201
202        $contentLength = (int) ($headers['content-length'] ?? 0);
203        $body          = substr($buffer, $headerEnd + 4);
204
205        if (strlen($body) < $contentLength) {
206            return null; // body still arriving
207        }
208
209        return new HttpRequest($requestLine[0], $requestLine[1], $headers, substr($body, 0, $contentLength));
210    }
