136    /**
137     * The current Inertia page object, or null when the page is not an
138     * Inertia page at all — which is itself worth distinguishing from
139     * "the component did not match".
140     *
141     * @return ?array<string, mixed>
142     */
143    public function inertiaPage(): ?array
144    {
145        $page = $this->script(InertiaRecorder::readExpression());
146
147        if (!is_array($page)) {
148            return null;
149        }
150
151        // A decoded JSON object, narrowed by inspection rather than
152        // asserted: the page object is keyed by name, and anything that
153        // is not simply is not part of it.
154        $named = [];
155
156        foreach ($page as $key => $value) {
157            if (is_string($key)) {
158                $named[$key] = $value;
159            }
160        }
161
162        return $named;
163    }

198    /**
199     * Walks a dot-separated path, returning null the moment it leaves
200     * the map — an absent prop and a null prop both read as null, and
201     * the assertion message names the path either way.
202     *
203     * @param array<mixed, mixed> $props decoded JSON of any shape: props nest
204     *                                   into lists as readily as into objects,
205     *                                   which is why `users.0.name` resolves
206     */
207    private static function dotted(array $props, string $key): mixed
208    {
209        $value = $props;
210
211        foreach (explode('.', $key) as $segment) {
212            if (!is_array($value) || !array_key_exists($segment, $value)) {
213                return null;
214            }
215
216            $value = $value[$segment];
217        }
218
219        return $value;
220    }

224    /**
225     * A Livewire component's current state, read from the `wire:snapshot`
226     * the component carries.
227     *
228     * No recorder here, and the asymmetry with Inertia is the finding
229     * rather than an oversight: Livewire **rewrites** `wire:snapshot` in
230     * the DOM on every round trip (observed against a real Livewire 4
231     * app — the counter's snapshot went from `count:0` to `count:1`,
232     * checksum and all), so the DOM is authoritative and an event
233     * listener would add a second source of truth for no gain. Inertia
234     * needed one precisely because it does the opposite.
235     *
236     * @param ?string $component name of the component to read; null = the first on the page
237     *
238     * @return ?array<string, mixed> null when the page carries no such component
239     */
240    public function wire(?string $component = null): ?array
241    {
242        $snapshot = $this->script(LivewireSnapshot::readExpression($component));
243
244        if (!is_array($snapshot)) {
245            return null;
246        }
247
248        $data = $snapshot['data'] ?? null;
249
250        if (!is_array($data)) {
251            return null;
252        }
253
254        $named = [];
255
256        foreach ($data as $key => $value) {
257            if (is_string($key)) {
258                $named[$key] = $value;
259            }
260        }
261
262        return $named;
263    }

660    /**
661     * Runs axe-core (the incumbent's engine — its failure messages
662     * carry axe rule links) against the page, reporting serious and
663     * critical violations (oracle-pinned: moderate landmark findings
664     * never appear in the incumbent's failures). The axe script is
665     * loaded from the SAME node_modules that provides Playwright:
666     * `npm install axe-core` — Crucible bundles no third-party code.
667     */
668    public function assertNoAccessibilityIssues(): self
669    {
670        $this->script($this->axeSource());
671        $results    = $this->script('axe.run(document)');
672        $violations = is_array($results) && is_array($results['violations'] ?? null) ? $results['violations'] : [];
673        $lines      = [];
674
675        foreach ($violations as $violation) {
676            if (!is_array($violation)) {
677                continue;
678            }
679
680            $impact = is_string($violation['impact'] ?? null) ? $violation['impact'] : 'unknown';
681
682            if ($impact !== 'serious' && $impact !== 'critical') {
683                continue;
684            }
685
686            $help    = is_string($violation['help'] ?? null) ? $violation['help'] : '';
687            $url     = is_string($violation['helpUrl'] ?? null) ? $violation['helpUrl'] : '';
688            $lines[] = sprintf('- [%s] %s %s', $impact, $help, $url);
689        }
690
691        Assert::assertSame([], $lines, "Accessibility issues found:\n" . implode("\n", $lines));
692
693        return $this;
694    }

714    private function axeSource(): string
715    {
716        $root      = $this->configuration->playwrightRoot;
717        $candidate = ($root ?? '.') . '/node_modules/axe-core/axe.min.js';
718
719        if ($root === null || !is_file($candidate)) {
720            throw new BrowserNotEnabledException(
721                "assertNoAccessibilityIssues needs axe-core next to the Playwright install:\n\n    npm install axe-core\n\nCrucible never bundles or downloads third-party code itself.",
722            );
723        }
724
725        $source = file_get_contents($candidate);
726
727        return $source === false ? '' : $source;
728    }

734        return $this->overMatches(Selector::resolve($target), 'els => els.length > 0') === true;

736
737    private function optionIsSelected(string $field, string $value): bool
738    {
739        return $this->overMatches(
740            Selector::field($field),
741            '(els, v) => [...els[0].options].some(o => o.selected && o.value === v)',
742            $value,
743        ) === true;
744    }
745
746    private static function linkSelector(string $text): string
747    {
748        return sprintf('a:has-text("%s")', str_replace('"', '\\"', $text));
749    }
750
751    /** One PHP_URL_* component of the current URL, '' when absent. */
752    private function urlPart(int $component): string
753    {
754        $part = parse_url($this->url(), $component);
755
756        if (is_string($part)) {
757            return $part;
758        }
759
760        return is_int($part) ? (string) $part : '';
761    }
762
763    /**
764     * @return array<string, string>
765     */
766    private function queryParameters(): array
767    {
768        $query      = $this->urlPart(PHP_URL_QUERY);
769        $parameters = [];
770
771        if ($query === '') {
772            return $parameters;
773        }
774
775        foreach (explode('&', $query) as $pair) {
776            [$name, $value]               = str_contains($pair, '=') ? explode('=', $pair, 2) : [$pair, ''];
777            $parameters[urldecode($name)] = urldecode($value);
778        }
779
780        return $parameters;
781    }
