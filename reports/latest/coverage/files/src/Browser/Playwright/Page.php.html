34/**
35 * A page, addressed through its main frame (the driver's navigation,
36 * interactions, and reads are frame methods; the page's initializer
37 * names its main frame — probe-pinned). The interaction surface
38 * follows the Pest browser vocabulary (spec/pest-api.md §6), each
39 * method one protocol call through the Selector grammar, each
40 * honoring the configured wait budget. Interactions and assertions
41 * return $this — the spec's tests are written as one fluent chain.
42 */

55    /**
56     * @param ?string $scopedFrameGuid a child frame to address instead of the
57     *                                 main frame — the withinFrame() surface
58     *                                 (D-064); every selector and read runs
59     *                                 inside that frame
60     */
61    public function __construct(
62        private Connection $connection,
63        public string $guid,
64        private BrowserConfiguration $configuration,
65        private string $contextGuid = '',
66        private ?string $scopedFrameGuid = null,
67    ) {
68        $this->mainFrameGuid = $scopedFrameGuid ?? $connection->object($guid)->referencedGuid('mainFrame');
69
70        if ($scopedFrameGuid === null) {
71            // Console events flow only once subscribed (probe-pinned);
72            // pageError events flow unconditionally. A frame-scoped
73            // surface shares the page's subscription.
74            $connection->subscribe($guid, 'console');
75
76            // Network events are subscribed here rather than inside
77            // waitForNetworkIdle() because subscribing is retroactive to
78            // nothing: an action that dispatches its request first — a
79            // click, a submit — would have its `request` event missed
80            // entirely, and the wait would see an idle page and return
81            // at once. That failure is silent, so the ordering trap is
82            // removed rather than documented.
83            foreach (['request', 'requestFinished', 'requestFailed'] as $event) {
84                $connection->subscribe($guid, $event);
85            }
86        }
87    }

261    /**
262     * Runs the callback against a surface scoped to an iframe: same
263     * vocabulary, every selector resolved inside the child frame.
264     *
265     * @param callable(self): mixed $callback
266     */
267    public function withinFrame(string $selector, callable $callback): self
268    {
269        $element = $this->frame('querySelector', ['selector' => Selector::resolve($selector)])['element'] ?? null;
270
271        if (!is_array($element) || !is_string($element['guid'] ?? null)) {
272            throw new BrowserProtocolException(sprintf('withinFrame(): no element matches "%s".', $selector));
273        }
274
275        $frame = $this->connection->call($element['guid'], 'contentFrame')['frame'] ?? null;
276
277        if (!is_array($frame) || !is_string($frame['guid'] ?? null)) {
278            throw new BrowserProtocolException(sprintf('withinFrame(): "%s" is not a frame.', $selector));
279        }
280
281        $callback(new self($this->connection, $this->guid, $this->configuration, $this->contextGuid, $frame['guid']));
282
283        return $this;
284    }

441    /**
442     * Chromium answers `Protocol error (Page.captureScreenshot): Unable
443     * to capture screenshot` when the renderer cannot produce a frame.
444     *
445     * ✓ Measured 2026-09-16 in a full run under concurrent load, which
446     * is where it fires. The session is ALIVE: it answers with a
447     * structured protocol error rather than dying, which disproves the
448     * previous diagnosis — "the driver connection dies under load" —
449     * that this code and BrowsingTest both carried.
450     *
451     * ⚠ The retry is not proven to clear it. The transient appeared once
452     * in roughly twenty full runs and could not be reproduced on demand:
453     * with attempts forced to 1, twelve runs under CPU saturation and
454     * eight concurrent Browser-suite runs all passed. So this gives the
455     * capture three chances where it had one, on the reading that a
456     * renderer too busy to paint will usually paint 250ms later. It is
457     * not a measured cure, and BrowsingTest keeps its skip as the
458     * backstop.
459     *
460     * Narrow on purpose: only that message retries. A closed page or a
461     * wrong path is not transient and must fail on the first attempt.
462     *
463     * @return array<string, mixed>
464     */
465    private function captureWithRetry(): array
466    {
467        $attempt = 0;
468
469        while (true) {
470            $attempt++;
471
472            try {
473                return $this->connection->call($this->guid, 'screenshot', [
474                    'timeout' => $this->configuration->timeoutMs,
475                    'type'    => 'png',
476                ]);
477            } catch (BrowserProtocolException $starved) {
478                if ($attempt >= self::SCREENSHOT_ATTEMPTS
479                    || !str_contains($starved->getMessage(), 'Unable to capture screenshot')
480                ) {
481                    throw $starved;
482                }
483
484                usleep(self::SCREENSHOT_RETRY_US);
485            }
486        }
487    }

524    /**
525     * Waits until the page stops talking to the network — no request in
526     * flight for a continuous quiet period.
527     *
528     * The primitive every framework round-trip needs: an Inertia visit
529     * or a Livewire update is a click followed by a request cycle, and
530     * asserting before it lands is the flake that `wait(0.5)` papers
531     * over. Built on the driver's own request/requestFinished/
532     * requestFailed events rather than a `performance` heuristic, so a
533     * request that fails still ends its own wait.
534     *
535     * The quiet period is what makes it usable: a handler that fires a
536     * request 50 ms after load would sail past a bare "nothing in
537     * flight right now" check.
538     *
539     * @param float $quietSeconds   how long the page must stay silent to count as settled
540     * @param float $timeoutSeconds how long to allow before calling it a failure — never silently
541     *
542     * @throws BrowserProtocolException when the page never settles
543     */
544    public function waitForNetworkIdle(float $quietSeconds = 0.25, float $timeoutSeconds = 5.0): self
545    {
546        foreach (['request', 'requestFinished', 'requestFailed'] as $event) {
547            $this->connection->subscribe($this->guid, $event);
548        }
549
550        $deadline = microtime(true) + $timeoutSeconds;
551        $quietAt  = null;
552
553        while (true) {
554            // The cheap round-trip that drains whatever the driver
555            // queued — the same idiom consoleLogs() uses.
556            $this->script('void 0');
557
558            $inFlight = $this->inFlightRequests();
559
560            if ($inFlight === 0) {
561                $quietAt ??= microtime(true);
562
563                if (microtime(true) - $quietAt >= $quietSeconds) {
564                    return $this;
565                }
566            } else {
567                $quietAt = null;
568            }
569
570            if (microtime(true) >= $deadline) {
571                throw new BrowserProtocolException(sprintf(
572                    'The page never went quiet: %d request(s) still in flight after %.1fs.',
573                    $inFlight,
574                    $timeoutSeconds,
575                ));
576            }
577
578            usleep(20_000);
579        }
580    }

582    /**
583     * Requests started but neither finished nor failed. Network events
584     * arrive on the browser context, not the page — probe-pinned, the
585     * same asymmetry consoleLogs() lives with.
586     */
587    private function inFlightRequests(): int
588    {
589        $started = 0;
590        $ended   = 0;
591
592        foreach ($this->connection->pageEventsFor($this->contextGuid) as $event) {
593            if ($event['method'] === 'request') {
594                ++$started;
595            } elseif ($event['method'] === 'requestFinished' || $event['method'] === 'requestFailed') {
596                ++$ended;
597            }
598        }
599
600        return max(0, $started - $ended);
601    }
