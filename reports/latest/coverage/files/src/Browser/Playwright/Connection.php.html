24/**
25 * The protocol layer over a transport, pinned black-box (probes
26 * 2026-07-16): requests are {id, guid, method, params, metadata} —
27 * metadata is mandatory (the driver crashes without it); replies are
28 * {id, result|error}; the remote object tree arrives as __create__ /
29 * __dispose__ events rooted at guid "" and is mirrored here so
30 * callers can read any object's initializer (the Playwright root
31 * carries the chromium/firefox/webkit guids, pages carry their main
32 * frame). Other events are ignored at this tier — the console/dialog
33 * streams become a listener seam when the assertions that need them
34 * land.
35 */
36final class Connection
37{
38    /**
39     * The id-less events worth keeping. Health data (D-057) plus the
40     * network lifecycle, which is what "has the page gone quiet?" is
41     * actually asking — request/finished/failed, never response, so a
42     * body that never arrives still counts as in flight.
43     */
44    private const array RECORDED_EVENTS = ['console', 'pageError', 'request', 'requestFinished', 'requestFailed'];
45
46    private int $nextId = 1;
47
48    /** @var array<string, true> guid|event pairs already subscribed */
49    private array $subscriptions = [];
50
51    /** Late-watches a Pumpable when the transport can pump (D-065). */
52    public function watch(Pumpable $pump): void
53    {
54        if ($this->transport instanceof DriverTransport) {
55            $this->transport->watch($pump);
56        }
57    }
58
59    /** @var array<string, RemoteObject> */
60    private array $objects = [];
61
62    /** @var list<array{guid: string, method: string, params: array<string, mixed>}> */
63    private array $pageEvents = [];
64
65    public function __construct(
66        private readonly Transport $transport,
67    ) {}
68
69    /**
70     * Send one request and block until its reply, folding object
71     * lifecycle events into the registry on the way.
72     *
73     * @param array<string, mixed> $params
74     *
75     * @return array<string, mixed>
76     */
77    public function call(string $guid, string $method, array $params = []): array
78    {
79        $id = $this->nextId++;
80
81        $this->transport->send([
82            'id'       => $id,
83            'guid'     => $guid,
84            'method'   => $method,
85            'params'   => $params === [] ? new stdClass() : $params,
86            'metadata' => new stdClass(),
87        ]);
88
89        while (true) {
90            $message = $this->transport->receive();
91
92            if ($this->absorbLifecycleEvent($message)) {
93                continue;
94            }
95
96            if (($message['id'] ?? null) !== $id) {
97                continue; // an event or a stale reply this tier does not track
98            }
99
100            if (isset($message['error'])) {
101                throw new BrowserProtocolException(sprintf(
102                    '%s.%s failed: %s',
103                    $guid,
104                    $method,
105                    $this->errorMessage($message['error']),
106                ));
107            }
108
109            $result = $message['result'] ?? [];
110
111            if (!is_array($result)) {
112                return [];
113            }
114
115            /** @var array<string, mixed> $result */
116            return $result;
117        }
118    }
119
120    public function object(string $guid): RemoteObject
121    {
122        return $this->objects[$guid] ?? throw new BrowserProtocolException(sprintf('Unknown remote object %s.', $guid));
123    }
124
125    /**
126     * Console/pageError events recorded for one emitter guid (they
127     * arrive on the browser-context — probe-pinned).
128     *
129     * @return list<array{method: string, params: array<string, mixed>}>
130     */
131    public function pageEventsFor(string $guid): array
132    {
133        $events = [];
134
135        foreach ($this->pageEvents as $event) {
136            if ($event['guid'] === $guid) {
137                $events[] = ['method' => $event['method'], 'params' => $event['params']];
138            }
139        }
140
141        return $events;
142    }
143
144    /**
145     * Ask the driver to start sending an event, at most once per
146     * emitter. Several events only flow once subscribed (probe-pinned),
147     * and the surfaces that need them are readonly, so remembering
148     * belongs here rather than in each caller.
149     */
150    public function subscribe(string $guid, string $event): void
151    {
152        $key = $guid . '|' . $event;
153
154        if (isset($this->subscriptions[$key])) {
155            return;
156        }
157
158        $this->subscriptions[$key] = true;
159
160        $this->call($guid, 'updateSubscription', ['event' => $event, 'enabled' => true]);
161    }
162
163    /** The first live object of a driver-side type (e.g. LocalUtils). */
164    public function objectOfType(string $type): RemoteObject
165    {
166        foreach ($this->objects as $object) {
167            if ($object->type === $type) {
168                return $object;
169            }
170        }
171
172        throw new BrowserProtocolException(sprintf('No remote object of type %s.', $type));
173    }
174
175    public function close(): void
176    {
177        $this->transport->close();
178    }
179
180    /**
181     * @param array<string, mixed> $message
182     */
183    private function absorbLifecycleEvent(array $message): bool
184    {
185        if (isset($message['id']) && is_int($message['id'])) {
186            return false;
187        }
188
189        $method = $message['method'] ?? null;
190        $params = $message['params'] ?? null;
191
192        if (in_array($method, self::RECORDED_EVENTS, true) && is_array($params) && is_string($message['guid'] ?? null)) {
193            /** @var array<string, mixed> $params */
194            $this->pageEvents[] = ['guid' => $message['guid'], 'method' => $method, 'params' => $params];
195
196            return true;
197        }
198
199        if ($method === '__create__' && is_array($params)
200            && is_string($params['guid'] ?? null) && is_string($params['type'] ?? null)) {
201            $initializer = $params['initializer'] ?? [];
202            /** @var array<string, mixed> $initializer */
203            $initializer = is_array($initializer) ? $initializer : [];
204            $parentGuid  = is_string($message['guid'] ?? null) ? $message['guid'] : '';
205
206            $this->objects[$params['guid']] = new RemoteObject($params['guid'], $params['type'], $parentGuid, $initializer);
207
208            return true;
209        }
210
211        if ($method === '__dispose__' && is_string($message['guid'] ?? null)) {
212            unset($this->objects[$message['guid']]);
213
214            return true;
215        }
216
217        return true; // every id-less message is an event; none others are tracked yet
218    }
219
220    private function errorMessage(mixed $error): string
221    {
222        if (is_array($error)) {
223            $inner = $error['error'] ?? $error;
224
225            if (is_array($inner) && is_string($inner['message'] ?? null)) {
226                return $inner['message'];
227            }
228        }
229
230        return (string) json_encode($error);
231    }
