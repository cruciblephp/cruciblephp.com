47/**
48 * The inline dialect frontend (growth G2c, DESIGN.md D-014/D-035):
49 * tests declared inside application source, not test files. Two tiers —
50 * `#[Check]` I/O tables on the function or method itself, and doctests:
51 * one expression per docblock tag, compiled into the file's namespace
52 * and run on the shared expect() surface. Both come out as the same
53 * dialect-neutral TestGroup every other frontend produces (D-008).
54 */
55final readonly class InlineBuilder
56{
57    public function __construct(
58        private ClassLocator $locator = new ClassLocator(),
59    ) {}
60
61    /**
62     * A doctest: `@crucible <expression>` on a starred docblock line — the
63     * one grammar the builder, the pre-filter and the PHPStan shadow read.
64     * `@crucible-equivalent` (D-134) and any other `@crucible-…` tag is not
65     * one: the tag must be followed by whitespace.
66     */
67    public const string DOCTEST = '/^[ \t]*\**[ \t]*@crucible[ \t]+(.+)$/m';
68
69    /**
70     * The discovery pre-filter: whether a source file can possibly
71     * declare inline tests, so discovery only loads files that opted in.
72     * A false positive is not merely one extra require: the file's classes
73     * are then declared before any test asks for them, and a mutated class
74     * can no longer load in their place — every mutant in the file would
75     * read as escaped (D-137). So the doctest test is the real grammar,
76     * not a substring.
77     */
78    public static function hasMarkers(string $source): bool
79    {
80        return str_contains($source, '#[Check')
81            || str_contains($source, 'Crucible\Attributes\Check')
82            || preg_match(self::DOCTEST, $source) === 1;
83    }

85    /**
86     * @param non-empty-string $file     absolute path
87     * @param non-empty-string $relative project-relative path, used for TestIds
88     */
89    public function build(string $file, string $relative): ?TestGroup
90    {
91        // Only what this require ADDS to the function table can have
92        // come from this file, so the candidates are a difference
93        // across it rather than every user function in the process. The
94        // require can autoload other files on the way, so the
95        // declaring-file check still decides — it just runs over a
96        // handful of names instead of thousands. A file something else
97        // already loaded adds nothing, and an empty difference would be
98        // wrong rather than merely small, so that case scans.
99        $resolved = realpath($file);
100        $loaded   = $resolved !== false && in_array($resolved, get_included_files(), true);
101        $before   = $loaded ? null : get_defined_functions()['user'];
102
103        require_once $file;
104
105        $declared = $this->functionsIn(
106            $file,
107            $before === null ? get_defined_functions()['user'] : array_diff(get_defined_functions()['user'], $before),
108        );
109
110        $definitions = [];
111
112        foreach ($this->locator->classesIn($file) as $class) {
113            if (!class_exists($class)) {
114                continue;
115            }
116
117            $reflection  = new ReflectionClass($class);
118            $classGroups = $this->groupsOf($reflection);
119
120            $definitions = [
121                ...$definitions,
122                ...$this->doctests($reflection->getDocComment(), $reflection->getShortName(), $reflection->getNamespaceName(), $classGroups, $relative),
123            ];
124
125            foreach ($reflection->getMethods() as $method) {
126                if ($method->getDeclaringClass()->getName() !== $class) {
127                    continue;
128                }
129
130                $name   = $reflection->getShortName() . '::' . $method->getName();
131                $groups = [...$classGroups, ...$this->groupsOf($method)];
132
133                $definitions = [
134                    ...$definitions,
135                    ...$this->checks($method, $reflection, $name, $groups, $relative),
136                    ...$this->doctests($method->getDocComment(), $name, $reflection->getNamespaceName(), $groups, $relative),
137                ];
138            }
139        }
140
141        foreach ($declared as $function) {
142            $name   = $function->getShortName();
143            $groups = $this->groupsOf($function);
144
145            if ($name === '') {
146                continue;
147            }
148
149            $definitions = [
150                ...$definitions,
151                ...$this->checks($function, null, $name, $groups, $relative),
152                ...$this->doctests($function->getDocComment(), $name, $function->getNamespaceName(), $groups, $relative),
153            ];
154        }
155
156        return $definitions === [] ? null : new TestGroup($relative, $definitions);
157    }

159    /**
160     * One test per #[Check] attribute, in declaration order. The claim
161     * hierarchy: `throws` expects the exception type, `returns`
162     * compares with the equality spec (toEqual semantics, matching
163     * table()), and a bare check claims the call completes.
164     *
165     * @param ?ReflectionClass<object> $class
166     * @param non-empty-string         $name
167     * @param list<Group>              $groups
168     * @param non-empty-string         $relative
169     *
170     * @return list<TestDefinition>
171     */
172    private function checks(ReflectionFunctionAbstract $target, ?ReflectionClass $class, string $name, array $groups, string $relative): array
173    {
174        $attributes = $target->getAttributes(Check::class);
175
176        if ($attributes === []) {
177            return [];
178        }
179
180        $metadata    = MetadataCollection::from(...$groups);
181        $definitions = [];
182        $seen        = [];
183
184        foreach ($attributes as $index => $attribute) {
185            $check = $attribute->newInstance();
186
187            if ($check->expectsReturn() && $check->throws !== null) {
188                throw new ConfigurationException(sprintf(
189                    '#[Check] on %s (%s) claims both returns: and throws: — pick one.',
190                    $name,
191                    $relative,
192                ));
193            }
194
195            $case = $check->name ?? 'check ' . ($index + 1);
196
197            if (isset($seen[$case])) {
198                throw new ConfigurationException(sprintf(
199                    '#[Check] on %s (%s): two checks named "%s".',
200                    $name,
201                    $relative,
202                    $case,
203                ));
204            }
205
206            $seen[$case] = true;
207            $invoker     = $this->invokerFor($target, $class, $name, $relative);
208
209            $definitions[] = new TestDefinition(
210                new TestId($relative, $name, $case),
211                static function (array $dependencyValues) use ($invoker, $check, $name): mixed {
212                    try {
213                        $result = $invoker()(...$check->args);
214                    } catch (Throwable $thrown) {
215                        if ($check->throws === null) {
216                            throw $thrown;
217                        }
218
219                        Assert::assertInstanceOf($check->throws, $thrown);
220
221                        return null;
222                    }
223
224                    if ($check->throws !== null) {
225                        Assert::fail(sprintf('Expected %s to be thrown by %s, but nothing was.', $check->throws, $name));
226                    }
227
228                    if ($check->expectsReturn()) {
229                        Assert::assertEquals($check->returns, $result);
230                    } else {
231                        Assert::countSatisfiedAssertion(); // completing without throwing is the claim
232                    }
233
234                    return $result;
235                },
236                $metadata,
237            );
238        }
239
240        return $definitions;
241    }

243    /**
244     * How a check reaches its target. The outer closure defers
245     * instance creation to run time, so every check gets a fresh
246     * instance and constructor failures land as that test's error.
247     * ReflectionMethod::getClosure() binds into the declaring scope,
248     * so non-public targets work — checks are not limited to the
249     * public surface.
250     *
251     * @param ?ReflectionClass<object> $class
252     * @param non-empty-string         $name
253     * @param non-empty-string         $relative
254     *
255     * @return Closure(): Closure
256     */
257    private function invokerFor(ReflectionFunctionAbstract $target, ?ReflectionClass $class, string $name, string $relative): Closure
258    {
259        if ($target instanceof ReflectionFunction) {
260            return $target->getClosure(...);
261        }
262
263        if (!$target instanceof ReflectionMethod || !$class instanceof ReflectionClass) {
264            throw new ConfigurationException(sprintf('#[Check] on %s (%s): not a checkable target.', $name, $relative));
265        }
266
267        if ($target->isStatic()) {
268            return $target->getClosure(...);
269        }
270
271        if ($target->isAbstract() || $class->isAbstract()) {
272            throw new ConfigurationException(sprintf(
273                '#[Check] on %s (%s): an abstract target cannot be called. Check a concrete subclass instead.',
274                $name,
275                $relative,
276            ));
277        }
278
279        $constructor = $class->getConstructor();
280
281        if ($constructor instanceof ReflectionMethod && $constructor->getNumberOfRequiredParameters() > 0) {
282            throw new ConfigurationException(sprintf(
283                '#[Check] on %s (%s): the constructor requires arguments, so no instance can be made. '
284                . 'Check a static factory or use a @crucible doctest.',
285                $name,
286                $relative,
287            ));
288        }
289
290        return static fn(): Closure => $target->getClosure($class->newInstance());
291    }

293    /**
294     * One test per @crucible docblock tag: the rest of the line is a
295     * single expression, compiled into the file's namespace now (a
296     * typo is a load error, not a silent skip) and evaluated at run
297     * time — documentation examples that cannot rot. `use` imports
298     * are not visible to the compiled expression; spell names
299     * namespace-relative or fully qualified.
300     *
301     * @param non-empty-string  $name
302     * @param list<Group>       $groups
303     * @param non-empty-string  $relative
304     * @return list<TestDefinition>
305     */
306    private function doctests(string|false $docComment, string $name, string $namespace, array $groups, string $relative): array
307    {
308        $expressions = $this->expressionsIn($docComment);
309
310        if ($expressions === []) {
311            return [];
312        }
313
314        PestBuilder::ensureUsable();
315
316        $metadata    = MetadataCollection::from(...$groups);
317        $definitions = [];
318
319        foreach ($expressions as $index => $expression) {
320            $closure = $this->compile($expression, $namespace, $name, $relative);
321
322            $definitions[] = new TestDefinition(
323                new TestId($relative, $name, 'crucible ' . ($index + 1)),
324                static fn(array $dependencyValues): mixed => $closure(),
325                $metadata,
326            );
327        }
328
329        return $definitions;
330    }

332    /**
333     * @return list<non-empty-string>
334     */
335    private function expressionsIn(string|false $docComment): array
336    {
337        if ($docComment === false || !str_contains($docComment, '@crucible')) {
338            return [];
339        }
340
341        preg_match_all(self::DOCTEST, $docComment, $matches);
342
343        $expressions = [];
344
345        foreach ($matches[1] as $expression) {
346            // A one-line docblock leaves its closer on the tag's line.
347            $expression = rtrim((string) preg_replace('~\s*\*/\s*$~', '', $expression));
348
349            if ($expression !== '') {
350                $expressions[] = $expression;
351            }
352        }
353
354        return $expressions;
355    }

357    /**
358     * @param non-empty-string $expression
359     * @param non-empty-string $where
360     * @param non-empty-string $relative
361     */
362    private function compile(string $expression, string $namespace, string $where, string $relative): Closure
363    {
364        $prefix = $namespace === '' ? '' : 'namespace ' . $namespace . '; ';
365
366        try {
367            $closure = GeneratedCode::evaluate(
368                // Parenthesised: this is the one generator whose payload
369                // is author text rather than Crucible-composed source,
370                // and the documented one-expression contract was
371                // otherwise unenforced. ✓ Neutral over PHP_INT_MAX,
372                // array literals, match, fn, yield, new, ?? and
373                // expect()->toBe(); an escape becomes a ParseError the
374                // catch below already reports against the docblock.
375                $prefix . 'return static function (): mixed { return (' . $expression . '); };',
376                'the @crucible doctest on ' . $where,
377            );
378        } catch (GeneratedCodeException $broken) {
379            // The seam's own message names the generated source, which
380            // is the right diagnosis when Crucible wrote it. Here the
381            // author wrote the expression, so the parse error is
382            // reported against THEIR docblock and the source we wrapped
383            // it in is noise. The ParseError travels as `previous`.
384            $parse = $broken->getPrevious();
385
386            throw new ConfigurationException(sprintf(
387                'The @crucible doctest on %s (%s) does not parse: %s — in `%s`.',
388                $where,
389                $relative,
390                $parse instanceof ParseError ? $parse->getMessage() : $broken->getMessage(),
391                $expression,
392            ), $broken->getCode(), $broken);
393        }
394
395        if (!$closure instanceof Closure) {
396            throw new ConfigurationException(sprintf('The @crucible doctest on %s (%s) is not one expression: `%s`.', $where, $relative, $expression));
397        }
398
399        return $closure;
400    }

402    /**
403     * @param ReflectionClass<object>|ReflectionFunctionAbstract $member
404     *
405     * @return list<Group>
406     */
407    private function groupsOf(ReflectionClass|ReflectionFunctionAbstract $member): array
408    {
409        return array_map(
410            static fn($attribute): Group => $attribute->newInstance(),
411            $member->getAttributes(Group::class),
412        );
413    }

415    /**
416     * Which of the candidate functions the file declares. Reflection
417     * rather than a token scan — precise, and no token games to tell
418     * functions from methods and closures, or to decide whether a
419     * conditionally declared one counts.
420     *
421     * The candidates are what `build()`'s own require added, so this
422     * runs over a handful of names; passed every user function in the
423     * process it still answers correctly, just slowly.
424     *
425     * @param non-empty-string   $file
426     * @param array<int, string> $candidates
427     *
428     * @return list<ReflectionFunction>
429     */
430    private function functionsIn(string $file, array $candidates): array
431    {
432        $resolved  = realpath($file);
433        $functions = [];
434
435        foreach ($candidates as $name) {
436            $reflection = new ReflectionFunction($name);
437            $declaredIn = $reflection->getFileName();
438
439            if ($declaredIn === $file || ($resolved !== false && $declaredIn === $resolved)) {
440                $functions[] = $reflection;
441            }
442        }
443
444        return $functions;
445    }
