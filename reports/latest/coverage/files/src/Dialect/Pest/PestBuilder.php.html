68/**
69 * The pest dialect frontend (growth G2): turns a *.pest.php file into
70 * the same dialect-neutral TestGroup every other frontend produces
71 * (D-008) — the engine cannot tell a test() closure from a TestCase
72 * method. Lifecycle is composed *inside* the definition closure:
73 * beforeEach/afterEach run bound to a fresh instance of the uses()
74 * class around each test body, so the engine needs no pest-specific
75 * hooks.
76 */
77final readonly class PestBuilder
78{
79    /**
80     * @param non-empty-string $file     absolute path
81     * @param non-empty-string $relative project-relative path, used for TestIds
82     */
83    public function build(string $file, string $relative): TestGroup
84    {
85        self::ensureUsable();
86        RealPhpUnitBootstrap::ensureConfigured();
87
88        // One separator from here on: on Windows the discovered path
89        // can arrive mixed (getcwd() gives '\', config joins with '/'),
90        // and every scope and dataset lookup below compares prefixes.
91        $file = WorkingDirectory::native($file);
92
93        // Suite-level configuration first: Pest.php and Datasets/*.php
94        // from every directory above the file, outermost first.
95        $root = rtrim(substr($file, 0, strlen($file) - strlen($relative)), '/' . DIRECTORY_SEPARATOR);
96
97        if ($root !== '') {
98            PestScopes::loadConfiguration($root, dirname($file));
99        }
100
101        PestRegistry::begin($file, $root);
102
103        require $file;
104
105        $state = $this->effectiveState(PestRegistry::drain(), $file);
106        $class = $state->uses ?? PestTestCase::class;
107
108        if (!class_exists($class)) {
109            throw new ConfigurationException(sprintf('uses(%s) in %s: the class does not exist.', $class, $relative));
110        }
111
112        $reflection = new ReflectionClass($class);
113
114        if ($reflection->isAbstract()) {
115            throw new ConfigurationException(sprintf('uses(%s) in %s: the class is abstract.', $class, $relative));
116        }
117
118        $class = TraitComposer::compose($class, [...$state->traits, ...AutoMixedTrait::forClass($reflection)]);
119        $class = SnapshotIdentityWrapper::wrap($class);
120
121        // Real Pest's own equivalent (spatie/pest-plugin-snapshots'
122        // override of SnapshotDirectoryAware) hardcodes this same
123        // 'tests' literal relative to the project root, not the
124        // individual test file's own directory — confirmed against a
125        // real pestphp/pest + spatie/pest-plugin-snapshots run
126        // (every snapshot landed in one project-root-level
127        // tests/__snapshots__/, regardless of how deeply nested the
128        // test file itself was). $testFileBasename mirrors real
129        // Pest's own TestCaseFactory::evaluate(), which names its
130        // generated class after the test file's own basename up to
131        // the first dot (so both SomeTest.php and SomeTest.pest.php
132        // become "SomeTest") — SnapshotIdAware's real, unoverridden
133        // id format bakes the running class's short name in, and
134        // real Pest's generated class is literally named after the
135        // file for exactly this reason.
136        $snapshotsDirectory = $root . '/tests/__snapshots__';
137        $testFileBasename   = basename($relative);
138        $dotPosition        = strpos($testFileBasename, '.');
139
140        if ($dotPosition !== false) {
141            $testFileBasename = substr($testFileBasename, 0, $dotPosition);
142        }
143
144        $definitions = [];
145
146        foreach ($state->calls as $call) {
147            // wip and done run their bodies like Pest; a todo, and any
148            // body-less call (nothing to run), stay pending. So a bare
149            // body-less call auto-todos, and a body-less wip/done is
150            // folded to a plain todo — its fields kept, its runnable
151            // status dropped. The marker travels as engine-visible
152            // metadata (D-045): --todos still selects on it and the
153            // runner blocks a todo before anything runs, every dialect
154            // alike.
155            $bodyless = !$call->test instanceof Closure && $call->chain === [];
156            $todo     = match (true) {
157                $bodyless && $call->todo instanceof Todo => new Todo(
158                    TodoStatus::Todo,
159                    $call->todo->assignee,
160                    $call->todo->issue,
161                    $call->todo->note,
162                ),
163                $bodyless => new Todo(),
164                default   => $call->todo,
165            };
166
167            $metadata = MetadataCollection::from(
168                ...array_map(
169                    static fn(string $group): Group => new Group($group),
170                    [...$state->groups, ...$call->groups],
171                ),
172                ...$this->coversOf($state),
173                ...($todo instanceof Todo ? [$todo] : []),
174                ...($call->expectedOutcome instanceof ExpectedOutcome ? [$call->expectedOutcome] : []),
175            );
176
177            // A dataset that throws while it is built fails its own test,
178            // not the run: real Pest reports the test and runs the rest of
179            // the suite. An uncaught throw here was a fatal for everything.
180            try {
181                $rows = $this->rows($call, dirname($file));
182            } catch (Throwable $datasetError) {
183                $definitions[] = new TestDefinition(
184                    new TestId($relative, $call->name()),
185                    static function () use ($datasetError): never {
186                        throw $datasetError;
187                    },
188                    $metadata,
189                    $call->dependsOn,
190                );
191
192                continue;
193            }
194
195            foreach ($rows as $datasetKey => $arguments) {
196                // PHP silently converts numeric-string array keys back
197                // to ints; the TestId wants the string form.
198                $datasetKey = (string) $datasetKey;
199
200                $id = new TestId($relative, $call->name(), $datasetKey === '' ? null : $datasetKey);
201
202                $definitions[] = new TestDefinition(
203                    $id,
204                    $this->definition($call, $arguments, $state, $class, $snapshotsDirectory, $testFileBasename, $id->dataset),
205                    $metadata,
206                    $call->dependsOn,
207                );
208            }
209        }
210
211        // A Pest file is a class (D-133): real Pest generates one extending
212        // the uses() class, and the runner calls its setUpBeforeClass() and
213        // tearDownAfterClass() around the file — the parent's first on the
214        // way in, the file's beforeAll() hooks after; afterAll() before the
215        // parent's on the way out. A base class that clears per-class state
216        // there (Orchestra Testbench) otherwise never clears it.
217        // The uses() class need not be a TestCase, so either may be absent.
218        $setUp    = method_exists($class, 'setUpBeforeClass') ? static fn(): mixed => $class::setUpBeforeClass() : null;
219        $tearDown = method_exists($class, 'tearDownAfterClass') ? static fn(): mixed => $class::tearDownAfterClass() : null;
220
221        return new TestGroup(
222            $relative,
223            $definitions,
224            $state->beforeAll === [] && $setUp === null ? null : static function () use ($state, $setUp): void {
225                if ($setUp !== null) {
226                    $setUp();
227                }
228
229                foreach ($state->beforeAll as $hook) {
230                    $hook();
231                }
232            },
233            $state->afterAll === [] && $tearDown === null ? null : static function () use ($state, $tearDown): void {
234                foreach ($state->afterAll as $hook) {
235                    $hook();
236                }
237
238                if ($tearDown !== null) {
239                    $tearDown();
240                }
241            },
242        );
243    }

245    /**
246     * Folds the matching Pest.php scopes into the file's own state:
247     * the file's uses() beats scoped extend(), traits and groups
248     * accumulate, global before-hooks run before file-level ones and
249     * global after-hooks after (spec §3). Bare registrations
250     * contribute hooks only.
251     *
252     * @param non-empty-string $file
253     */
254    private function effectiveState(PestFileState $state, string $file): PestFileState
255    {
256        $scopes = PestScopes::matching($file);
257
258        if ($scopes === []) {
259            return $state;
260        }
261
262        $class      = null;
263        $traits     = [];
264        $groups     = [];
265        $beforeEach = [];
266        $afterEach  = [];
267        $beforeAll  = [];
268        $afterAll   = [];
269
270        foreach ($scopes as $scope) {
271            if ($scope->globs !== null) {
272                $class  = $scope->class ?? $class;
273                $traits = [...$traits, ...$scope->traits];
274                $groups = [...$groups, ...$scope->groups];
275            }
276
277            $beforeEach = [...$beforeEach, ...$scope->beforeEach];
278            $afterEach  = [...$afterEach, ...$scope->afterEach];
279            $beforeAll  = [...$beforeAll, ...$scope->beforeAll];
280            $afterAll   = [...$afterAll, ...$scope->afterAll];
281        }
282
283        return new PestFileState(
284            $state->calls,
285            [...$beforeEach, ...$state->beforeEach],
286            [...$state->afterEach, ...$afterEach],
287            [...$beforeAll, ...$state->beforeAll],
288            [...$state->afterAll, ...$afterAll],
289            $state->uses ?? $class,
290            [...$traits, ...$state->traits],
291            [...$groups, ...$state->groups],
292            $state->covers,
293            $state->mutates,
294        );
295    }

297    /**
298     * covers() as the attributes the coverage reader already consumes.
299     * A target that names a live function becomes CoversFunction, and
300     * everything else CoversClass — the same split #[CoversClass] and
301     * #[CoversFunction] express, spelled file-scoped.
302     *
303     * @return list<CoversClass|CoversFunction|MutatesClass>
304     */
305    private function coversOf(PestFileState $state): array
306    {
307        $attributes = [];
308
309        foreach ($state->mutates as $target) {
310            if (!class_exists($target)) {
311                throw new ConfigurationException(sprintf(
312                    'mutates(%s): no such class. mutates() names the source to mutate, so a target '
313                    . 'that does not exist would silently narrow the run to nothing.',
314                    $target,
315                ));
316            }
317
318            $attributes[] = new MutatesClass($target);
319        }
320
321        foreach ($state->covers as $target) {
322            if (function_exists($target) && !class_exists($target)) {
323                $attributes[] = new CoversFunction($target);
324
325                continue;
326            }
327
328            if (class_exists($target) || interface_exists($target) || trait_exists($target)) {
329                $attributes[] = new CoversClass($target);
330
331                continue;
332            }
333
334            throw new ConfigurationException(sprintf(
335                'covers(%s): no such class or function. covers() names the code under test, so a '
336                . 'target that does not exist would silently cover nothing.',
337                $target,
338            ));
339        }
340
341        return $attributes;
342    }

350    /**
351     * Whether this process defined the pest vocabulary itself.
352     *
353     * The invariant D-019 protects is "never mask a real package", and
354     * the thing that would do the masking is REDEFINING a name the real
355     * package owns — not the package being present on disk. Keying the
356     * refusal on installation was a proxy for that, and it was wrong in
357     * both directions: it refused where the names were free, and it
358     * would have said nothing if some other package had taken them.
359     */
360    private static function vocabularyIsOurs(bool ...$set): bool
361    {
362        /** @var bool $ours */
363        static $ours = false;
364
365        if ($set !== []) {
366            $ours = $set[0];
367        }
368
369        return $ours;
370    }

392    public static function ensureUsable(): void
393    {
394        // `test()` already declared and not by us: the real package (or
395        // something else) owns the name, and PHP has no function_alias()
396        // to stand beside it. Redeclaring is an uncatchable fatal, so
397        // this is a refusal, not a preference.
398        if (function_exists('test') && !self::vocabularyIsOurs()) {
399            throw new ConfigurationException(
400                'the pest dialect stays off: test() is already declared, so its global functions '
401                . 'would mask whatever owns that name (pestphp/pest, normally). PHP has no '
402                . 'function_alias(), so they cannot coexist in one process. Either remove '
403                . 'pestphp/pest, or run Crucible with the vocabulary prelude, which asks Composer '
404                . "not to load Pest's two function files while leaving its classes intact:"
405                . PHP_EOL . PHP_EOL
406                . '    php -d auto_prepend_file=' . __DIR__ . '/vocabulary-prelude.php vendor/bin/crucible'
407                . PHP_EOL,
408            );
409        }
410
411        self::loadVocabulary();
412
413        // Pest\Laravel\* and Pest\Livewire\livewire() (growth G2
414        // follow-up): the real plugin packages that export these
415        // hard-require pestphp/pest itself, which the guard above
416        // already refuses to coexist with — so they can never
417        // actually be installed alongside this dialect, and these
418        // proxy files are safe to load unconditionally rather than
419        // gated on some other package's presence.
420        if (!function_exists('Pest\Laravel\mock')) {
421            require dirname(__DIR__, 2) . '/Bridge/PestLaravel/functions.php';
422        }
423
424        if (!function_exists('Pest\Livewire\livewire')) {
425            require dirname(__DIR__, 2) . '/Bridge/PestLivewire/functions.php';
426        }
427
428        if (!function_exists('Spatie\Snapshots\assertMatchesSnapshot')) {
429            require dirname(__DIR__, 2) . '/Bridge/PestSnapshots/functions.php';
430        }
431    }

433    /**
434     * Dataset rows keyed for TestIds: '' means no dataset. A list row
435     * spreads as positional arguments, an associative row binds to
436     * closure parameters by name (spec §4), anything else is one
437     * argument. Multiple ->with() calls multiply (the spec's
438     * Cartesian product; keys join with ' / '), and ->repeat(n) is
439     * one more factor.
440     *
441     * PHP degrades numeric-string keys to ints at runtime, so the
442     * declared key type is array-key even though only strings are
443     * ever assigned.
444     *
445     * @param non-empty-string $fileDirectory
446     *
447     * @return array<array-key, array<array-key, mixed>>
448     */
449    private function rows(TestCall $call, string $fileDirectory): array
450    {
451        $rows = ['' => []];
452
453        foreach ($call->datasets as $dataset) {
454            $factor = $this->materializeFactor($dataset, $fileDirectory);
455            $next   = [];
456
457            foreach ($rows as $key => $arguments) {
458                foreach ($factor as $rowKey => $row) {
459                    $rowArguments = is_array($row) ? $row : [$row];
460
461                    $next[$this->joinKeys($key, (string) $rowKey)] = [...$arguments, ...$rowArguments];
462                }
463            }
464
465            $rows = $next;
466        }
467
468        if ($call->repetitions > 1) {
469            $next = [];
470
471            foreach ($rows as $key => $arguments) {
472                for ($repetition = 1; $repetition <= $call->repetitions; ++$repetition) {
473                    $label = sprintf('repetition %d of %d', $repetition, $call->repetitions);
474
475                    $next[$this->joinKeys($key, $label)] = $arguments;
476                }
477            }
478
479            $rows = $next;
480        }
481
482        return $rows;
483    }

485    /**
486     * PHP degrades numeric-string array keys to ints, so the left
487     * side (an accumulated key read back from an array) arrives as
488     * either type.
489     */
490    private function joinKeys(int|string $left, string $right): string
491    {
492        return $left === '' ? $right : $left . ' / ' . $right;
493    }

495    /**
496     * One ->with() factor as rows: inline arrays pass through, a
497     * string resolves a shared dataset() (nearest folder wins), a
498     * closure is a lazy factory invoked now.
499     *
500     * @param array<array-key, mixed>|non-empty-string|Closure $dataset
501     * @param non-empty-string                                 $fileDirectory
502     *
503     * @return array<array-key, mixed>
504     */
505    private function materializeFactor(array|string|Closure $dataset, string $fileDirectory): array
506    {
507        if (is_string($dataset)) {
508            return PestScopes::resolveDataset($dataset, $fileDirectory);
509        }
510
511        if ($dataset instanceof Closure) {
512            $rows = $dataset();
513
514            if (!is_iterable($rows)) {
515                throw new ConfigurationException('with(closure): the closure must return an iterable of rows.');
516            }
517
518            return PestScopes::materialize($rows);
519        }
520
521        return $dataset;
522    }

524    /**
525     * @param array<array-key, mixed> $arguments dataset arguments (string keys bind by name), before dependency injections
526     * @param class-string            $class
527     * @param non-empty-string        $snapshotsDirectory
528     * @param ?non-empty-string       $dataset
529     *
530     * @return Closure(list<mixed>): mixed
531     */
532    private function definition(TestCall $call, array $arguments, PestFileState $state, string $class, string $snapshotsDirectory, string $testFileBasename, ?string $dataset = null): Closure
533    {
534        // The real PHPUnit\Framework\TestCase::__construct() requires
535        // its $name argument (Orchestra Testbench's TestCase inherits
536        // it unchanged, being final); Crucible's own TestCase declares
537        // no constructor at all, so passing an argument it doesn't
538        // expect would itself be the error. Checked once per call, at
539        // build time, not per invocation.
540        $classReflection = new ReflectionClass($class);
541        $needsName       = ($classReflection->getConstructor()?->getNumberOfRequiredParameters() ?? 0) > 0;
542
543        // #[Before]/#[PreCondition]/#[PostCondition]/#[After] — real
544        // PHPUnit's own attributes too (HookPlanner, via
545        // MetadataParser/RealPhpUnitHookAttributes), not just
546        // Crucible's. A separate mechanism from beforeEach/afterEach
547        // above: those are pest's own dialect-level hooks; these are
548        // attributed methods on the uses() class itself, or on a
549        // trait mixed into it (Spatie\Snapshots\MatchesSnapshots's
550        // real #[PostCondition] method, the case this was built for
551        // — verified against a real phpunit/phpunit run first, then
552        // confirmed Crucible produced a different result before this
553        // existed). Computed once per TestCall, reused across every
554        // dataset row, matching $needsName above.
555        $hooks = HookPlanner::forClass($classReflection);
556
557        return static function (array $dependencyValues) use ($call, $arguments, $state, $class, $needsName, $hooks, $snapshotsDirectory, $testFileBasename, $dataset): mixed {
558            /** @var list<mixed> $dependencyValues */
559            if ($call->skipped === true || is_string($call->skipped)) {
560                throw new SkippedTestError(is_string($call->skipped) ? $call->skipped : 'Skipped.');
561            }
562
563            $instance = $needsName ? new $class($call->name()) : new $class();
564
565            // Named before setUp(), matching invokeTest()'s contract —
566            // the phpunit dialect's seam for the same thing. Without
567            // it a pest test has no name to answer with, which is what
568            // the snapshot bridge's identity is built from.
569            if ($instance instanceof TestCase) {
570                $instance->nameTest($call->name(), $dataset);
571            }
572
573            $invoke = static function (Closure $closure, array $args = []) use ($instance, $class): mixed {
574                $bound = @Closure::bind($closure, $instance, $class);
575
576                return ($bound ?? $closure)(...$args);
577            };
578
579            // A uses() class may be a real setUp()/tearDown()-based
580            // TestCase — Crucible's own, or (via phpunit/phpunit,
581            // e.g. Orchestra Testbench) the real PHPUnit one. Real
582            // Pest runs those as genuine PHPUnit test cases, so their
583            // setUp() is what boots the app; skipping it silently
584            // starved every Laravel-package Pest suite of its
585            // container (confirmed against spatie/laravel-data).
586            // Duck-typed, not an instanceof check, since either
587            // TestCase family satisfies the same contract. Invoked by
588            // reflection, not a bound closure, since setUp/tearDown
589            // are protected and PHP has allowed reflection to reach
590            // non-public methods without setAccessible() since 8.1.
591            $hasLifecycle  = method_exists($instance, 'setUp') && method_exists($instance, 'tearDown');
592            $isRealPhpUnit = method_exists($instance, 'numberOfAssertionsPerformed');
593
594            // Scoped for Pest\Laravel\* (src/Bridge/PestLaravel), which
595            // reaches "the instance currently running a test" the same
596            // way real Pest's own plugin proxies do (test()'s zero-arg
597            // form) — cleared in the finally block below regardless of
598            // outcome, so a failure never leaks the instance into the
599            // next test.
600            CurrentTest::set($instance);
601
602            // Scoped the same way, for SnapshotIdentityWrapper's
603            // generated overrides (src/Bridge/PestSnapshots) — see
604            // SnapshotFileContext's own docblock for why this exists
605            // at all.
606            SnapshotFileContext::set($snapshotsDirectory, $testFileBasename);
607
608            if ($hasLifecycle && $isRealPhpUnit) {
609                RealPhpUnitBootstrap::resetAssertionCount();
610            }
611
612            // The before list carries setUp() at its priority-0 place
613            // (HookPlan): the #[Before] hooks ahead of it run first, as
614            // under the incumbent, and beforeEach runs right after it,
615            // since real Pest's setUp() is what calls beforeEach. The
616            // hooks behind setUp() wait for the closure-form skip.
617            $setUpAt     = array_search(HookPlan::SET_UP, $hooks->before, true);
618            $beforeSetUp = $setUpAt === false ? [] : array_slice($hooks->before, 0, $setUpAt);
619            $behindSetUp = $setUpAt === false ? $hooks->before : array_slice($hooks->before, $setUpAt + 1);
620
621            foreach ($beforeSetUp as $hook) {
622                (new ReflectionMethod($instance, $hook))->invoke($instance);
623            }
624
625            if ($hasLifecycle) {
626                (new ReflectionMethod($instance, 'setUp'))->invoke($instance);
627            }
628
629            foreach ($state->beforeEach as $hook) {
630                $invoke($hook);
631            }
632
633            try {
634                // The spec evaluates closure-form skips after beforeEach,
635                // so the condition can read hook-provided state.
636                if ($call->skipped instanceof Closure) {
637                    $verdict = $invoke($call->skipped);
638
639                    if ($verdict !== false && $verdict !== null) {
640                        throw new SkippedTestError(match (true) {
641                            is_string($verdict)      => $verdict,
642                            $call->skipReason !== '' => $call->skipReason,
643                            default                  => 'Skipped.',
644                        });
645                    }
646                }
647
648                foreach ($behindSetUp as $hook) {
649                    (new ReflectionMethod($instance, $hook))->invoke($instance);
650                }
651
652                foreach ($hooks->preConditions as $hook) {
653                    self::invokeHook($instance, $hook);
654                }
655
656                // Bound rows (spec §4): a closure-valued dataset argument
657                // resolves after beforeEach, bound to the instance — but
658                // only when BOTH: it takes zero required parameters
659                // (`fn() => new ArrayObject([$this->x])`, DESIGN.md's own
660                // example — a closure requiring parameters can't be
661                // zero-arg-invoked at all, confirmed against a real
662                // spatie/laravel-data case, ->with(fn () => yield [...,
663                // fn (LazyData $data) => $data->include('name'), ...]));
664                // AND the test's own parameter at this position isn't
665                // itself typed to accept a callable — `Closure
666                // $temporaryPartial` (another real spatie/laravel-data
667                // case) or `callable|null $setDefaults` wants the closure
668                // itself, not what calling it produces, even when the
669                // closure happens to take zero arguments. Named arguments
670                // go last so positional ones stay positional.
671                $positional = [];
672                $named      = [];
673
674                foreach ($arguments as $key => $argument) {
675                    $argument = $argument instanceof Closure
676                        && (new ReflectionFunction($argument))->getNumberOfRequiredParameters() === 0
677                        && !self::targetParameterAcceptsCallable($call->test, $key)
678                        ? $invoke($argument)
679                        : $argument;
680
681                    if (is_int($key)) {
682                        $positional[] = $argument;
683                    } else {
684                        $named[$key] = $argument;
685                    }
686                }
687
688                $args = [...$positional, ...$dependencyValues, ...$named];
689                $body = self::body($call, $instance, $class);
690
691                if ($call->fails) {
692                    try {
693                        $body($args);
694
695                        // Verification failures are test failures, so
696                        // fails() must cover them too (D-046).
697                        if ($instance instanceof TestCase) {
698                            $instance->verifyTestDoubles();
699                        }
700                    } catch (SkippedTestError|IncompleteTestError $signal) {
701                        throw $signal;
702                    } catch (Throwable $thrown) {
703                        if ($call->failsMessage !== null) {
704                            Assert::assertStringContainsString($call->failsMessage, $thrown->getMessage());
705                        }
706
707                        Assert::countSatisfiedAssertion(); // the expected failure happened
708
709                        return null;
710                    }
711
712                    Assert::fail('Expected the test to fail, but it passed.');
713                }
714
715                try {
716                    $value = $body($args);
717                } catch (SkippedTestError|IncompleteTestError $signal) {
718                    throw $signal;
719                } catch (Throwable $thrown) {
720                    if ($call->throws !== null) {
721                        self::assertThrown($thrown, $call->throws, $call->throwsMessage);
722
723                        return null;
724                    }
725
726                    // Real PHPUnit's own $this->expectException() family
727                    // (PHPUnit\Framework\Assert, on a real
728                    // PHPUnit-descended uses() class) — a separate
729                    // mechanism from ->throws() above, only ever
730                    // verified by real PHPUnit's own runner, which
731                    // PestBuilder never delegates to. Rethrows its own
732                    // AssertionFailedError when set but mismatched,
733                    // same as real PHPUnit would fail the test rather
734                    // than silently accept a wrong-shaped exception.
735                    if (RealPhpUnitExceptionExpectations::wasExpected($instance, $thrown)) {
736                        return null;
737                    }
738
739                    throw $thrown;
740                }
741
742                if ($call->throws !== null) {
743                    Assert::fail(sprintf('Expected %s to be thrown, but nothing was.', $call->throws));
744                }
745
746                RealPhpUnitExceptionExpectations::verifyNotUnraised($instance);
747
748                if ($call->throwsNothing) {
749                    Assert::countSatisfiedAssertion(); // the absence was verified
750                }
751
752                // End-of-test double settlement (D-046) — the seam the
753                // phpunit dialect gets from invokeTest. Skipped on the
754                // expected-exception paths above, matching that spec.
755                if ($instance instanceof TestCase) {
756                    $instance->verifyTestDoubles();
757                }
758
759                // Success-path only, same as TestCase::invokeTest()'s
760                // own placement — unlike before/preConditions, a
761                // postCondition does not run on the fails()/throws()
762                // early-return paths above, matching real PHPUnit:
763                // it only fires once the test method itself completed
764                // normally.
765                foreach ($hooks->postConditions as $hook) {
766                    self::invokeHook($instance, $hook);
767                }
768
769                return $value;
770            } finally {
771                // Unconditional, matching TestCase::invokeTest()'s own
772                // "the after phase always runs" contract. tearDown()
773                // holds its priority-0 place in the list (HookPlan), and
774                // afterEach runs right before it, since real Pest's
775                // tearDown() is what calls afterEach.
776                foreach ($hooks->after as $hook) {
777                    if ($hook !== HookPlan::TEAR_DOWN) {
778                        (new ReflectionMethod($instance, $hook))->invoke($instance);
779
780                        continue;
781                    }
782
783                    foreach ($state->afterEach as $afterEach) {
784                        $invoke($afterEach);
785                    }
786
787                    if ($hasLifecycle) {
788                        (new ReflectionMethod($instance, 'tearDown'))->invoke($instance);
789                    }
790                }
791
792                // Real PHPUnit's own TestCase tracks assertions
793                // separately from Crucible's. Two sources feed it,
794                // both invisible to Crucible's own counter without
795                // bridging: Orchestra Testbench's Mockery integration
796                // verifies expectations during tearDown() and reports
797                // them via addToAssertionCount() directly; and
798                // $this->assertSame()-style calls made anywhere in the
799                // test count through a separate static counter that
800                // only ever reaches the instance via
801                // RealPhpUnitBootstrap::bridgeAssertionCount() (see its
802                // own docblock). Without either, such tests are wrongly
803                // flagged risky. Bridged the same way real Laravel test
804                // helpers report into PHPUnit's counter
805                // (Assert::addToAssertionCount()'s own docblock).
806                if ($isRealPhpUnit) {
807                    RealPhpUnitBootstrap::bridgeAssertionCount($instance);
808
809                    $performed = $instance->numberOfAssertionsPerformed();
810
811                    if (is_int($performed)) {
812                        (new class extends Assert {})->addToAssertionCount($performed);
813                    }
814                }
815
816                // Cleared last, after afterEach/tearDown have both had
817                // their chance to still reach it — matching real Pest's
818                // own scoping, where the instance stays "current" for
819                // the whole test lifecycle, not just its body.
820                CurrentTest::set(null);
821                SnapshotFileContext::set(null, null);
822            }
823        };
824    }

826    /**
827     * Whether the test closure's own parameter at this dataset slot is
828     * itself typed to accept a callable — `Closure`/`callable`, bare or
829     * inside a union — in which case a zero-arg closure-valued row must
830     * be passed through as-is, not auto-invoked (see definition()'s
831     * bound-rows comment). Only positional slots resolve; a named row
832     * key can't be matched to a parameter by position.
833     */
834    /**
835     * A condition hook, or the condition template of its phase — which a
836     * uses() class may not have (it need not be a TestCase), and which
837     * then has nothing to run.
838     */
839    private static function invokeHook(object $instance, string $hook): void
840    {
841        if (HookPlan::isTemplate($hook) && !method_exists($instance, $hook)) {
842            return;
843        }
844
845        (new ReflectionMethod($instance, $hook))->invoke($instance);
846    }
847
848    private static function targetParameterAcceptsCallable(?Closure $test, int|string $key): bool
849    {
850        if (!$test instanceof Closure || !is_int($key)) {
851            return false;
852        }
853
854        $parameters = (new ReflectionFunction($test))->getParameters();
855        $type       = ($parameters[$key] ?? null)?->getType();
856
857        return $type !== null && self::typeAcceptsCallable($type);
858    }
859
860    private static function typeAcceptsCallable(ReflectionType $type): bool
861    {
862        if ($type instanceof ReflectionNamedType) {
863            return in_array($type->getName(), ['callable', 'Closure'], true);
864        }
865
866        if (!$type instanceof ReflectionUnionType) {
867            // An intersection type: callable/Closure can't be a member
868            // of one (both are non-interface types), so it never matches.
869            return false;
870        }
871
872        foreach ($type->getTypes() as $member) {
873            if ($member instanceof ReflectionNamedType && self::typeAcceptsCallable($member)) {
874                return true;
875            }
876        }
877
878        return false;
879    }

881    /**
882     * The runnable body: the test closure, or — higher-order tests —
883     * the collected chain replayed against the instance. `->expect()`
884     * re-roots the chain on an Expectation (its closure receives the
885     * dataset/dependency arguments); every other step is called on
886     * the running subject, so `get('/')->assertStatus(200)` reads the
887     * way the spec writes it.
888     *
889     * @param class-string $class
890     *
891     * @return Closure(array<array-key, mixed>): mixed
892     */
893    private static function body(TestCall $call, object $instance, string $class): Closure
894    {
895        $test = $call->test;
896
897        if ($test instanceof Closure) {
898            return static function (array $args) use ($test, $instance, $class): mixed {
899                $bound = @Closure::bind($test, $instance, $class);
900
901                return ($bound ?? $test)(...$args);
902            };
903        }
904
905        return static function (array $args) use ($call, $instance, $class): mixed {
906            $subject = $instance;
907
908            foreach ($call->chain as $position => [$member, $stepArguments, $isProperty]) {
909                if ($isProperty) {
910                    $subject = $subject->{$member};
911
912                    continue;
913                }
914
915                $stepArguments = array_map(
916                    static fn(mixed $argument): mixed => $argument instanceof Closure
917                        ? (@Closure::bind($argument, $instance, $class) ?? $argument)
918                        : $argument,
919                    $stepArguments,
920                );
921
922                if ($subject === $instance && $member === 'expect') {
923                    $value = $stepArguments[0] ?? null;
924
925                    // toThrow needs the closure unevaluated — it invokes
926                    // and observes it itself. Everything else expects the
927                    // produced value.
928                    $subject = new Expectation(
929                        $value instanceof Closure && !self::nextMatcherIsToThrow($call->chain, $position)
930                            ? $value(...$args)
931                            : $value,
932                    );
933
934                    continue;
935                }
936
937                $subject = $subject->{$member}(...$stepArguments);
938            }
939
940            return null;
941        };
942    }

944    /**
945     * Whether the first matcher after the expect step (skipping
946     * property reads like ->not) is toThrow.
947     *
948     * @param list<array{non-empty-string, list<mixed>, bool}> $chain
949     */
950    private static function nextMatcherIsToThrow(array $chain, int $expectPosition): bool
951    {
952        for ($i = $expectPosition + 1, $count = count($chain); $i < $count; ++$i) {
953            [$member, , $isProperty] = $chain[$i];
954
955            if ($isProperty) {
956                continue;
957            }
958
959            return $member === 'toThrow';
960        }
961
962        return false;
963    }

965    /**
966     * The spec's throws() dual reading: a class name checks the type,
967     * anything else is a message fragment.
968     *
969     * @param non-empty-string $exception
970     */
971    private static function assertThrown(Throwable $thrown, string $exception, ?string $message): void
972    {
973        if (class_exists($exception) || interface_exists($exception)) {
974            Assert::assertInstanceOf($exception, $thrown);
975        } else {
976            $message ??= $exception;
977        }
978
979        if ($message !== null) {
980            Assert::assertStringContainsString($message, $thrown->getMessage());
981        }
982    }
