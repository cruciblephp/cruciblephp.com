28/**
29 * pest()->use(Trait::class) needs the trait mixed into the binding
30 * class, and PHP has no runtime trait application — so a subclass is
31 * generated once per (class, traits) combination and cached, the
32 * same eval-once pattern the doubles generator uses (D-017).
33 */
34final class TraitComposer
35{
36    /** @var array<string, class-string> */
37    private static array $composed = [];
38
39    /**
40     * @param class-string       $class
41     * @param list<class-string> $traits
42     *
43     * @return class-string
44     */
45    public static function compose(string $class, array $traits): string
46    {
47        $traits = array_values(array_unique($traits));
48
49        if ($traits === []) {
50            return $class;
51        }
52
53        sort($traits);
54
55        $key = $class . '|' . implode(',', $traits);
56
57        if (isset(self::$composed[$key])) {
58            return self::$composed[$key];
59        }
60
61        $reflection = new ReflectionClass($class);
62
63        if ($reflection->isFinal()) {
64            throw new ConfigurationException(sprintf(
65                'Cannot mix traits into %s: the class is final.',
66                $class,
67            ));
68        }
69
70        // Both checks below refuse what PHP would refuse anyway. The
71        // point is not the verdict, which is identical, but that an
72        // uncatchable compile fatal has no exit code and no message of
73        // ours: ✓ measured, the collision died at 255 naming
74        // ComposedTestCase_<md5> and "eval()'d code" — Crucible's
75        // internals, for a mistake in the user's uses() call. The
76        // incumbent does not do that either: ✓ pest 5.1.1 catches the
77        // same fatal in a shutdown handler and reports it at exit 1.
78        // Refusing before eval() is what makes a chosen exit code and a
79        // message about the user's own file possible at all.
80        self::guardMethods($reflection, $traits);
81        self::guardProperties($reflection, $traits);
82
83        $short = 'ComposedTestCase_' . md5($key);
84        $fqcn  = __NAMESPACE__ . '\\Composed\\' . $short;
85
86        if (!class_exists($fqcn, false)) {
87            GeneratedCode::evaluate(
88                sprintf(
89                    'namespace %s\\Composed; class %s extends \\%s { use \\%s; }',
90                    __NAMESPACE__,
91                    $short,
92                    $class,
93                    implode(', \\', $traits),
94                ),
95                $class . ' composed with ' . implode(', ', $traits),
96            );
97        }
98
99        /** @var class-string $fqcn */
100        return self::$composed[$key] = $fqcn;
101    }
102    /**
103     * Everything about the METHODS that PHP would refuse, in one pass
104     * over them.
105     *
106     * Three rules read the same reflection — a collision between two
107     * traits, a trait bringing a method the base sealed, and an
108     * abstract requirement nobody answers — so they share the walk
109     * rather than each opening their own. What differs between them is
110     * the question, not the data.
111     *
112     * @param ReflectionClass<object> $reflection
113     * @param list<class-string>      $traits
114     */
115    private static function guardMethods(ReflectionClass $reflection, array $traits): void
116    {
117        $sealed   = [];
118        $provided = [];
119        $required = [];
120
121        foreach ($reflection->getMethods() as $method) {
122            $name = strtolower($method->getName());
123
124            if ($method->isFinal()) {
125                $sealed[$name] = $method->getDeclaringClass()->getName();
126            }
127
128            if ($method->isAbstract()) {
129                $required[$name] = $reflection->getName() . '::' . $method->getName() . '()';
130
131                continue;
132            }
133
134            $provided[$name] = true;
135        }
136
137        /** @var array<string, class-string> $declaredBy method name => the trait that brought it */
138        $declaredBy = [];
139
140        foreach ($traits as $trait) {
141            foreach ((new ReflectionClass($trait))->getMethods() as $method) {
142                $name = strtolower($method->getName());
143
144                // An abstract trait method is a REQUIREMENT, not a
145                // rival declaration: ✓ measured, an abstract greeting()
146                // beside a concrete one composes.
147                if ($method->isAbstract()) {
148                    $required[$name] ??= $trait . '::' . $method->getName() . '()';
149
150                    continue;
151                }
152
153                if (isset($declaredBy[$name])) {
154                    throw new ConfigurationException(sprintf(
155                        'Cannot mix %s and %s together: both declare %s(), and PHP resolves that only with '
156                            . 'an insteadof clause, which uses() cannot write. Bind one of them, or rename '
157                            . 'the method.',
158                        $declaredBy[$name],
159                        $trait,
160                        $method->getName(),
161                    ));
162                }
163
164                if (isset($sealed[$name])) {
165                    throw new ConfigurationException(sprintf(
166                        'Cannot mix %s into %s: it declares %s(), which %s declares final.',
167                        $trait,
168                        $reflection->getName(),
169                        $method->getName(),
170                        $sealed[$name],
171                    ));
172                }
173
174                $declaredBy[$name] = $trait;
175                $provided[$name]   = true;
176            }
177        }
178
179        $unimplemented = [];
180
181        foreach ($required as $name => $spelling) {
182            if (!isset($provided[$name])) {
183                $unimplemented[] = $spelling;
184            }
185        }
186
187        if ($unimplemented !== []) {
188            throw new ConfigurationException(sprintf(
189                'Cannot mix traits into %s: nothing here implements %s, and the composed class is '
190                    . 'concrete, which PHP refuses.',
191                $reflection->getName(),
192                implode(', ', $unimplemented),
193            ));
194        }
195    }

197    /**
198     * A property may be declared twice only if the two declarations are
199     * IDENTICAL.
200     *
201     * ✓ Measured: same name and type but a different default is fatal,
202     * so is a different visibility, so is a different type — against
203     * another trait or against the base. Identical is fine, which is
204     * why the comparison is a signature rather than a name.
205     *
206     * @param ReflectionClass<object> $reflection
207     * @param list<class-string>      $traits
208     */
209    private static function guardProperties(ReflectionClass $reflection, array $traits): void
210    {
211        /** @var array<string, array{0: string, 1: string}> $seen name => [signature, who declared it] */
212        $seen = [];
213
214        foreach ($reflection->getProperties() as $property) {
215            $seen[$property->getName()] = [
216                self::propertySignature($property),
217                $property->getDeclaringClass()->getName(),
218            ];
219        }
220
221        foreach ($traits as $trait) {
222            foreach ((new ReflectionClass($trait))->getProperties() as $property) {
223                $signature = self::propertySignature($property);
224                $known     = $seen[$property->getName()] ?? null;
225
226                if ($known === null) {
227                    $seen[$property->getName()] = [$signature, $trait];
228
229                    continue;
230                }
231
232                if ($known[0] !== $signature) {
233                    throw new ConfigurationException(sprintf(
234                        'Cannot mix %s into %s: it declares $%s as "%s", and %s declares it as "%s". PHP '
235                            . 'allows the same property twice only when the two declarations are identical.',
236                        $trait,
237                        $reflection->getName(),
238                        $property->getName(),
239                        $signature,
240                        $known[1],
241                        $known[0],
242                    ));
243                }
244            }
245        }
246    }
247
248    /**
249     * What has to match, spelled out: visibility, static, readonly,
250     * type, and the default.
251     */
252    private static function propertySignature(ReflectionProperty $property): string
253    {
254        $visibility = match (true) {
255            $property->isPrivate()   => 'private',
256            $property->isProtected() => 'protected',
257            default                  => 'public',
258        };
259
260        return implode(' ', [
261            $visibility,
262            $property->isStatic() ? 'static' : 'instance',
263            $property->isReadOnly() ? 'readonly' : 'mutable',
264            (string) $property->getType(),
265            $property->hasDefaultValue() ? var_export($property->getDefaultValue(), true) : '<no default>',
266        ]);
267    }
