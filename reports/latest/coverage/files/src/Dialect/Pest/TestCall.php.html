33/**
34 * What test()/it() return: the chainable handle the pest spec hangs
35 * datasets, skips, groups, exception expectations, and dependencies
36 * on. Mutable while its file is being loaded; the builder reads the
37 * final state once the file has finished executing.
38 *
39 * A call without a body is either a todo (no chain) or a higher-order
40 * test: unknown chained methods are collected via __call and replayed
41 * against the test-case instance at run time.
42 */

106    /**
107     * Anything that is not a declared chainable is a higher-order
108     * step — but only on a body-less test; on a test with a body it
109     * can only be a typo, and failing at load time beats a confusing
110     * run-time error.
111     *
112     * @param non-empty-string $name
113     * @param list<mixed>      $arguments
114     */
115    public function __call(string $name, array $arguments): self
116    {
117        if ($this->test instanceof Closure) {
118            throw new ConfigurationException(sprintf(
119                '->%s() is not a pest chainable; higher-order chaining is only available on tests without a body.',
120                $name,
121            ));
122        }
123
124        $this->chain[] = [$name, $arguments, false];
125
126        return $this;
127    }
128
129    /**
130     * Property reads chain too — `check(fn() => 5)->not->toBe(6)`
131     * records ->not for replay on the expectation.
132     *
133     * @param non-empty-string $name
134     */
135    public function __get(string $name): self
136    {
137        if ($this->test instanceof Closure) {
138            throw new ConfigurationException(sprintf(
139                '->%s is not a pest chainable; higher-order chaining is only available on tests without a body.',
140                $name,
141            ));
142        }
143
144        $this->chain[] = [$name, [], true];
145
146        return $this;
147    }

149    /**
150     * @param iterable<array-key, mixed>|non-empty-string|Closure $dataset rows, a shared dataset name, or a lazy factory
151     */
152    public function with(iterable|string|Closure $dataset): self
153    {
154        if (is_string($dataset) || $dataset instanceof Closure) {
155            $this->datasets[] = $dataset;
156
157            return $this;
158        }
159
160        $rows = [];
161
162        foreach ($dataset as $key => $row) {
163            $rows[$key] = $row;
164        }
165
166        $this->datasets[] = $rows;
167
168        return $this;
169    }

171    public function skip(bool|string|Closure $condition = true, string $reason = ''): self
172    {
173        if ($condition === false) {
174            return $this;
175        }
176
177        if ($condition instanceof Closure) {
178            $this->skipped    = $condition;
179            $this->skipReason = $reason;
180
181            return $this;
182        }
183
184        $this->skipped = match (true) {
185            $condition === true => $reason !== '' ? $reason : true,
186            default             => $condition,
187        };
188
189        return $this;
190    }

232    /**
233     * @param non-empty-string $constraint an operator-prefixed version, e.g. '>=8.0.0'
234     */
235    public function skipOnPhp(string $constraint): self
236    {
237        if (preg_match('/^(<=|>=|<>|!=|<|>|==?)?\s*(.+)$/', $constraint, $matches) !== 1) {
238            throw new ConfigurationException(sprintf('skipOnPhp(%s): not a version constraint.', $constraint));
239        }
240
241        $operator = match ($matches[1]) {
242            '', '=' => '==',
243            default => $matches[1],
244        };
245
246        return $this->skip(
247            version_compare(PHP_VERSION, $matches[2], $operator),
248            'Skipped on PHP ' . $constraint . '.',
249        );
250    }

384    /**
385     * The declared name inside the file: describe path joined with
386     * the spec's "it " prefix applied.
387     *
388     * @return non-empty-string
389     */
390    public function name(): string
391    {
392        $leaf = ($this->kind === 'it' ? 'it ' : '') . $this->description;
393
394        return $this->describePath === []
395            ? $leaf
396            : implode(' > ', $this->describePath) . ' > ' . $leaf;
