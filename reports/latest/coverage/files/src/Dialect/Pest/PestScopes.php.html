36/**
37 * The suite-level side of the pest dialect: Pest.php scope
38 * registrations and shared datasets, discovered by walking the
39 * directories above each test file (outermost first, each loaded
40 * once per process). This is where pest()->extend()->in() and
41 * dataset() deposit what they declare.
42 */
43final class PestScopes
44{
45    /** @var list<ScopeRegistration> */
46    private static array $registrations = [];

48    /**
49     * Dataset name → declarations; several directories may own the
50     * same name (the spec's per-folder scoping).
51     *
52     * @var array<string, list<array{non-empty-string, array<array-key, mixed>|Closure}>>
53     */
54    private static array $datasets = [];

59    /**
60     * Non-null while a Pest.php or Datasets file is being required.
61     *

64    private static ?string $configDir = null;

66    /**
67     * Loads Pest.php and Datasets/*.php from every directory between
68     * the project root and the test file's directory, top-down —
69     * outer configuration registers before inner and therefore
70     * applies first.
71     *
72     * @param non-empty-string $root
73     * @param non-empty-string $directory
74     */
75    public static function loadConfiguration(string $root, string $directory): void
76    {
77        foreach (self::chain(WorkingDirectory::native($root), WorkingDirectory::native($directory)) as $dir) {
78            if (isset(self::$loaded[$dir])) {
79                continue;
80            }
81
82            self::$loaded[$dir] = true;
83
84            if (is_file($dir . '/Pest.php')) {
85                self::requireConfiguration($dir . '/Pest.php', $dir);
86            }
87
88            $datasetFiles = glob($dir . '/Datasets/*.php');
89
90            foreach ($datasetFiles === false ? [] : $datasetFiles as $datasetFile) {
91                self::requireConfiguration($datasetFile, $dir);
92            }
93        }
94    }

137    /**
138     * @param non-empty-string                     $name
139     * @param iterable<array-key, mixed>|Closure   $rows
140     */
141    public static function dataset(string $name, iterable|Closure $rows): void
142    {
143        $dir = self::$configDir ?? PestRegistry::currentFileDirectory() ?? throw new ConfigurationException(
144            'dataset() can only be called from a Pest.php, Datasets file, or while a *.pest.php file is loading.',
145        );
146
147        self::$datasets[$name][] = [WorkingDirectory::native($dir), $rows instanceof Closure ? $rows : self::materialize($rows)];
148    }

150    /**
151     * Per-folder scoping: among declarations whose directory contains
152     * the test file, the nearest wins. Closure-valued datasets are
153     * re-invoked per use, so generators replay.
154     *
155     * @param non-empty-string $name
156     * @param non-empty-string $fileDirectory
157     *
158     * @return array<array-key, mixed>
159     */
160    public static function resolveDataset(string $name, string $fileDirectory): array
161    {
162        $best          = null;
163        $bestLength    = -1;
164        $fileDirectory = WorkingDirectory::native($fileDirectory);
165
166        foreach (self::$datasets[$name] ?? [] as [$baseDir, $rows]) {
167            $contains = $fileDirectory === $baseDir
168                || str_starts_with($fileDirectory . DIRECTORY_SEPARATOR, $baseDir . DIRECTORY_SEPARATOR);
169
170            if ($contains && strlen($baseDir) > $bestLength) {
171                $best       = $rows;
172                $bestLength = strlen($baseDir);
173            }
174        }
175
176        if ($best === null) {
177            throw new ConfigurationException(sprintf(
178                "Unknown dataset '%s' — declare it via dataset() in a Datasets/*.php file or Pest.php above the test file.",
179                $name,
180            ));
181        }
182
183        if ($best instanceof Closure) {
184            $rows = $best();
185
186            if (!is_iterable($rows)) {
187                throw new ConfigurationException(sprintf("Dataset '%s': the closure must return an iterable.", $name));
188            }
189
190            return self::materialize($rows);
191        }
192
193        return $best;
194    }

196    /**
197     * Registrations that apply to a test file, in declaration order
198     * (outer configuration first). Bare registrations contribute
199     * hooks only — the builder reads $globs to know.
200     *
201     * @param non-empty-string $file absolute path
202     *
203     * @return list<ScopeRegistration>
204     */
205    public static function matching(string $file): array
206    {
207        $result = [];
208        $file   = WorkingDirectory::native($file);
209
210        foreach (self::$registrations as $registration) {
211            if (!str_starts_with($file, $registration->baseDir . DIRECTORY_SEPARATOR)) {
212                continue;
213            }
214
215            if ($registration->globs === null) {
216                if ($registration->hasHooks()) {
217                    $result[] = $registration;
218                }
219
220                continue;
221            }
222
223            // Globs are written with '/' whatever the OS, and
224            // FNM_PATHNAME only treats '/' as a separator.
225            $relative = str_replace(DIRECTORY_SEPARATOR, '/', substr($file, strlen($registration->baseDir) + 1));
226
227            foreach ($registration->globs as $glob) {
228                if (self::matches($glob, $relative)) {
229                    $result[] = $registration;
230
231                    break;
232                }
233            }
234        }
235
236        return $result;
237    }

239    /**
240     * Iterators may yield any key type; only array keys survive.
241     *
242     * A generator's key is preserved only when the *first* one is a
243     * string — a deliberate, author-chosen case label — matching real
244     * Pest's own DatasetsRepository::processDatasets() exactly
245     * (`iterator_to_array($generator, preserveKeys: is_string($generator->key()))`).
246     * Implicit int keys are NOT preserved: a real-world dataset
247     * commonly composes many sub-generators via `yield from`
248     * (confirmed against spatie/laravel-data's own
249     * tests/Datasets/RulesDataset.php, ~35 of them), each restarting
250     * its own 0-based numbering — preserving those would silently
251     * drop the vast majority of rows to array-key collision (found
252     * the hard way: 129 materialized rows where real Pest — and the
253     * real project's own test suite — produces 413, because only the
254     * last colliding sub-generator's row for each int key survived).
255     *
256     * @param iterable<mixed, mixed> $rows
257     *
258     * @return array<array-key, mixed>
259     */
260    public static function materialize(iterable $rows): array
261    {
262        if (is_array($rows)) {
263            return $rows;
264        }
265
266        $materialized = [];
267        $preserveKeys = null;
268
269        foreach ($rows as $key => $row) {
270            if (!is_int($key) && !is_string($key)) {
271                throw new ConfigurationException('Dataset keys must be ints or strings.');
272            }
273
274            $preserveKeys ??= is_string($key);
275
276            if ($preserveKeys) {
277                $materialized[$key] = $row;
278            } else {
279                $materialized[] = $row;
280            }
281        }
282
283        return $materialized;
284    }

286    /**
287     * A plain name selects a directory subtree; anything with a
288     * wildcard is an fnmatch pattern where * does not cross '/'.
289     */
290    private static function matches(string $glob, string $relative): bool
291    {
292        // ->in('.') is the real-world idiom for "this whole directory,
293        // no further restriction" — confirmed against
294        // spatie/laravel-data's own Pest.php
295        // (uses(TestCase::class)->in('.')), boilerplate that
296        // spatie/laravel-package-tools' starter template ships to
297        // every Laravel package built from it. '.' has no glob
298        // metacharacters, so without this it fell into the literal-
299        // prefix branch below and matched nothing.
300        if ($glob === '.') {
301            return true;
302        }
303
304        if (!str_contains($glob, '*') && !str_contains($glob, '?') && !str_contains($glob, '[')) {
305            return $relative === $glob || str_starts_with($relative, $glob . '/');
306        }
307
308        return fnmatch($glob, $relative, FNM_PATHNAME);
309    }

311    /**
312     * The directories from $root down to $directory, both inclusive.
313     *
314     * @param non-empty-string $root
315     * @param non-empty-string $directory
316     *
317     * @return list<non-empty-string>
318     */
319    private static function chain(string $root, string $directory): array
320    {
321        if ($directory === $root || !str_starts_with($directory, $root . DIRECTORY_SEPARATOR)) {
322            return [$directory];
323        }
324
325        $chain   = [$root];
326        $current = $root;
327
328        foreach (explode(DIRECTORY_SEPARATOR, trim(substr($directory, strlen($root)), DIRECTORY_SEPARATOR)) as $segment) {
329            $current .= DIRECTORY_SEPARATOR . $segment;
330            $chain[] = $current;
331        }
332
333        return $chain;
334    }

341        self::$configDir = $baseDir;
