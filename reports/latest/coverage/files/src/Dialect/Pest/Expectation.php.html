116/**
117 * The pest spec's expect() chain, compiled onto Crucible's constraint
118 * engine (D-008: every dialect's assertions funnel through the same
119 * constraints, so failures look identical everywhere and risky
120 * detection counts them for free).
121 *
122 * `->not` negates exactly the next matcher; `->and($value)` starts a
123 * fresh chain on a new value; `->each` fans the following matchers
124 * out over every item of an iterable value. Property access that is
125 * not a modifier is a higher-order expectation: `expect($user)->name`
126 * is `expect($user->name)`.
127 *
128 * The value's type rides the chain for the analyser (D-128): see
129 * `ExpectationChainReturnTypeExtension`.
130 *
131 * @template-covariant TValue = mixed
132 *
133 * @property-read Expectation<TValue> $each
134 * @property-read Expectation<TValue> $not
135 */

221    /**
222     * With a callback, runs it against an expectation of every item;
223     * without one, returns an expectation whose following matchers
224     * apply to every item (the `->each->toBeInt()` form).
225     *
226     * @param ?Closure(self): mixed $callback
227     */
228    public function each(?Closure $callback = null): self
229    {
230        if (!$callback instanceof Closure) {
231            $each         = new self($this->value);
232            $each->spread = true;
233
234            return $each;
235        }
236
237        foreach ($this->items('each()') as $item) {
238            $callback(new self($item));
239        }
240
241        return $this;
242    }

244    /**
245     * One expectation per item, in iteration order: a closure receives
246     * (expectation, key); any other value means toEqual. The counts
247     * must agree — a silent partial match would hide missing items.
248     */
249    public function sequence(mixed ...$expectations): self
250    {
251        $items = $this->items('sequence()');
252
253        if (count($items) !== count($expectations)) {
254            Assert::fail(sprintf(
255                'sequence() received %d expectations for %d items.',
256                count($expectations),
257                count($items),
258            ));
259        }
260
261        $position = 0;
262
263        foreach ($items as $key => $item) {
264            $expected = $expectations[$position++];
265
266            if ($expected instanceof Closure) {
267                $expected(new self($item), $key);
268            } else {
269                (new self($item))->toEqual($expected);
270            }
271        }
272
273        return $this;
274    }

357    /**
358     * `expect($user)->name` is `expect($user->name)`; on arrays the
359     * name is a key. `->each` is the spread modifier; `->not` negates
360     * exactly the next matcher.
361     */
362    public function __get(string $name): self
363    {
364        if ($name === 'each') {
365            return $this->each();
366        }
367
368        if ($name === 'not') {
369            return $this->negate();
370        }
371
372        $root  = $this->root ?? $this;
373        $child = new self($this->member($this->shouldReset ? $root->value : $this->value, $name));
374
375        $child->root = $root;
376
377        return $child;
378    }
379
380    /**
381     * Custom matchers registered via extend() win; any other unknown
382     * method call is forwarded to the underlying value and the chain
383     * continues on its return value.
384     *
385     * @param list<mixed> $arguments
386     */
387    public function __call(string $name, array $arguments): self
388    {
389        $extension = self::$extensions[$name] ?? null;
390
391        if ($extension !== null) {
392            $bound = Closure::bind($extension, $this, self::class) ?? $extension;
393            $bound(...$arguments);
394
395            return $this;
396        }
397
398        $root    = $this->root ?? $this;
399        $subject = $this->shouldReset ? $root->value : $this->value;
400
401        if (!is_object($subject)) {
402            Assert::fail(sprintf(
403                'Method %s() is neither a matcher nor callable on the expectation value (%s).',
404                $name,
405                Exporter::describe($subject),
406            ));
407        }
408
409        $child       = new self($subject->{$name}(...$arguments));
410        $child->root = $root;
411
412        return $child;
413    }

789    /**
790     * The arch matchers, and the one reading D1 settles.
791     *
792     * A subject is resolved the way `arch()` resolves a target — the
793     * declared symbol it names, PLUS everything beneath it in the
794     * configured source — through the one rule both spellings share, so
795     * they cannot drift into meaning different things by the same
796     * words. That subsumes the value reading rather than competing with
797     * it: `expect(Foo::class)` resolves to exactly `{Foo}`, which is the
798     * same answer asking about the string itself gave.
799     *
800     * A symbol outside the configured source still answers for itself.
801     * Pure targeting would have said "not in your source, therefore
802     * nothing to check" about a vendor class someone named on purpose.
803     *
804     * **An empty match passes, and warns.** That is the incumbent's
805     * behaviour — measured: an arch expectation matching nothing passes
806     * vacuously there — so a Pest suite gets Pest's verdict and Pest's
807     * exit code. Crucible disagrees out loud instead of silently: the
808     * warning names the target, `--fail-on-warning` turns it into a
809     * failure, and D-088's position that a rule targeting nothing
810     * enforces nothing stays available without being imposed on a
811     * borrowed suite.
812     *
813     * The empty case short-circuits BEFORE negation reaches it, but only
814     * once negation has become a predicate of its own. ✓ Measured
815     * 2026-09-07: a hand-written `not->` form is an ordinary rule, so an
816     * empty layer passes it vacuously; the `__call` fallback runs the
817     * positive rule and fails when it passes, so there an empty layer is
818     * a FAILURE. `toBeCasedCorrectly` is the only arch matcher on it.
819     *
820     * ✓ `->not` is not a logical inversion here. pest 5.1.1's
821     * OppositeExpectation hand-writes a SECOND per-class predicate per
822     * matcher and asserts it over every target, keeping the universal
823     * quantifier. A matcher with no $negated form keeps the inversion,
824     * which is what the incumbent's `__call` fallback does.
825     *
826     * @param Closure(class-string): bool  $predicate
827     * @param non-empty-string             $description
828     * @param ?Closure(class-string): bool $negated     the incumbent's own `not->` predicate, where it has one
829     * @param ?non-empty-string            $negatedDescription
830     */
831    private function arch(
832        Closure $predicate,
833        string $description,
834        ?Closure $negated = null,
835        ?string $negatedDescription = null,
836    ): self {
837        if ($this->negated && $negated instanceof Closure) {
838            $this->negated = false;
839            $predicate     = $negated;
840            $description   = $negatedDescription ?? $description;
841        }
842
843        return $this->archOverTargets(static fn(array $targets): Closure => $predicate, $description);
844    }

846    /**
847     * The same, for a matcher whose answer depends on the whole target
848     * set rather than on one class at a time: the dependency matchers,
849     * where a reference between two TARGETED classes does not cross the
850     * boundary the rule is about. The factory is called once the
851     * targets are known, so the set is the resolved one even when the
852     * subject is spread.
853     *
854     * @param Closure(list<class-string>): (Closure(class-string): bool) $factory
855     * @param non-empty-string                                           $description
856     */
857    private function archOverTargets(Closure $factory, string $description): self
858    {
859        if (is_string($this->value) && $this->value !== '' && !$this->spread && self::targetsOf($this->value) === []) {
860            trigger_error(
861                sprintf('Nothing matches %s, so this expectation asserts nothing.', $this->value),
862                E_USER_WARNING,
863            );
864
865            // Still negated means no `not->` predicate to swap in, so the
866            // incumbent is on __call and inverts the vacuous pass.
867            if (!$this->negated) {
868                $this->shouldReset = true;
869
870                return $this;
871            }
872        }
873
874        return $this->assert(new Callback(
875            static function (mixed $value) use ($factory): bool {
876                if (!is_string($value) || $value === '') {
877                    return false;
878                }
879
880                $targets = self::targetsOf($value);
881
882                if ($targets === []) {
883                    return true;
884                }
885
886                $predicate = $factory($targets);
887
888                foreach ($targets as $class) {
889                    if (!$predicate($class)) {
890                        return false;
891                    }
892                }
893
894                return true;
895            },
896            $description,
897        ));
898    }

2027    /** Higher-order member access: object property, or array key. */
2028    /**
2029     * A missing property or key is null, not a failure — real Pest's
2030     * own retrieve() is `$value->$key ?? $default`, the same silent
2031     * fallback (confirmed against Pest's Retrievable trait). A real
2032     * spatie/laravel-data test relies on exactly this: it chains
2033     * ->dataCollectionClass (the real property, on DataPropertyType,
2034     * is spelled dataCollectABLEClass — the test has had this typo
2035     * for a while) ->toBeNull(), which only ever passed because the
2036     * missing-property fallback IS null.
2037     */
2038    private function member(mixed $subject, string $name): mixed
2039    {
2040        // property_exists() is true for a declared-but-uninitialized
2041        // typed property, but reading it directly still fatals
2042        // ("must not be accessed before initialization") — real Pest's
2043        // own retrieve() is $value->$key ?? $default specifically to
2044        // stay safe there: ?? uses isset() semantics, which treats an
2045        // uninitialized typed property as unset instead of fataling
2046        // (confirmed against a real spatie/laravel-data case: a
2047        // property left uninitialized by disabled name-mapping).
2048        if (is_object($subject) && property_exists($subject, $name)) {
2049            return $subject->{$name} ?? null;
2050        }
2051
2052        if (is_array($subject) && array_key_exists($name, $subject)) {
2053            return $subject[$name];
2054        }
2055
2056        if ($subject instanceof ArrayAccess && $subject->offsetExists($name)) {
2057            return $subject[$name];
2058        }
