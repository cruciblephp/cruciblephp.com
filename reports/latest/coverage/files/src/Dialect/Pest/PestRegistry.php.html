149    public static function drain(): PestFileState
150    {
151        self::$collecting = false;
152        self::$file       = null;
153
154        // Fold the file's uses() chains: last class wins, traits and
155        // groups accumulate, chained hooks join the file-level ones.
156        $class      = null;
157        $traits     = [];
158        $groups     = [];
159        $beforeEach = self::$beforeEach;
160        $afterEach  = self::$afterEach;
161        $beforeAll  = self::$beforeAll;
162        $afterAll   = self::$afterAll;
163
164        foreach (self::$registrations as $registration) {
165            $class      = $registration->class ?? $class;
166            $traits     = [...$traits, ...$registration->traits];
167            $groups     = [...$groups, ...$registration->groups];
168            $beforeEach = [...$beforeEach, ...$registration->beforeEach];
169            $afterEach  = [...$afterEach, ...$registration->afterEach];
170            $beforeAll  = [...$beforeAll, ...$registration->beforeAll];
171            $afterAll   = [...$afterAll, ...$registration->afterAll];
172        }
173
174        return new PestFileState(
175            self::$calls,
176            $beforeEach,
177            $afterEach,
178            $beforeAll,
179            $afterAll,
180            $class,
181            $traits,
182            $groups,
183            self::$covers,
184            self::$mutates,
185        );
186    }

219    /**
220     * The path to a fixture file: `tests/Fixtures/` under the project
221     * root, always, whatever directory the calling test file sits in.
222     *
223     * Measured against Pest 5.1.1 rather than guessed, because the
224     * obvious rule is the wrong one. Crucible first resolved this
225     * against the *nearest* `Fixtures/` at or above the test file,
226     * which returns a different file whenever a project has more than
227     * one — and the incumbent has no such rule. Probed from
228     * `tests/nested/deep/`, with `tests/nested/Fixtures/sample.txt`
229     * present and nearer, it still resolved `tests/Fixtures/sample.txt`;
230     * probed from a second test-suite directory, it still looked under
231     * `tests/`, not that suite's own.
232     *
233     * The name is joined to that directory, so `fixture('Fixtures/x')`
234     * is an error there and here — the incumbent prepends `Fixtures/`
235     * itself.
236     *
237     * A missing file is refused: a fixture path that silently does not
238     * exist fails later, somewhere else, as a confusing read error.
239     *
240     * @param non-empty-string $path
241     *
242     * @return non-empty-string absolute path
243     */
244    public static function fixture(string $path): string
245    {
246        if (self::$file === null) {
247            throw new ConfigurationException(
248                'fixture() is only available while a test file is being collected.',
249            );
250        }
251
252        $candidate = self::$root . '/tests/Fixtures/' . $path;
253
254        if (!file_exists($candidate)) {
255            throw new ConfigurationException(sprintf('fixture(%s): the fixture file [%s] does not exist.', $path, $candidate));
256        }
257
258        $real = realpath($candidate);
259
260        return $real === false ? $candidate : $real;
261    }

263    /**
264     * @param non-empty-string $description
265     * @param 'test'|'it'      $kind
266     */
267    public static function test(string $description, ?Closure $test, string $kind): TestCall
268    {
269        self::ensureCollecting($kind === 'it' ? 'it()' : 'test()');
270
271        $call = new TestCall($description, $kind, $test, self::$describePath);
272        $name = $call->name();
273
274        // ✓ Measured against Pest 5.1.1, not inferred: a second test
275        // with the same composed name is refused at COLLECTION, with
276        // Pest\Exceptions\TestAlreadyExist and exit 1, and nothing in
277        // the file runs -- neither body, not even the first. Crucible
278        // used to accept both, and since a name IS the TestId that
279        // quietly filed two tests under one id: the shape where a
280        // result stops meaning what it says.
281        //
282        // Composed, because scoping is what the incumbent counts.
283        // ✓ Measured in the same run: `group A > shared` and
284        // `group B > shared` coexist, and `it shared` is not `shared`.
285        if (isset(self::$takenNames[$name])) {
286            throw new ConfigurationException(sprintf(
287                'A test named [%s] already exists in [%s]. Please give this test a different description.',
288                $name,
289                self::$file ?? '',
290            ));
291        }
292
293        self::$takenNames[$name] = true;
294        self::$calls[]           = $call;
295
296        return $call;
297    }

300     * An architecture rule (D-088): a fluent rule that registers itself
301     * as an ordinary test. The closure captures the rule object, so
302     * expectations chained on after this call are visible by the time
303     * the test runs — built at collection time, asserted at execution
304     * time, like every other deferred body here.
305     *
306     * @param ?non-empty-string $description

308    public static function arch(?string $description = null): ArchRule
309    {
310        self::ensureCollecting('arch()');
311
312        $rule = Architecture::rule();
313
314        self::test($description ?? 'architecture', static function () use ($rule): void {
315            $rule->assert();

338    /**
339     * The crucible dialect's property test (G5): generators first, the
340     * property closure last, one test on the engine's property runner.
341     *
342     * @param non-empty-string $description
343     * @param Gen<mixed>|Closure ...$arguments
344     */
345    public static function property(string $description, Gen|Closure ...$arguments): TestCall
346    {
347        self::ensureCollecting('property()');
348
349        $arguments = array_values($arguments);
350        $body      = array_pop($arguments);
351
352        if (!$body instanceof Closure) {
353            throw new ConfigurationException('property() takes the property closure as its last argument.');
354        }
355
356        $generators = [];
357
358        foreach ($arguments as $generator) {
359            if (!$generator instanceof Gen) {
360                throw new ConfigurationException('property() takes generators between the description and the closure.');
361            }
362
363            $generators[] = $generator;
364        }
365
366        if ($generators === []) {
367            throw new ConfigurationException('property() needs at least one generator.');
368        }
369
370        return self::test($description, static function () use ($generators, $body): void {
371            Property::forAll(...$generators)->check($body);
372        }, 'test');
373    }

375    /**
376     * The crucible dialect's I/O table (G2b): one test per row, the last
377     * element being the expected value (toEqual semantics), everything
378     * before it the arguments. A string row key names the case.
379     *
380     * @param iterable<array-key, mixed> $rows
381     */
382    public static function table(callable $subject, iterable $rows): DescribeCall
383    {
384        self::ensureCollecting('table()');
385
386        $base  = self::callableName($subject);
387        $calls = [];
388
389        foreach ($rows as $key => $row) {
390            if (!is_array($row) || $row === []) {
391                throw new ConfigurationException(
392                    'table() rows are arrays of [arguments..., expected value] — got a ' . get_debug_type($row) . ' row.',
393                );
394            }
395
396            $arguments = array_values($row);
397            $expected  = array_pop($arguments);
398
399            $name = is_string($key)
400                ? $base . ': ' . $key
401                : $base . '(' . implode(', ', array_map(Exporter::describe(...), $arguments)) . ') = ' . Exporter::describe($expected);
402
403            $call          = self::test(self::uniqueName($name), null, 'test');
404            $call->chain[] = ['expect', [static fn(): mixed => $subject(...$arguments)], false];
405            $call->chain[] = ['toEqual', [$expected], false];
406
407            $calls[] = $call;
408        }
409
410        return new DescribeCall($calls);
411    }

417    {
418        self::ensureCollecting('describe()');
419

423

490    /**
491     * A check()'s name, in order of preference: the trailing comment
492     * on its line (`check(...)->toBe(3); // adds small numbers`), the
493     * source line itself, the line number when the source is
494     * unreadable.
495     *
496     * @return non-empty-string
497     */
498    private static function autoName(Closure $test): string
499    {
500        $reflection = new ReflectionFunction($test);
501        $file       = $reflection->getFileName();
502        $line       = $reflection->getStartLine();
503
504        $snippet = $file === false || $line === false ? '' : trim(self::sourceLine($file, $line));
505        $comment = self::trailingComment($snippet);
506
507        if ($comment !== null) {
508            return self::uniqueName($comment);
509        }
510
511        if (mb_strlen($snippet) > 100) {
512            $snippet = mb_substr($snippet, 0, 100) . '…';
513        }
514
515        return self::uniqueName($snippet === '' ? 'check on line ' . (int) $line : $snippet);
516    }

518    /**
519     * The text of a trailing comment on the line, tokenizer-parsed so
520     * a '//' inside a string can never false-match. Handles the //,
521     * #, and inline /* *​/ forms.
522     *
523     * @return ?non-empty-string
524     */
525    private static function trailingComment(string $line): ?string
526    {
527        if ($line === '' || (!str_contains($line, '//') && !str_contains($line, '#') && !str_contains($line, '/*'))) {
528            return null;
529        }
530
531        foreach (@token_get_all('<?php ' . $line) as $token) {
532            if (!is_array($token) || $token[0] !== T_COMMENT) {
533                continue;
534            }
535
536            $text = $token[1];
537
538            $text = str_starts_with($text, '/*')
539                ? substr($text, 2, -2)
540                : ltrim($text, '/#');
541
542            $text = trim($text);
543
544            return $text === '' ? null : $text;
545        }
546
547        return null;
548    }

550    /**
551     * Names must be unique within a file — they are the TestId. Two
552     * identical snippets (or table rows) get an occurrence suffix.
553     *
554     * @param non-empty-string $name
555     *
556     * @return non-empty-string
557     */
558    private static function uniqueName(string $name): string
559    {
560        $occurrence = self::$autoNames[$name] = (self::$autoNames[$name] ?? 0) + 1;
561
562        return $occurrence > 1 ? $name . ' #' . $occurrence : $name;
563    }

565    private static function sourceLine(string $file, int $line): string
566    {
567        $lines = self::$sources[$file] ??= (static function (string $file): array {
568            $contents = file_get_contents($file);
569
570            return $contents === false ? [] : explode("\n", $contents);
571        })($file);
572
573        return $lines[$line - 1] ?? '';
574    }

576    /** @return non-empty-string */
577    private static function callableName(callable $subject): string
578    {
579        if (is_string($subject)) {
580            return $subject;
581        }
582
583        if (is_array($subject)) {
584            [$classOrObject, $method] = $subject;
585
586            return (is_object($classOrObject) ? $classOrObject::class : $classOrObject) . '::' . $method;
587        }
588
589        if ($subject instanceof Closure) {
590            $reflection = new ReflectionFunction($subject);
591            $name       = $reflection->getName();
592
593            if (!str_contains($name, '{closure')) {
594                return $name;
595            }
596
597            return 'table on line ' . (int) $reflection->getStartLine();
598        }
599
600        return get_debug_type($subject) . '::__invoke';
601    }
