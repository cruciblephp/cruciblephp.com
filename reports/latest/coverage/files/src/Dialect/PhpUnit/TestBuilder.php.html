56/**
57 * The phpunit dialect frontend (D-008): turns a TestCase class into a
58 * dialect-neutral TestGroup. A test method is a public method named
59 * test*, carrying #[Test], or tagged with the legacy `@test` docblock;
60 * parameterized tests expand to one TestDefinition per dataset row at
61 * discovery time.
62 */
63final readonly class TestBuilder
64{
65    public function __construct(
66        private MetadataParser $parser = new MetadataParser(),
67    ) {}
68
69    /**
70     * @param class-string<TestCase> $className
71     * @param non-empty-string       $file      project-relative path, used for TestIds
72     */
73    public function build(string $className, string $file): TestGroup
74    {
75        $class       = new ReflectionClass($className);
76        $hooks       = HookPlanner::forClass($class);
77        $definitions = [];
78
79        foreach ($class->getMethods(ReflectionMethod::IS_PUBLIC) as $method) {
80            if ($method->isStatic() || $method->isAbstract()) {
81                continue;
82            }
83
84            $methodName = $method->getName();
85
86            /** @var non-empty-string $methodName */
87            $metadata = $this->parser->forClassAndMethod($className, $methodName);
88
89            // Class-level first, then method-level (PHPUnit's own
90            // precedence: a class-wide requirement applies to every
91            // test unless the method narrows it further) — matches
92            // Monolog's own MongoDBHandlerTest, whose `@requires
93            // extension mongodb` sits on the class docblock, not any
94            // individual method.
95            $docblockRequirements = [
96                ...$this->docblockRequirements($class->getDocComment()),
97                ...$this->docblockRequirements($method->getDocComment()),
98            ];
99
100            if ($docblockRequirements !== []) {
101                $metadata = $metadata->mergedWith(\LucianoPereira\Crucible\Metadata\MetadataCollection::from(...$docblockRequirements));
102            }
103
104            if (!str_starts_with($methodName, 'test') && !$metadata->has(Test::class) && !$this->docblockMarksAsTest($method->getDocComment())) {
105                continue;
106            }
107
108            $dependencies = $this->dependenciesOf($metadata, $method);
109
110            foreach ($this->datasets($className, $methodName) as $datasetName => $arguments) {
111                $id = new TestId(
112                    $file,
113                    $methodName,
114                    $datasetName === '' ? null : $datasetName,
115                );
116
117                $definitions[] = new TestDefinition(
118                    $id,
119
120                    // Spec order: dataset arguments first, then the
121                    // return values of depended-upon tests.
122                    static fn(array $dependencyValues): mixed => (new $className())->invokeTest($methodName, [...$arguments, ...$dependencyValues], $hooks, $id->dataset),
123                    $metadata,
124                    $dependencies,
125                );
126            }
127        }
128
129        return new TestGroup(
130            $className,
131            $this->sortedByDependencies($definitions),
132            static function () use ($className): void {
133                $className::setUpBeforeClass();
134            },
135            static function () use ($className): void {
136                $className::tearDownAfterClass();
137            },
138        );
139    }

141    /**
142     * @return list<non-empty-string>
143     */
144    private function dependenciesOf(\LucianoPereira\Crucible\Metadata\MetadataCollection $metadata, ReflectionMethod $method): array
145    {
146        $dependencies = [];
147
148        foreach ($metadata->ofType(Depends::class) as $depends) {
149            $dependencies[] = $depends->methodName;
150        }
151
152        foreach ($metadata->ofType(DependsUsingDeepClone::class) as $depends) {
153            $dependencies[] = $depends->methodName;
154        }
155
156        foreach ($metadata->ofType(DependsUsingShallowClone::class) as $depends) {
157            $dependencies[] = $depends->methodName;
158        }
159
160        // PHPUnit-compat fallback: the legacy `@depends` docblock form
161        // still appears in real, actively-maintained PHPUnit suites
162        // alongside the modern #[Depends] attribute (confirmed against
163        // Monolog's own tests) — the compat surface's whole point is
164        // running that code unmodified, so both forms are honored.
165        foreach ($this->docblockDependencies($method) as $dependency) {
166            $dependencies[] = $dependency;
167        }
168
169        return $dependencies;
170    }

172    /**
173     * PHPUnit-compat fallback: the legacy `@test` docblock tag marks a
174     * method as a test regardless of its name, still used alongside
175     * the modern #[Test] attribute and test-prefixed naming (confirmed
176     * against Monolog's own GoogleCloudLoggingFormatterTest, whose
177     * descriptively-named methods rely on it — without this, they're
178     * silently never discovered at all, not even as skipped). `\b`
179     * after `test` keeps this from matching `@testWith`/`@testdox`
180     * (oracle-verified against PHPUnit's own tag tokenizer, DocBlock.php).
181     */
182    private function docblockMarksAsTest(string|false $doc): bool
183    {
184        return $doc !== false && preg_match('/@test\b/', $doc) === 1;
185    }

187    /**
188     * @return list<non-empty-string>
189     */
190    private function docblockDependencies(ReflectionMethod $method): array
191    {
192        $doc = $method->getDocComment();
193
194        if ($doc === false) {
195            return [];
196        }
197
198        $dependencies = [];
199
200        if (preg_match_all('/@depends\s+(?:clone\s+|shallowClone\s+)?(\S+)/', $doc, $matches) > 0) {
201            /** @var list<non-empty-string> $names */
202            $names = $matches[1];
203
204            foreach ($names as $name) {
205                $dependencies[] = $name;
206            }
207        }
208
209        return $dependencies;
210    }

212    /**
213     * PHPUnit-compat fallback: the legacy `@requires` docblock form
214     * (extension/function/PHP/OS/OSFAMILY/setting), still common
215     * alongside the modern #[Requires*] attributes (confirmed against
216     * Monolog's own tests — e.g. its MongoDB handler tests skip via
217     * `@requires extension mongodb`, never migrated to attributes).
218     * Synthesizes the same attribute instances Requirements::unmet()
219     * already reads, so no downstream code needs to change.
220     *
221     * @return list<\LucianoPereira\Crucible\Metadata\CrucibleAttribute>
222     */
223    private function docblockRequirements(string|false $doc): array
224    {
225        if ($doc === false) {
226            return [];
227        }
228
229        $requirements = [];
230
231        if (preg_match_all('/@requires\s+(\S+)[ \t]*([^\r\n]*)/', $doc, $matches, PREG_SET_ORDER) === false) {
232            return [];
233        }
234
235        foreach ($matches as $match) {
236            $kind = mb_strtolower($match[1]);
237            $rest = trim($match[2]);
238
239            if ($rest === '') {
240                continue;
241            }
242
243            $requirement = match ($kind) {
244                'php'       => new RequiresPhp($rest),
245                'os'        => new RequiresOperatingSystem($rest),
246                'osfamily'  => new RequiresOperatingSystemFamily($rest),
247                'extension' => $this->extensionRequirement($rest),
248                'function'  => $this->functionOrMethodRequirement($rest),
249                'setting'   => $this->settingRequirement($rest),
250                default     => null,
251            };
252
253            if ($requirement !== null) {
254                $requirements[] = $requirement;
255            }
256        }
257
258        return $requirements;
259    }

261    private function extensionRequirement(string $rest): ?RequiresPhpExtension
262    {
263        $parts = preg_split('/\s+/', $rest, 2);
264
265        if ($parts === false || $parts[0] === '') {
266            return null;
267        }
268
269        return new RequiresPhpExtension($parts[0], isset($parts[1]) && $parts[1] !== '' ? $parts[1] : null);
270    }

272    private function functionOrMethodRequirement(string $rest): RequiresFunction|RequiresMethod|null
273    {
274        $name = ltrim($rest, '\\');
275
276        if ($name === '') {
277            return null;
278        }
279
280        if (str_contains($name, '::')) {
281            [$className, $methodName] = explode('::', $name, 2);
282
283            if ($className === '' || $methodName === '') {
284                return null;
285            }
286
287            /** @var class-string $className */
288            return new RequiresMethod($className, $methodName);
289        }
290
291        return new RequiresFunction($name);
292    }

294    private function settingRequirement(string $rest): ?RequiresSetting
295    {
296        $parts = preg_split('/\s+/', $rest, 2);
297
298        if ($parts === false || $parts[0] === '' || !isset($parts[1]) || $parts[1] === '') {
299            return null;
300        }
301
302        return new RequiresSetting($parts[0], $parts[1]);
303    }

305    /**
306     * Stable topological order: dependencies before dependents.
307     * Unresolvable references (cycles, unknown names) keep declaration
308     * order and are reported at run time.
309     *
310     * @param list<TestDefinition> $definitions
311     *
312     * @return list<TestDefinition>
313     */
314    private function sortedByDependencies(array $definitions): array
315    {
316        $remainingRows = [];
317
318        foreach ($definitions as $definition) {
319            $remainingRows[$definition->id->name] = ($remainingRows[$definition->id->name] ?? 0) + 1;
320        }
321
322        $ordered = [];
323        $pending = $definitions;
324
325        while ($pending !== []) {
326            $progress = false;
327            $next     = [];
328
329            foreach ($pending as $definition) {
330                $ready = true;
331
332                foreach ($definition->dependencies as $dependency) {
333                    if (($remainingRows[$dependency] ?? 0) > 0 && $dependency !== $definition->id->name) {
334                        $ready = false;
335
336                        break;
337                    }
338                }
339
340                if ($ready) {
341                    $ordered[] = $definition;
342                    $remainingRows[$definition->id->name]--;
343                    $progress = true;
344                } else {
345                    $next[] = $definition;
346                }
347            }
348
349            if (!$progress) {
350                // Cycle: append what is left in declaration order.
351                foreach ($next as $definition) {
352                    $ordered[] = $definition;
353                }
354
355                break;
356            }
357
358            $pending = $next;
359        }
360
361        return $ordered;
362    }

364    /**
365     * Resolves every dataset source of a method into named argument
366     * rows. A method without parameterization yields one unnamed row.
367     *
368     * @param class-string<TestCase> $className
369     * @param non-empty-string       $methodName
370     *
371     * @return iterable<string, list<mixed>>
372     */
373    private function datasets(string $className, string $methodName): iterable
374    {
375        $metadata = $this->parser->forMethod($className, $methodName);
376
377        $providers = [];
378
379        foreach ($metadata->ofType(DataProvider::class) as $provider) {
380            $providers[] = [$className, $provider->methodName];
381        }
382
383        foreach ($metadata->ofType(DataProviderExternal::class) as $provider) {
384            $providers[] = [$provider->className, $provider->methodName];
385        }
386
387        $inline     = $metadata->ofType(TestWith::class);
388        $inlineJson = $metadata->ofType(TestWithJson::class);
389
390        if ($providers === [] && $inline === [] && $inlineJson === []) {
391            yield '' => [];
392
393            return;
394        }
395
396        $index = 0;
397
398        foreach ($providers as [$providerClass, $providerMethod]) {
399            $rows = $this->rowsFromProvider($providerClass, $providerMethod);
400
401            foreach ($rows as $key => $row) {
402                if (!is_array($row)) {
403                    throw new ConfigurationException(sprintf(
404                        'Data provider %s::%s() must yield arrays of arguments.',
405                        $providerClass,
406                        $providerMethod,
407                    ));
408                }
409
410                $name = is_string($key) ? $key : (string) $index;
411
412                yield $name => array_values($row);
413
414                $index++;
415            }
416        }
417
418        foreach ($inline as $row) {
419            yield ($row->name ?? (string) $index) => array_values($row->data);
420
421            $index++;
422        }
423
424        foreach ($inlineJson as $row) {
425            $decoded = json_decode($row->json, true, flags: JSON_THROW_ON_ERROR);
426
427            if (!is_array($decoded)) {
428                throw new ConfigurationException(sprintf(
429                    'TestWithJson on %s::%s() must decode to an array of arguments.',
430                    $className,
431                    $methodName,
432                ));
433            }
434
435            yield ($row->name ?? (string) $index) => array_values($decoded);
436
437            $index++;
438        }
439    }

441    /**
442     * @param class-string     $providerClass
443     * @param non-empty-string $providerMethod
444     *
445     * @return iterable<mixed>
446     */
447    private function rowsFromProvider(string $providerClass, string $providerMethod): iterable
448    {
449        $method = (new ReflectionClass($providerClass))->getMethod($providerMethod);
450
451        if (!$method->isStatic()) {
452            throw new ConfigurationException(sprintf(
453                'Data provider %s::%s() must be static.',
454                $providerClass,
455                $providerMethod,
456            ));
457        }
458
459        $rows = $method->invoke(null);
460
461        if (!is_iterable($rows)) {
462            throw new ConfigurationException(sprintf(
463                'Data provider %s::%s() must return an iterable.',
464                $providerClass,
465                $providerMethod,
466            ));
467        }
468
469        return $rows;
470    }
