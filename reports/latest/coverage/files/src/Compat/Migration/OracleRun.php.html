33/**
34 * Runs the project's real, installed PHPUnit binary as a subprocess —
35 * the oracle side of `compat-check` (the same black-box comparison
36 * conformance/run.php does for Crucible's own maintainer-only
37 * fixtures, D-018, adapted here for a real end-user suite). Keyed by
38 * "<relative file path>::method" (plus "#dataset") — matching
39 * Crucible's own NDJSON `id` shape exactly (confirmed empirically: it
40 * identifies a test by its source file, not its FQCN), not the
41 * FQCN-based keying an earlier version of this class assumed, which
42 * silently matched nothing at all across an entire real suite.
43 *
44 * On a pest project the oracle is `vendor/bin/pest`, because ✓ measured
45 * the real phpunit binary cannot run a pest suite at all — pest refuses
46 * from inside its own TestSuite and collects nothing (D-114). Pest
47 * reports through the same JUnit logger, in its own shape: the `file`
48 * attribute already carries `<relative file>::<test name>`, which is
49 * Crucible's id exactly, so those rows need no translation beyond
50 * their dataset suffix.
51 */
52final readonly class OracleRun
53{
54    /**
55     * @param non-empty-string       $binary           path to the incumbent's binary
56     * @param list<non-empty-string> $extraArgs        forwarded CLI args (--filter, --testsuite, ...)
57     * @param non-empty-string       $junitFile        where to write --log-junit
58     * @param ?non-empty-string      $eventsFile       where to write --log-events-text, for the rows JUnit cannot locate
59     *
60     * @return array{outcomes: array<string, string>, unmapped: list<string>} keyed "<relative file>::method#dataset" => pass|fail|error|risky|skip
61     */
62    public static function outcomes(string $binary, array $extraArgs, WorkingDirectory $workingDirectory, string $junitFile, ?string $eventsFile = null): array
63    {
64        // Both logs from ONE run. The events log is asked for only
65        // where JUnit is known to lose the file — the pest oracle — so
66        // an older phpunit is never handed an option it would refuse.
67        $command = [$binary, '--log-junit', $junitFile];
68
69        if ($eventsFile !== null) {
70            $command[] = '--log-events-text';
71            $command[] = $eventsFile;
72        }
73
74        self::run([...$command, ...$extraArgs], $workingDirectory->path);
75
76        return self::parse($junitFile, $workingDirectory);
77    }
78
79    /**
80     * @param list<string> $command
81     */
82    private static function run(array $command, string $cwd): void
83    {
84        $process = proc_open($command, [1 => ['pipe', 'w'], 2 => ['pipe', 'w']], $pipes, $cwd);
85
86        if (!is_resource($process)) {
87            return;
88        }
89
90        stream_get_contents($pipes[1]);
91        stream_get_contents($pipes[2]);
92        proc_close($process);
93    }

95    /**
96     * The oracle binary for this project: pest when it is installed,
97     * the real phpunit otherwise, null when neither is.
98     *
99     * ✓ Measured: `vendor/bin/phpunit` cannot run a pest suite at all —
100     * pest refuses from inside its own TestSuite and collects nothing,
101     * so on a pest project the phpunit binary is not a narrower oracle,
102     * it is an empty one (D-114). Pest is preferred wherever it is
103     * present, because a project that has it is a project whose tests
104     * it owns.
105     *
106     *
107     * @return ?non-empty-string
108     */
109    public static function binaryIn(WorkingDirectory $workingDirectory): ?string
110    {
111        $root = rtrim($workingDirectory->path, '/');
112
113        foreach ([$root . '/vendor/bin/pest', $root . '/vendor/bin/phpunit'] as $binary) {
114            if (is_file($binary)) {
115                return $binary;
116            }
117        }
118
119        return null;
120    }

122    /**
123     * The pure parsing half, public so it can be unit-tested against a
124     * fixture JUnit file without a real PHPUnit subprocess.
125     *
126     * Both answers come from ONE parse. They used to be two public
127     * methods over the same file, so a single compat-check run read and
128     * built the XML tree twice to learn two things about the same rows.
129     *
130     * The second answer is the rows no Crucible id can be built for.
131     * They are returned rather than dropped, and rather than guessed
132     * at: silently leaving them out would shrink the comparison without
133     * saying so, which is a vacuous pass one size down — the tool would
134     * look like it compared a suite when it compared part of one.
135     *
136     * @param non-empty-string $junitFile
137     *
138     * @return array{outcomes: array<string, string>, unmapped: list<string>}
139     */
140    public static function parse(string $junitFile, WorkingDirectory $workingDirectory): array
141    {
142        $outcomes = [];
143        $unmapped = [];
144
145        foreach (self::testcases($junitFile, $workingDirectory) as [$key, $outcome, $name]) {
146            if ($key === null) {
147                $unmapped[] = $name;
148
149                continue;
150            }
151
152            $outcomes[$key] = $outcome;
153        }
154
155        return ['outcomes' => $outcomes, 'unmapped' => $unmapped];
156    }

158    /**
159     * @param non-empty-string $junitFile
160     *
161     * @return list<array{0: ?non-empty-string, 1: string, 2: string}> key (null when unmappable), outcome, reported name
162     */
163    private static function testcases(string $junitFile, WorkingDirectory $workingDirectory): array
164    {
165        $xml = simplexml_load_string((string) file_get_contents($junitFile));
166
167        if ($xml === false) {
168            return [];
169        }
170
171        $rows = [];
172
173        /** @var array<string, int> $ordinals */
174        $ordinals = [];
175
176        foreach ($xml->xpath('//testcase') ?? [] as $testcase) {
177            $outcome = 'pass';
178
179            if (property_exists($testcase, 'failure') && $testcase->failure !== null) {
180                $outcome = 'fail';
181            } elseif (property_exists($testcase, 'error') && $testcase->error !== null) {
182                $type    = (string) $testcase->error['type'];
183                $outcome = stripos($type, 'risky') !== false ? 'risky' : 'error';
184            } elseif (property_exists($testcase, 'skipped') && $testcase->skipped !== null) {
185                $outcome = 'skip';
186            }
187
188            $rows[] = [self::key($testcase, $workingDirectory, $ordinals), $outcome, (string) $testcase['name']];
189        }
190
191        return $rows;
192    }

194    /**
195     * "<relative file>::method" (plus "#dataset"), matching Crucible's
196     * own NDJSON `id` shape directly, so no translation is needed to
197     * compare the two sides.
198     *
199     * @param array<string, int> &$ordinals         dataset counters, per test, in execution order
200     *
201     * @return ?non-empty-string
202     */
203    private static function key(SimpleXMLElement $testcase, WorkingDirectory $workingDirectory, array &$ordinals): ?string
204    {
205        $name = (string) $testcase['name'];
206        $file = (string) $testcase['file'];
207
208        // Pest's shape: the `file` attribute already holds
209        // "<relative file>::<test name>" — Crucible's id, dataset
210        // aside. ✓ Measured: for a classic PHPUnit class inside a pest
211        // run the same attribute holds pest's DESCRIPTION of the class
212        // ("Classic (Tests\Classic)"), which names no file anywhere,
213        // and pest's teamcity locationHint says the same. That is why
214        // those rows come back unmappable instead of guessed at.
215        if ($name !== '' && str_ends_with($file, '::' . $name)) {
216            return self::pestKey(substr($file, 0, -strlen('::' . $name)), $name, $workingDirectory, $ordinals);
217        }
218
219        preg_match('/^(?<m>.+?)(?: with data set (?:"(?<n>.+)"|#(?<i>\d+)))?$/s', $name, $match);
220
221        $method  = $match['m'] ?? $name;
222        $dataset = ($match['n'] ?? '') !== '' ? $match['n'] : (($match['i'] ?? '') !== '' ? $match['i'] : '');
223
224        return sprintf('%s::%s%s', self::relative($file, $workingDirectory), $method, $dataset !== '' ? '#' . $dataset : '');
225    }

227    /**
228     * The pest-shaped key, or null when the path names no file.
229     *
230     * @param array<string, int> &$ordinals
231     *
232     * @return ?non-empty-string
233     */
234    private static function pestKey(string $path, string $name, WorkingDirectory $workingDirectory, array &$ordinals): ?string
235    {
236        $relative = self::relative($path, $workingDirectory);
237        $absolute = str_starts_with($relative, '/') ? $relative : rtrim($workingDirectory->path, '/') . '/' . $relative;
238
239        if ($relative === '' || !is_file($absolute)) {
240            return null;
241        }
242
243        if (preg_match('/^(?<t>.*) with data set "(?<label>.*)"$/s', $name, $match) !== 1) {
244            return $relative . '::' . $name;
245        }
246
247        $test = $relative . '::' . $match['t'];
248
249        // A named set carries its own name, and Crucible's id uses that
250        // name. A positional one is labelled with its VALUES — "(1)" —
251        // where the id holds the ORDINAL, which the label never
252        // contains. JUnit lists testcases in execution order and the
253        // ordinal counts in that same order, so they are counted as
254        // they arrive rather than parsed out of a label that has no
255        // ordinal in it.
256        if (preg_match('/^dataset "(?<n>.*)"$/s', $match['label'], $named) === 1) {
257            return $test . '#' . $named['n'];
258        }
259
260        $ordinal         = $ordinals[$test] ?? 0;
261        $ordinals[$test] = $ordinal + 1;
262
263        return $test . '#' . $ordinal;
264    }
265
266    private static function relative(string $absoluteFile, WorkingDirectory $workingDirectory): string
267    {
268        $prefix = rtrim($workingDirectory->path, '/') . '/';
269
270        return str_starts_with($absoluteFile, $prefix) ? substr($absoluteFile, strlen($prefix)) : $absoluteFile;
271    }
