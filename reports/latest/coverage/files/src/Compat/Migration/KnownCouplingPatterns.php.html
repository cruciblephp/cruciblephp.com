19/**
20 * The seeded table `compat-check --auto-fix` actually knows how to
21 * rewrite. Every pattern here was verified against a real suite
22 * (Monolog 3.9.0, 2026-07-28 session) and re-checked against the real
23 * PHPUnit oracle after rewriting, so a fix doesn't just weaken the
24 * test into a vacuous pass — including a real correction found by
25 * that same re-check: a first version of the vendor-path pattern
26 * matched ANY frame index, but only frame 0 (the immediate call site)
27 * is structurally guaranteed to be some TestCase.php in every engine.
28 * Frame 1+ reflects each engine's own internal call depth, which does
29 * NOT line up — real PHPUnit's frame 1 is also TestCase.php, but
30 * Crucible's is a different file entirely. The two non-immediate-frame
31 * patterns below exist specifically to still close that case, honestly
32 * (weakened to "this frame exists", never claiming which file it's
33 * in) rather than either guessing wrong or leaving it unfixed.
34 *
35 * A fourth real shape found in that same audit is deliberately NOT
36 * here: a composite `assertEquals()` against a whole array/object
37 * literal with a PHPUnit-internal value buried inside one of its
38 * fields. Fixing that needs understanding the surrounding structure,
39 * not just one call's literal argument — outside what a single-call
40 * token rewrite can safely do. It stays diagnosed, not auto-fixed.
41 * Extend this table only with patterns verified the same way: fixed
42 * by hand against a real failure, then re-checked against the real
43 * oracle — never by guessing at a plausible-looking transform.
44 */
45final readonly class KnownCouplingPatterns
46{
47    /**
48     * PHPUnit-only internal filenames with no Crucible equivalent at
49     * all — no path prefix to relax, so containment can only be
50     * weakened to a structural "some stacktrace frame exists" check.
51     */
52    private const array NO_EQUIVALENT_FILENAMES = ['TestSuite.php'];
53
54    /**
55     * Order matters: `SourceRewriter` tries these in array order and
56     * stops at the first match, so the two non-immediate-frame
57     * patterns (more specific) must come before the general one
58     * (which would otherwise wrongly claim a specific file for a
59     * frame that does not structurally guarantee one).
60     *
61     * @return list<CouplingPattern>
62     */
63    public static function all(): array
64    {
65        return [
66            // Frame N>=1 with an explicit "#N " marker inside the
67            // literal itself (LineFormatterTest::testBasePath's
68            // shape). The frame-number text is real and verifiable in
69            // both engines; the specific file at that frame is not
70            // (real PHPUnit's frame 1 is also TestCase.php -- two
71            // consecutive TestCase.php frames, its own internal
72            // quirk -- but Crucible's frame 1 is TestBuilder.php, a
73            // different file). Weaken to "this frame exists", not
74            // which file it names.
75            new CouplingPattern(
76                description: 'Asserts a literal vendor/phpunit/phpunit path at a non-immediate stack frame',
77                methods: ['assertStringContainsString', 'assertStringStartsWith'],
78                matchesLiteral: static fn(string $literal, string $restArgs): bool => str_contains($literal, 'vendor/phpunit/phpunit')
79                    && preg_match('/^.*?#\d+\s+vendor\/phpunit\/phpunit/', $literal) === 1
80                    && !self::isImmediateFrame($literal, $restArgs),
81                rewrite: static function (string $literal, string $restArgs): string {
82                    preg_match('/^(.*?#\d+\s+)/', $literal, $match);
83
84                    return 'assertMatchesRegularExpression(\'/' . preg_quote($match[1] ?? '', '/') . '\S+/\', ' . $restArgs . ')';
85                },
86            ),
87            // Frame N>=1 via a trailing "[N]" array index with NO
88            // frame marker in the literal at all
89            // (NormalizerFormatterTest's shape: the identical literal
90            // is reused for every frame, only the expression's index
91            // distinguishes them). Nothing in the literal is worth
92            // preserving for a non-immediate frame, so the honest
93            // weakening is confirming that frame has some string
94            // content, not what it says.
95            new CouplingPattern(
96                description: 'Asserts a literal vendor/phpunit/phpunit path at a non-immediate array-indexed frame',
97                methods: ['assertStringContainsString', 'assertStringStartsWith'],
98                matchesLiteral: static fn(string $literal, string $restArgs): bool => str_contains($literal, 'vendor/phpunit/phpunit')
99                    && preg_match('/\[\s*(\d+)\s*]\s*$/', $restArgs, $match) === 1
100                    && $match[1] !== '0',
101                rewrite: static fn(string $literal, string $restArgs): string => 'assertIsString(' . $restArgs . ')',
102            ),
103            // Frame 0 (the immediate call site) -- structurally
104            // guaranteed to be some TestCase.php in every engine, so
105            // preserving the file-name suffix is safe.
106            new CouplingPattern(
107                description: 'Asserts a literal vendor/phpunit/phpunit install path',
108                methods: ['assertStringContainsString', 'assertStringStartsWith'],
109                matchesLiteral: static fn(string $literal, string $restArgs): bool => str_contains($literal, 'vendor/phpunit/phpunit')
110                    && self::isImmediateFrame($literal, $restArgs),
111                rewrite: static function (string $literal, string $restArgs): string {
112                    $pattern = str_replace(
113                        preg_quote('vendor/phpunit/phpunit', '/'),
114                        '\S*',
115                        preg_quote($literal, '/'),
116                    );
117
118                    return 'assertMatchesRegularExpression(\'/' . $pattern . '/\', ' . $restArgs . ')';
119                },
120            ),
121            new CouplingPattern(
122                description: 'Asserts a PHPUnit-only internal filename Crucible has no equivalent for',
123                methods: ['assertStringContainsString'],
124                matchesLiteral: static fn(string $literal, string $restArgs): bool => in_array($literal, self::NO_EQUIVALENT_FILENAMES, true),
125                rewrite: static fn(string $literal, string $restArgs): string => 'assertMatchesRegularExpression(\'/#\d+ /\', ' . $restArgs . ')',
126            ),
127        ];
128    }
129
130    /**
131     * Whether this occurrence is checking the immediate call frame
132     * (index 0) specifically, as opposed to a deeper one — the only
133     * case verified safe to rewrite (see class docblock). Checks both
134     * signals a real occurrence might carry: an explicit "#N " frame
135     * marker inside the literal itself (LineFormatterTest's shape), or
136     * a trailing "[N]" array index in the asserted-on expression
137     * (NormalizerFormatterTest's shape, where the literal is identical
138     * for every frame and only the expression distinguishes them). No
139     * signal present at all (neither marker) is treated as frame 0,
140     * matching every real occurrence seen so far.
141     */
142    private static function isImmediateFrame(string $literal, string $restArgs): bool
143    {
144        if (preg_match('/#(\d+)\s/', $literal, $match) === 1 && $match[1] !== '0') {
145            return false;
146        }
147
148        if (preg_match('/\[\s*(\d+)\s*]\s*$/', $restArgs, $match) === 1 && $match[1] !== '0') {
149            return false;
150        }
151
152        return true;
153    }
154}
