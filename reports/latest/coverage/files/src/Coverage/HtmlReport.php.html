34/**
35 * The HTML coverage report (D-062) — Crucible-native presentation, no
36 * parity constraint: a self-contained static directory (inline CSS,
37 * zero scripts, zero requests) with an index of per-file bars and one
38 * annotated source page per file. Line classes follow the driver's
39 * value convention; branch analysis, when the run collected it, adds
40 * per-line branch badges and the index column.
41 */
42final readonly class HtmlReport
43{
44    /**
45     * @param bool $withoutClassView the spec's --without-class-view: drop the per-class table
46     * @param bool $withoutFileView  the spec's --without-file-view: drop the annotated source pages
47     */
48    public function __construct(
49        private bool $withoutClassView = false,
50        private bool $withoutFileView = false,
51    ) {}

74    /**
75     * @param non-empty-string $root      absolute project root
76     * @param non-empty-string $directory output directory
77     * @param non-empty-string $driver
78     */
79    public function write(CoverageData $data, string $root, string $directory, string $driver): void
80    {
81        if (!is_dir($directory)) {
82            mkdir($directory, 0o777, true);
83        }
84
85        $prefix = $root . '/';
86        $lines  = $data->lines;
87
88        ksort($lines);
89
90        $rows         = '';
91        $classRows    = '';
92        $withBranches = $data->branches !== [];
93
94        foreach ($lines as $file => $values) {
95            $relative = str_starts_with($file, $prefix) ? substr($file, strlen($prefix)) : $file;
96
97            $covered    = 0;
98            $executable = 0;
99
100            foreach ($values as $value) {
101                if ($value === -2) {
102                    continue;
103                }
104
105                $executable++;
106
107                if ($value > 0) {
108                    $covered++;
109                }
110            }
111
112            $branches = CoverageData::branchCounts($data->branches[$file] ?? []);
113            $percent  = $executable > 0 ? 100 * $covered / $executable : 0.0;
114
115            if (!$this->withoutFileView) {
116                $this->writeFilePage($directory, $relative, $file, $values, $data->branches[$file] ?? []);
117            }
118
119            if (!$this->withoutClassView && $file !== '') {
120                $classRows .= $this->classRows($file, $relative, $values);
121            }
122
123            // Without the file view there is no page to link to, so the
124            // name is plain text rather than a link that 404s.
125            $escaped = htmlspecialchars($relative, ENT_QUOTES);
126            $label   = $this->withoutFileView
127                ? $escaped
128                : sprintf('<a href="files/%s.html">%s</a>', $escaped, $escaped);
129
130            $rows .= sprintf(
131                "<tr><td>%s</td><td><span class=\"bar\"><span style=\"width:%.0f%%\"></span></span></td><td class=\"num\">%.2f%%</td><td class=\"num\">%d/%d</td>%s</tr>\n",
132                $label,
133                $percent,
134                $percent,
135                $covered,
136                $executable,
137                $withBranches
138                    ? sprintf('<td class="num">%s</td>', $branches['total'] > 0 ? sprintf('%d/%d', $branches['covered'], $branches['total']) : '—')
139                    : '',
140            );
141        }
142
143        $totals       = $data->totals();
144        $branchTotals = $data->branchTotals();
145        $totalPercent = $totals['executable'] > 0 ? 100 * $totals['covered'] / $totals['executable'] : 0.0;
146
147        $index = sprintf(
148            "<!DOCTYPE html>\n<html lang=\"en\">\n<head><meta charset=\"utf-8\"><title>Crucible coverage</title><style>%s</style></head>\n<body>\n<h1>%sCrucible coverage <small>(%s)</small></h1>\n<table>\n<tr><th>File</th><th></th><th>Lines</th><th>Covered</th>%s</tr>\n%s<tr class=\"total\"><td>Total</td><td><span class=\"bar\"><span style=\"width:%.0f%%\"></span></span></td><td class=\"num\">%.2f%%</td><td class=\"num\">%d/%d</td>%s</tr>\n</table>\n</body>\n</html>\n",
149            self::STYLE,
150            Version::logoHtml(),
151            htmlspecialchars($driver, ENT_QUOTES),
152            $withBranches ? '<th>Branches</th>' : '',
153            $rows,
154            $totalPercent,
155            $totalPercent,
156            $totals['covered'],
157            $totals['executable'],
158            $withBranches
159                ? sprintf('<td class="num">%d/%d</td>', $branchTotals['covered'], $branchTotals['total'])
160                : '',
161        );
162
163        if ($classRows !== '') {
164            $index = str_replace(
165                '</body>',
166                sprintf(
167                    "<h1>By method</h1>\n<table>\n<tr><th>Class</th><th>Method</th><th>Complexity</th><th>CRAP</th><th></th><th>Covered</th><th>File</th></tr>\n%s</table>\n</body>",
168                    $classRows,
169                ),
170                $index,
171            );
172        }
173
174        file_put_contents($directory . '/index.html', $index);
175    }

177    /**
178     * The per-class table: what a file-level bar cannot show, which is
179     * which class inside a covered file is the uncovered one. CRAP sits
180     * beside the percentage because a complicated class at 60% is a
181     * different problem from a simple one at the same number.
182     *
183     * @param non-empty-string $file
184     * @param array<int, int>  $values
185     */
186    private function classRows(string $file, string $relative, array $values): string
187    {
188        $rows = '';
189
190        foreach (SourceAnalysis::of($file)->classes as $qualified => $class) {
191            foreach ($class->methods as $method) {
192                $coverage = $method->coverage($values);
193
194                $rows .= sprintf(
195                    "<tr><td>%s</td><td>%s</td><td class=\"num\">%d</td><td class=\"num\">%.2f</td><td><span class=\"bar\"><span style=\"width:%.0f%%\"></span></span></td><td class=\"num\">%.2f%%</td><td>%s</td></tr>\n",
196                    htmlspecialchars($qualified, ENT_QUOTES),
197                    htmlspecialchars($method->name, ENT_QUOTES),
198                    $method->complexity,
199                    $method->crap($values),
200                    $coverage,
201                    $coverage,
202                    htmlspecialchars($relative, ENT_QUOTES),
203                );
204            }
205        }
206
207        return $rows;
208    }

210    /**
211     * One annotated source page, mirrored under files/ so relative
212     * links survive any nesting depth.
213     *
214     * @param non-empty-string                           $directory
215     * @param array<int, int>                            $values
216     * @param array<string, array{line: int, hit: int}> $branches
217     */
218    private function writeFilePage(string $directory, string $relative, string $file, array $values, array $branches): void
219    {
220        $target = $directory . '/files/' . $relative . '.html';
221        $parent = dirname($target);
222
223        if (!is_dir($parent)) {
224            mkdir($parent, 0o777, true);
225        }
226
227        $source = is_file($file) ? file_get_contents($file) : false;
228
229        if ($source === false) {
230            // The source moved since collection — an index entry
231            // without a page beats a fatal at report time.
232            return;
233        }
234
235        $branchesOnLine = [];
236
237        foreach ($branches as $branch) {
238            $branchesOnLine[$branch['line']]['total'] = ($branchesOnLine[$branch['line']]['total'] ?? 0) + 1;
239
240            if ($branch['hit'] > 0) {
241                $branchesOnLine[$branch['line']]['covered'] = ($branchesOnLine[$branch['line']]['covered'] ?? 0) + 1;
242            }
243        }
244
245        $body = '';
246
247        foreach (explode("\n", $source) as $index => $code) {
248            $line  = $index + 1;
249            $value = $values[$line] ?? null;
250
251            $class = match (true) {
252                $value === null => '',
253                $value === -2   => 'dead',
254                $value > 0      => 'covered',
255                default         => 'missed',
256            };
257
258            $badge     = '';
259            $condition = $branchesOnLine[$line] ?? null;
260
261            if ($condition !== null) {
262                $conditionCovered = $condition['covered'] ?? 0;
263
264                $badge = sprintf(
265                    '<span class="badge %s">%d/%d branches</span>',
266                    $conditionCovered === $condition['total'] ? 'full' : ($conditionCovered > 0 ? 'partial' : 'none'),
267                    $conditionCovered,
268                    $condition['total'],
269                );
270            }
271
272            $body .= sprintf(
273                "<span class=\"%s\"><span class=\"ln\">%d</span>%s%s</span>\n",
274                $class,
275                $line,
276                htmlspecialchars($code, ENT_QUOTES),
277                $badge,
278            );
279        }
280
281        // files/ pages link home through their own depth.
282        $home = str_repeat('../', substr_count($relative, '/') + 1) . 'index.html';
283
284        file_put_contents($target, sprintf(
285            "<!DOCTYPE html>\n<html lang=\"en\">\n<head><meta charset=\"utf-8\"><title>%s — Crucible coverage</title><style>%s</style></head>\n<body>\n<h1>%s<a href=\"%s\">Crucible coverage</a> / %s</h1>\n<pre>%s</pre>\n</body>\n</html>\n",
286            htmlspecialchars($relative, ENT_QUOTES),
287            self::STYLE,
288            Version::logoHtml(),
289            $home,
290            htmlspecialchars($relative, ENT_QUOTES),
291            $body,
292        ));
293    }
