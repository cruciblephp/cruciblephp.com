23/**
24 * The Clover XML emitter (D-041) — CI interop only, string-built like
25 * the JUnit writer, output-only (the no-XML rule bans XML inputs;
26 * opt-in emitters for CI consumers are the sanctioned exception).
27 *
28 * The document is grouped the way Clover's consumers read it: packages
29 * hold files, files declare their classes, and each method contributes
30 * a `type="method"` line beside the statement lines. That structure is
31 * the report — a document carrying only statement lines parses fine and
32 * renders as a project with no methods and no code, which is how this
33 * emitter used to fail: it reported loc="0" for every file it described.
34 * The counts come from the same source analysis the XML report's class
35 * view uses, so the two reports cannot disagree about a file.
36 */
37final readonly class CloverWriter
38{
39    /**
40     * @param bool $openClover the spec's --coverage-openclover: the same document in the
41     *                         maintained fork's framing — the file's basename beside its
42     *                         path, a short class name, a method line carrying its
43     *                         signature, and complexity in every metrics block
44     */
45    public function __construct(private bool $openClover = false) {}
46
47    /**
48     * @param int<0, max> $timestamp
49     */
50    public function write(CoverageData $data, int $timestamp): string
51    {
52        $lines = $data->lines;
53
54        ksort($lines);
55
56        $packages = [];
57        $totals   = $this->zero();
58
59        foreach ($lines as $file => $values) {
60            if ($file === '') {
61                continue;
62            }
63
64            ksort($values);
65
66            $analysis = SourceAnalysis::of($file);
67            $package  = '';
68            $classes  = '';
69            $rows     = '';
70            $file_    = $this->zero();
71
72            $file_['loc']     = $analysis->lines['total'];
73            $file_['ncloc']   = $analysis->lines['total'] - $analysis->lines['comments'];
74            $file_['classes'] = 0;
75
76            // Method lines first in document order is not required; what
77            // is required is that every method has one, since a Clover
78            // reader counts methods by reading them.
79            $methodRows = [];
80
81            foreach ($analysis->classes as $qualified => $class) {
82                $package = $package === '' ? $class->namespace : $package;
83                $unit    = $this->zero();
84
85                foreach ($class->methods as $method) {
86                    $own = $this->counts($values, $method->startLine, $method->endLine);
87
88                    $unit['complexity'] += $method->complexity;
89                    $unit['methods']++;
90
91                    // Covered means the body ran, not that the
92                    // signature line did — a declaration is not an
93                    // executable line, so keying on it reports every
94                    // method uncovered while the shapes still match.
95                    $covered = $own['executed'] > 0;
96
97                    if ($covered) {
98                        $unit['coveredmethods']++;
99                    }
100
101                    $unit['statements'] += $own['executable'];
102                    $unit['coveredstatements'] += $own['executed'];
103
104                    $methodRows[$method->startLine] = $this->openClover
105                        ? sprintf(
106                            "        <line num=\"%d\" type=\"method\" complexity=\"%d\" count=\"%d\" signature=\"%s\" visibility=\"%s\"/>\n",
107                            $method->startLine,
108                            $method->complexity,
109                            $covered ? 1 : 0,
110                            $this->escape($method->signature()),
111                            $this->escape($method->visibility),
112                        )
113                        : sprintf(
114                            "        <line num=\"%d\" type=\"method\" name=\"%s\" visibility=\"%s\" complexity=\"%d\" crap=\"%s\" count=\"%d\"/>\n",
115                            $method->startLine,
116                            $this->escape($method->name),
117                            $this->escape($method->visibility),
118                            $method->complexity,
119                            $this->number($method->crap($values)),
120                            $covered ? 1 : 0,
121                        );
122                }
123
124                $unit['elements']        = $unit['statements'] + $unit['methods'];
125                $unit['coveredelements'] = $unit['coveredstatements'] + $unit['coveredmethods'];
126
127                $classes .= $this->openClover
128                    ? sprintf(
129                        "        <class name=\"%s\">\n          %s\n        </class>\n",
130                        $this->escape($class->name),
131                        $this->metrics($unit, ['complexity', 'elements', 'coveredelements', 'conditionals', 'coveredconditionals', 'statements', 'coveredstatements', 'methods', 'coveredmethods']),
132                    )
133                    : sprintf(
134                        "        <class name=\"%s\" namespace=\"%s\">\n          %s\n        </class>\n",
135                        $this->escape($qualified),
136                        $this->escape($class->namespace),
137                        $this->metrics($unit, ['complexity', 'methods', 'coveredmethods', 'conditionals', 'coveredconditionals', 'statements', 'coveredstatements', 'elements', 'coveredelements']),
138                    );
139
140                $file_['classes']++;
141                $file_['complexity'] += $unit['complexity'];
142                $file_['methods'] += $unit['methods'];
143                $file_['coveredmethods'] += $unit['coveredmethods'];
144            }
145
146            foreach ($values as $line => $value) {
147                if ($value === -2 || isset($methodRows[$line])) {
148                    continue;
149                }
150
151                $count = max($value, 0);
152
153                $file_['statements']++;
154
155                if ($count > 0) {
156                    $file_['coveredstatements']++;
157                }
158
159                $rows .= sprintf("        <line num=\"%d\" type=\"stmt\" count=\"%d\"/>\n", $line, $count);
160            }
161
162            ksort($methodRows);
163
164            // The conditionals metric is the branch analysis (D-062);
165            // without it the counts stay honestly zero.
166            $branches                     = CoverageData::branchCounts($data->branches[$file] ?? []);
167            $file_['conditionals']        = $branches['total'];
168            $file_['coveredconditionals'] = $branches['covered'];
169            $file_['elements']            = $file_['statements'] + $file_['methods'] + $branches['total'];
170            $file_['coveredelements']     = $file_['coveredstatements'] + $file_['coveredmethods'] + $branches['covered'];
171
172            $escaped    = $this->escape($file);
173            $attributes = $this->openClover
174                ? sprintf('name="%s" path="%s"', $this->escape(basename($file)), $escaped)
175                : sprintf('name="%s"', $escaped);
176
177            $fileMetrics = $this->openClover
178                ? $this->metrics($file_, ['loc', 'ncloc', 'classes', 'complexity', 'elements', 'coveredelements', 'conditionals', 'coveredconditionals', 'statements', 'coveredstatements', 'methods', 'coveredmethods'])
179                : $this->metrics($file_, ['loc', 'ncloc', 'classes', 'methods', 'coveredmethods', 'conditionals', 'coveredconditionals', 'statements', 'coveredstatements', 'elements', 'coveredelements']);
180
181            $body = $this->openClover
182                ? sprintf("      <file %s>\n        %s\n%s%s%s      </file>\n", $attributes, $fileMetrics, $classes, implode('', $methodRows), $rows)
183                : sprintf("      <file %s>\n%s%s%s        %s\n      </file>\n", $attributes, $classes, implode('', $methodRows), $rows, $fileMetrics);
184
185            $packages[$package][] = $body;
186
187            foreach ($file_ as $key => $value) {
188                $totals[$key] += $value;
189            }
190
191            $totals['files']++;
192        }
193
194        $rendered = '';
195
196        foreach ($packages as $name => $bodies) {
197            $rendered .= $this->openClover
198                ? sprintf(
199                    "    <package name=\"%s\">\n      %s\n%s    </package>\n",
200                    $this->escape($name),
201                    $this->metrics($totals, ['complexity', 'elements', 'coveredelements', 'conditionals', 'coveredconditionals', 'statements', 'coveredstatements', 'methods', 'coveredmethods']),
202                    implode('', $bodies),
203                )
204                : sprintf("    <package name=\"%s\">\n%s    </package>\n", $this->escape($name), implode('', $bodies));
205        }
206
207        $projectMetrics = $this->openClover
208            ? $this->metrics($totals, ['files', 'loc', 'ncloc', 'classes', 'complexity', 'elements', 'coveredelements', 'conditionals', 'coveredconditionals', 'statements', 'coveredstatements', 'methods', 'coveredmethods'])
209            : $this->metrics($totals, ['files', 'loc', 'ncloc', 'classes', 'methods', 'coveredmethods', 'conditionals', 'coveredconditionals', 'statements', 'coveredstatements', 'elements', 'coveredelements']);
210
211        return $this->openClover
212            ? sprintf(
213                "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<coverage clover=\"3.2.0\" generated=\"%d\">\n  <project timestamp=\"%d\" name=\"OpenClover Coverage\">\n    %s\n%s  </project>\n</coverage>\n",
214                $timestamp,
215                $timestamp,
216                $projectMetrics,
217                $rendered,
218            )
219            : sprintf(
220                "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<coverage generated=\"%d\">\n  <project timestamp=\"%d\" name=\"Clover Coverage\">\n%s    %s\n  </project>\n</coverage>\n",
221                $timestamp,
222                $timestamp,
223                $rendered,
224                $projectMetrics,
225            );
226    }

228    /**
229     * @return array<string, int>
230     */
231    private function zero(): array
232    {
233        return [
234            'files'      => 0, 'loc' => 0, 'ncloc' => 0, 'classes' => 0, 'complexity' => 0,
235            'methods'    => 0, 'coveredmethods' => 0, 'conditionals' => 0, 'coveredconditionals' => 0,
236            'statements' => 0, 'coveredstatements' => 0, 'elements' => 0, 'coveredelements' => 0,
237        ];
238    }
239
240    /**
241     * @param array<string, int> $counts
242     * @param list<string>       $keys
243     */
244    private function metrics(array $counts, array $keys): string
245    {
246        $attributes = '';
247
248        foreach ($keys as $key) {
249            $attributes .= sprintf(' %s="%d"', $key, $counts[$key] ?? 0);
250        }
251
252        return '<metrics' . $attributes . '/>';
253    }
254
255    /**
256     * @param array<int, int> $values
257     *
258     * @return array{executable: int, executed: int}
259     */
260    private function counts(array $values, int $from, int $to): array
261    {
262        $executable = 0;
263        $executed   = 0;
264
265        for ($line = $from; $line <= $to; $line++) {
266            if (!isset($values[$line]) || $values[$line] === -2) {
267                continue;
268            }
269
270            $executable++;
271
272            if ($values[$line] > 0) {
273                $executed++;
274            }
275        }
276
277        return ['executable' => $executable, 'executed' => $executed];
278    }
279
280    private function number(float $value): string
281    {
282        return sprintf('%.2F', $value);
283    }
284
285    private function escape(string $value): string
286    {
287        return htmlspecialchars($value, ENT_XML1 | ENT_QUOTES);
288    }
