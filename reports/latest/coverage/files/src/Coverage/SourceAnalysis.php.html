28/**
29 * The class and method structure of a source file, with each method's
30 * cyclomatic complexity — what a line-coverage map cannot say on its
31 * own, and what the CRAP index and the per-class report views need.
32 *
33 * Read from tokens, not reflection: the report runs in the parent after
34 * a run whose workers loaded the code, so nothing about which classes
35 * happen to be declared here is a safe basis for a report. Tokens are
36 * also the only source that stays correct for code that never ran.
37 *
38 * Complexity is the standard decision-point count: one, plus every
39 * branch a path could take.
40 */
41final readonly class SourceAnalysis
42{
43    /**
44     * Every token that opens another independent path through a method,
45     * matching the spec's set exactly (sebastian/complexity): `??` is
46     * not a branch in it, `match` counts once per *arm* rather than once
47     * for the construct, and a do-while's trailing `while` is the same
48     * loop already counted at `do`.
49     */
50    private const array DECISIONS = [
51        T_IF, T_ELSEIF, T_FOR, T_FOREACH, T_WHILE, T_CASE, T_CATCH,
52        T_BOOLEAN_AND, T_BOOLEAN_OR, T_LOGICAL_AND, T_LOGICAL_OR,
53    ];
54
55    /** What a `?` follows when it opens a ternary rather than a nullable type. */
56    private const array ENDS_AN_EXPRESSION = [
57        T_STRING, T_VARIABLE, T_CONSTANT_ENCAPSED_STRING, T_LNUMBER, T_DNUMBER,
58        T_ARRAY, T_STRING_VARNAME, T_NUM_STRING, T_ENCAPSED_AND_WHITESPACE,
59        T_NAME_QUALIFIED, T_NAME_FULLY_QUALIFIED, T_NAME_RELATIVE,
60        T_CLASS_C, T_FUNC_C, T_METHOD_C, T_LINE, T_FILE, T_DIR, T_NS_C,
61    ];

77    /**
78     * @param non-empty-string $file
79     */
80    public static function of(string $file): self
81    {
82        if (!is_file($file)) {
83            return new self([]);
84        }
85
86        // Every report that needs method structure asks for the same
87        // files, and a run may ask several times over. Keyed by mtime
88        // rather than path alone: an edited file is a different file.
89        $key    = $file . '@' . filemtime($file);
90        $cached = SourceAnalysisCache::get($key);
91
92        if ($cached instanceof self) {
93            return $cached;
94        }
95
96        $source   = file_get_contents($file);
97        $analysis = $source === false ? new self([]) : self::parse($source);
98
99        SourceAnalysisCache::put($key, $analysis);
100
101        return $analysis;
102    }

104    /**
105     * The spec's --warm-coverage-cache: parse now, so the run that
106     * needs the structure does not pay for it. Returns how many files
107     * were analysed.
108     *
109     * @param list<non-empty-string> $files
110     */
111    public static function warm(array $files): int
112    {
113        $warmed = 0;
114
115        foreach ($files as $file) {
116            if (is_file($file)) {
117                self::of($file);
118                $warmed++;
119            }
120        }
121
122        return $warmed;
123    }

125    public static function parse(string $source): self
126    {
127        /** @var list<array{int, string, int}|string> $tokens */
128        $tokens = token_get_all($source);
129
130        $namespace = '';
131        $classes   = [];
132
133        /** @var ?array{name: string, namespace: string, methods: array<string, SourceMethod>, depth: int} $current */
134        $current = null;
135
136        /** @var ?array{name: string, start: int, complexity: int, depth: int, static: bool, visibility: string} $method */
137        $method = null;
138
139        $depth     = 0;
140        $modifiers = ['static' => false, 'visibility' => 'public'];
141
142        /** @var list<array{depth: int, brackets: int}> $matchDepths brace and bracket depth inside each open match block */
143        $matchDepths = [];
144
145        // A `match` seen but not yet opened, and a `do` whose trailing
146        // `while` closes the same loop rather than starting another.
147        $pendingMatch = false;
148        $pendingDo    = 0;
149
150        // An arrow function's `=>` separates its body; written inside a
151        // match it is not another arm.
152        $pendingFn = 0;
153
154        // `[` depth, so the `=>` of an array literal written inside a
155        // match arm is not mistaken for another arm; `(` depth, so the
156        // named argument `default:` is not mistaken for a case label.
157        $brackets = 0;
158        $parens   = 0;
159        $counter  = count($tokens);
160
161        for ($i = 0; $i < $counter; $i++) {
162            $token = $tokens[$i];
163
164            if ($token === '{') {
165                $depth++;
166
167                if ($pendingMatch) {
168                    $matchDepths[] = ['depth' => $depth, 'brackets' => $brackets];
169                    $pendingMatch  = false;
170                }
171
172                continue;
173            }
174
175            if ($token === '}') {
176                if ($matchDepths !== [] && $matchDepths[count($matchDepths) - 1]['depth'] === $depth) {
177                    array_pop($matchDepths);
178                }
179
180                $depth--;
181
182                if ($method !== null && $depth === $method['depth']) {
183                    $line = self::lineOf($tokens, $i);
184
185                    /** @var array{name: string, namespace: string, methods: array<string, SourceMethod>, depth: int} $current */
186                    $current['methods'][$method['name']] = new SourceMethod(
187                        $method['name'],
188                        $method['start'],
189                        $line,
190                        $method['complexity'],
191                        $method['visibility'],
192                        $method['static'],
193                    );
194
195                    $method = null;
196                }
197
198                if ($current !== null && $method === null && $depth === $current['depth']) {
199                    $classes[self::qualify($current['namespace'], $current['name'])] = new SourceClass(
200                        $current['name'],
201                        $current['namespace'],
202                        $current['methods'],
203                    );
204
205                    $current = null;
206                }
207
208                continue;
209            }
210
211            if ($token === '[') {
212                $brackets++;
213
214                continue;
215            }
216
217            if ($token === ']') {
218                $brackets--;
219
220                continue;
221            }
222
223            if ($token === '(') {
224                $parens++;
225
226                continue;
227            }
228
229            if ($token === ')') {
230                $parens--;
231
232                continue;
233            }
234
235            if (!is_array($token)) {
236                // `?` opens a ternary when it follows a complete
237                // expression, and marks a nullable type when it does
238                // not — `?int $x` against `$a ? $b : $c`.
239                if ($token === '?' && $method !== null && self::endsAnExpression($tokens, $i)) {
240                    $method['complexity']++;
241                }
242
243                continue;
244            }
245
246            if ($method !== null && in_array($token[0], self::DECISIONS, true) && !self::isMemberName($tokens, $i)) {
247                // The `while` of a do-while closes the loop `do` opened.
248                if ($token[0] === T_WHILE && $pendingDo > 0) {
249                    $pendingDo--;
250
251                    continue;
252                }
253
254                $method['complexity']++;
255
256                continue;
257            }
258
259            // An attribute opens with "#[" as one token and closes with a
260            // bare "]", so its opener has to be counted the same way.
261            if ($token[0] === T_ATTRIBUTE) {
262                $brackets++;
263
264                continue;
265            }
266
267            // String interpolation opens a brace with a token but closes
268            // it with a bare "}", so the opener has to be counted or the
269            // depth drifts negative and every later boundary is wrong.
270            if ($token[0] === T_CURLY_OPEN || $token[0] === T_DOLLAR_OPEN_CURLY_BRACES) {
271                $depth++;
272
273                continue;
274            }
275
276            if ($token[0] === T_DO) {
277                $pendingDo++;
278
279                continue;
280            }
281
282            if ($token[0] === T_MATCH) {
283                $pendingMatch = true;
284
285                continue;
286            }
287
288            if ($token[0] === T_FN) {
289                $pendingFn++;
290
291                continue;
292            }
293
294            // One arm, one path — but only the arms of the match itself,
295            // never the `=>` of an array literal written inside one.
296            if ($token[0] === T_DOUBLE_ARROW && $pendingFn > 0) {
297                $pendingFn--;
298
299                continue;
300            }
301
302            if ($token[0] === T_DOUBLE_ARROW
303                && $method !== null
304                && $matchDepths !== []
305                && $matchDepths[count($matchDepths) - 1]['depth'] === $depth
306                && $matchDepths[count($matchDepths) - 1]['brackets'] === $brackets) {
307                $method['complexity']++;
308
309                continue;
310            }
311
312            // A switch's `default` is a case like any other; a match's
313            // is an arm, already counted by its `=>`; and one inside
314            // parentheses is the named argument `default:`, not a label.
315            if ($token[0] === T_DEFAULT
316                && $method !== null
317                && $parens === 0
318                && !self::isMemberName($tokens, $i)
319                && ($matchDepths === [] || $matchDepths[count($matchDepths) - 1]['depth'] !== $depth)) {
320                $method['complexity']++;
321
322                continue;
323            }
324
325            switch ($token[0]) {
326                case T_NAMESPACE:
327                    $namespace = self::nameAfter($tokens, $i);
328
329                    break;
330
331                case T_CLASS:
332                case T_TRAIT:
333                case T_INTERFACE:
334                case T_ENUM:
335                    // An anonymous class has no name to key a report by.
336                    $name = self::nameAfter($tokens, $i);
337
338                    if ($name !== '' && $current === null) {
339                        $current = ['name' => $name, 'namespace' => $namespace, 'methods' => [], 'depth' => $depth];
340                    }
341
342                    break;
343
344                case T_STATIC:
345                    $modifiers['static'] = true;
346
347                    break;
348
349                case T_PUBLIC:
350                case T_PROTECTED:
351                case T_PRIVATE:
352                    $modifiers['visibility'] = $token[1];
353
354                    break;
355
356                case T_FUNCTION:
357                    $name = self::nameAfter($tokens, $i);
358
359                    if ($name !== '' && $current !== null && $method === null) {
360                        $method = [
361                            'name'       => $name,
362                            'start'      => $token[2],
363                            'complexity' => 1,
364                            'depth'      => $depth,
365                            'static'     => $modifiers['static'],
366                            'visibility' => $modifiers['visibility'],
367                        ];
368                    }
369
370                    $modifiers = ['static' => false, 'visibility' => 'public'];
371
372                    break;
373
374                case T_WHITESPACE:
375                case T_COMMENT:
376                case T_DOC_COMMENT:
377                    break;
378
379                default:
380                    // Anything else ends a modifier run that named no
381                    // function: a property, a constant, a return type.
382                    if (!in_array($token[0], [T_READONLY, T_ABSTRACT, T_FINAL], true)) {
383                        $modifiers = ['static' => false, 'visibility' => 'public'];
384                    }
385            }
386        }
387
388        $braceLines = self::braceLines($source);
389
390        return new self(self::extendToBraces($classes, $braceLines), self::lineCounts($source, $tokens), $braceLines);
391    }

393    /**
394     * A method ends at its closing brace, not at its last statement.
395     *
396     * The token walk cannot see it: token_get_all hands back a bare
397     * string for `}` with no line attached, so lineOf() reports the last
398     * token that had one. The implicit return lives on that brace, the
399     * incumbent reports it as the method's end, and for a body that is
400     * only a comment it is the sole line coverage can measure — so a
401     * method whose range stops short reports itself untested.
402     *
403     * @param array<string, SourceClass> $classes
404     * @param array<int, true>           $braceLines
405     *
406     * @return array<string, SourceClass>
407     */
408    private static function extendToBraces(array $classes, array $braceLines): array
409    {
410        $extended = [];
411
412        foreach ($classes as $qualified => $class) {
413            $methods = [];
414
415            foreach ($class->methods as $name => $method) {
416                $closing = null;
417
418                foreach (array_keys($braceLines) as $line) {
419                    if ($line > $method->endLine && ($closing === null || $line < $closing)) {
420                        $closing = $line;
421                    }
422                }
423
424                $methods[$name] = $closing === null ? $method : new SourceMethod(
425                    $method->name,
426                    $method->startLine,
427                    $closing,
428                    $method->complexity,
429                    $method->visibility,
430                    $method->static,
431                );
432            }
433
434            $extended[$qualified] = new SourceClass($class->name, $class->namespace, $methods);
435        }
436
437        return $extended;
438    }

440    /**
441     * Lines whose only code is a closing brace.
442     *
443     * Xdebug reports a function's closing brace as executable — the
444     * implicit return lives there — and reports it executed. Nobody
445     * wrote a statement on that line, so counting it makes a two-line
446     * body measure three and inflates every ratio derived from it.
447     * PhpToken rather than token_get_all: the latter hands back a bare
448     * string for `}` with no line number attached.
449     *
450     * @return array<int, true>
451     */
452    private static function braceLines(string $source): array
453    {
454        $significant = [];
455        $braces      = [];
456
457        foreach (PhpToken::tokenize($source) as $token) {
458            if ($token->is([T_WHITESPACE, T_COMMENT, T_DOC_COMMENT, T_OPEN_TAG, T_CLOSE_TAG, T_INLINE_HTML])) {
459                continue;
460            }
461
462            $significant[$token->line] = ($significant[$token->line] ?? 0) + 1;
463
464            if ($token->text === '}') {
465                $braces[$token->line] = ($braces[$token->line] ?? 0) + 1;
466            }
467        }
468
469        $lines = [];
470
471        foreach ($braces as $line => $count) {
472            if ($count === $significant[$line]) {
473                $lines[$line] = true;
474            }
475        }
476
477        return $lines;
478    }

480    /**
481     * Total, comment and code lines — counted from the same token pass
482     * rather than a second one. A line holding both code and a trailing
483     * comment counts as code, which is what the incumbent's metric does.
484     *
485     * @param list<array{int, string, int}|string> $tokens
486     *
487     * @return array{total: int, comments: int, code: int}
488     */
489    private static function lineCounts(string $source, array $tokens): array
490    {
491        // A trailing newline opens a last line, and it counts: the
492        // incumbent reports 49 for a 48-newline file and every coverage
493        // percentage is a ratio against this number.
494        $total = $source === '' ? 0 : substr_count($source, "\n") + 1;
495
496        /** @var array<int, true> $commentLines */
497        $commentLines = [];
498
499        /** @var array<int, true> $codeLines */
500        $codeLines = [];
501
502        foreach ($tokens as $token) {
503            if (!is_array($token)) {
504                continue;
505            }
506
507            $lines = substr_count($token[1], "\n");
508
509            for ($line = $token[2]; $line <= $token[2] + $lines; $line++) {
510                if ($token[0] === T_COMMENT || $token[0] === T_DOC_COMMENT) {
511                    $commentLines[$line] = true;
512                } elseif (!in_array($token[0], [T_WHITESPACE, T_OPEN_TAG, T_CLOSE_TAG], true)) {
513                    $codeLines[$line] = true;
514                }
515            }
516        }
517
518        // A line that is both is code: the comment rode along with it.
519        $comments = count(array_diff_key($commentLines, $codeLines));
520
521        return ['total' => $total, 'comments' => $comments, 'code' => count($codeLines)];
522    }

524    /**
525     * PHP lets a reserved word name a member, so `Suite::for(...)` and
526     * `$x->match()` carry control-structure tokens that open no branch.
527     *
528     * @param list<array{int, string, int}|string> $tokens
529     */
530    private static function isMemberName(array $tokens, int $index): bool
531    {
532        for ($i = $index - 1; $i >= 0; $i--) {
533            $token = $tokens[$i];
534
535            if (!is_array($token)) {
536                return false;
537            }
538
539            if (in_array($token[0], [T_WHITESPACE, T_COMMENT, T_DOC_COMMENT], true)) {
540                continue;
541            }
542
543            return in_array(
544                $token[0],
545                [T_DOUBLE_COLON, T_OBJECT_OPERATOR, T_NULLSAFE_OBJECT_OPERATOR, T_FUNCTION, T_CONST],
546                true,
547            );
548        }
549
550        return false;
551    }

553    /**
554     * @param list<array{int, string, int}|string> $tokens
555     */
556    private static function endsAnExpression(array $tokens, int $index): bool
557    {
558        for ($i = $index - 1; $i >= 0; $i--) {
559            $token = $tokens[$i];
560
561            if (!is_array($token)) {
562                return in_array($token, [')', ']', '}'], true);
563            }
564
565            if (in_array($token[0], [T_WHITESPACE, T_COMMENT, T_DOC_COMMENT], true)) {
566                continue;
567            }
568
569            return in_array($token[0], self::ENDS_AN_EXPRESSION, true);
570        }
571
572        return false;
573    }

575    /**
576     * @param list<array{int, string, int}|string> $tokens
577     */
578    private static function nameAfter(array $tokens, int $index): string
579    {
580        $counter = count($tokens);
581        for ($i = $index + 1; $i < $counter; $i++) {
582            $token = $tokens[$i];
583
584            if (!is_array($token)) {
585                // `function (` is a closure, `class {` an anonymous one:
586                // neither is something a report can name.
587                return '';
588            }
589
590            if (in_array($token[0], [T_WHITESPACE, T_COMMENT, T_DOC_COMMENT, T_ATTRIBUTE], true)) {
591                continue;
592            }
593
594            if ($token[0] === T_STRING || $token[0] === T_NAME_QUALIFIED) {
595                return $token[1];
596            }
597
598            // `function &foo()` returns by reference; `namespace\` is a
599            // relative name, not a declaration.
600            return '';
601        }
602
603        return '';
604    }

606    /**
607     * @param list<array{int, string, int}|string> $tokens
608     */
609    private static function lineOf(array $tokens, int $index): int
610    {
611        for ($i = $index; $i >= 0; $i--) {
612            $token = $tokens[$i];
613
614            if (is_array($token)) {
615                return $token[2];
616            }
617        }
618
619        return 0;
620    }
