22/**
23 * The Crap4J XML emitter: CRAP (Change Risk Anti-Patterns) per method,
24 * the metric that pairs cyclomatic complexity with coverage so a
25 * complicated method nobody tests scores worse than either number says
26 * alone. Method boundaries and complexity come from {@see SourceAnalysis}.
27 */
28final readonly class Crap4jWriter
29{
30    /**
31     * @param int<1, max> $threshold the CRAP value at which a method counts as crappy
32     */
33    public function __construct(private int $threshold = 30) {}
34
35    /**
36     * @param non-empty-string $root absolute project root
37     */
38    public function write(CoverageData $data, string $root): string
39    {
40        $lines = $data->lines;
41
42        ksort($lines);
43
44        $methods     = '';
45        $methodCount = 0;
46        $crapCount   = 0;
47        $totalCrap   = 0.0;
48        $totalLoad   = 0.0;
49
50        foreach ($lines as $file => $values) {
51            if ($file === '') {
52                continue;
53            }
54
55            foreach (SourceAnalysis::of($file)->classes as $qualified => $class) {
56                foreach ($class->methods as $method) {
57                    $crap     = $method->crap($values);
58                    $coverage = $method->coverage($values);
59
60                    $methodCount++;
61                    $totalCrap += $crap;
62
63                    if ($crap >= $this->threshold) {
64                        $crapCount++;
65                        $totalLoad += $method->complexity * (1.0 - $coverage / 100) + $method->complexity / $this->threshold;
66                    }
67
68                    $methods .= sprintf(
69                        "      <method>\n        <package>%s</package>\n        <className>%s</className>\n        <methodName>%s</methodName>\n        <methodSignature>%s</methodSignature>\n        <fullMethod>%s</fullMethod>\n        <crap>%s</crap>\n        <complexity>%d</complexity>\n        <coverage>%s</coverage>\n        <crapLoad>%d</crapLoad>\n      </method>\n",
70                        $this->escape($class->namespace),
71                        $this->escape($qualified),
72                        $this->escape($method->name),
73                        $this->escape($method->signature()),
74                        $this->escape($method->signature()),
75                        $this->number($crap),
76                        $method->complexity,
77                        $this->number($coverage),
78                        (int) round($crap >= $this->threshold ? $method->complexity * (1.0 - $coverage / 100) + $method->complexity / $this->threshold : 0.0),
79                    );
80                }
81            }
82        }
83
84        return sprintf(
85            "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<crap_result>\n  <project>%s</project>\n  <timestamp>%s</timestamp>\n  <stats>\n    <name>Method Crap Stats</name>\n    <methodCount>%d</methodCount>\n    <crapMethodCount>%d</crapMethodCount>\n    <crapLoad>%d</crapLoad>\n    <totalCrap>%s</totalCrap>\n    <crapMethodPercent>%s</crapMethodPercent>\n  </stats>\n  <methods>\n%s  </methods>\n</crap_result>\n",
86            $this->escape($root),
87            date('Y-m-d H:i:s'),
88            $methodCount,
89            $crapCount,
90            (int) round($totalLoad),
91            $this->number($totalCrap),
92            $this->number($methodCount === 0 ? 0.0 : 100 * $crapCount / $methodCount),
93            $methods,
94        );
95    }
96
97    private function escape(string $value): string
98    {
99        return htmlspecialchars($value, ENT_XML1 | ENT_QUOTES);
100    }
101
102    private function number(float $value): string
103    {
104        return sprintf('%01.2F', round($value, 2));
105    }
106}
