40/**
41 * A test's coverage-target claim, resolved from its Covers and Uses
42 * metadata to file line ranges (D-063, oracle-pinned semantics):
43 * Covers* ranges are what the test may CONTRIBUTE to the aggregate;
44 * Uses* ranges extend only what it may EXECUTE under
45 * beStrictAboutCoverageMetadata; CoversNothing declares a target,
46 * contributes nothing, and is exempt from the strict check. A target
47 * naming an undeclared symbol resolves to no range — never an engine
48 * error; strict runs surface the resulting strays instead.
49 */
50final readonly class CoversTargets
51{
52    /**
53     * @param list<array{string, int, int}> $covered file, first line, last line
54     * @param list<array{string, int, int}> $allowed covered + the Uses* ranges
55     */
56    private function __construct(
57        public bool $declared,
58        public bool $coversNothing,
59        private array $covered,
60        private array $allowed,
61    ) {}
62
63    public static function from(MetadataCollection $metadata): self
64    {
65        $covered = [];
66        $allowed = [];
67
68        foreach ($metadata->ofType(CoversClass::class) as $covers) {
69            self::addType($covered, $covers->className);
70        }
71
72        foreach ($metadata->ofType(CoversTrait::class) as $covers) {
73            self::addType($covered, $covers->traitName);
74        }
75
76        foreach ($metadata->ofType(CoversMethod::class) as $covers) {
77            self::addMethod($covered, $covers->className, $covers->methodName);
78        }
79
80        foreach ($metadata->ofType(CoversFunction::class) as $covers) {
81            self::addFunction($covered, $covers->functionName);
82        }
83
84        $allowed = $covered;
85
86        foreach ($metadata->ofType(UsesClass::class) as $uses) {
87            self::addType($allowed, $uses->className);
88        }
89
90        foreach ($metadata->ofType(UsesTrait::class) as $uses) {
91            self::addType($allowed, $uses->traitName);
92        }
93
94        foreach ($metadata->ofType(UsesMethod::class) as $uses) {
95            self::addMethod($allowed, $uses->className, $uses->methodName);
96        }
97
98        foreach ($metadata->ofType(UsesFunction::class) as $uses) {
99            self::addFunction($allowed, $uses->functionName);
100        }
101
102        $coversNothing = $metadata->has(CoversNothing::class);
103
104        $declared = $coversNothing
105            || $metadata->has(CoversClass::class) || $metadata->has(CoversTrait::class)
106            || $metadata->has(CoversMethod::class) || $metadata->has(CoversFunction::class)
107            || $metadata->has(UsesClass::class) || $metadata->has(UsesTrait::class)
108            || $metadata->has(UsesMethod::class) || $metadata->has(UsesFunction::class);
109
110        return new self($declared, $coversNothing, $covered, $allowed);
111    }

113    /**
114     * The window this test may add to the aggregate report: identity
115     * without a covers claim; otherwise executed lines OUTSIDE the
116     * covered ranges (all of them, under CoversNothing) are DEMOTED
117     * to executable-missed, never dropped — the oracle keeps stray
118     * lines in the denominators (probe-pinned, D-063). The per-test
119     * map is NOT this — the impact and mutation artifacts keep
120     * observed truth.
121     */
122    public function contribution(CoverageWindow $window): CoverageWindow
123    {
124        if (!$this->declared) {
125            return $window;
126        }
127
128        $lines = [];
129
130        foreach ($window->lines as $file => $values) {
131            foreach ($values as $line => $value) {
132                $lines[$file][$line] = $value > 0 && ($this->coversNothing || !$this->inRanges($this->covered, $file, $line))
133                    ? -1
134                    : $value;
135            }
136        }
137
138        return new CoverageWindow($lines, $this->claimed($window->branches), $this->claimed($window->paths));
139    }
140
141    /**
142     * A hit outside what this test claims to cover contributes nothing
143     * to the aggregate — the same rule the lines follow, applied to
144     * branch and path entries alike.
145     *
146     * @param array<string, array<string, array{line: int, hit: int}>> $entries
147     *
148     * @return array<string, array<string, array{line: int, hit: int}>>
149     */
150    private function claimed(array $entries): array
151    {
152        $kept = [];
153
154        foreach ($entries as $file => $byId) {
155            foreach ($byId as $id => $entry) {
156                $kept[$file][$id] = $entry['hit'] > 0 && ($this->coversNothing || !$this->inRanges($this->covered, $file, $entry['line']))
157                    ? ['line' => $entry['line'], 'hit' => 0]
158                    : $entry;
159            }
160        }
161
162        return $kept;
163    }

165    /**
166     * The beStrictAboutCoverageMetadata check (D-063): executed lines
167     * outside every covered/used range, named by their declaring unit
168     * — the oracle reports class names, not files. Lines outside any
169     * known unit (file-level code) cannot be named and are not strays.
170     *
171     * @return list<non-empty-string>
172     */
173    public function strays(CoverageWindow $window): array
174    {
175        if (!$this->declared || $this->coversNothing) {
176            return [];
177        }
178
179        /** @var array<non-empty-string, true> $strays */
180        $strays = [];
181        $index  = null;
182
183        foreach ($window->lines as $file => $values) {
184            foreach ($values as $line => $value) {
185                if ($value <= 0 || $this->inRanges($this->allowed, $file, $line)) {
186                    continue;
187                }
188
189                // Built once per check, and only when a stray exists.
190                $index ??= $this->unitIndex();
191
192                foreach ($index[$file] ?? [] as $unit => [$start, $end]) {
193                    if ($unit !== '' && $line >= $start && $line <= $end) {
194                        $strays[$unit] = true;
195
196                        break;
197                    }
198                }
199            }
200        }
201
202        $names = array_keys($strays);
203
204        sort($names);
205
206        return $names;
207    }

209    /**
210     * Whether the named type resolves — attempting the load, but never
211     * letting the attempt end the run.
212     *
213     * class_exists() autoloads, and autoloading a class whose parent is
214     * absent throws out of the loader. Crucible's own laravel bridge
215     * does exactly that wherever illuminate is not installed, and one
216     * #[CoversClass] naming it killed every coverage run over this
217     * suite before the first report was written. Measured against the
218     * oracle (phpunit 13.3.1, xdebug, a covered class extending a
219     * missing parent): the run finishes at exit 0 and the target simply
220     * resolves to nothing. So an unloadable name lands where an
221     * undeclared one already does — no range, no engine error.
222     *
223     * @phpstan-assert-if-true class-string $name
224     */
225    private static function resolves(string $name): bool
226    {
227        try {
228            return class_exists($name) || trait_exists($name) || interface_exists($name);
229        } catch (Throwable) {
230            return false;
231        }
232    }
233
234    /**
235     * @param list<array{string, int, int}> $target
236     */
237    private static function addType(array &$target, string $name): void
238    {
239        if (!self::resolves($name)) {
240            return;
241        }
242
243        $reflection = new ReflectionClass($name);
244        $file       = $reflection->getFileName();
245        $start      = $reflection->getStartLine();
246        $end        = $reflection->getEndLine();
247
248        if ($file !== false && is_int($start) && is_int($end)) {
249            $target[] = [$file, $start, $end];
250        }
251    }

254     * @param list<array{string, int, int}> $target
255     */
256    private static function addMethod(array &$target, string $class, string $method): void
257    {
258        if (!self::resolves($class) || !method_exists($class, $method)) {
259            return;
260        }
261
262        $reflection = new ReflectionMethod($class, $method);
263        $file       = $reflection->getFileName();
264        $start      = $reflection->getStartLine();
265        $end        = $reflection->getEndLine();
266
267        if ($file !== false && is_int($start) && is_int($end)) {
268            $target[] = [$file, $start, $end];
269        }
270    }

281        $reflection = new ReflectionFunction($function);
282        $file       = $reflection->getFileName();
283        $start      = $reflection->getStartLine();
284        $end        = $reflection->getEndLine();
285
286        if ($file !== false && is_int($start) && is_int($end)) {
287            $target[] = [$file, $start, $end];

291    /**
292     * @param list<array{string, int, int}> $ranges
293     */
294    private function inRanges(array $ranges, string $file, int $line): bool
295    {
296        foreach ($ranges as [$rangeFile, $start, $end]) {
297            if ($rangeFile === $file && $line >= $start && $line <= $end) {
298                return true;
299            }
300        }
301
302        return false;
303    }

305    /**
306     * A per-file index of every declared class, trait, and user
307     * function — the stray lines' declaring units. Built only when
308     * the strict check actually finds a stray.
309     *
310     * @return array<string, array<string, array{int, int}>> file => unit name => range
311     */
312    private function unitIndex(): array
313    {
314        $index = [];
315
316        foreach ([...get_declared_classes(), ...get_declared_traits()] as $type) {
317            $reflection = new ReflectionClass($type);
318            $file       = $reflection->getFileName();
319            $start      = $reflection->getStartLine();
320            $end        = $reflection->getEndLine();
321
322            if ($file !== false && is_int($start) && is_int($end)) {
323                $index[$file][$type] = [$start, $end];
324            }
325        }
326
327        foreach (get_defined_functions()['user'] as $function) {
328            $reflection = new ReflectionFunction($function);
329            $file       = $reflection->getFileName();
330            $start      = $reflection->getStartLine();
331            $end        = $reflection->getEndLine();
332
333            if ($file !== false && is_int($start) && is_int($end)) {
334                $index[$file][$function] = [$start, $end];
335            }
336        }
337
338        return $index;
339    }
