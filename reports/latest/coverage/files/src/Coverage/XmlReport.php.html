43/**
44 * The spec's --coverage-xml: a directory of XML, an index naming every
45 * file and one document per file carrying its classes, their methods
46 * with line ranges and CRAP, and the per-line coverage with the tests
47 * that reached each line.
48 *
49 * Output-only, like the Clover and Cobertura emitters — the no-XML rule
50 * bans XML *inputs*; emitters for consumers that speak XML are the
51 * sanctioned exception (D-041).
52 */
53final readonly class XmlReport
54{

97    /**
98     * @param non-empty-string $root      absolute project root
99     * @param non-empty-string $directory output directory
100     * @param non-empty-string $driver
101     * @param list<array{id: string, status: string, time: float, size?: string}> $tests
102     *                                                                                   how each test ended, for the index's <tests>; empty when the caller has no run to describe
103     */
104    public function write(CoverageData $data, string $root, string $directory, string $driver, array $tests = []): void
105    {
106        if (!is_dir($directory)) {
107            mkdir($directory, 0o777, true);
108        }
109
110        $prefix = $root . '/';
111        $lines  = $data->lines;
112
113        ksort($lines);
114
115        // Which tests reached which line, inverted from the per-test map
116        // so a line can name its own tests without re-scanning.
117        $byLine = [];
118
119        foreach ($data->tests as $testId => $files) {
120            foreach ($files as $file => $executed) {
121                foreach ($executed as $line) {
122                    $byLine[$file][$line][] = $testId;
123                }
124            }
125        }
126
127        $files = '';
128
129        foreach ($lines as $file => $values) {
130            if ($file === '') {
131                continue;
132            }
133
134            $relative = str_starts_with($file, $prefix) ? substr($file, strlen($prefix)) : $file;
135            $name     = $this->documentName($relative);
136
137            file_put_contents(
138                $directory . '/' . $name,
139                $this->fileDocument($file, $relative, $values, $byLine[$file] ?? []),
140            );
141
142            $files .= sprintf(
143                "    <file name=\"%s\" href=\"%s\" hash=\"%s\"/>\n",
144                $this->escape($relative),
145                $this->escape($name),
146                sha1_file($file) === false ? '' : sha1_file($file),
147            );
148        }
149
150        file_put_contents($directory . '/index.xml', sprintf(
151            "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<phpunit xmlns=\"https://schema.phpunit.de/coverage/1.0\">\n"
152            . "  <build time=\"%s\" phpunit=\"%s\" coverage=\"crucible %s\">\n"
153            . "    <runtime name=\"PHP\" version=\"%s\" url=\"https://www.php.net/\"/>\n"
154            . "    <driver name=\"%s\" version=\"%s\"/>\n"
155            . "  </build>\n"
156            . "  <project source=\"%s\">\n%s    <directory name=\"/\">\n%s%s    </directory>\n  </project>\n</phpunit>\n",
157            date('D M j G:i:s T Y'),
158            Version::COMPATIBILITY_SERIES,
159            Version::NUMBER,
160            PHP_VERSION,
161            $this->escape($driver),
162            $this->escape(phpversion($driver) === false ? 'unknown' : phpversion($driver)),
163            $this->escape($root),
164            $this->tests($tests),
165            $this->totals($data),
166            $files,
167        ));
168    }

170    /**
171     * @param non-empty-string                $file
172     * @param array<int, int>                 $values
173     * @param array<int, list<string>>        $byLine
174     */
175    private function fileDocument(string $file, string $relative, array $values, array $byLine): string
176    {
177        $units = '';
178
179        foreach (SourceAnalysis::of($file)->classes as $qualified => $class) {
180            $executable = 0;
181            $executed   = 0;
182            $start      = 0;
183            $complexity = 0;
184            $methods    = '';
185
186            foreach ($class->methods as $method) {
187                $start = $start === 0 ? $method->startLine : $start;
188                $own   = $this->counts($values, $method->startLine, $method->endLine);
189                $executable += $own['executable'];
190                $executed += $own['executed'];
191                $complexity += $method->complexity;
192
193                $methods .= sprintf(
194                    "      <method name=\"%s\" signature=\"%s\" start=\"%d\" end=\"%d\" crap=\"%s\" executable=\"%d\" executed=\"%d\" coverage=\"%s\"/>\n",
195                    $this->escape($method->name),
196                    $this->escape($method->signature()),
197                    $method->startLine,
198                    $method->endLine,
199                    $this->number($method->crap($values)),
200                    $own['executable'],
201                    $own['executed'],
202                    $this->number($method->coverage($values)),
203                );
204            }
205
206            $units .= sprintf(
207                "    <class name=\"%s\" start=\"%d\" executable=\"%d\" executed=\"%d\" crap=\"%s\">\n      <namespace name=\"%s\"/>\n%s    </class>\n",
208                $this->escape($qualified),
209                $start,
210                $executable,
211                $executed,
212                $this->number($this->crap($complexity, $executable === 0 ? 0.0 : 100 * $executed / $executable)),
213                $this->escape($class->namespace),
214                $methods,
215            );
216        }
217
218        $coverage = '';
219        $numbers  = $values;
220
221        ksort($numbers);
222
223        foreach ($numbers as $line => $value) {
224            if ($value <= 0) {
225                continue;
226            }
227
228            $covering = '';
229
230            $reachedBy = $byLine[$line] ?? [];
231
232            foreach ($reachedBy as $testId) {
233                $covering .= sprintf(
234                    "        <covered by=\"%s\" count=\"%d\"/>\n",
235                    $this->escape($testId),
236                    count($reachedBy),
237                );
238            }
239
240            // A covered line's hit counts are on its <covered> children,
241            // where the incumbent puts them, so repeating the total on
242            // the parent would say nothing new. A line executed by no
243            // recorded test has no children to carry it, and there the
244            // count is the only hit information the document has.
245            $coverage .= $covering === ''
246                ? sprintf("      <line nr=\"%d\" count=\"%d\"/>\n", $line, $value)
247                : sprintf("      <line nr=\"%d\">\n%s      </line>\n", $line, $covering);
248        }
249
250        $source = '';
251
252        if (!$this->excludeSource) {
253            $contents = file_get_contents($file);
254            $source   = $this->source($contents === false ? '' : $contents);
255        }
256
257        return sprintf(
258            "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<phpunit xmlns=\"https://schema.phpunit.de/coverage/1.0\">\n  <file name=\"%s\" path=\"%s\" hash=\"%s\">\n%s    <coverage>\n%s    </coverage>\n%s  </file>\n</phpunit>\n",
259            $this->escape(basename($relative)),
260            $this->escape(dirname($relative)),
261            sha1_file($file) === false ? '' : sha1_file($file),
262            $units,
263            $coverage,
264            $source,
265        );
266    }

268    private function totals(CoverageData $data): string
269    {
270        $totals = $data->totals();
271
272        // The loc and unit counts the incumbent's consumers read, from
273        // the analysis the CRAP and class views already need — a report
274        // that only carried executable/executed would be a narrower
275        // document wearing the same schema.
276        $lines   = ['total' => 0, 'comments' => 0, 'code' => 0];
277        $classes = 0;
278        $traits  = 0;
279        $methods = 0;
280        $tested  = 0;
281
282        foreach ($data->lines as $file => $values) {
283            if ($file === '') {
284                continue;
285            }
286
287            $analysis = SourceAnalysis::of($file);
288
289            foreach ($lines as $key => $count) {
290                $lines[$key] = $count + $analysis->lines[$key];
291            }
292
293            foreach ($analysis->classes as $class) {
294                $classes++;
295
296                foreach ($class->methods as $method) {
297                    $methods++;
298
299                    if ($method->coverage($values) > 0.0) {
300                        $tested++;
301                    }
302                }
303            }
304        }
305
306        return sprintf(
307            "      <totals>\n"
308            . "        <lines total=\"%d\" comments=\"%d\" code=\"%d\" executable=\"%d\" executed=\"%d\" percent=\"%s\"/>\n"
309            . "        <methods count=\"%d\" tested=\"%d\" percent=\"%s\"/>\n"
310            . "        <functions count=\"0\" tested=\"0\" percent=\"0\"/>\n"
311            . "        <classes count=\"%d\" tested=\"%d\" percent=\"%s\"/>\n"
312            . "        <traits count=\"%d\" tested=\"0\" percent=\"0\"/>\n"
313            . "      </totals>\n",
314            $lines['total'],
315            $lines['comments'],
316            $lines['code'],
317            $totals['executable'],
318            $totals['covered'],
319            $this->number($totals['executable'] === 0 ? 0.0 : 100 * $totals['covered'] / $totals['executable']),
320            $methods,
321            $tested,
322            $this->number($methods === 0 ? 0.0 : 100 * $tested / $methods),
323            $classes,
324            0,
325            $this->number(0.0),
326            $traits,
327        );
328    }

330    /**
331     * @param array<int, int> $values
332     *
333     * @return array{executable: int, executed: int}
334     */
335    private function counts(array $values, int $from, int $to): array
336    {
337        $executable = 0;
338        $executed   = 0;
339
340        for ($line = $from; $line <= $to; $line++) {
341            if (!isset($values[$line]) || $values[$line] === -2) {
342                continue;
343            }
344
345            $executable++;
346
347            if ($values[$line] > 0) {
348                $executed++;
349            }
350        }
351
352        return ['executable' => $executable, 'executed' => $executed];
353    }

355    /**
356     * One flat directory, so a path separator cannot escape it and two

364    /**
365     * The index's <tests>: how each test ended and how long it took.
366     * Omitted rather than emitted empty, because an empty element would
367     * describe a run that reported nothing.
368     *
369     * @param list<array{id: string, status: string, time: float, size?: string}> $tests
370     */
371    private function tests(array $tests): string
372    {
373        if ($tests === []) {
374            return '';
375        }
376
377        $rows = '';
378
379        foreach ($tests as $test) {
380            $rows .= sprintf(
381                "      <test name=\"%s\" size=\"%s\" status=\"%s\" time=\"%F\"/>\n",
382                $this->escape($test['id']),
383                $this->escape($test['size'] ?? 'unknown'),
384                $this->escape($test['status']),
385                $test['time'],
386            );
387        }
388
389        return sprintf("    <tests>\n%s    </tests>\n", $rows);
390    }

392    /**
393     * The <source> dump: every line of the file carrying the tokens on
394     * it, which is what a consumer rendering annotated source reads.
395     * A token spanning lines is split at each newline, so every fragment
396     * sits under the line it actually appears on and a line that holds
397     * nothing is still present as an empty element.
398     */
399    private function source(string $contents): string
400    {
401        if ($contents === '') {
402            return "    <source/>\n";
403        }
404
405        $tokens = [];
406        $line   = 1;
407
408        foreach (token_get_all($contents) as $token) {
409            if (is_array($token)) {
410                $name = $this->tokenName($token[0]);
411                $text = $token[1];
412                $line = $token[2];
413            } else {
414                $name = self::PUNCTUATION[$token] ?? 'T_UNKNOWN';
415                $text = $token;
416            }
417
418            foreach (explode("\n", $text) as $offset => $fragment) {
419                if ($fragment !== '') {
420                    $tokens[$line + $offset][] = sprintf(
421                        '<token name="%s">%s</token>',
422                        $name,
423                        $this->escape($fragment),
424                    );
425                }
426            }
427
428            $line += substr_count($text, "\n");
429        }
430
431        $body = '';
432
433        for ($number = 1, $total = substr_count($contents, "\n") + 1; $number <= $total; ++$number) {
434            $body .= isset($tokens[$number])
435                ? sprintf(
436                    "      <line no=\"%d\">\n        %s\n      </line>\n",
437                    $number,
438                    implode("\n        ", $tokens[$number]),
439                )
440                : sprintf("      <line no=\"%d\"/>\n", $number);
441        }
442
443        return sprintf("    <source>\n%s    </source>\n", $body);
444    }

446    /**
447     * PHP's own name for a token, except the one nobody outside the
448     * parser spells that way: `::` reaches consumers as T_DOUBLE_COLON.
449     */
450    private function tokenName(int $token): string
451    {
452        $name = token_name($token);
453
454        return $name === 'T_PAAMAYIM_NEKUDOTAYIM' ? 'T_DOUBLE_COLON' : $name;
455    }

457    private function escape(string $value): string
458    {
459        return htmlspecialchars($value, ENT_XML1 | ENT_QUOTES);
460    }

462    /**
463     * The class-level CRAP: the same curve a method's uses, over the
464     * class's summed complexity and its own covered proportion.
465     */
466    private function crap(int $complexity, float $coverage): float
467    {
468        return match (true) {
469            $coverage === 0.0 => $complexity ** 2 + $complexity,
470            $coverage >= 95   => $complexity,
471            default           => $complexity ** 2 * (1 - $coverage / 100) ** 3 + $complexity,
472        };
473    }
474
475    private function number(float $value): string
476    {
477        return sprintf('%01.2F', $value);
478    }
