74/**
75 * The assertion API of the PHPUnit 13 spec, implemented on Crucible's
76 * constraint engine. Every method funnels through assertThat(), which
77 * is also where assertions are counted.
78 *
79 * Coverage status vs the ~176-method spec surface is tracked in
80 * ROADMAP.md Phase 2.
81 */
82abstract class Assert
83{
84    /** @var int<0, max> */
85    private static int $count = 0;
86
87    public static function assertThat(mixed $value, Constraint $constraint, string $message = ''): void
88    {
89        self::$count++;
90
91        $constraint->evaluate($value, $message);
92    }
93
94    public static function fail(string $message = ''): never
95    {
96        self::$count++;
97
98        throw new AssertionFailedError($message !== '' ? $message : 'Failed asserting that a test did not fail.');
99    }
100
101    /**
102     * Spec method: frameworks layering their own assertion helpers on
103     * top (Laravel's TestResponse, for one) report their assertions
104     * through this, keeping risky-test detection honest.
105     */
106    public function addToAssertionCount(int $count): void
107    {
108        if ($count > 0) {
109            self::$count += $count;
110        }
111    }
112
113    /**
114     * Engine-internal: count one satisfied assertion without
115     * evaluating anything — the "this observed outcome IS the claim"
116     * tick (an expected failure happened, a snapshot was recorded).
117     * Introduced by D-049 to replace the assertTrue(true) idiom,
118     * which the narrowing extension rightly reports as a tautology.
119     * User helpers layering assertions use addToAssertionCount().
120     */
121    public static function countSatisfiedAssertion(): void
122    {
123        self::$count++;
124    }

1001     * @param list<array-key>         $keys
1002     *
1003     * @return array<array-key, mixed>
1004     */
1005    private static function withoutKeys(array $array, array $keys): array
1006    {

1009        }
1010
1011        return $array;

1014    /**
1015     * @param array<array-key, mixed> $array
1016     * @param list<array-key>         $keys
1017     *
1018     * @return array<array-key, mixed>
1019     */
1020    private static function onlyKeys(array $array, array $keys): array
1021    {
1022        $kept = [];
1023
1024        foreach ($keys as $key) {
1025            if (array_key_exists($key, $array)) {
1026                $kept[$key] = $array[$key];
1027            }
1028        }
1029
1030        return $kept;
1031    }

1365    private static function readFile(string $path): string
1366    {
1367        if (!is_file($path) || !is_readable($path)) {
1368            throw new AssertionFailedError(sprintf('Failed asserting that file "%s" exists and is readable.', $path));
1369        }
1370
1371        $contents = file_get_contents($path);
1372
1373        if ($contents === false) {
1374            throw new AssertionFailedError(sprintf('Failed reading file "%s".', $path));
1375        }
1376
1377        return $contents;
1378    }

1394    /**
1395     * The inline snapshot assertion (D-076): the expected value lives in
1396     * this call's own argument, not a `.snap` file. Passing nothing
1397     * records it under --update-snapshots (the source is rewritten);
1398     * once recorded, the exported value must equal the stored literal.
1399     * The call site is captured here — this method's immediate caller is
1400     * the user's test line.
1401     */
1402    public static function assertMatchesInlineSnapshot(mixed $value, ?string $expected = null): void
1403    {
1404        $frame = array_first(debug_backtrace(DEBUG_BACKTRACE_IGNORE_ARGS, 1));
1405
1406        Snapshots::matchInline($value, $expected, is_string($frame['file'] ?? null) ? $frame['file'] : '', is_int($frame['line'] ?? null) ? $frame['line'] : 0);
