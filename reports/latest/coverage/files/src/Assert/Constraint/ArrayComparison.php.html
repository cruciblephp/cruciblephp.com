23/**
24 * The assertArrays*() family, which is four independent questions rather
25 * than eight assertions — each name picks an answer to each:
26 *
27 * - strict: are values compared with === or ==
28 * - ignoreKeys: "Have*Values" drops the keys and compares the values;
29 *   "Are*" keeps them, so a differing key is a difference
30 * - ignoreOrder: with keys, it is the *key order* that stops mattering
31 *   (the pairing still does); without keys, the values become a multiset
32 *
33 * Observed, not assumed: `assertArraysAreEqual([1,2],[2,1])` fails, even
34 * though both hold the same values, because the pairs 0=>1 and 0=>2
35 * disagree — "ignoring order" never means "ignoring which key holds
36 * what". The multiset readings keep duplicate counts, so [1,1,2] and
37 * [1,2,2] stay unequal.
38 */
39final class ArrayComparison extends Constraint
40{
41    /**
42     * @param array<array-key, mixed> $expected
43     */
44    public function __construct(
45        private readonly array $expected,
46        private readonly bool $strict,
47        private readonly bool $ignoreKeys,
48        private readonly bool $ignoreOrder,
49    ) {}
50
51    #[Override]
52    public function matches(mixed $other): bool
53    {
54        if (!is_array($other)) {
55            return false;
56        }
57
58        if ($this->ignoreKeys) {
59            return $this->ignoreOrder
60                ? $this->sameMultiset(array_values($this->expected), array_values($other), $this->strict)
61                : $this->sameSequence(array_values($this->expected), array_values($other), $this->strict);
62        }
63
64        if (!$this->strict) {
65            // PHP's own == on arrays is already this question: same
66            // pairs, key order irrelevant, values loose.
67            return $this->expected == $other;
68        }
69
70        if (!$this->ignoreOrder) {
71            return $this->expected === $other;
72        }
73
74        $expected = $this->expected;
75        $actual   = $other;
76        ksort($expected);
77        ksort($actual);
78
79        return $expected === $actual;
80    }
81
82    public function toString(): string
83    {
84        $what = $this->strict ? 'identical to' : 'equal to';
85
86        $ignoring = match (true) {
87            $this->ignoreKeys && $this->ignoreOrder => ' while ignoring keys and order',
88            $this->ignoreKeys                       => ' while ignoring keys',
89            $this->ignoreOrder                      => ' while ignoring order',
90            default                                 => '',
91        };
92
93        return 'is ' . $what . ' ' . Exporter::describe($this->expected) . $ignoring;
94    }
95
96    #[Override]
97    protected function failureDescription(mixed $other): string
98    {
99        $what = $this->strict ? 'identical' : 'equal';
100
101        $ignoring = match (true) {
102            $this->ignoreKeys && $this->ignoreOrder => ' while ignoring keys and order',
103            $this->ignoreKeys                       => ' while ignoring keys',
104            $this->ignoreOrder                      => ' while ignoring order',
105            default                                 => '',
106        };
107
108        return 'Failed asserting that two arrays are ' . $what . $ignoring . '.';
109    }
110
111    #[Override]
112    protected function comparison(mixed $other): ?ComparisonFailure
113    {
114        if (!is_array($other)) {
115            return null;
116        }
117
118        $expected = Exporter::export($this->expected);
119        $actual   = Exporter::export($other);
120
121        return new ComparisonFailure($expected, $actual, Differ::diff($expected, $actual));
122    }
123
124    /**
125     * @param list<mixed> $expected
126     * @param list<mixed> $actual
127     */
128    private function sameSequence(array $expected, array $actual, bool $strict): bool
129    {
130        if (count($expected) !== count($actual)) {
131            return false;
132        }
133
134        foreach ($expected as $index => $value) {
135            if ($strict ? $value !== $actual[$index] : $value != $actual[$index]) {
136                return false;
137            }
138        }
139
140        return true;
141    }
142
143    /**
144     * Multiset equality by consuming matches, rather than by sorting:
145     * a mixed-type list has no total order to sort by, and duplicate
146     * counts have to survive the comparison.
147     *
148     * @param list<mixed> $expected
149     * @param list<mixed> $actual
150     */
151    private function sameMultiset(array $expected, array $actual, bool $strict): bool
152    {
153        if (count($expected) !== count($actual)) {
154            return false;
155        }
156
157        foreach ($expected as $value) {
158            $found = false;
159
160            foreach ($actual as $index => $candidate) {
161                if ($strict ? $value === $candidate : $value == $candidate) {
162                    unset($actual[$index]);
163                    $found = true;
164
165                    break;
166                }
167            }
168
169            if (!$found) {
170                return false;
171            }
172        }
173
174        return true;
175    }
176}
