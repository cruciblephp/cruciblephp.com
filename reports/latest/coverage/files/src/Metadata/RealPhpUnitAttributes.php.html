26/**
27 * Real PHPUnit's own #[CoversClass]/#[UsesClass]/#[RequiresPhp]/
28 * #[Group]/#[Depends]/#[DataProvider]/#[TestWith]/
29 * #[ExcludeGlobalVariableFromBackup] and their siblings
30 * (PHPUnit\Framework\Attributes\*) are, like the hook attributes
31 * RealPhpUnitHookAttributes already handles, a genuinely different
32 * class from Crucible's own equivalents whenever aliasing isn't
33 * actually active (PhpUnitCompatibility::isLoaded() false — the
34 * coexistence/migration scenario, phpunit/phpunit installed with no
35 * opt-in). MetadataParser's CrucibleAttribute-instanceof scan never
36 * matches them, so a project's real, un-aliased Covers, Uses,
37 * Requires, Group, Depends, DataProvider, TestWith and
38 * Exclude*Backup usage silently does nothing — verified directly
39 * against MetadataParser first (none of these recognized on a class
40 * carrying the real attributes), a genuine gap, not fidelity to some
41 * PHPUnit limitation.
42 *
43 * Reads ReflectionAttribute::getArguments() directly rather than
44 * instantiating the real attribute class — no need to touch its
45 * actual public-property/getter shape at all, only the literal
46 * arguments written at the attribute's use site, forwarded into
47 * Crucible's own equivalent constructor (verified to match every
48 * candidate's real parameter names/order). filteredArguments() drops
49 * any argument the target constructor doesn't declare — the one
50 * known delta is DataProvider/DataProviderExternal's extra trailing
51 * validateArgumentCount, which Crucible's own equivalents don't
52 * accept.
53 *
54 * Before/PreCondition/PostCondition/After are deliberately excluded
55 * from the glob below: RealPhpUnitHookAttributes already translates
56 * them via their real priority() method (they don't expose it as a
57 * constructor-matching property the way this file's candidates do),
58 * and running both would fire every hook twice.
59 *
60 * Isolated in its own file, excluded from phpstan (phpstan.neon), for
61 * the same reason as RealPhpUnitHookAttributes: compiles against
62 * PHPUnit\Framework\Attributes\* classes deliberately not a
63 * dev-dependency of the engine.
64 */
65final class RealPhpUnitAttributes
66{
67    private const array EXCLUDED = ['Before', 'PreCondition', 'PostCondition', 'After'];
68
69    /**
70     * @param ReflectionClass<object> $class
71     *
72     * @return list<CrucibleAttribute>
73     */
74    public static function forClass(ReflectionClass $class): array
75    {
76        if (PhpUnitCompatibility::isLoaded()) {
77            return [];
78        }
79
80        $translated = [];
81
82        foreach (self::candidates() as $name => $crucibleClass) {
83            foreach ($class->getAttributes('PHPUnit\\Framework\\Attributes\\' . $name) as $attribute) {
84                $translated[] = self::instantiate($attribute, $crucibleClass);
85            }
86        }
87
88        return $translated;
89    }
90
91    /**
92     * @return list<CrucibleAttribute>
93     */
94    public static function forMethod(ReflectionMethod $method): array
95    {
96        if (PhpUnitCompatibility::isLoaded()) {
97            return [];
98        }
99
100        $translated = [];
101
102        foreach (self::candidates() as $name => $crucibleClass) {
103            foreach ($method->getAttributes('PHPUnit\\Framework\\Attributes\\' . $name) as $attribute) {
104                $translated[] = self::instantiate($attribute, $crucibleClass);
105            }
106        }
107
108        return $translated;
109    }
110
111    /**
112     * Same glob-by-basename convention as
113     * PhpUnitCompatibility::aliases() — every file in src/Attributes/
114     * is assumed name-identical to its real PHPUnit counterpart when
115     * one exists; when it doesn't (a handful of Crucible-only
116     * attributes have no PHPUnit equivalent at all), getAttributes()
117     * against the never-used real name simply returns nothing, so no
118     * separate allowlist is needed.
119     *
120     * @return array<string, class-string<CrucibleAttribute>>
121     */
122    private static function candidates(): array
123    {
124        // The attribute files do not change during a run, and this is asked
125        // once per method of every test class: a glob per call was hundreds
126        // of thousands of them on a Pest suite over Laravel's TestCase,
127        // which never finished discovering.
128        static $memo = null;
129
130        if (is_array($memo)) {
131            return $memo;
132        }
133
134        $files = glob(__DIR__ . '/../Attributes/*.php');
135
136        $candidates = [];
137
138        foreach ($files === false ? [] : $files as $file) {
139            $name = basename($file, '.php');
140
141            if (in_array($name, self::EXCLUDED, true)) {
142                continue;
143            }
144
145            /** @var class-string<CrucibleAttribute> $crucibleClass */
146            $crucibleClass = 'LucianoPereira\\Crucible\\Attributes\\' . $name;
147
148            $candidates[$name] = $crucibleClass;
149        }
150
151        return $memo = $candidates;
152    }
153
154    /**
155     * @param ReflectionAttribute<object>     $attribute
156     * @param class-string<CrucibleAttribute> $crucibleClass
157     */
158    private static function instantiate(ReflectionAttribute $attribute, string $crucibleClass): CrucibleAttribute
159    {
160        $arguments = self::filteredArguments($attribute->getArguments(), $crucibleClass);
161
162        return new $crucibleClass(...$arguments);
163    }
164
165    /**
166     * @param array<int|string, mixed>        $arguments
167     * @param class-string<CrucibleAttribute> $crucibleClass
168     *
169     * @return array<int|string, mixed>
170     */
171    private static function filteredArguments(array $arguments, string $crucibleClass): array
172    {
173        $parameters    = (new ReflectionMethod($crucibleClass, '__construct'))->getParameters();
174        $acceptedNames = array_map(static fn($parameter) => $parameter->getName(), $parameters);
175        $acceptedCount = count($parameters);
176
177        $filtered = [];
178
179        foreach ($arguments as $key => $value) {
180            if (is_int($key)) {
181                if ($key < $acceptedCount) {
182                    $filtered[$key] = $value;
183                }
184
185                continue;
186            }
187
188            if (in_array($key, $acceptedNames, true)) {
189                $filtered[$key] = $value;
190            }
191        }
192
193        return $filtered;
194    }
195}
