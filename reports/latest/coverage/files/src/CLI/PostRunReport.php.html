65/**
66 * The post-run reporting cluster split off `RunTestsCommand`
67 * (`improvements.md` #1): everything that runs after the suite
68 * finishes and only prints or computes from what already happened —
69 * run-scoped check tallies, flakiness/quarantine notes, the DeFlaker
70 * "outside the diff" hint, deprecation-budget breaches, coverage
71 * rendering, and the exit-code policy. Kept as one class, not one per
72 * method, because every method here was already independently
73 * testable; the split is about getting them out of a 1,350-line file,
74 * not about finding further boundaries within this cluster.
75 */
76final readonly class PostRunReport
77{

79     * Reading `time()` here bypassed the Clock the rest of the engine

81     * could not be made reproducible however the run was configured.

86    /**
87     * The run's instant as the coverage formats want it. Clamped at zero
88     * because they type the field as non-negative and a clock can be set
89     * before 1970 — the epoch `--reproducible` freezes to is exactly the
90     * boundary.
91     *
92     * @return int<0, max>
93     */
94    private function stamp(): int
95    {
96        return max(0, $this->clock->now()->getTimestamp());
97    }

99    /**
100     * Prints the run-scoped checks summary — a tally kept separate from
101     * the tests — and returns whether any check voted the exit code. A
102     * failing check carries its named reason (never a bare non-zero).
103     */
104    public function printChecks(CheckLog $checkLog): bool
105    {
106        $checks = $checkLog->checks();
107
108        if ($checks === []) {
109            return false;
110        }
111
112        $failing = $checkLog->failing();
113
114        printf('Checks: %d run, %d failed.' . PHP_EOL, count($checks), count($failing));
115
116        foreach ($failing as $check) {
117            if ($check->reason !== null) {
118                print $check->reason . PHP_EOL;
119            }
120        }
121
122        return $failing !== [];
123    }

125    /**
126     * The spec's <source> element is the coverage scope: include
127     * directories (as absolute prefixes) and files, excludes removed
128     * at collection by prefix.
129     *
130     * @param list<non-empty-string> $filter the spec's --coverage-filter
131     *
132     * @return list<non-empty-string>
133     */
134    public function coverageScope(Configuration $configuration, WorkingDirectory $workingDirectory, array $filter = []): array
135    {
136        $scope = [];
137
138        // The spec's --coverage-filter reads "include <dir> in code
139        // coverage reporting", so it adds to the configured scope
140        // rather than replacing it — and with no <source> configured it
141        // is the whole scope, which is the bare "just this directory"
142        // use the option exists for.
143        foreach ($filter as $directory) {
144            $absolute = realpath($workingDirectory->absolute($directory));
145
146            if ($absolute !== false) {
147                $scope[] = is_dir($absolute) ? $absolute . '/' : $absolute;
148            }
149        }
150
151        foreach ($configuration->source->includeDirectories as $directory) {
152            $absolute = realpath($workingDirectory->absolute($directory));
153
154            if ($absolute !== false) {
155                $scope[] = $absolute . '/';
156            }
157        }
158
159        foreach ($configuration->source->includeFiles as $file) {
160            $absolute = realpath($workingDirectory->absolute($file));
161
162            if ($absolute !== false) {
163                $scope[] = $absolute;
164            }
165        }
166
167        return $scope;
168    }

170    /**
171     * Folds every artifact matching a glob into the collection and
172     * removes it, returning how many were merged. Unreadable is
173     * skipped rather than fatal, and still removed: an artifact nobody
174     * can read is not evidence, and leaving it would have every later
175     * run try again.
176     *
177     * @return int how many artifacts contributed
178     */
179    private function mergeArtifacts(CoverageData $data, string $pattern): int
180    {
181        $artifacts = glob($pattern);
182        $merged    = 0;
183
184        foreach ($artifacts === false ? [] : $artifacts as $artifact) {
185            $contents = file_get_contents($artifact);
186
187            if ($contents !== false) {
188                $data->merge(CoverageData::fromJson($contents));
189                $merged++;
190            }
191
192            unlink($artifact);
193        }
194
195        return $merged;
196    }

198    /**
199     * After the run: merge worker and external artifacts into the
200     * in-process collection, render the requested reports, and refresh
201     * the observed-edges map the impact graph reads (D-041).
202     *
203     * @param non-empty-string $cacheDirectory
204     * @param list<array{id: string, status: string, time: float, size?: string}> $tests
205     *                                                                                   how each test ended, for the XML index; the other writers have nowhere to put it
206     *
207     * @return CoverageData the merged collection, so the caller can judge it against a floor
208     */
209    public function reportCoverage(CoverageCollector $collector, CliOptions $options, WorkingDirectory $workingDirectory, string $cacheDirectory, array $tests = []): CoverageData
210    {
211        $data  = $collector->data();
212        $cache = $workingDirectory->absolute($cacheDirectory);
213
214        // Two directories because the two lifetimes differ. A worker's
215        // artifact belongs to this run and is swept as stale before the
216        // run starts; an external one is written deliberately BEFORE a
217        // run, by another process under its own driver, and that sweep
218        // would delete it every time. Both are consumed exactly once.
219        $this->mergeArtifacts($data, $cache . '/coverage.tmp/*.json');
220        $external = $this->mergeArtifacts($data, $cache . '/coverage.external/*.json');
221
222        if ($external > 0) {
223            // Folding another process's work in silently would leave the
224            // total describing more than the run it prints under.
225            printf('Merged %d coverage artifact(s) from processes outside this run.' . PHP_EOL, $external);
226        }
227
228        $textReport = new TextReport($options->onlySummaryForCoverageText, $options->showUncoveredForCoverageText);
229
230        if ($options->coverage || $options->coverageBranch) {
231            print PHP_EOL . $textReport->render($data, $workingDirectory->path, $collector->driverName());
232        }
233
234        // --coverage-text is the same report to a chosen sink: bare
235        // means the terminal, which is the spec's documented default.
236        if ($options->coverageText !== null) {
237            $rendered = $textReport->render($data, $workingDirectory->path, $collector->driverName());
238
239            if ($options->coverageText === '') {
240                print PHP_EOL . $rendered;
241            } else {
242                file_put_contents($workingDirectory->absolute($options->coverageText), $rendered);
243                printf('Text coverage written to %s.' . PHP_EOL, $options->coverageText);
244            }
245        }
246
247        if ($options->coverageOpenClover !== null) {
248            file_put_contents(
249                $workingDirectory->absolute($options->coverageOpenClover),
250                (new CloverWriter(openClover: true))->write($data, $this->stamp()),
251            );
252
253            printf('OpenClover coverage written to %s.' . PHP_EOL, $options->coverageOpenClover);
254        }
255
256        if ($options->coveragePhp !== null) {
257            file_put_contents(
258                $workingDirectory->absolute($options->coveragePhp),
259                (new PhpWriter())->write($data, $collector->driverName(), $options->includeGitInformation ? GitInformation::read($workingDirectory) : null, $this->clock->now()),
260            );
261
262            printf('Serialized coverage written to %s.' . PHP_EOL, $options->coveragePhp);
263        }
264
265        if ($options->coverageClover !== null) {
266            file_put_contents(
267                $workingDirectory->absolute($options->coverageClover),
268                (new CloverWriter())->write($data, $this->stamp()),
269            );
270
271            printf('Clover coverage written to %s.' . PHP_EOL, $options->coverageClover);
272        }
273
274        if ($options->coverageCobertura !== null) {
275            file_put_contents(
276                $workingDirectory->absolute($options->coverageCobertura),
277                (new CoberturaWriter())->write($data, $workingDirectory->path, $this->stamp()),
278            );
279
280            printf('Cobertura coverage written to %s.' . PHP_EOL, $options->coverageCobertura);
281        }
282
283        if ($options->coverageCrap4j !== null) {
284            file_put_contents(
285                $workingDirectory->absolute($options->coverageCrap4j),
286                (new Crap4jWriter())->write($data, $workingDirectory->path),
287            );
288
289            printf('Crap4J coverage written to %s.' . PHP_EOL, $options->coverageCrap4j);
290        }
291
292        if ($options->coverageXml !== null) {
293            (new XmlReport($options->excludeSourceFromXmlCoverage))->write(
294                $data,
295                $workingDirectory->path,
296                $workingDirectory->absolute($options->coverageXml),
297                $collector->driverName(),
298                $tests,
299            );
300
301            printf('XML coverage written to %s/index.xml.' . PHP_EOL, $options->coverageXml);
302        }
303
304        if ($options->coverageHtml !== null) {
305            // Annotating every source file is the longest silence a run
306            // has after the suite itself, and it happens once the tally
307            // has already printed — so it reads as a hang.
308            (new Spinner('Rendering the HTML coverage report'))->spin(
309                function () use ($options, $data, $workingDirectory, $collector): void {
310                    (new HtmlReport($options->withoutClassView, $options->withoutFileView))->write(
311                        $data,
312                        $workingDirectory->path,
313                        $workingDirectory->absolute($options->coverageHtml),
314                        $collector->driverName(),
315                    );
316                },
317            );
318
319            printf('HTML coverage written to %s/index.html.' . PHP_EOL, $options->coverageHtml);
320        }
321
322        CoverageMap::save(
323            $workingDirectory->absolute($cacheDirectory) . '/coverage-map.json',
324            CoverageMap::fromData($data, $workingDirectory->path),
325        );
326
327        // The per-test line map (D-047): the mutation query's data,
328        // refreshed beside the impact map on every coverage run.
329        TestLineMap::fromData($data, $workingDirectory->path)
330            ->save($workingDirectory->absolute($cacheDirectory) . '/coverage-lines.json');
331
332        return $data;
333    }

335    /**
336     * The coverage floor (`--min`): a breach message when line coverage
337     * came in under the percentage asked for, null when it did not.
338     *
339     * The number judged is the one the text report *prints* — rounded to
340     * two places — not the raw quotient. A gate that failed a run whose
341     * report reads `Total: 90.00%` against `--min=90` would be arguing
342     * with the user about a third decimal they cannot see.
343     *
344     * An empty scope is a breach rather than a pass. Zero executable
345     * lines means the run measured nothing, and a gate that reports
346     * success when it measured nothing is the failure mode the gate
347     * exists to prevent.
348     *
349     * @return ?non-empty-string
350     */
351    public function coverageFloor(CoverageData $data, ?float $minimum): ?string
352    {
353        if ($minimum === null) {
354            return null;
355        }
356
357        // A floor of zero demands nothing, so nothing can fail it. The
358        // empty-scope rule below is about an ask that CANNOT be met;
359        // this one is met by anything, including nothing measured.
360        if ($minimum <= 0.0) {
361            return null;
362        }
363
364        $totals = $data->totals();
365
366        if ($totals['executable'] === 0) {
367            return 'Coverage floor cannot be judged: the coverage scope contains no executable lines.';
368        }
369
370        $percent = round(100 * $totals['covered'] / $totals['executable'], 2);
371
372        if ($percent >= $minimum) {
373            return null;
374        }
375
376        return sprintf(
377            'Coverage below the required minimum: %.2f%% of %d executable lines covered, at least %s%% required.',
378            $percent,
379            $totals['executable'],
380            $minimum,
381        );
382    }

384    /**
385     * The G4 post-run notes: name what was flaky, what quarantine
386     * absorbed, and which quarantined tests look ready for release.
387     */
388    public function printFlakiness(FlakinessLog $flakiness): void
389    {
390        if ($flakiness->flakyCount() > 0) {
391            printf(PHP_EOL . 'Flaky: %d test(s) passed only on retry:' . PHP_EOL, $flakiness->flakyCount());
392
393            foreach ($flakiness->flakyTests() as $id) {
394                print '  - ' . $id . PHP_EOL;
395            }
396        }
397
398        if ($flakiness->quarantinedFailureCount() > 0) {
399            printf(PHP_EOL . 'Quarantined: %d failing test(s) did not affect the result:' . PHP_EOL, $flakiness->quarantinedFailureCount());
400
401            foreach ($flakiness->quarantinedFailures() as $id) {
402                print '  - ' . $id . PHP_EOL;
403            }
404        }
405
406        if ($flakiness->quarantinedPasses() !== []) {
407            print PHP_EOL . 'Quarantined but passing — candidates for release:' . PHP_EOL;
408
409            foreach ($flakiness->quarantinedPasses() as $id) {
410                print '  - ' . $id . PHP_EOL;
411            }
412        }
413    }

415    /**
416     * The DeFlaker hint (D-048): a fresh failure that executed none
417     * of the changed code was probably not caused by the change.
418     * Advisory only — it never touches outcomes or the exit code, and
419     * every missing ingredient (no failures, no line map from a prior
420     * coverage run, no git answer, no comparable diff) is silence.
421     *
422     * @param non-empty-string $cacheDirectory
423     */
424    public function printFailuresOutsideDiff(
425        FlakinessLog $flakiness,
426        ResultCache $cache,
427        Configuration $configuration,
428        CliOptions $options,
429        WorkingDirectory $workingDirectory,
430        string $cacheDirectory,
431    ): void {
432        if ($flakiness->failures() === []) {
433            return;
434        }
435
436        // Only fresh failures qualify (DeFlaker's "new failure"): a
437        // test that already failed last run was broken before this
438        // diff existed, and the diff can be neither blamed nor cleared.
439        $fresh = [];
440
441        foreach ($flakiness->failures() as $id) {
442            $previous = $cache->outcomes($id)[1] ?? null;
443
444            if ($previous !== Outcome::Failed && $previous !== Outcome::Errored) {
445                $fresh[] = $id;
446            }
447        }
448
449        if ($fresh === []) {
450            return;
451        }
452
453        $map = TestLineMap::load($workingDirectory->absolute($cacheDirectory) . '/coverage-lines.json');
454
455        if ($map->tests === []) {
456            return;
457        }
458
459        $reference = $options->changed ?? 'HEAD';
460        $changed   = ChangedFiles::fromGit($workingDirectory, $reference);
461
462        if (is_string($changed)) {
463            return;
464        }
465
466        // Deleted PHP is code the map may still reference but no run
467        // can observe; changed PHP outside the project the same. Both
468        // make "executed no changed code" unanswerable.
469        foreach ($changed->deleted as $name) {
470            if (str_ends_with($name, '.php')) {
471                return;
472            }
473        }
474
475        $prefix     = $workingDirectory->path . '/';
476        $changedPhp = [];
477
478        foreach ($changed->files as $absolute) {
479            if (!str_ends_with($absolute, '.php')) {
480                continue;
481            }
482
483            if (!str_starts_with($absolute, $prefix)) {
484                return;
485            }
486
487            $relative = substr($absolute, strlen($prefix));
488
489            if ($relative !== '') {
490                $changedPhp[] = $relative;
491            }
492        }
493
494        $scope = [];
495
496        foreach ($this->coverageScope($configuration, $workingDirectory) as $absoluteScope) {
497            if (str_starts_with($absoluteScope, $prefix)) {
498                $relative = substr($absoluteScope, strlen($prefix));
499
500                if ($relative !== '') {
501                    $scope[] = $relative;
502                }
503            }
504        }
505
506        $suspects = (new FailureOutsideDiff($map))->suspects($fresh, $changedPhp, $scope);
507
508        if ($suspects === []) {
509            return;
510        }
511
512        printf(
513            PHP_EOL . 'Outside the diff: %d fresh failure(s) executed none of the changed files (against %s) — likely flaky or environmental, not this change. `crucible flakes` classifies:' . PHP_EOL,
514            count($suspects),
515            $reference,
516        );
517
518        foreach ($suspects as $id) {
519            print '  - ' . $id . PHP_EOL;
520        }
521    }

524    /**
525     * The spec's --display-* family: what the tally counted, listed.
526     *
527     * The run summary says "Deprecations: 3"; these say which three, where
528     * they were triggered, and — for a deprecation — whether it came from
529     * the project's own code, from a dependency it called, or from one
530     * dependency calling another.
531     */
532    public function printIssueDisplays(CliOptions $options, IssueLog $issueLog, OutcomeLog $outcomes): void
533    {
534        $kinds = array_filter([
535            IssueKind::Deprecation->value => $options->displayDeprecations,
536            IssueKind::Notice->value      => $options->displayNotices,
537            IssueKind::Warning->value     => $options->displayWarnings,
538        ]);
539
540        foreach (array_keys($kinds) as $kind) {
541            $matching = array_values(array_filter(
542                $issueLog->issues(),
543                static fn(Issue $issue): bool => $issue->kind->value === $kind,
544            ));
545
546            if ($matching === []) {
547                continue;
548            }
549
550            printf(PHP_EOL . '%d %s(s):' . PHP_EOL, count($matching), $kind);
551
552            foreach ($matching as $issue) {
553                printf(
554                    '  %s:%d  %s%s' . PHP_EOL,
555                    $issue->file,
556                    $issue->line,
557                    $issue->message,
558                    $issue->scope instanceof DeprecationScope ? ' [' . $issue->scope->value . ']' : '',
559                );
560            }
561        }
562
563        $listings = array_filter([
564            'errored'    => $options->displayErrors,
565            'skipped'    => $options->displaySkipped,
566            'incomplete' => $options->displayIncomplete,
567        ]);
568
569        foreach (array_keys($listings) as $name) {
570            $outcome = Outcome::from($name === 'errored' ? 'error' : ($name === 'skipped' ? 'skip' : 'incomplete'));
571            $tests   = $outcomes->of($outcome);
572
573            if ($tests === []) {
574                continue;
575            }
576
577            printf(PHP_EOL . '%d %s test(s):' . PHP_EOL, count($tests), $name);
578
579            foreach ($tests as $test) {
580                printf('  %s%s' . PHP_EOL, $test['id'], $test['reason'] !== null ? '  ' . $test['reason'] : '');
581            }
582        }
583    }

585    /**
586     * The per-scope deprecation budgets (DESIGN.md D-026): typed form
587     * of the Symfony bridge's max[self|direct|indirect] grammar.
588     *
589     * @return list<non-empty-string>
590     */
591    public function thresholdBreaches(Configuration $configuration, IssueLog $issueLog): array
592    {
593        $scopes   = $issueLog->deprecationsByScope();
594        $breaches = [];
595
596        $budgets = [
597            'self'     => $configuration->maxSelfDeprecations,
598            'direct'   => $configuration->maxDirectDeprecations,
599            'indirect' => $configuration->maxIndirectDeprecations,
600        ];
601
602        foreach ($budgets as $scope => $budget) {
603            $count = $scopes[$scope] ?? 0;
604
605            if ($budget !== null && $count > $budget) {
606                $breaches[] = sprintf(
607                    'Deprecation budget exceeded for scope "%s": %d observed, at most %d allowed.',
608                    $scope,
609                    $count,
610                    $budget,
611                );
612            }
613        }
614
615        return $breaches;
616    }

618    public function exitCode(RunSummary $summary, Configuration $configuration, CliOptions $options, FlakinessLog $flakiness = new FlakinessLog(), IssueLog $issueLog = new IssueLog()): int
619    {
620        // Quarantined failures are reported but never judged (G4) —
621        // subtract them before the verdict.
622        $errored = $summary->errored - $flakiness->quarantinedErrorCount();
623        $failed  = $summary->failed - ($flakiness->quarantinedFailureCount() - $flakiness->quarantinedErrorCount());
624
625        if ($errored > 0) {
626            return 2;
627        }
628
629        $failOnRisky       = $options->failOnRisky ?? $configuration->failOnRisky;
630        $failOnSkipped     = $options->failOnSkipped ?? $configuration->failOnSkipped;
631        $failOnIncomplete  = $options->failOnIncomplete ?? $configuration->failOnIncomplete;
632        $failOnDeprecation = $options->failOnDeprecation ?? $configuration->failOnDeprecation;
633        $failOnNotice      = $options->failOnNotice ?? $configuration->failOnNotice;
634        $failOnWarning     = $options->failOnWarning ?? $configuration->failOnWarning;
635
636        // The scoped deprecation policies read the attribution the
637        // collector already made (DeprecationScope): a trigger in the
638        // project's own code is self, one in a dependency is direct when
639        // project code called it and indirect otherwise. Same counts the
640        // D-026 budgets use, judged as a yes/no rather than a ceiling.
641        $scopes = $issueLog->deprecationsByScope();
642
643        $failOnScope = static fn(string $scope, ?bool $cli, bool $configured): bool => ($cli ?? $configured)
644            && ($scopes[$scope] ?? 0) > 0;
645
646        $failedByPolicy = ($failOnRisky && $summary->risky > 0)
647            || ($failOnSkipped && $summary->skipped > 0)
648            || ($failOnIncomplete && $summary->incomplete > 0)
649            || ($failOnDeprecation && $summary->deprecations > 0)
650            || ($failOnNotice && $summary->notices > 0)
651            || ($failOnWarning && $summary->warnings > 0)
652            || $failOnScope('self', $options->failOnSelfDeprecation, $configuration->failOnSelfDeprecation)
653            || $failOnScope('direct', $options->failOnDirectDeprecation, $configuration->failOnDirectDeprecation)
654            || $failOnScope('indirect', $options->failOnIndirectDeprecation, $configuration->failOnIndirectDeprecation)
655            || (($options->failOnEmptyTestSuite ?? $configuration->failOnEmptyTestSuite) && $summary->total() === 0)
656            || ($options->failOnFlaky === true && $flakiness->flakyCount() > 0);
657
658        if ($failed > 0 || $failedByPolicy) {
659            return 1;
660        }
661
662        return 0;
663    }
