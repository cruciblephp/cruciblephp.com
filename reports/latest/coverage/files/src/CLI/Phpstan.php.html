33/**
34 * Running PHPStan, once, for everything that does (D-137): `lint-inline`
35 * (D-052), type tests (D-130), and the tests and probes that prove the
36 * extension. Each used to locate the binary, decide the extension
37 * include, spawn the process and read the JSON on its own — four copies,
38 * one of which read stdout then stderr from two pipes, the order that
39 * blocks forever once a child fills the pipe nobody is reading.
40 *
41 * Output goes to files, not pipes; the report is read defensively, once.
42 */
43final readonly class Phpstan
44{
45    private const array EXTENSION = ['vendor/cruciblephp/crucible/phpstan/extension.neon', 'phpstan/extension.neon'];
46
47    /**
48     * Crucible's extension.neon as the project sees it: installed under
49     * vendor/, or this checkout's own. Null when neither exists.
50     *
51     * @return ?non-empty-string relative to the working directory
52     */
53    public static function extension(WorkingDirectory $workingDirectory): ?string
54    {
55        return array_find(self::EXTENSION, static fn(string $candidate): bool => is_file($workingDirectory->path . '/' . $candidate));
56    }
57
58    /**
59     * The project's own PHPStan configuration, by PHPStan's own file names.
60     *
61     * @return ?non-empty-string relative to the working directory
62     */
63    public static function projectConfiguration(WorkingDirectory $workingDirectory): ?string
64    {
65        return array_find(['phpstan.neon', 'phpstan.neon.dist', 'phpstan.dist.neon'], static fn(string $candidate): bool => is_file($workingDirectory->path . '/' . $candidate));
66    }
67
68    /**
69     * The include a written configuration needs for Crucible's extension:
70     * none when phpstan/extension-installer already loads it (D-127) —
71     * including it twice makes PHPStan refuse to run — or when it cannot
72     * be found.
73     *
74     * @return list<non-empty-string> absolute: the extension, or nothing
75     */
76    public static function extensionIncludes(WorkingDirectory $workingDirectory): array
77    {
78        $extension = self::extension($workingDirectory);
79
80        return $extension === null || PhpstanNeon::installerLoadsExtension($workingDirectory)
81            ? []
82            : [$workingDirectory->path . '/' . $extension];
83    }
84
85    /**
86     * One analysis. The report, or why there is none: PHPStan missing,
87     * not started, or writing no JSON (its stderr, trimmed, is the reason).
88     *
89     * @param non-empty-string       $binary
90     * @param list<non-empty-string> $arguments after `analyse`: --configuration, paths, …
91     *
92     * @return PhpstanReport|non-empty-string
93     */
94    public static function analyse(string $binary, array $arguments, WorkingDirectory $workingDirectory): PhpstanReport|string
95    {
96        if (!is_file($binary)) {
97            return 'PHPStan is not installed at ' . $binary . ': composer require --dev phpstan/phpstan';
98        }
99
100        $out = tempnam(sys_get_temp_dir(), 'crucible-phpstan-');
101        $err = tempnam(sys_get_temp_dir(), 'crucible-phpstan-err-');
102
103        if ($out === false || $err === false) {
104            return 'Temporary files for the PHPStan report could not be created.';
105        }
106
107        $process = @proc_open(
108            [PHP_BINARY, $binary, 'analyse', '--error-format=json', '--no-progress', '--no-interaction', ...$arguments],
109            [['file', '/dev/null', 'r'], ['file', $out, 'w'], ['file', $err, 'w']],
110            $pipes,
111            $workingDirectory->path,
112        );
113
114        if (!is_resource($process)) {
115            @unlink($out);
116            @unlink($err);
117
118            return 'PHPStan could not be started.';
119        }
120
121        proc_close($process);
122
123        $decoded = json_decode((string) file_get_contents($out), true);
124        $stderr  = trim((string) file_get_contents($err));
125
126        @unlink($out);
127        @unlink($err);
128
129        if (!is_array($decoded) || !is_array($decoded['files'] ?? null)) {
130            return 'PHPStan wrote no report' . ($stderr !== '' ? ': ' . substr($stderr, 0, 500) : '.');
131        }
132
133        $messages = [];
134
135        foreach ($decoded['files'] as $file => $entry) {
136            $entries = is_array($entry) && is_array($entry['messages'] ?? null) ? $entry['messages'] : [];
137
138            foreach ($entries as $message) {
139                if (!is_array($message) || !is_string($message['message'] ?? null) || $message['message'] === '') {
140                    continue;
141                }
142
143                $identifier = $message['identifier'] ?? null;
144
145                $messages[] = new PhpstanMessage(
146                    (string) $file,
147                    is_int($message['line'] ?? null) && $message['line'] >= 0 ? $message['line'] : 0,
148                    is_string($identifier) && $identifier !== '' ? $identifier : null,
149                    $message['message'],
150                );
151            }
152        }
153
154        $general = [];
155
156        foreach (is_array($decoded['errors'] ?? null) ? $decoded['errors'] : [] as $error) {
157            if (is_string($error) && $error !== '') {
158                $general[] = $error;
159            }
160        }
161
162        return new PhpstanReport($messages, $general);
163    }
