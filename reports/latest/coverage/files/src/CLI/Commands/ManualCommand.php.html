33/**
34 * `crucible manual`: a fullscreen, Norton-Guides-style reader over
35 * Crucible's own documentation. Local `[text](other.md)` links are
36 * followed with a back stack (number keys open, ⌫ goes back, q quits);
37 * a non-interactive terminal (CI, piped output) gets the rendered text
38 * once instead.
39 *
40 * ⚠ Rooted in the package, not in the working directory. Installed in
41 * someone's project, reading `./README.md` opened THEIR repository's
42 * readme — the one document guaranteed not to be about this command.
43 */
44final class ManualCommand
45{
46    /** Columns the PDF lays the text out at — wider than a terminal, narrower than the page. */
47    private const int PDF_COLUMNS = 96;
48
49    public function execute(?string $out = null): int
50    {
51        // ⚠ The package's own docs, not the working directory's. Installed
52        // in someone's project, reading `./README.md` opened THEIR
53        // repository's readme — the one document guaranteed not to be
54        // about this command.
55        $path = dirname(__DIR__, 3) . '/MANUAL.md';
56
57        if (! is_file($path)) {
58            print 'Crucible\'s MANUAL.md was not found beside the installed package.' . PHP_EOL;
59
60            return 1;
61        }
62
63        if ($out !== null) {
64            return $this->writePdf($path, $out);
65        }
66
67        $terminal = Runtime::terminal();
68
69        if (! $terminal->supportsInteractivity()) {
70            $width = max(20, $terminal->columns());
71            $terminal->write(Markdown::render((string) file_get_contents($path), $width)->plainText() . PHP_EOL);
72
73            return 0;
74        }
75
76        /** @var list<non-empty-string> $stack */
77        $stack   = [];
78        $current = $path;
79
80        while (true) {
81            $width    = max(20, $terminal->columns() - 4);
82            $document = Markdown::render((string) file_get_contents($current), $width);
83            $viewer   = new Pager($document->lines, basename($current), $document->links);
84
85            Runtime::run($viewer);
86
87            if ($viewer->quit) {
88                return 0;
89            }
90
91            if ($viewer->goBack) {
92                if ($stack !== []) {
93                    $current = array_pop($stack);
94                }
95
96                continue;
97            }
98
99            if ($viewer->followLink === null || ! isset($document->links[$viewer->followLink])) {
100                return 0;
101            }
102
103            $target = $this->resolveDocPath($document->links[$viewer->followLink], dirname($current));
104
105            if ($target !== null) {
106                $stack[] = $current;
107                $current = $target;
108            }
109        }
110    }
111
112    /**
113     * The manual as a monospaced PDF, laid out exactly as the pager
114     * lays it out — the same text, on paper.
115     */
116    private function writePdf(string $source, string $out): int
117    {
118        $text = Markdown::render((string) file_get_contents($source), self::PDF_COLUMNS)->plainText();
119
120        if (file_put_contents($out, (new TextPdf())->render($text, 'Crucible PHP — the manual', Version::AUTHOR)) === false) {
121            printf('Could not write %s.' . PHP_EOL, $out);
122
123            return 1;
124        }
125
126        printf('Wrote the manual to %s.' . PHP_EOL, $out);
127
128        return 0;
129    }
130
131    /**
132     * Resolve a local documentation link relative to the current
133     * document's directory. External (scheme://) links and missing
134     * files return null.
135     *
136     * @return non-empty-string|null
137     */
138    private function resolveDocPath(string $target, string $baseDir): ?string
139    {
140        if (preg_match('#^[a-z]+://#i', $target) === 1) {
141            return null;
142        }
143
144        $candidate = str_starts_with($target, '/') ? $target : $baseDir . '/' . $target;
145        $real      = realpath($candidate);
146
147        return $real !== false && is_file($real) ? $real : null;
148    }
