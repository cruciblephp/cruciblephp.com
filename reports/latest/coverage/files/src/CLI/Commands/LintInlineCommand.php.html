42/**
43 * `crucible lint-inline` (D-052): every `@crucible` doctest re-homed
44 * into a shadow file in the cache directory, analyzed by the
45 * real PHPStan through the extension, every reported line mapped
46 * back to its origin. Exit 1 when the doctests have findings.
47 */
48final class LintInlineCommand
49{
50    public function execute(CliOptions $options, WorkingDirectory $workingDirectory): int
51    {
52        try {
53            $loaded = (new Loader())->load($workingDirectory, $options->configuration);
54        } catch (Exception $e) {
55            print $e->getMessage() . PHP_EOL;
56
57            return 1;
58        }
59
60        if (!is_file($workingDirectory->path . '/vendor/bin/phpstan')) {
61            // An opt-in feature owes the user the way in (D-041's rule).
62            print 'lint-inline needs PHPStan:' . PHP_EOL;
63            print '  composer require --dev phpstan/phpstan' . PHP_EOL;
64
65            return 1;
66        }
67
68        $cacheDirectory = $options->cacheDirectory ?? $loaded->configuration->cacheDirectory;
69        $shadowRoot     = $workingDirectory->absolute($cacheDirectory) . '/lint-inline';
70
71        if (!is_dir($shadowRoot)) {
72            mkdir($shadowRoot, 0o777, true);
73        }
74
75        $stale = glob($shadowRoot . '/*.php');
76
77        foreach ($stale === false ? [] : $stale as $leftover) {
78            unlink($leftover);
79        }
80
81        /** @var array<string, array{origin: non-empty-string, lines: array<int, int>}> $shadows shadow path => mapping */
82        $shadows = [];
83        $count   = 0;
84
85        foreach ((new TestDiscoverer())->inlineFiles($loaded->configuration->source, $workingDirectory) as $file) {
86            $source = file_get_contents($file);
87
88            if ($source === false) {
89                continue;
90            }
91
92            $shadow = DoctestShadow::fromSource($source);
93
94            if (!$shadow instanceof DoctestShadow) {
95                continue;
96            }
97
98            $relative = str_starts_with($file, $workingDirectory->path . '/')
99                ? substr($file, strlen($workingDirectory->path) + 1)
100                : $file;
101            $shadowFile = $shadowRoot . '/' . str_replace('/', '__', $relative);
102
103            file_put_contents($shadowFile, $shadow->content);
104
105            $shadows[$shadowFile] = ['origin' => $relative, 'lines' => $shadow->lines];
106            $count += $shadow->count;
107        }
108
109        if ($shadows === []) {
110            print 'No @crucible doctests found in the configured source.' . PHP_EOL;
111
112            return 0;
113        }
114
115        // The origin sources are scanned, not analyzed: the doctests
116        // reference their own file's symbols, and scanning keeps them
117        // resolvable even where no autoloader covers them.
118        $scanDirectories = array_values(array_filter(
119            array_map($workingDirectory->absolute(...), $loaded->configuration->source->includeDirectories),
120            is_dir(...),
121        ));
122        $scanFiles = array_values(array_filter(
123            array_map($workingDirectory->absolute(...), $loaded->configuration->source->includeFiles),
124            is_file(...),
125        ));
126
127        $neon = PhpstanNeon::analysis(Phpstan::extensionIncludes($workingDirectory), [
128            'level'           => 'max',
129            'paths'           => [$shadowRoot],
130            'tmpDir'          => $shadowRoot . '/.cache',
131            'scanDirectories' => $scanDirectories,
132            'scanFiles'       => $scanFiles,
133        ]);
134
135        file_put_contents($shadowRoot . '/phpstan.neon', $neon);
136
137        $arguments = ['--configuration=' . $shadowRoot . '/phpstan.neon'];
138
139        if (is_file($workingDirectory->path . '/vendor/autoload.php')) {
140            $arguments[] = '--autoload-file=' . $workingDirectory->path . '/vendor/autoload.php';
141        }
142
143        $report = Phpstan::analyse($workingDirectory->path . '/vendor/bin/phpstan', $arguments, $workingDirectory);
144
145        if (is_string($report)) {
146            print $report . PHP_EOL;
147
148            return 1;
149        }
150
151        // Errors PHPStan places in no file — a broken include, an
152        // unreadable path — are findings too: a doctest run that did not
153        // analyse is not clean.
154        foreach ($report->general as $error) {
155            print $error . PHP_EOL;
156        }
157
158        $findings = count($report->general);
159
160        foreach ($report->messages as $message) {
161            $findings++;
162
163            $mapping = $shadows[$message->file] ?? null;
164
165            if ($mapping !== null) {
166                printf('%s:%d %s' . PHP_EOL, $mapping['origin'], $mapping['lines'][$message->line] ?? $message->line, $message->message);
167            } else {
168                printf('%s:%d %s' . PHP_EOL, $message->file, $message->line, $message->message);
169            }
170        }
171
172        if ($findings > 0) {
173            printf(PHP_EOL . '%d finding(s) in %d doctest(s) across %d file(s).' . PHP_EOL, $findings, $count, count($shadows));
174
175            return 1;
176        }
177
178        printf('Doctests analyse clean: %d expression(s) across %d file(s).' . PHP_EOL, $count, count($shadows));
179
180        return 0;
181    }
182}
