38/**
39 * `crucible phpstan-init` (D-049): wire Crucible's PHPStan extension
40 * into the project's analysis config — create a phpstan.neon
41 * derived from the crucible configuration, or conservatively insert
42 * the one includes line into the existing file. Anything
43 * structurally surprising prints the line instead of guessing.
44 */
45final class PhpstanInitCommand
46{
47    public function execute(CliOptions $options, WorkingDirectory $workingDirectory): int
48    {
49        try {
50            $loaded = (new Loader())->load($workingDirectory, $options->configuration);
51        } catch (Exception $e) {
52            print $e->getMessage() . PHP_EOL;
53            print 'phpstan-init derives the analysis paths from crucible.php — create one with crucible --init first.' . PHP_EOL;
54
55            return 1;
56        }
57
58        $include = Phpstan::extension($workingDirectory);
59
60        if ($include === null) {
61            print 'Cannot find Crucible\'s extension.neon (expected vendor/cruciblephp/crucible/phpstan/extension.neon).' . PHP_EOL;
62
63            return 1;
64        }
65        $target = Phpstan::projectConfiguration($workingDirectory);
66
67        // Crucible declares its extension to phpstan/extension-installer
68        // (D-127), which then loads it for every analysis. An include
69        // written as well loads it twice, and PHPStan refuses to run.
70        $installer = PhpstanNeon::installerLoadsExtension($workingDirectory);
71
72        if ($installer) {
73            $existing = $target === null ? false : file_get_contents($workingDirectory->path . '/' . $target);
74
75            if (is_string($existing) && PhpstanNeon::alreadyWired($existing, $include)) {
76                printf('phpstan/extension-installer already loads the Crucible extension, and %s includes it too.' . PHP_EOL, $target);
77                print 'PHPStan refuses a file included twice. Remove this line from ' . $target . ':' . PHP_EOL;
78                print PHP_EOL . '    - ' . $include . PHP_EOL;
79
80                return 1;
81            }
82
83            if ($target !== null) {
84                print 'phpstan/extension-installer already loads the Crucible extension — nothing to add.' . PHP_EOL;
85
86                return $this->wireDialect($loaded->configuration, $workingDirectory, $target, $include);
87            }
88        }
89
90        if ($target === null) {
91            $paths = $loaded->configuration->source->includeDirectories;
92            foreach ($loaded->configuration->source->includeFiles as $file) {
93                $paths[] = $file;
94            }
95            foreach ($loaded->configuration->testSuites as $suite) {
96                foreach ([...$suite->directories, ...$suite->files] as $path) {
97                    $paths[] = $path;
98                }
99            }
100            $paths  = array_values(array_unique($paths));
101            $target = 'phpstan.neon';
102            file_put_contents($workingDirectory->path . '/' . $target, PhpstanNeon::create($installer ? null : $include, $paths));
103            printf(
104                'Created phpstan.neon — %s, paths from %s.' . PHP_EOL,
105                $installer ? 'the installer wires the extension' : 'extension wired',
106                $loaded->path,
107            );
108        } else {
109            $existing = file_get_contents($workingDirectory->path . '/' . $target);
110
111            if ($existing === false) {
112                printf('Cannot read %s.' . PHP_EOL, $target);
113
114                return 1;
115            }
116
117            if (PhpstanNeon::alreadyWired($existing, $include)) {
118                printf('%s already includes the Crucible extension — nothing to do.' . PHP_EOL, $target);
119
120                return $this->wireDialect($loaded->configuration, $workingDirectory, $target, $include);
121            }
122
123            $wired = PhpstanNeon::wire($existing, $include);
124
125            if ($wired === null) {
126                printf('%s has a shape this command will not edit. Add the include yourself:' . PHP_EOL, $target);
127                print PHP_EOL . 'includes:' . PHP_EOL . '    - ' . $include . PHP_EOL;
128
129                return 1;
130            }
131
132            file_put_contents($workingDirectory->path . '/' . $target, $wired);
133
134            printf('Updated %s — added the Crucible extension include.' . PHP_EOL, $target);
135        }
136
137        // An opt-in feature owes the user the way in (D-041's rule).
138        if (!is_file($workingDirectory->path . '/vendor/bin/phpstan')) {
139            print PHP_EOL . 'PHPStan itself is not installed yet:' . PHP_EOL;
140            print '  composer require --dev phpstan/phpstan' . PHP_EOL;
141        }
142
143        return $this->wireDialect($loaded->configuration, $workingDirectory, $target, $include);
144    }
145
146    /**
147     * check(), property() and table() are scanned by crucible-dialect.neon,
148     * not by the extension every project gets (D-128). A suite that writes
149     * the dialect — it holds *.crucible.php files — needs that include
150     * too; the installer never adds it.
151     *
152     * @param non-empty-string $target
153     * @param non-empty-string $include the extension.neon include in use
154     */
155    private function wireDialect(Configuration $configuration, WorkingDirectory $workingDirectory, string $target, string $include): int
156    {
157        if (!$this->writesDialect($configuration, $workingDirectory)) {
158            return 0;
159        }
160
161        $dialect  = str_replace('extension.neon', 'crucible-dialect.neon', $include);
162        $existing = file_get_contents($workingDirectory->path . '/' . $target);
163
164        if ($existing === false) {
165            printf('Cannot read %s.' . PHP_EOL, $target);
166
167            return 1;
168        }
169
170        if (PhpstanNeon::alreadyWired($existing, $dialect)) {
171            return 0;
172        }
173
174        $wired = PhpstanNeon::wire($existing, $dialect);
175
176        if ($wired === null) {
177            printf('%s has a shape this command will not edit. Your suite writes the crucible dialect; add:' . PHP_EOL, $target);
178            print PHP_EOL . 'includes:' . PHP_EOL . '    - ' . $dialect . PHP_EOL;
179
180            return 1;
181        }
182
183        file_put_contents($workingDirectory->path . '/' . $target, $wired);
184        printf('Updated %s — the suite writes the crucible dialect, so check()/property()/table() are declared too.' . PHP_EOL, $target);
185
186        return 0;
187    }
188
189    /** Whether any configured test directory holds a *.crucible.php file. */
190    private function writesDialect(Configuration $configuration, WorkingDirectory $workingDirectory): bool
191    {
192        foreach ($configuration->testSuites as $suite) {
193            foreach ($suite->directories as $directory) {
194                $absolute = $workingDirectory->absolute($directory);
195
196                if (!is_dir($absolute)) {
197                    continue;
198                }
199
200                foreach (new RecursiveIteratorIterator(new RecursiveDirectoryIterator($absolute, FilesystemIterator::SKIP_DOTS)) as $file) {
201                    if ($file instanceof SplFileInfo && str_ends_with($file->getFilename(), '.crucible.php')) {
202                        return true;
203                    }
204                }
205            }
206        }
207
208        return false;
209    }
