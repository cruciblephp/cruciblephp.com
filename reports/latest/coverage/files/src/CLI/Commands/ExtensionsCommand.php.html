48/**
49 * `crucible extensions`: introspects the report-format, subscriber, and
50 * progress-view plugins registered in `crucible.php` — what's wired,
51 * whether each is available in this environment, and (via `--preview`)
52 * a live, code-driven sample, so a plugin's actual output is never
53 * something you have to trust documentation for. All three kinds
54 * share the same class-string + `#[Attribute]` + reflection
55 * registration shape (`AbstractRegistry`), so `list()`/`detail()` walk
56 * them through one `array<string, AbstractRegistry>` map rather than a
57 * per-kind branch that grows with every new kind — adding a fourth
58 * kind later touches one map entry, not a new branch here.
59 */
60final class ExtensionsCommand
61{
62    /**
63     * @param list<string>     $argv
64     */
65    public function execute(CliOptions $options, array $argv, WorkingDirectory $workingDirectory): int
66    {
67        try {
68            $loaded = (new Loader())->load($workingDirectory, $options->configuration);
69        } catch (Exception $e) {
70            print $e->getMessage() . PHP_EOL;
71
72            return 1;
73        }
74
75        $reportFormats = new ReportFormatRegistry($loaded->configuration->reportFormats);
76        $subscribers   = new SubscriberRegistry($loaded->configuration->subscribers);
77        $progressViews = new ProgressViewRegistry($loaded->configuration->progressViews);
78
79        /** @var array<string, AbstractRegistry> */
80        $registries = [
81            'report-format' => $reportFormats,
82            'subscriber'    => $subscribers,
83            'progress-view' => $progressViews,
84        ];
85
86        if ($options->preview) {
87            return $this->preview($reportFormats, $subscribers, $progressViews, $loaded->configuration->subscribers, $loaded->configuration->progressViews, $options);
88        }
89
90        if ($options->key !== null) {
91            return $this->detail($registries, $options->key, $options->json);
92        }
93
94        return $this->list($registries, $options->json);
95    }

97    /** @param array<string, AbstractRegistry> $registries */
98    private function list(array $registries, bool $json): int
99    {
100        /** @var list<array{kind: string, descriptor: PluginDescriptor}> $rows */
101        $rows = [];
102
103        foreach ($registries as $kind => $registry) {
104            foreach ($registry->all() as $d) {
105                $rows[] = ['kind' => $kind, 'descriptor' => $d];
106            }
107        }
108
109        if ($json) {
110            print json_encode(array_map(
111                static fn(array $row): array => ['kind' => $row['kind']] + $row['descriptor']->toArray(),
112                $rows,
113            ), JSON_PRETTY_PRINT) . PHP_EOL;
114
115            return 0;
116        }
117
118        if ($rows === []) {
119            print 'Nothing registered — add a report format via ->reportFormat(), a subscriber via ->subscriber(), or a progress view via ->progressView() in crucible.php.' . PHP_EOL;
120
121            return 0;
122        }
123
124        $tableRows = [];
125
126        foreach ($rows as $row) {
127            $d = $row['descriptor'];
128
129            $tableRows[] = [
130                $row['kind'],
131                $d->key,
132                $d->description,
133                $d->available ? 'yes' : 'no',
134                $d->available ? $this->orDash(implode(', ', $d->requires)) : implode(', ', $d->missingRequirements),
135            ];
136        }
137
138        print (new Table(['KIND', 'KEY', 'DESCRIPTION', 'AVAILABLE', 'REQUIREMENTS'], $tableRows))->toString();
139
140        return 0;
141    }

143    /** @param array<string, AbstractRegistry> $registries */
144    private function detail(array $registries, string $key, bool $json): int
145    {
146        $d = null;
147
148        foreach ($registries as $registry) {
149            if ($registry->has($key)) {
150                $d = $registry->describe($key);
151
152                break;
153            }
154        }
155
156        if ($d === null) {
157            printf('"%s" is not registered as a report format, a subscriber, or a progress view.' . PHP_EOL, $key);
158
159            return 1;
160        }
161
162        if ($json) {
163            print json_encode($d->toArray(), JSON_PRETTY_PRINT) . PHP_EOL;
164
165            return 0;
166        }
167
168        printf('%s' . PHP_EOL, $d->key);
169        printf('  %s' . PHP_EOL, $d->description);
170
171        if ($d->comment !== null) {
172            printf(PHP_EOL . '%s' . PHP_EOL, $d->comment);
173        }
174
175        printf(
176            PHP_EOL . 'Available: %s' . PHP_EOL,
177            $d->available ? 'yes' : sprintf('no (%s)', implode(', ', $d->missingRequirements)),
178        );
179
180        if ($d->params !== []) {
181            $paramRows = [];
182
183            foreach ($d->params as $p) {
184                $paramRows[] = [$p['name'], $p['type'], $p['required'] ? 'required' : 'optional', $p['description']];
185            }
186
187            print PHP_EOL . 'Params:' . PHP_EOL;
188            print (new Table(['NAME', 'TYPE', 'REQUIRED', 'DESCRIPTION'], $paramRows))->toString();
189        }
190
191        return 0;
192    }

194    /**
195     * @param array<string, array{class: class-string, params: array<string, mixed>}> $configuredSubscribers
196     * @param array<string, array{class: class-string, params: array<string, mixed>}> $configuredViews
197     */
198    private function preview(
199        ReportFormatRegistry $reportFormats,
200        SubscriberRegistry $subscribers,
201        ProgressViewRegistry $progressViews,
202        array $configuredSubscribers,
203        array $configuredViews,
204        CliOptions $options,
205    ): int {
206        try {
207            $key = $options->key ?? $this->chooseKey($reportFormats, $subscribers, $progressViews);
208        } catch (NonInteractiveException) {
209            // The prompt refused to answer for a script, and this command
210            // knows the flag that would have. Its own contract is kept:
211            // the same message and the same exit code it always gave.
212            print '--preview requires --key=<name>.' . PHP_EOL;
213
214            return 1;
215        }
216
217        if ($key === null) {
218            print 'Nothing is registered to preview — add one via ->reportFormat(), ->subscriber(), or ->progressView() in crucible.php.' . PHP_EOL;
219
220            return 1;
221        }
222
223        if ($reportFormats->has($key)) {
224            return $this->previewReportFormat($reportFormats, $key, $options->out);
225        }
226
227        if ($subscribers->has($key)) {
228            return $this->previewSubscriber($configuredSubscribers, $key, $options->out);
229        }
230
231        if ($progressViews->has($key)) {
232            return $this->previewProgressView($configuredViews, $key, $options->out);
233        }
234
235        printf('"%s" is not registered as a report format, a subscriber, or a progress view.' . PHP_EOL, $key);
236
237        return 1;
238    }

240    /**
241     * Ask which plugin to preview, when the command line did not say.
242     *
243     * The prompt carries no default on purpose: off a terminal it raises
244     * {@see \LucianoPereira\Crucible\Console\Exceptions\NonInteractiveException}
245     * rather than previewing whichever plugin happened to register first.
246     * Null here means the list itself is empty, which is a different answer.
247     */
248    private function chooseKey(AbstractRegistry ...$registries): ?string
249    {
250        $choices = [];
251
252        foreach ($registries as $registry) {
253            foreach ($registry->all() as $d) {
254                $choices[$d->key] = $d->available ? $d->key : $d->key . '  (unavailable)';
255            }
256        }
257
258        if ($choices === []) {
259            return null;
260        }
261
262        $prompt = new SelectPrompt(
263            label: 'Which plugin should be previewed?',
264            options: $choices,
265            hint: 'Or pass --key=<name>.',
266        );
267
268        Runtime::run($prompt);
269
270        $chosen = $prompt->value();
271
272        return $chosen === null ? null : (string) $chosen;
273    }

275    private function previewReportFormat(ReportFormatRegistry $registry, string $key, ?string $out): int
276    {
277        $d = $registry->describe($key);
278
279        if (!$d->available) {
280            printf('Report format "%s" is not available: %s.' . PHP_EOL, $key, implode(', ', $d->missingRequirements));
281
282            return 1;
283        }
284
285        /** @var array<string, mixed> $params */
286        $params = [];
287
288        foreach ($d->params as $param) {
289            if (array_key_exists('default', $param)) {
290                $params[$param['name']] = $param['default'];
291            }
292        }
293
294        $instance = $registry->instantiate($key);
295        $document = SampleDocument::build();
296        $context  = new ReportContext('Sample Report', Version::AUTHOR, sprintf('Crucible %s', Version::NUMBER), null, 0.0);
297        $rendered = $instance->render($document, $context, $params);
298
299        $path = $out ?? sprintf('%s/crucible-preview-%s.out', sys_get_temp_dir(), $key);
300
301        if (file_put_contents($path, $rendered) === false) {
302            printf('Cannot write %s.' . PHP_EOL, $path);
303
304            return 1;
305        }
306
307        printf('Wrote preview to %s (%d bytes).' . PHP_EOL, $path, strlen($rendered));
308
309        return 0;
310    }

312    /**
313     * A subscriber has no static Document to render — it only shows
314     * what it does by actually receiving events (via {@see SampleRun}).
315     * Unlike a report format, whose "output" param is vestigial at
316     * preview time (the caller writes the returned string itself),
317     * "output" is what a subscriber's constructor actually opens, so
318     * it is always forced to the computed preview path here — the
319     * schema's own default (if any) is not enough to make it write
320     * anywhere real.
321     *
322     * @param array<string, array{class: class-string, params: array<string, mixed>}> $configuredSubscribers
323     */
324    private function previewSubscriber(array $configuredSubscribers, string $key, ?string $out): int
325    {
326        $d = (new SubscriberRegistry($configuredSubscribers))->describe($key);
327
328        if (!$d->available) {
329            printf('Subscriber "%s" is not available: %s.' . PHP_EOL, $key, implode(', ', $d->missingRequirements));
330
331            return 1;
332        }
333
334        $path = $out ?? sprintf('%s/crucible-preview-%s.out', sys_get_temp_dir(), $key);
335
336        $overridden = $configuredSubscribers;
337
338        if (in_array('output', array_column($d->params, 'name'), true)) {
339            $overridden[$key] = [
340                'class'  => $overridden[$key]['class'],
341                'params' => [...$overridden[$key]['params'], 'output' => $path],
342            ];
343        }
344
345        $instance = (new SubscriberRegistry($overridden))->resolve($key);
346
347        SampleRun::through($instance);
348
349        $size = filesize($path);
350
351        printf('Wrote preview to %s (%d bytes).' . PHP_EOL, $path, $size === false ? 0 : $size);
352
353        return 0;
354    }

356    /**
357     * A progress view writes to a stream, not a user-chosen path — so
358     * unlike a subscriber's preview, there's no "output" param to
359     * force. Buffered through a memory stream instead (mirroring
360     * {@see previewReportFormat()}'s buffer-then-write shape), then
361     * written to the preview path once the run through
362     * {@see SampleRun} has finished.
363     *
364     * @param array<string, array{class: class-string, params: array<string, mixed>}> $configuredViews
365     */
366    private function previewProgressView(array $configuredViews, string $key, ?string $out): int
367    {
368        $registry = new ProgressViewRegistry($configuredViews);
369        $d        = $registry->describe($key);
370
371        if (!$d->available) {
372            printf('Progress view "%s" is not available: %s.' . PHP_EOL, $key, implode(', ', $d->missingRequirements));
373
374            return 1;
375        }
376
377        $stream = fopen('php://memory', 'w+');
378
379        if ($stream === false) {
380            print 'Cannot open an in-memory stream for the preview.' . PHP_EOL;
381
382            return 1;
383        }
384
385        $instance = $registry->resolve($key, $stream);
386
387        SampleRun::through($instance);
388
389        rewind($stream);
390        $rendered = (string) stream_get_contents($stream);
391        fclose($stream);
392
393        $path = $out ?? sprintf('%s/crucible-preview-%s.out', sys_get_temp_dir(), $key);
394
395        if (file_put_contents($path, $rendered) === false) {
396            printf('Cannot write %s.' . PHP_EOL, $path);
397
398            return 1;
399        }
400
401        printf('Wrote preview to %s (%d bytes).' . PHP_EOL, $path, strlen($rendered));
402
403        return 0;
404    }

406    private function orDash(string $value): string
407    {
408        return $value === '' ? '—' : $value;
