48/**
49 * `crucible compat-check`: runs the project's real, installed PHPUnit
50 * alongside Crucible, finds tests the oracle passes but Crucible
51 * doesn't, and diagnoses why — the productized version of a manual
52 * compat audit. Auto-fix only ever touches the small, explicit table
53 * in KnownCouplingPatterns; anything else is diagnosed, never guessed
54 * at, and the interactive path always asks before writing.
55 */
56final class CompatCheckCommand
57{
58    /**
59     * @param list<string>     $argv
60     */
61    public function execute(CliOptions $options, array $argv, WorkingDirectory $workingDirectory): int
62    {
63        $manifest = new FixManifest(
64            $workingDirectory->path . '/' . ($options->cacheDirectory ?? '.crucible.cache') . '/compat-fixes.json',
65        );
66
67        if ($options->revert) {
68            return $this->revert($manifest);
69        }
70
71        // Every exit below is either 1 — the engines disagree, which
72        // is a finding about the code — or 2, the run could not be
73        // performed, which is a finding about the setup. HELP.md's
74        // table already draws that line ("no tests found" is a 2), and
75        // a CI job acts on the two differently: fix the tests, or fix
76        // the configuration.
77        if (!PhpUnitCompatibility::phpUnitIsInstalled()) {
78            print 'compat-check needs the real phpunit/phpunit installed alongside Crucible '
79                . '(it is the oracle being compared against) — composer require --dev phpunit/phpunit.' . PHP_EOL;
80
81            return 2;
82        }
83
84        // Pest when the project has it, phpunit otherwise: measured,
85        // the phpunit binary cannot run a pest suite at all, so on a
86        // pest project it is not a narrower oracle but an empty one
87        // (D-114).
88        $oracleBinary = OracleRun::binaryIn($workingDirectory);
89
90        if ($oracleBinary === null) {
91            printf('Cannot find an oracle binary — looked for vendor/bin/pest and vendor/bin/phpunit in %s.' . PHP_EOL, $workingDirectory->path);
92
93            return 2;
94        }
95
96        $crucibleBinary = realpath($argv[0] ?? '');
97
98        if ($crucibleBinary === false) {
99            print 'Cannot resolve the crucible binary path for the comparison run.' . PHP_EOL;
100
101            return 2;
102        }
103
104        $forwarded = $this->forwardedArgs($argv);
105        $tmp       = sys_get_temp_dir() . '/crucible-compat-check-' . getmypid();
106
107        // The events log is asked for only from pest, the one oracle
108        // whose JUnit loses the file (D-118). An older phpunit is never
109        // handed an option it would refuse.
110        $eventsFile = basename($oracleBinary) === 'pest' ? $tmp . '-events.txt' : null;
111
112        printf('Running the real %s oracle...' . PHP_EOL, basename($oracleBinary));
113        ['outcomes' => $oracle, 'unmapped' => $unmapped] = OracleRun::outcomes(
114            $oracleBinary,
115            $forwarded,
116            $workingDirectory,
117            $tmp . '-junit.xml',
118            $eventsFile,
119        );
120
121        if (is_file($tmp . '-junit.xml')) {
122            unlink($tmp . '-junit.xml');
123        }
124
125        print 'Running Crucible...' . PHP_EOL;
126        $crucible = CrucibleRun::outcomes($crucibleBinary, $forwarded, $workingDirectory, $tmp . '-events.ndjson');
127
128        print PHP_EOL;
129
130        // What pest described instead of locating is asked of the
131        // engine that owns it, before any verdict — recovering those
132        // rows widens the comparison, and what stays unrecovered
133        // narrows what the verdict is ABOUT.
134        $lost = 0;
135
136        if ($unmapped !== [] && $eventsFile !== null && is_file($eventsFile)) {
137            $placed = PhpUnitEventLog::keyedByFile(
138                PhpUnitEventLog::parse($eventsFile),
139                self::classesInCandidateFiles($oracle, $crucible, $workingDirectory),
140            );
141
142            $oracle = [...$oracle, ...$placed['outcomes']];
143
144            $lost = $this->reportUnmapped($unmapped, $placed['outcomes'], $placed['unlocated']);
145        } elseif ($unmapped !== []) {
146            $lost = $this->reportUnmapped($unmapped, [], []);
147        }
148
149        if ($eventsFile !== null && is_file($eventsFile)) {
150            unlink($eventsFile);
151        }
152
153        $drifted = OutcomeDiff::driftedToFailure($oracle, $crucible);
154        $missing = OutcomeDiff::missingFromCrucible($oracle, $crucible);
155
156        // And before that: an oracle that collected nothing leaves both
157        // lists empty, which is exactly what agreement is spelled with.
158        $vacuous = self::nothingToCompare($oracle, $crucible);
159
160        if ($vacuous !== null) {
161            print $vacuous;
162
163            return 2;
164        }
165
166        // Before any verdict: a test Crucible never ran is not a test
167        // Crucible agrees about. Comparing outcomes alone cannot see
168        // that, so nothing below is meaningful until it is ruled out.
169        if ($missing !== []) {
170            return $this->reportMissing($oracle, $crucible, $missing);
171        }
172
173        if ($drifted === []) {
174            // A partial comparison does not get the clean sentence
175            // first and a caveat after it. What a reader takes from a
176            // skim is the verdict, so the qualifier is IN the verdict:
177            // the unearned green this command exists to refuse is one
178            // that reads green.
179            if ($lost > 0) {
180                printf(
181                    'No drift among the %d test(s) both engines could be asked about — but %d could not be, '
182                        . 'so the suite is not proven.' . PHP_EOL,
183                    count($oracle),
184                    $lost,
185                );
186
187                return 2;
188            }
189
190            // The count is part of the verdict, not decoration: this
191            // command's failure mode is agreeing about nothing, and a
192            // clean run that does not say how much it compared cannot
193            // be told from one that compared almost none of it.
194            printf(
195                'No drift: all %d test(s) the real %s oracle passes, Crucible also passes.' . PHP_EOL,
196                count($oracle),
197                basename($oracleBinary),
198            );
199
200            return 0;
201        }
202
203        return $this->handleDrift($drifted, $manifest, $workingDirectory, $options->autoFix, basename($oracleBinary));
204    }

206    /**
207     * The classes declared by the files the oracle could not key.
208     *
209     * The candidate set is implied by the two runs already made: every
210     * file Crucible reported an id in, minus every file the oracle
211     * managed to key. Only those files are scanned, and the scan is
212     * ClassLocator's token pass — a file that declares a class says so
213     * in its own grammar, so this needs no loading and no inference.
214     *
215     * A pest-shaped file is deliberately NOT excluded here. It was, for
216     * as long as the recovery meant handing the file to phpunit, which
217     * dies inside pest on such a file — and that exclusion is what
218     * capped the comparison at 3.3% of a real suite (D-117), because a
219     * classic class living in a pest-shaped file is exactly the case
220     * that needed recovering. Reading the run's own event log asks no
221     * engine to load anything, so the guard has nothing left to guard.
222     *
223     * @param array<string, string> $oracle
224     * @param array<string, string> $crucible
225     *
226     * @return array<string, non-empty-string> class => project-relative file
227     */
228    public static function classesInCandidateFiles(array $oracle, array $crucible, WorkingDirectory $workingDirectory): array
229    {
230        $answered = [];
231
232        foreach (array_keys($oracle) as $key) {
233            $answered[explode('::', $key, 2)[0]] = true;
234        }
235
236        $locator = new ClassLocator();
237        $classes = [];
238        $seen    = [];
239
240        foreach (array_keys($crucible) as $key) {
241            $file = explode('::', $key, 2)[0];
242
243            if ($file === '' || isset($answered[$file]) || isset($seen[$file])) {
244                continue;
245            }
246
247            $seen[$file] = true;
248            $absolute    = rtrim($workingDirectory->path, '/') . '/' . $file;
249
250            if (!is_file($absolute)) {
251                continue;
252            }
253
254            foreach ($locator->classesIn($absolute) as $class) {
255                $classes[$class] = $file;
256            }
257        }
258
259        return $classes;
260    }

262    /**
263     * What the first oracle pass could not key, and what became of it.
264     *
265     * Anything the second pass did not recover leaves the comparison
266     * and is named here rather than dropped: quietly shrinking the
267     * compared suite is [nothingToCompare]'s defect one size down, and
268     * the harder one to notice, because the number that shrinks is
269     * never printed. When there was nothing to retry the cause is
270     * usually one thing, and it is said instead of shrugged at.
271     *
272     * @param list<string>          $unmapped
273     * @param array<string, string> $recovered
274     * @param list<string>          $unlocated classes the event log named that no candidate file declares
275     *
276     * @return int the number left out of the comparison
277     */
278    private function reportUnmapped(array $unmapped, array $recovered, array $unlocated): int
279    {
280        if ($unmapped === []) {
281            return 0;
282        }
283
284        printf(
285            '%d oracle test(s) came back without a source file — pest reports a classic PHPUnit class by '
286                . 'description, not by file.' . PHP_EOL,
287            count($unmapped),
288        );
289
290        if ($recovered !== []) {
291            printf(
292                '  Read %d of them from the run\'s own event log, which reports a class by name rather than by '
293                    . 'description, and placed them by the file that declares that class.' . PHP_EOL,
294                count($recovered),
295            );
296        }
297
298        $lost = count($unmapped) - count($recovered);
299
300        if ($lost <= 0) {
301            print PHP_EOL;
302
303            return 0;
304        }
305
306        printf('  %d are NOT part of what follows:' . PHP_EOL, $lost);
307
308        foreach (array_slice($unmapped, 0, 5) as $name) {
309            printf('    %s' . PHP_EOL, $name);
310        }
311
312        if (count($unmapped) > 5) {
313            printf('    ... and %d more.' . PHP_EOL, count($unmapped) - 5);
314        }
315
316        if ($unlocated !== []) {
317            // The residual case, and it has the opposite remedy to a
318            // limit: the event log named a class, so the oracle DID run
319            // it — no candidate file declares it, which means Crucible
320            // ran nothing in that file. That is discovery, not a
321            // structural bar, and it is said rather than shrugged at.
322            printf(
323                '  %d of them name a class no file Crucible ran declares, so the two runs cannot be lined up '
324                    . 'there. Crucible discovering that file is what closes it — check the suites in '
325                    . 'crucible.php, and ->phpunitCompatibility() if the class extends PHPUnit\'s TestCase '
326                    . 'while phpunit/phpunit is installed (D-019):' . PHP_EOL,
327                count($unlocated),
328            );
329
330            foreach (array_slice($unlocated, 0, 5) as $id) {
331                printf('    %s' . PHP_EOL, $id);
332            }
333        }
334
335        print PHP_EOL;
336
337        return $lost;
338    }

340    /**
341     * Why this comparison proves nothing, or null when it proves
342     * something.    /**
343     * Why this comparison proves nothing, or null when it proves
344     * something.
345     *
346     * "No drift" is spelled with two empty lists — no missing tests, no
347     * drifted ones — and an oracle that collected nothing produces
348     * exactly that pair. The green then reports the strength of the
349     * evidence as agreement, which is the one failure mode a parity
350     * tool must not have: the same output for "they agree everywhere"
351     * and "nothing was asked". The Crucible-side hole was already
352     * guarded; this is its mirror.
353     *
354     * @param array<string, string> $oracle
355     * @param array<string, string> $crucible
356     */
357    public static function nothingToCompare(array $oracle, array $crucible): ?string
358    {
359        if ($oracle !== []) {
360            return null;
361        }
362
363        return sprintf(
364            'The oracle ran no tests at all, while Crucible ran %d. Nothing can be compared, so this '
365                . 'is not "no drift".' . PHP_EOL . PHP_EOL
366                . 'Check that the real phpunit collects anything here on its own (vendor/bin/phpunit), that '
367                . 'a forwarded --filter matches something — it is passed to both engines verbatim — and that '
368                . 'this is the directory its configuration belongs to.' . PHP_EOL,
369            count($crucible),
370        );
371    }

373    /**
374     * @param array<string, string>  $oracle
375     * @param array<string, string>  $crucible
376     * @param list<non-empty-string> $missing
377     */
378    private function reportMissing(array $oracle, array $crucible, array $missing): int
379    {
380        if ($crucible === []) {
381            printf(
382                'Crucible ran no tests at all, while the oracle ran %d. Nothing can be compared, so this '
383                    . 'is not "no drift".' . PHP_EOL . PHP_EOL
384                    . 'The usual cause: phpunit/phpunit is installed, so the PHPUnit-namespace compatibility '
385                    . 'aliases stand down (D-019) and a suite extending PHPUnit\'s TestCase is discovered by '
386                    . 'nobody. Give the suite Crucible\'s TestCase, opt in with ->phpunitCompatibility(), or '
387                    . 'run the comparison from a checkout where the dialect applies.' . PHP_EOL,
388                count($oracle),
389            );
390
391            // The setup side of the line: nothing was compared, and the
392            // cause named above is a configuration one. Its neighbour
393            // below stays a 1 — tests that ARE missing from a Crucible
394            // run that happened is a finding about discovery, and D-112
395            // is the proof that it can be a real bug.
396            return 2;
397        }
398
399        printf(
400            '%d test(s) the oracle passes produced no Crucible outcome at all — they were never run, which '
401                . 'is not the same as passing:' . PHP_EOL,
402            count($missing),
403        );
404
405        foreach (array_slice($missing, 0, 10) as $key) {
406            printf('  %s' . PHP_EOL, $key);
407        }
408
409        if (count($missing) > 10) {
410            printf('  ... and %d more.' . PHP_EOL, count($missing) - 10);
411        }
412
413        return 1;
414    }

416    /**
417     * @param list<non-empty-string> $drifted
418     */
419    private function handleDrift(array $drifted, FixManifest $manifest, WorkingDirectory $workingDirectory, bool $autoFix, string $oracle): int
420    {
421        $patterns = KnownCouplingPatterns::all();
422        $unfixed  = 0;
423
424        foreach ($drifted as $key) {
425            [$relativeFile, $rest] = explode('::', $key, 2);
426            $method                = explode('#', $rest, 2)[0];
427
428            printf('%s::%s fails under Crucible, passes under real %s.' . PHP_EOL, $relativeFile, $method, $oracle);
429            print "  Likely coupled to PHPUnit's own internals (file layout, internal class/method names) rather than your code's behavior." . PHP_EOL;
430
431            $file = rtrim($workingDirectory->path, '/') . '/' . $relativeFile;
432
433            if (!is_file($file)) {
434                print '  Could not locate the source file — review manually.' . PHP_EOL . PHP_EOL;
435                $unfixed++;
436
437                continue;
438            }
439
440            $source = (string) file_get_contents($file);
441            $span   = SourceRewriter::locateMethod($source, $method);
442
443            if ($span === null) {
444                print '  Could not locate the method in its source file to check for a known fix — review manually.' . PHP_EOL . PHP_EOL;
445                $unfixed++;
446
447                continue;
448            }
449
450            [$startLine, $endLine] = $span;
451            $preview               = SourceRewriter::preview($source, $startLine, $endLine, $patterns);
452
453            if ($preview === []) {
454                print '  No automatic fix available for this pattern — review manually, or run this specific test under PHPUnit.' . PHP_EOL . PHP_EOL;
455                $unfixed++;
456
457                continue;
458            }
459
460            foreach ($preview as $match) {
461                printf('  %s (line %d):' . PHP_EOL, $match['description'], $match['line']);
462                printf('    - %s' . PHP_EOL, $match['before']);
463                printf('    + %s' . PHP_EOL, $match['after']);
464            }
465
466            if (!$autoFix && !$this->confirm('  Rewrite this test?')) {
467                print '  Skipped.' . PHP_EOL . PHP_EOL;
468                $unfixed++;
469
470                continue;
471            }
472
473            $manifest->backup($file, $source);
474
475            $result = SourceRewriter::apply($source, $startLine, $endLine, $patterns);
476            file_put_contents($file, $result['source']);
477
478            // "Rewrote", not "Fixed": this only confirms the known
479            // pattern(s) above were applied, not that the whole test
480            // now passes — the same method can carry other coupling
481            // this table does not recognize (that is exactly what
482            // happened to LineFormatterTest::testBasePath's second
483            // assertion during this feature's own real-world
484            // verification: fixing frame 0 left frame 1 still
485            // failing, silently, until the mandatory re-run caught
486            // it). Re-running compat-check is how success is
487            // actually confirmed, not this message.
488            print '  Rewrote the pattern(s) above.' . PHP_EOL . PHP_EOL;
489        }
490
491        if ($unfixed > 0) {
492            printf('%d test(s) still drifted after this run. Re-run crucible compat-check to verify, or --revert to undo applied fixes.' . PHP_EOL, $unfixed);
493
494            return 1;
495        }
496
497        print 'Rewrote every drifted test that matched a known pattern. Re-run crucible compat-check to confirm they now pass — a rewrite here is not itself proof of that.' . PHP_EOL;
498
499        return 0;
500    }

502    private function revert(FixManifest $manifest): int
503    {
504        if ($manifest->isEmpty()) {
505            print 'Nothing to revert.' . PHP_EOL;
506
507            return 0;
508        }
509
510        $restored = $manifest->revert();
511
512        print 'Restored:' . PHP_EOL;
513
514        foreach ($restored as $file) {
515            print '  - ' . $file . PHP_EOL;
516        }
517
518        return 0;
519    }

521    /**
522     * @param list<string> $argv
523     *
524     * @return list<non-empty-string>
525     */
526    private function forwardedArgs(array $argv): array
527    {
528        $forwarded = [];
529
530        foreach (array_slice($argv, 1) as $argument) {
531            if (in_array($argument, ['compat-check', '--auto-fix', '--revert', ''], true)) {
532                continue;
533            }
534
535            $forwarded[] = $argument;
536        }
537
538        return $forwarded;
539    }

541    private function confirm(string $prompt): bool
542    {
543        print $prompt . ' [y/N] ';
544
545        $answer = fgets(STDIN);
546
547        return $answer !== false && strtolower(trim($answer)) === 'y';
548    }
