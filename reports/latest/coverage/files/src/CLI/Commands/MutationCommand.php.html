67/**
68 * The `crucible mutation-index` and `crucible mutate` commands
69 * (D-077, D-082): the per-line covering-tests index, and the
70 * mutation-testing run built on top of it.
71 */
72final class MutationCommand
73{
74    /**
75     * `crucible mutation-index` (D-077): emit the mutation-facing query
76     * index as JSON — for every covered source line, the tests that ran
77     * it, fastest-first. Assembled from the per-test line map
78     * (`--coverage`) and the result cache's timings; a mutation tool
79     * loads it to run a mutant's covering tests cheapest-first.
80     *
81     */
82    public function index(CliOptions $options, WorkingDirectory $workingDirectory): int
83    {
84        try {
85            $loaded = (new Loader())->load($workingDirectory, $options->configuration);
86        } catch (Exception $e) {
87            print $e->getMessage() . PHP_EOL;
88
89            return 1;
90        }
91
92        $cacheDirectory = $options->cacheDirectory ?? $loaded->configuration->cacheDirectory;
93        $base           = $workingDirectory->absolute($cacheDirectory);
94        $map            = TestLineMap::load($base . '/coverage-lines.json');
95
96        if ($map->tests === []) {
97            // An opt-in query owes the user the way in (D-041's rule).
98            print 'mutation-index needs a per-test line map — run the suite once with --coverage first.' . PHP_EOL;
99
100            return 1;
101        }
102
103        $cache     = ResultCache::load($base . '/results.json');
104        $durations = [];
105
106        foreach (array_keys($map->tests) as $id) {
107            if ($id === '') {
108                continue;
109            }
110
111            $duration = $cache->duration($id);
112
113            if ($duration !== null) {
114                $durations[$id] = $duration;
115            }
116        }
117
118        print json_encode(
119            MutationIndex::build($map, $durations)->toArray(),
120            JSON_THROW_ON_ERROR | JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES,
121        ) . PHP_EOL;
122
123        return 0;
124    }

126    /**
127     * `crucible mutate`: generate mutants across the covered source, run each
128     * against its covering tests (fastest-first, D-077), and report the
129     * mutation score. Cold execution — portable, no pcntl — in v1; the
130     * warm fork is the follow-up the router already accommodates. Exits 1
131     * when any mutant escaped: an escaped mutant is a gap in the suite.
132     *
133     * @param list<string>     $argv
134     */
135    public function mutate(CliOptions $options, array $argv, WorkingDirectory $workingDirectory): int
136    {
137        try {
138            $loaded = (new Loader())->load($workingDirectory, $options->configuration);
139        } catch (Exception $e) {
140            print $e->getMessage() . PHP_EOL;
141
142            return 1;
143        }
144
145        $cacheDirectory = $options->cacheDirectory ?? $loaded->configuration->cacheDirectory;
146        $base           = $workingDirectory->absolute($cacheDirectory);
147        $mapFile        = $base . '/coverage-lines.json';
148        $map            = TestLineMap::load($mapFile);
149
150        if ($map->tests === []) {
151            print 'mutate needs a per-test line map — run the suite once with --coverage first.' . PHP_EOL;
152
153            return 1;
154        }
155
156        $cache     = ResultCache::load($base . '/results.json');
157        $durations = [];
158
159        foreach (array_keys($map->tests) as $id) {
160            if ($id === '') {
161                continue;
162            }
163
164            $duration = $cache->duration($id);
165
166            if ($duration !== null) {
167                $durations[$id] = $duration;
168            }
169        }
170
171        $index   = MutationIndex::build($map, $durations);
172        $covered = array_keys($index->toArray());
173
174        $this->warnIfCoverageStale($mapFile, $covered, $workingDirectory);
175
176        // Discovery is hoisted above mutant generation because mutates()
177        // lives in test metadata, and the warm executor below needs the
178        // same discovered set — one discovery serves both.
179        try {
180            $discovered = (new TestDiscoverer())->discover($loaded->configuration, $workingDirectory);
181        } catch (Exception $e) {
182            print $e->getMessage() . PHP_EOL;
183
184            return 1;
185        }
186
187        $declared = $this->declaredMutationScope($discovered);
188        $covered  = $this->narrowToDeclaredMutationScope($declared, $covered, $workingDirectory);
189
190        // Asked to mutate named classes and none of them is covered: the
191        // ask cannot be met. Reporting "no mutants" and exiting 0 here
192        // would say the mutation run succeeded having tested nothing,
193        // which is the silent pass this command exists to prevent.
194        if ($declared !== [] && $covered === []) {
195            printf(
196                'mutates(): none of the declared classes is covered, so there is nothing to mutate: %s' . PHP_EOL,
197                implode(', ', array_keys($declared)),
198            );
199
200            return 1;
201        }
202
203        $mutants = $this->generateMutants($covered, $workingDirectory);
204        $binary  = realpath($argv[0] ?? '');
205
206        if ($mutants === []) {
207            print 'No mutants generated from the covered source.' . PHP_EOL;
208
209            return 0;
210        }
211
212        if ($binary === false) {
213            print 'Cannot resolve the crucible binary path for the cold mutation workers.' . PHP_EOL;
214
215            return 1;
216        }
217
218        // Resume before anything is executed. The journal is keyed on
219        // mutated content, so a source edit drops its own entries and
220        // only the untouched files carry over.
221        $journal     = MutationJournal::load($base . '/mutation-verdicts.ndjson');
222        $coveringIds = function (Mutant $mutant) use ($index, $workingDirectory): array {
223            $relative = $this->relativeTo($mutant->file, $workingDirectory);
224            $ids      = [];
225
226            foreach ($index->coveringTests($relative, $mutant->line) as $test) {
227                if ($test['id'] !== '') {
228                    $ids[] = $test['id'];
229                }
230            }
231
232            return $ids;
233        };
234
235        $sources  = [];
236        $judgedBy = [];
237        $done     = [];
238        $pending  = [];
239
240        foreach ($mutants as $mutant) {
241            $fingerprint                               = $this->judgedBy($coveringIds($mutant), $workingDirectory, $sources);
242            $judgedBy[MutationJournal::keyOf($mutant)] = $fingerprint;
243
244            if ($journal->has($mutant, $fingerprint)) {
245                $done[] = $mutant;
246
247                continue;
248            }
249
250            $pending[] = $mutant;
251        }
252
253        printf('Mutation testing: %d mutant(s) across %d covered file(s).' . PHP_EOL, count($mutants), count($covered));
254
255        if ($done !== []) {
256            printf(
257                'Resuming: %d already decided, %d to run. Delete %s to start over.' . PHP_EOL,
258                count($done),
259                count($pending),
260                $this->relativeTo($base . '/mutation-verdicts.ndjson', $workingDirectory),
261            );
262        }
263
264        print PHP_EOL;
265
266        // Warm forking where the platform allows it; the suite is discovered
267        // here in the parent, which never runs a test — so the source under
268        // mutation stays unloaded and every mutant is warm-applicable. A
269        // box without pcntl has no warm executor and routes every mutant
270        // cold (the router's contract). D-082.
271        $warm = null;
272
273        if (MutantApplier::isSupported()) {
274            $covering = new CoveringTestRunner($discovered, $workingDirectory, $cache);
275            $warm     = new WarmMutantExecutor(new MutantApplier(), $covering->firstKiller(...));
276        }
277
278        $cold = new ColdMutantExecutor($binary, $loaded->path, $workingDirectory);
279
280        // The bar counts the whole run, resumed mutants included, so a
281        // resumed run does not restart at 0%.
282        $bar          = new Progress('Mutation testing', count($mutants));
283        $bar->current = count($done);
284        $bar->start();
285
286        (new MutationRunner($cold, $warm, $coveringIds))
287            ->run($pending, $this->progress($bar, $workingDirectory, $journal, $judgedBy, count($done), count($mutants)));
288
289        $bar->finish();
290
291        // The report is the WHOLE run, read back from the journal: the
292        // verdicts this process just appended plus the ones an earlier
293        // one did. Scoring $report alone would rate a resumed run on
294        // only the mutants it happened to finish.
295        return $this->printMutationReport(new MutationReport($journal->verdictsFor($mutants)), $workingDirectory);
296    }

298    /**
299     * The classes the suite declared with mutates(), if any.
300     *
301     * @param list<TestGroup> $discovered
302     *
303     * @return array<class-string, true>
304     */
305    private function declaredMutationScope(array $discovered): array
306    {
307        $declared = [];
308
309        foreach ($discovered as $group) {
310            foreach ($group->tests as $test) {
311                foreach ($test->metadata->ofType(MutatesClass::class) as $mutates) {
312                    $declared[$mutates->className] = true;
313                }
314            }
315        }
316
317        return $declared;
318    }

320    /**
321     * Narrow the covered source to what mutates() declared, if anything did.
322     *
323     * Declaring nothing keeps the whole covered scope: a suite that never
324     * says mutates() must not silently mutate nothing.
325     *
326     * @param array<class-string, true> $declared
327     * @param list<string>              $covered  relative paths
328     *
329     * @return list<string>
330     */
331    private function narrowToDeclaredMutationScope(array $declared, array $covered, WorkingDirectory $workingDirectory): array
332    {
333        if ($declared === []) {
334            return $covered;
335        }
336
337        // Compare file -> class rather than resolving class -> file: the
338        // covered files are already in hand, and this needs no autoload
339        // of the declared target.
340        $locator  = new ClassLocator();
341        $narrowed = array_values(array_filter(
342            $covered,
343            static function (string $relative) use ($locator, $declared, $workingDirectory): bool {
344                $class = $locator->classIn($workingDirectory->absolute($relative));
345
346                return $class !== null && isset($declared[$class]);
347            },
348        ));
349
350        printf(
351            'mutates(): narrowed the mutation scope to %d of %d covered file(s).' . PHP_EOL,
352            count($narrowed),
353            count($covered),
354        );
355
356        return $narrowed;
357    }

359    /**
360     * Generate mutants over the covered source files (relative paths),
361     * resolving each file's class as the autoload key. Files without a
362     * declared class, or that cannot be read, are skipped.
363     *
364     * @param list<string>     $coveredFiles relative paths
365     *
366     * @return list<Mutant>
367     */
368    private function generateMutants(array $coveredFiles, WorkingDirectory $workingDirectory): array
369    {
370        $generator = new MutantGenerator();
371        $locator   = new ClassLocator();
372        $mutants   = [];
373
374        foreach ($coveredFiles as $relative) {
375            if ($relative === '') {
376                continue;
377            }
378
379            $absolute = $workingDirectory->absolute($relative);
380
381            // A covered "file" can be an eval()'d-code pseudo-path (inline
382            // doctests) — not a real file to mutate.
383            if (!is_file($absolute)) {
384                continue;
385            }
386
387            $source = @file_get_contents($absolute);
388            $class  = $locator->classIn($absolute);
389
390            if ($source === false || $class === null) {
391                continue;
392            }
393
394            $fileMutants = $generator->generate($absolute, $class, $source);
395
396            foreach ($fileMutants as $mutant) {
397                $mutants[] = $mutant;
398            }
399
400            // A marker that declares nothing is reported, not trusted
401            // (D-134): one covering no mutant (the code moved or the
402            // mutators changed), or one giving no reason.
403            foreach (EquivalentMarkers::scan($source)->ranges as $range) {
404                $covers = array_filter($fileMutants, static fn(Mutant $m): bool => $m->line >= $range['from'] && $m->line <= $range['to']) !== [];
405
406                if ($range['reason'] === null || !$covers) {
407                    printf(
408                        '  STALE    %s:%d  crucible-equivalent %s' . PHP_EOL,
409                        $relative,
410                        $range['line'],
411                        $range['reason'] === null ? 'gives no reason, so declares nothing' : 'covers no mutant; remove it',
412                    );
413                }
414            }
415        }
416
417        return $mutants;
418    }

420    /**
421     * Coverage that predates the source it maps is a lie — an escaped
422     * mutant might just be a covering test that no longer runs. Warn (not
423     * refuse: the user may know better) and name the drift.
424     *
425     * @param list<string>     $coveredFiles relative paths
426     */
427    private function warnIfCoverageStale(string $mapFile, array $coveredFiles, WorkingDirectory $workingDirectory): void
428    {
429        $coverageTime = @filemtime($mapFile);
430
431        if ($coverageTime === false) {
432            return;
433        }
434
435        $stale = [];
436
437        foreach ($coveredFiles as $relative) {
438            if ($relative === '') {
439                continue;
440            }
441
442            $sourceTime = @filemtime($workingDirectory->absolute($relative));
443
444            if ($sourceTime !== false && $sourceTime > $coverageTime) {
445                $stale[] = $relative;
446            }
447        }
448
449        if ($stale !== []) {
450            printf(
451                'Warning: %d covered file(s) changed since coverage was collected — results may be inaccurate; re-run with --coverage. First: %s' . PHP_EOL . PHP_EOL,
452                count($stale),
453                $stale[0],
454            );
455        }
456    }

458    /**
459     * Live output for a run measured in hours.
460     *
461     * Escapes print the moment they are found rather than at the end,
462     * because an escape is the finding — a run stopped halfway has still
463     * told the user which mutants their suite does not catch. The
464     * counter is throttled to once a second: 4083 lines of progress is
465     * not progress.
466     *
467     * Every verdict is journalled here too, before anything is printed:
468     * the point of the journal is to survive the kill, so it must be
469     * written at the moment the verdict exists, not at the end.
470     *
471     * @param array<string, string> $judgedBy         mutant key => covering-tests fingerprint
472     * @param int                   $resumed          verdicts an earlier run already decided
473     * @param int                   $overall          mutants in the whole run, resumed included
474     *
475     * @return Closure(MutationVerdict, int, int): void
476     */
477    private function progress(Progress $bar, WorkingDirectory $workingDirectory, MutationJournal $journal, array $judgedBy, int $resumed, int $overall): Closure
478    {
479        $started = microtime(true);
480        $last    = 0.0;
481        $escaped = 0;
482
483        return function (MutationVerdict $verdict, int $done, int $total) use ($bar, $workingDirectory, $journal, $judgedBy, $resumed, $overall, $started, &$last, &$escaped): void {
484            $journal->append($verdict->mutant, $verdict, $judgedBy[MutationJournal::keyOf($verdict->mutant)] ?? '');
485
486            // $done counts THIS process; the rate must too. Dividing the
487            // elapsed time of a resumed run by a count that includes the
488            // resumed mutants reads as a machine several times faster
489            // than the one doing the work.
490            $rate  = $done === 0 ? 0.0 : (microtime(true) - $started) / $done;
491            $done  = $resumed + $done;
492            $total = $overall;
493
494            $bar->current = $done;
495
496            if ($verdict->outcome === MutationOutcome::Escaped) {
497                $escaped++;
498
499                // Through the bar, not print: an escape is history and has
500                // to scroll away above a frame that stays put. Without a
501                // bar on screen this is the same line it always was.
502                $bar->interrupt(sprintf(
503                    '  ESCAPED  %s:%d  %s',
504                    $this->relativeTo($verdict->mutant->file, $workingDirectory),
505                    $verdict->mutant->line,
506                    $verdict->mutant->mutatorId,
507                ));
508            }
509
510            $now = microtime(true);
511
512            if ($now - $last < 1.0 && $done !== $total) {
513                return;
514            }
515
516            $last    = $now;
517            $elapsed = $now - $started;
518            $left    = $rate * ($total - $done);
519
520            // ⚠ The piped form is unchanged, deliberately: a `tee`d
521            // mutation log is how a run that takes hours gets read back,
522            // and these are the lines that get grepped. The bar is what
523            // replaces them at a terminal, where nothing is being kept.
524            if (! $bar->isActive()) {
525                printf(
526                    '  %d/%d mutants — %d escaped — %s elapsed, ~%s left' . PHP_EOL,
527                    $done,
528                    $total,
529                    $escaped,
530                    $this->duration($elapsed),
531                    $this->duration($left),
532                );
533
534                return;
535            }
536
537            $bar->hint(sprintf(
538                '%d/%d · %d escaped · %s elapsed, ~%s left',
539                $done,
540                $total,
541                $escaped,
542                $this->duration($elapsed),
543                $this->duration($left),
544            ));
545        };
546    }

548    /**
549     * A fingerprint of the tests that will judge this mutant.
550     *
551     * Their CONTENTS, not their names: rewriting a test's assertions
552     * without renaming it is exactly the change that must invalidate a
553     * recorded verdict, and it is the change that slipped through when
554     * the journal keyed on the mutant alone.
555     *
556     * Per FILE rather than per test id — a test id is `file.php::method`
557     * and a file's methods share its bytes — with the digests cached
558     * across mutants, since one test file covers hundreds of them.
559     *
560     * @param list<non-empty-string> $testIds
561     * @param array<string, string>  $sources   digest cache, by file
562     */
563    private function judgedBy(array $testIds, WorkingDirectory $workingDirectory, array &$sources): string
564    {
565        $digests = [];
566
567        foreach ($testIds as $id) {
568            $file = explode('::', $id)[0];
569
570            if ($file === '') {
571                continue;
572            }
573
574            if (!isset($sources[$file])) {
575                $contents       = @file_get_contents($workingDirectory->absolute($file));
576                $sources[$file] = $contents === false ? '' : md5($contents);
577            }
578
579            $digests[$file] = $sources[$file];
580        }
581
582        ksort($digests);
583
584        return md5(implode('|', $digests) . '#' . implode('|', array_keys($digests)));
585    }

587    /** @return non-empty-string */
588    private function duration(float $seconds): string
589    {
590        $whole = (int) $seconds;
591
592        return $whole < 60
593            ? $whole . 's'
594            : sprintf('%dm%02ds', intdiv($whole, 60), $whole % 60);
595    }

597    private function printMutationReport(MutationReport $report, WorkingDirectory $workingDirectory): int
598    {
599        $escaped = array_values(array_filter(
600            $report->verdicts,
601            static fn(MutationVerdict $verdict): bool => $verdict->outcome === MutationOutcome::Escaped,
602        ));
603
604        foreach (array_slice($escaped, 0, 20) as $verdict) {
605            printf('  ESCAPED  %s:%d  %s' . PHP_EOL, $this->relativeTo($verdict->mutant->file, $workingDirectory), $verdict->mutant->line, $verdict->mutant->mutatorId);
606        }
607
608        if (count($escaped) > 20) {
609            printf('  … and %d more escaped.' . PHP_EOL, count($escaped) - 20);
610        }
611
612        if ($escaped !== []) {
613            print PHP_EOL;
614        }
615
616        $equivalent = array_values(array_filter(
617            $report->verdicts,
618            static fn(MutationVerdict $verdict): bool => $verdict->outcome === MutationOutcome::Equivalent,
619        ));
620
621        // Declared equivalents are listed, never hidden (D-134): the
622        // decision stays in front of whoever reads the report.
623        foreach (array_slice($equivalent, 0, 20) as $verdict) {
624            printf('  EQUIVALENT  %s:%d  %s — %s' . PHP_EOL, $this->relativeTo($verdict->mutant->file, $workingDirectory), $verdict->mutant->line, $verdict->mutant->mutatorId, (string) $verdict->reason);
625        }
626
627        if (count($equivalent) > 20) {
628            printf('  … and %d more declared equivalent.' . PHP_EOL, count($equivalent) - 20);
629        }
630
631        if ($equivalent !== []) {
632            print PHP_EOL;
633        }
634
635        printf(
636            'Killed: %d  Escaped: %d  Errored: %d  Timed out: %d  Not covered: %d  Declared equivalent: %d' . PHP_EOL,
637            $report->count(MutationOutcome::Killed),
638            count($escaped),
639            $report->count(MutationOutcome::Errored),
640            $report->count(MutationOutcome::TimedOut),
641            $report->count(MutationOutcome::NotCovered),
642            count($equivalent),
643        );
644        printf('MSI: %.1f%% over %d covered mutant(s).' . PHP_EOL, $report->score(), $report->covered());
645
646        return $escaped === [] ? 0 : 1;
647    }

649    private function relativeTo(string $file, WorkingDirectory $workingDirectory): string
650    {
651        $prefix = rtrim($workingDirectory->path, '/') . '/';
652
653        return str_starts_with($file, $prefix) ? substr($file, strlen($prefix)) : $file;
654    }
