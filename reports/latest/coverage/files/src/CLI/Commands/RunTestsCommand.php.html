159/**
160 * `crucible` with no subcommand: run the configured test suite. Owns
161 * both the parent-process orchestration (`execute()`) and the child
162 * worker protocol (`worker()`) — the two ends of the same process
163 * split that process isolation and parallelism use throughout a run.
164 */

181            // not being read, which is the only thing it can prove here —
182            // Crucible takes no positional paths to run instead.

277                return 1;

284        // subscriber registered the same way in crucible.php) —

304        }

307            $selectedSubscribers['junit'] = $options->logJunit;
308        }

310        // The documentation view's file targets take the same path: one
311        // dispatch mechanism under every spelling.
312        foreach ([
313            'otr'             => $options->logOtr,
314            'testdox-text'    => $options->testdoxText,
315            'testdox-html'    => $options->testdoxHtml,
316            'testdox-summary' => $options->testdoxSummary,
317        ] as $key => $path) {
318            if ($path !== null) {
319                $selectedSubscribers[$key] = $path;
320            }
321        }

323        if ($selectedSubscribers !== []) {
324            $subscribers = $configuration->subscribers;
325
326            foreach ($selectedSubscribers as $key => $path) {
327                if (!isset($subscribers[$key])) {
328                    printf('Subscriber "%s" is not registered — add it via ->subscriber() in crucible.php.' . PHP_EOL, $key);

330                    return 1;
331                }
332
333                $subscribers[$key]['params']['output'] = $path;
334            }
335
336            $subscriberRegistry = new SubscriberRegistry($subscribers);
337
338            try {
339                foreach (array_keys($selectedSubscribers) as $key) {
340                    $emitter->subscribe($subscriberRegistry->resolve($key));
341                }
342            } catch (Exception $e) {
343                print $e->getMessage() . PHP_EOL;
344
345                return 1;
346            }

350        // party's own format registered the same way in crucible.php)
351        // — --log-pdf=/--log-markdown= are convenience sugar for the
352        // same underlying "key:path" selection --report= takes, so
353        // there's exactly one dispatch mechanism under both spellings.

355
356        foreach ($options->report as $entry) {
357            [$key, $path] = array_pad(explode(':', $entry, 2), 2, null);
358
359            if ($key === null || $path === null || $key === '' || $path === '') {

362                return 1;
363            }

371
372        if ($options->logMarkdown !== null) {
373            $selections[] = ['markdown', $options->logMarkdown];

377        // be dropped without a word — refused instead, naming both.

379
380        foreach ($selections as [$format, $output]) {
381            $already = $selectedReportFormats[$format] ?? null;
382
383            if ($already !== null && $already !== $output) {
384                printf('Report format "%s" is asked for twice, at %s and at %s: one path per format.' . PHP_EOL, $format, $already, $output);
385
386                return 1;
387            }
388
389            $selectedReportFormats[$format] = $output;
390        }

392        if ($selectedReportFormats !== []) {
393            $reportFormats = $configuration->reportFormats;
394
395            foreach ($selectedReportFormats as $key => $path) {
396                if (!isset($reportFormats[$key])) {
397                    printf('Report format "%s" is not registered — add it via ->reportFormat() in crucible.php.' . PHP_EOL, $key);

399                    return 1;
400                }
401
402                $reportFormats[$key]['params']['output'] = $path;
403            }

406
407            try {
408                foreach (array_keys($selectedReportFormats) as $key) {
409                    $registry->resolve($key);
410                }
411            } catch (Exception $e) {
412                print $e->getMessage() . PHP_EOL;
413
414                return 1;
415            }
416
417            $emitter->subscribe(new GenericReportWriter($registry, array_keys($selectedReportFormats), $configuration->reportTitle, $options->reproducible));
418        }

420        $order = $configuration->executionOrder;
421
422        if ($options->orderBy !== null) {
423            $order = ExecutionOrder::tryFrom($options->orderBy);
424
425            if (!$order instanceof ExecutionOrder) {
426                printf('Unknown --order-by value "%s": use default, defects, duration, random, reverse or size.' . PHP_EOL, $options->orderBy);
427
428                return 1;
429            }

432        $cacheDirectory = $options->cacheDirectory ?? $configuration->cacheDirectory;
433        $cacheResult    = $options->cacheResult ?? $configuration->cacheResult;
434        $cacheFile      = $workingDirectory->absolute($cacheDirectory) . '/results.json';
435        $cache          = $cacheResult ? ResultCache::load($cacheFile) : new ResultCache();
436
437        $propertyFile = $workingDirectory->absolute($cacheDirectory) . '/property-failures.json';
438
439        if ($cacheResult) {

447        // for one with recording off is asking for nothing to happen —
448        // said out loud rather than silently ignored, as in the spec.

450            printf(
451                'Tests cannot be ordered by %s because recording of the test run history is disabled.' . PHP_EOL . PHP_EOL,
452                $order->value,

454        }

457            $seed = $options->randomOrderSeed ?? random_int(0, 2147483647);

464                ...array_map(static fn($suite): string => $suite->name, $configuration->testSuites),
465                ...array_map(static fn(VitestSuite $suite): string => $suite->name, $configuration->vitest),
466                ...array_map(static fn(TypeTestSuite $suite): string => $suite->name, $configuration->typeTests),
467            ];

470                if (!in_array($requested, $known, true)) {
471                    printf('The requested test suite "%s" is not configured.' . PHP_EOL, $requested);
472
473                    return 1;
474                }
475            }

479        // test file is parsed before a single one executes. The grid
480        // opens onto that silence rather than after it, so the frame is
481        // already up, ticking, saying what it is waiting for — and the

487            $configuration,
488            $workingDirectory,
489            $options->testsuite,
490            $options->excludeTestsuite,
491            $options->testSuffixes,
492        );

494        try {
495            $groups = $opened instanceof MapView ? $discover() : $this->discoveryMark()->sweepWhile($discover);

497            print $e->getMessage() . PHP_EOL;

502            // which claims tests ran and disagreed. A CI job acts on
503            // those differently: fix the tests, or fix the setup.
504            return 2;

508        // pest()->printer()->compact() declaration is known (D-066);
509        // nothing is emitted before the run itself. --view=<key> is
510        // the general selection mechanism (any registered progress
511        // view, third-party or built-in) and wins outright, being the
512        // most explicit; absent that, the existing precedence holds:

514        // testdox/teamcity/console remain mutually exclusive — exactly
515        // one view is ever subscribed.

518
519        // The default is the map wherever a screen can be redrawn, and the
520        // dot-per-test view wherever one cannot — a pipe, a CI log, a dumb
521        // terminal. {@see Capabilities::animation()} draws that line, and
522        // {@see MapView::isDefaultFor()} is the same question the banner
523        // asked earlier, so the two cannot disagree about what is coming.
524        $selectedView = match (true) {
525            $options->view !== null                      => $options->view,
526            $testdoxWanted                               => 'testdox',
527            $options->teamcity === true                  => 'teamcity',
528            Capabilities::animation(Runtime::terminal()) => MapView::DEFAULT_KEY,
529            default                                      => 'console',
530        };
531
532        $progressViews = $configuration->progressViews;
533        $viewRegistry  = new ProgressViewRegistry($progressViews);
534
535        if (!$viewRegistry->has($selectedView)) {
536            printf('Progress view "%s" is not registered — add it via ->progressView() in crucible.php.' . PHP_EOL, $selectedView);
537
538            return 1;
539        }
540
541        // These are not user-configured params — they are engine-computed
542        // from the command line the same run already resolves — so they are
543        // injected here, mirroring how --log-junit's path is injected into
544        // a subscriber's params. A view that does not declare one simply
545        // does not receive it, which is how a third-party view stays free
546        // of options it never agreed to.
547        $computed = $this->viewParams($options, $configuration, $workingDirectory->relative($loaded->path));
548        $declared = array_column($viewRegistry->describe($selectedView)->params, 'name');
549        $injected = [];
550
551        foreach ($computed as $name => $value) {
552            if ($value !== null && in_array($name, $declared, true)) {
553                $injected[$name] = $value;
554            }
555        }
556
557        // A view that names the configuration itself is not told it twice:
558        // printed here, the line would be covered by the frame that opens
559        // a moment later and left under the report as though it were part
560        // of the run's output.
561        if (!$options->noOutput && !in_array('configuration', $declared, true)) {
562            printf('Configuration: %s' . PHP_EOL . PHP_EOL, $loaded->path);
563        }
564
565        if ($injected !== []) {
566            $progressViews[$selectedView] = [
567                'class'  => $progressViews[$selectedView]['class'],
568                'params' => [...$progressViews[$selectedView]['params'], ...$injected],
569            ];
570            $viewRegistry = new ProgressViewRegistry($progressViews);
571        }
572
573        // --no-output means no output: the view is never subscribed, so
574        // nothing writes a character. Everything else about the run —
575        // the exit code, the log targets, the reports — is unchanged.
576        if (!$options->noOutput) {
577            try {
578                // The grid may already be on screen, holding the frame
579                // discovery ran on. Resolving a second one would open a
580                // second surface over the first and leave it behind.
581                $emitter->subscribe($opened ?? $viewRegistry->resolve($selectedView, $options->stderr ? STDERR : STDOUT));
582            } catch (Exception $e) {
583                print $e->getMessage() . PHP_EOL;
584
585                return 1;
586            }
587        }
588
589        // --log-teamcity is the teamcity view sent to a file instead of the
590        // terminal, resolved through the same registry entry --teamcity
591        // uses. It does not replace the progress output, and --no-output
592        // does not silence it: it is a log target, not output.
593        if ($options->logTeamcity !== null) {
594            $stream = fopen($options->logTeamcity, 'w');
595
596            if ($stream === false) {
597                printf('Cannot open %s for the TeamCity log.' . PHP_EOL, $options->logTeamcity);
598
599                return 1;
600            }
601
602            try {
603                $emitter->subscribe($viewRegistry->resolve('teamcity', $stream));
604            } catch (Exception $e) {
605                print $e->getMessage() . PHP_EOL;
606
607                return 1;
608            }
609        }
610
611        // Subscribed after the progress printer so its block lands below
612        // the summary rather than interleaved with the dots.
613        if ($options->profile) {
614            $emitter->subscribe(new ProfileReporter(STDOUT, new Style($this->colorsEnabled($options, $configuration))));
615        }
616
617        // The Vitest suites this run folds in (D-079). Full by default;
618        // narrowed to the change set under impact selection below, where

898            && $options->excludeGroups === []

900            && $options->assignee === null
901            && $options->issue === null

903            && $options->testsuite === null

906            && $options->shard === null

946            $configuration,
947            stopOnDefect: $options->stopOnDefect,
948            stopOnError: $options->stopOnError,
949            stopOnFailure: $options->stopOnFailure,
950            stopOnRisky: $options->stopOnRisky,
951            stopOnSkipped: $options->stopOnSkipped,
952            stopOnIncomplete: $options->stopOnIncomplete,
953            stopOnDeprecation: $options->stopOnDeprecation,
954            stopOnNotice: $options->stopOnNotice,
955            stopOnWarning: $options->stopOnWarning,
956            projectDirectories: $this->projectDirectories($configuration, $workingDirectory),

1112                seed: $seed,

1172        // must not be able to change a verdict, so the work runs either
1173        // way and only what it wrote is discarded.
1174        if ($options->noOutput) {
1175            ob_start();
1176        }
1177
1178        $postRun->printIssueDisplays($options, $issueLog, $outcomeLog);
1179
1180        $breaches = $postRun->thresholdBreaches($configuration, $issueLog);
1181
1182        if ($coverageFloorBreach !== null) {
1183            $breaches[] = $coverageFloorBreach;
1184        }
1185
1186        foreach ($breaches as $breach) {
1187            print $breach . PHP_EOL;
1188        }
1189
1190        if ($snapshotLog->created() > 0 || $snapshotLog->updated() > 0) {
1191            printf('Snapshots: %d created, %d updated.' . PHP_EOL, $snapshotLog->created(), $snapshotLog->updated());
1192        }
1193
1194        $postRun->printFlakiness($flakiness);
1195        $postRun->printFailuresOutsideDiff($flakiness, $cache, $configuration, $options, $workingDirectory, $cacheDirectory);
1196
1197        // Run-scoped checks (command gates) already ran inside the run
1198        // bracket and rode the stream; here we print their separate
1199        // tally and let a failing one vote the exit code with its reason.
1200        $checkFailed = $postRun->printChecks($checkLog);
1201
1202        if ($options->noOutput) {
1203            ob_end_clean();
1204        }
1205
1206        $exitCode = $postRun->exitCode($summary, $configuration, $options, $flakiness, $issueLog);
1207
1208        return $breaches !== [] || $checkFailed ? max(1, $exitCode) : $exitCode;
1209    }

1228    /**

1237    {
1238        $prefixes = [];
1239
1240        foreach ($configuration->testSuites as $suite) {
1241            foreach ($suite->directories as $directory) {
1242                $prefixes[] = $workingDirectory->absolute($directory) . '/';
1243            }

1894        }
1895
1896        // tests and test-ids: the same list, named for a human or for a
1897        // --run-test-id round trip.
1898        foreach ($groups as $group) {
1899            foreach ($group->tests as $test) {
1900                print $what === 'test-ids'
1901                    ? $test->id->toString() . PHP_EOL
1902                    : ' - ' . $group->name . '::' . $test->id->name . PHP_EOL;
1903            }
1904        }
1905    }

1907    /**
1908     * Instantiates the classes named by --extension. They take no
1909     * constructor arguments: a plugin that needs configuring is registered
1910     * with it in `crucible.php`, where the values have types.
1911     *
1912     * @return list<Extension>|non-empty-string the extensions, or an error message
1913     */
1914    private function extensionsFromCli(CliOptions $options): array|string
1915    {
1916        $extensions = [];
1917
1918        foreach ($options->extensionClasses as $class) {
1919            if (!class_exists($class)) {
1920                return sprintf('Extension class "%s" does not exist.', $class);
1921            }
1922
1923            $extension = new $class();
1924
1925            if (!$extension instanceof Extension) {
1926                return sprintf('Extension class "%s" does not implement %s.', $class, Extension::class);
1927            }
1928
1929            $extensions[] = $extension;
1930        }
1931
1932        return $extensions;
1933    }

1935    /**
1936     * The configuration values this command line overrides, gathered once
1937     * so the parent, the workers, and the runner all resolve them the
1938     * same way (D-019: the spec's spellings win over the file, and a
1939     * parallel run must behave like its sequential twin).
1940     *
1941     */
1942    private function overrides(CliOptions $options, WorkingDirectory $workingDirectory): Overrides
1943    {
1944        // Which baseline this run reads, and whether it reads one at all.
1945        // --generate-baseline is a *generate*: the run must see every
1946        // deprecation, so it reads nothing and writes fresh, where
1947        // --update-deprecations-baseline merges into the one it read.
1948        $baseline = $options->useBaseline ?? $options->generateBaseline;
1949
1950        return new Overrides(
1951            bootstrap: $options->bootstrap,
1952            backupGlobals: $options->backupGlobals,
1953            backupStaticProperties: $options->backupStaticProperties,
1954            beStrictAboutChangesToGlobalState: $options->strictGlobalState,
1955            requireCoverageMetadata: $options->requireCoverageMetadata,
1956            reportUselessTests: $options->reportUselessTests ? null : false,
1957            diffContext: $options->diffContext,
1958            baselineFile: $baseline === null
1959                ? null
1960                : $workingDirectory->absolute($baseline),
1961            ignoreBaseline: $options->ignoreBaseline || $options->generateBaseline !== null
1962                ? true
1963                : null,
1964            resolveDependencies: $options->resolveDependencies,
1965            coverageTargeting: $options->disableCoverageTargeting ? false : null,
1966            coverageFilter: $options->coverageFilter,
1967            repeat: $options->repeat,
1968            disallowTestOutput: $options->disallowTestOutput ? true : null,
1969            enforceTimeLimit: $options->enforceTimeLimit ? true : null,

1971        );
1972    }

1974    /**
1975     * @param list<non-empty-string> $includePaths the spec's --include-path
1976     */
1977    private function applyPhpSettings(Configuration $configuration, WorkingDirectory $workingDirectory, Overrides $overrides = new Overrides(), array $includePaths = []): void
1978    {
1979        // Before the bootstrap: a suite that relies on include_path expects
1980        // it set by the time its own loader runs.
1981        if ($includePaths !== []) {
1982            $absolute = array_map(
1983                $workingDirectory->absolute(...),
1984                $includePaths,
1985            );
1986
1987            set_include_path(implode(PATH_SEPARATOR, [...$absolute, get_include_path()]));
1988        }
1989
1990        // Run-wide display state, set before any test can render a failure.
1991        Differ::context($overrides->diffContext);
1992
1993        $bootstrap = $overrides->bootstrap ?? $configuration->bootstrap;
1994
1995        if ($bootstrap !== null) {
1996            if (!str_starts_with($bootstrap, '/')) {
1997                $bootstrap = $workingDirectory->path . '/' . $bootstrap;
1998            }
1999
2000            if (is_file($bootstrap)) {
2001                require_once $bootstrap;
2002            }
2003        }
2004
2005        foreach ($configuration->php->ini as $name => $value) {
2006            ini_set($name, $value);
2007        }
2008
2009        foreach ($configuration->php->env as $name => $value) {
2010            // All three channels, like the spec's <env> handler:
2011            // frameworks resolve env from $_SERVER first, and a parent
2012            // process (artisan test) may have exported conflicting
2013            // values there.
2014            $this->exportEnv($name, $value);
2015        }
2016
2017        foreach ($configuration->php->constants as $name => $value) {
2018            if (!defined($name)) {
2019                define($name, $value);
2020            }
2021        }
2022    }
