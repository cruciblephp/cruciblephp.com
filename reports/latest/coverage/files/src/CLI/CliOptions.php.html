369    /**
370     * Whether anything on this command line asks for a coverage driver.
371     * One question with one answer, so a new report target cannot be
372     * wired to a writer and then silently never collected for.
373     */
374    public function wantsCoverage(): bool
375    {
376        return $this->coverage
377            || $this->coverageBranch
378            || $this->pathCoverage
379            || $this->coverageClover !== null
380            || $this->coverageCobertura !== null
381            || $this->coverageHtml !== null
382            || $this->coverageOpenClover !== null
383            || $this->coveragePhp !== null
384            || $this->coverageCrap4j !== null
385            || $this->coverageXml !== null
386            || $this->coverageText !== null
387            // A floor is a request to measure: you cannot ask for one and
388            // not want the measurement it is a floor on.
389            || $this->minCoverage !== null;
390    }

392    /**
393     * A switch with both spellings present: --fail-on-x turns the policy
394     * on, --no-fail-on-x turns it off, neither leaves it null so the
395     * configuration decides.
396     *
397     * @param array<string, true> $flags
398     */
399    private static function tristate(array $flags, string $on, string $off): ?bool
400    {
401        return match (true) {
402            isset($flags[$on])  => true,
403            isset($flags[$off]) => false,
404            default             => null,
405        };
406    }

408    /**
409     * Every name the parser accepts: options, their aliases, commands.
410     *
411     * ⚠ Read from the same constants the parse loop reads, so a
412     * generated completion script cannot drift from what the binary
413     * takes. `--worker` is the supervisor's private handshake with its
414     * own children and is not a thing a user types.
415     *
416     * @return list<string>
417     */
418    public static function accepted(): array
419    {
420        $names = [];
421
422        foreach ([self::FLAGS, self::VALUED, self::REPEATABLE, self::COMMANDS] as $set) {
423            foreach ($set as $name) {
424                $names[$name] = true;
425            }
426        }
427
428        foreach (array_keys(self::ALIASES) as $name) {
429            $names[$name] = true;
430        }
431
432        // Parsed by hand because their value is optional.
433        foreach (['--changed', '--colors', '--coverage-text'] as $byHand) {
434            $names[$byHand] = true;
435        }
436
437        unset($names['--worker']);
438
439        return array_keys($names);
440    }

442    /**
443     * @param list<string> $argv
444     *
445     * @return self|non-empty-string the options, or an error message
446     */
447    public static function fromArgv(array $argv): self|string
448    {
449        /** @var array<string, true> $flags */
450        $flags = [];
451
452        /** @var array<string, non-empty-string> $values */
453        $values = [];
454
455        /** @var array<string, list<non-empty-string>> $lists */
456        $lists = [];
457
458        /** @var ?('auto'|'always'|'never') $colors */
459        $colors = null;
460
461        /** The command seen, and the bare word following it. */
462        $command         = null;
463        $commandArgument = null;
464
465        /** @var ?non-empty-string $changed */
466        $changed = null;
467
468        /** @var ?string $coverageText '' = the terminal, non-empty = that file */
469        $coverageText = null;
470
471        // ⚠ The alias is resolved on the NAME, not on the whole
472        // argument. Every alias used to be a flag, so `--alias=value`
473        // never came up; the spec's spellings for valued options can be
474        // written that way, and matching the whole token would have let
475        // them through unresolved to be rejected as unknown.
476        $arguments = array_map(
477            static function (string $argument): string {
478                if (str_starts_with($argument, '--') && str_contains($argument, '=')) {
479                    [$name, $value] = explode('=', $argument, 2);
480
481                    return (self::ALIASES[$name] ?? $name) . '=' . $value;
482                }
483
484                return self::ALIASES[$argument] ?? $argument;
485            },
486            array_slice($argv, 1),
487        );
488
489        for ($i = 0; isset($arguments[$i]); $i++) {
490            $argument = $arguments[$i];
491            $name     = $argument;
492            $inline   = null;
493
494            if (str_starts_with($argument, '--') && str_contains($argument, '=')) {
495                [$name, $inline] = explode('=', $argument, 2);
496            }
497
498            if (in_array($name, self::FLAGS, true)) {
499                if ($inline !== null) {
500                    return sprintf('Option %s does not take a value.', $name);
501                }
502
503                $flags[$name] = true;
504
505                continue;
506            }
507
508            if (in_array($name, self::VALUED, true) || in_array($name, self::REPEATABLE, true)) {
509                $value = $inline;
510
511                if ($value === null && isset($arguments[$i + 1])) {
512                    $value = $arguments[$i + 1];
513                    $i++;
514                }
515
516                if ($value === null || $value === '') {
517                    return sprintf('Option %s requires a value.', $name);
518                }
519
520                if (in_array($name, self::REPEATABLE, true)) {
521                    foreach (explode(',', $value) as $item) {
522                        if ($item !== '') {
523                            $lists[$name][] = $item;
524                        }
525                    }
526
527                    continue;
528                }
529
530                // Last occurrence wins, matching the spec's CLI.
531                $values[$name] = $value;
532
533                continue;
534            }
535
536            if (in_array($name, self::COMMANDS, true)) {
537                $flags[$name] = true;
538                $command      = $name;
539
540                continue;
541            }
542
543            // ⚠ Only after a command, and only the first one. A bare word
544            // with no command before it is still an error, so a mistyped
545            // option cannot be silently swallowed as an argument.
546            if ($command !== null && $commandArgument === null && !str_starts_with($name, '-')) {
547                $commandArgument = $name;
548
549                continue;
550            }
551
552            // --changed takes an optional value: bare means HEAD (the
553            // working tree's uncommitted changes), and like --colors
554            // it never consumes the next argument (only the = form).
555            if ($name === '--changed') {
556                $reference = $inline ?? 'HEAD';
557
558                if ($reference === '') {
559                    return 'Option --changed expects a git reference.';
560                }
561
562                $changed = $reference;
563
564                continue;
565            }
566
567            // --coverage-text takes an optional value, the spec spelling
568            // it --coverage-text=<file> for exactly that reason: bare
569            // means the terminal, so it never consumes the next argument.
570            if ($name === '--coverage-text') {
571                $coverageText = $inline ?? '';
572
573                continue;
574            }
575
576            // --colors takes an optional value: bare means auto, and
577            // it never consumes the next argument (only the = form).
578            if ($name === '--colors') {
579                $when = $inline ?? 'auto';
580
581                if (!in_array($when, ['auto', 'always', 'never'], true)) {
582                    return sprintf('Option --colors expects auto, always or never, got "%s".', $when);
583                }
584
585                $colors = $when;
586
587                continue;
588            }
589
590            return str_starts_with($argument, '-')
591                ? sprintf('Unknown option "%s". Run crucible --help for the list of options.', $argument)
592                : sprintf('Unrecognized argument "%s".', $argument);
593        }
594
595        foreach (['--random-order-seed', '--parallel', '--retries', '--retry', '--rounds', '--diff-context'] as $numeric) {
596            if (isset($values[$numeric]) && !ctype_digit($values[$numeric])) {
597                return sprintf('Option %s expects a non-negative integer, got "%s".', $numeric, $values[$numeric]);
598            }
599        }
600
601        // A percentage, not an integer: --min=99.5 is a real ask, and the
602        // number the gate compares is the one the report prints.
603        if (isset($values['--min'])) {
604            if (!is_numeric($values['--min'])) {
605                return sprintf('Option --min expects a percentage between 0 and 100, got "%s".', $values['--min']);
606            }
607
608            $minimum = (float) $values['--min'];
609
610            if ($minimum < 0.0 || $minimum > 100.0) {
611                return sprintf('Option --min expects a percentage between 0 and 100, got "%s".', $values['--min']);
612            }
613
614            // Refused rather than resolved: --no-coverage says measure
615            // nothing, --min says fail below a measurement. Letting one
616            // win silently would drop a CI gate and report success.
617            if (isset($flags['--no-coverage'])) {
618                return 'Options --min and --no-coverage are mutually exclusive.';
619            }
620        }
621
622        if (isset($flags['--cache-result'], $flags['--do-not-cache-result'])) {
623            return 'Options --cache-result and --do-not-cache-result are mutually exclusive.';
624        }
625
626        $switches = [
627            'deprecation', 'incomplete', 'notice', 'risky', 'skipped', 'warning',
628            'direct', 'indirect', 'self',
629            'empty-test-suite',
630            'phpunit-deprecation', 'phpunit-notice', 'phpunit-warning',
631        ];
632
633        foreach ($switches as $issue) {
634            if (isset($flags['--fail-on-' . $issue], $flags['--no-fail-on-' . $issue])) {
635                return sprintf('Options --fail-on-%1$s and --no-fail-on-%1$s are mutually exclusive.', $issue);
636            }
637        }
638
639        // --fail-on-all-issues turns on every policy the run has, which is
640        // what the spec means by "all". A negation still wins: the point of
641        // pairing it with --no-fail-on-skipped is "everything but that".
642        if (isset($flags['--fail-on-all-issues'])) {
643            foreach ($switches as $issue) {
644                if (!isset($flags['--no-fail-on-' . $issue])) {
645                    $flags['--fail-on-' . $issue] = true;
646                }
647            }
648        }
649
650        if (isset($flags['--php-advisory'], $flags['--no-php-advisory'])) {
651            return 'Options --php-advisory and --no-php-advisory are mutually exclusive.';
652        }
653
654        if (isset($flags['--resolve-dependencies'], $flags['--ignore-dependencies'])) {
655            return 'Options --resolve-dependencies and --ignore-dependencies are mutually exclusive.';
656        }
657
658        if (isset($values['--generate-baseline'], $values['--use-baseline'])) {
659            return 'Options --generate-baseline and --use-baseline are mutually exclusive.';
660        }
661
662        if (isset($values['--retries'], $values['--retry'])) {
663            return 'Options --retries and --retry are mutually exclusive.';
664        }
665
666        // Repeating and retrying answer opposite questions — "run it
667        // again regardless" against "run it again only if it failed" —
668        // so the spec refuses the pair rather than ordering them.
669        if (isset($values['--repeat']) && (isset($values['--retries']) || isset($values['--retry']))) {
670            return 'Options --repeat and --retries/--retry are mutually exclusive.';
671        }
672
673        // The oracle's --retry counts *total* attempts ("attempt each test up
674        // to N times"); Crucible's --retries counts the extra ones, following
675        // nextest (D-043), which is also what ->retries() and #[Retry] mean.
676        // Both spellings stay exact by converting here rather than by moving
677        // either vocabulary onto the other's.
678        $retries = match (true) {
679            isset($values['--retries']) => max(0, (int) $values['--retries']),
680            isset($values['--retry'])   => max(0, (int) $values['--retry'] - 1),
681            default                     => null,
682        };
683
684        // --display-all-issues is the spec's "every --display-* at once",
685        // resolved here so the command sees one already-answered question
686        // per kind rather than an "or the all switch" at every use.
687        $displayAll = isset($flags['--display-all-issues']);
688
689        // --columns takes a number or the word "max", which the spec uses
690        // for "as wide as the terminal". Without a terminal to ask, max is
691        // the widest line worth writing to a pipe.
692        $columns = null;
693
694        if (isset($values['--columns'])) {
695            if ($values['--columns'] === 'max') {
696                $columns = 120;
697            } elseif (ctype_digit($values['--columns']) && (int) $values['--columns'] > 0) {
698                $columns = (int) $values['--columns'];
699            } else {
700                return sprintf('Option --columns expects a positive integer or "max", got "%s".', $values['--columns']);
701            }
702        }
703
704        // A single --include-path may carry several entries, separated the
705        // way PHP itself separates them.
706        $includePaths = [];
707
708        foreach (isset($values['--include-path']) ? explode(PATH_SEPARATOR, $values['--include-path']) : [] as $path) {
709            if ($path !== '') {
710                $includePaths[] = $path;
711            }
712        }
713
714        // The listing family is one question — what to print instead of
715        // running — so it resolves to one value rather than six booleans
716        // the command would have to re-derive an order for.
717        $list = null;
718
719        foreach (['tests', 'groups', 'suites', 'test-files', 'test-ids', 'tests-xml'] as $what) {
720            if (isset($flags['--list-' . $what])) {
721                if ($list !== null) {
722                    return sprintf('Options --list-%s and --list-%s are mutually exclusive.', $list, $what);
723                }
724
725                $list = $what;
726            }
727        }
728
729        // The spec's --no-coverage suppresses coverage for the whole run, so
730        // a CI line can append it to a command that already asks for reports.
731        // --no-logging and --no-extensions are the same shape over the log
732        // targets and the plugins: a blanket off switch appended to a command
733        // that already asks for them.
734        $noCoverage   = isset($flags['--no-coverage']);
735        $noLogging    = isset($flags['--no-logging']);
736        $noExtensions = isset($flags['--no-extensions']);
737
738        $cacheResult = null;
739
740        if (isset($flags['--cache-result'])) {
741            $cacheResult = true;
742        } elseif (isset($flags['--do-not-cache-result'])) {
743            $cacheResult = false;
744        }
745
746        return new self(
747            version: isset($flags['--version']),
748            help: isset($flags['--help']) || isset($flags['-h']),
749            init: isset($flags['--init']),
750            worker: isset($flags['--worker']),
751            migrateConfig: isset($flags['migrate-config']),
752            phpstanInit: isset($flags['phpstan-init']),
753            lintInline: isset($flags['lint-inline']),
754            mutationIndex: isset($flags['mutation-index']),
755            mutate: isset($flags['mutate']),
756            compatCheck: isset($flags['compat-check']),
757            autoFix: isset($flags['--auto-fix']),
758            revert: isset($flags['--revert']),
759            extensions: isset($flags['extensions']),
760            manual: isset($flags['manual']),
761            completion: isset($flags['completion']),
762            argument: $commandArgument,
763            preview: isset($flags['--preview']),
764            json: isset($flags['--json']),
765            key: $values['--key'] ?? null,
766            out: $values['--out'] ?? null,
767            view: $values['--view'] ?? null,
768            configuration: $values['--configuration'] ?? null,
769            logEventsJson: $noLogging ? null : ($values['--log-events-json'] ?? null),
770            logJunit: $noLogging ? null : ($values['--log-junit'] ?? null),
771            logMarkdown: $noLogging ? null : ($values['--log-markdown'] ?? null),
772            logPdf: $noLogging ? null : ($values['--log-pdf'] ?? null),
773            orderBy: $values['--order-by'] ?? null,
774            randomOrderSeed: isset($values['--random-order-seed']) ? (int) $values['--random-order-seed'] : null,
775            parallel: isset($values['--parallel']) ? (int) $values['--parallel'] : null,
776            filter: $values['--filter'] ?? null,

782            stopOnDefect: isset($flags['--stop-on-defect']) ? true : null,
783            stopOnError: isset($flags['--stop-on-error']) ? true : null,
784            stopOnFailure: isset($flags['--stop-on-failure']) ? true : null,
785            stopOnRisky: isset($flags['--stop-on-risky']) ? true : null,

789            stopOnNotice: isset($flags['--stop-on-notice']) ? true : null,
790            stopOnWarning: isset($flags['--stop-on-warning']) ? true : null,

792            failOnIncomplete: self::tristate($flags, '--fail-on-incomplete', '--no-fail-on-incomplete'),
