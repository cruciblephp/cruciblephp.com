59    /** @param list<string> $argv */
60    public function run(array $argv): int
61    {
62        $options = CliOptions::fromArgv($argv);
63
64        if (is_string($options)) {
65            print $options . PHP_EOL;
66
67            return 1;
68        }
69
70        // Worker mode prints nothing: stdout is the NDJSON protocol.
71        if ($options->worker) {
72            return (new RunTestsCommand())->worker();
73        }
74
75        $wantsBanner = $this->wantsBanner($options);
76        $banner      = $wantsBanner ? $this->startBanner($options->version) : null;
77
78        if ($options->version) {
79            return 0;
80        }
81
82        // The spec's two version gates. --atleast-version is a CI guard:
83        // it answers with an exit code, not a message. --check-version asks
84        // whether a newer release exists, which Crucible cannot answer —
85        // there is no update channel and it does not phone home — so it
86        // says so rather than pretending the running version is current.
87        if ($options->atLeastVersion !== null) {
88            $banner?->settle();
89
90            return version_compare(Version::NUMBER, $options->atLeastVersion, '>=') ? 0 : 1;
91        }
92
93        if ($options->checkVersion) {
94            $banner?->settle();
95
96            print 'Crucible has no update channel and does not check for new versions.' . PHP_EOL;
97
98            return 0;
99        }
100
101        // The spec's php.ini audit: advisory about the environment, so
102        // it answers and exits rather than running anything. Exit 1
103        // when a setting is not as recommended, which is what makes it
104        // usable as a CI gate on the machine rather than the code.
105        if ($options->checkPhpConfiguration) {
106            $banner?->settle();
107
108            return $this->printPhpConfiguration();
109        }
110
111        $banner?->settle();
112
113        // The separator belongs to the banner, so it goes when the banner
114        // does — a suppressed banner must not leave its blank line behind.
115        if ($wantsBanner) {
116            print PHP_EOL;
117        }
118
119        if ($options->help) {
120            $this->help();
121
122            return 0;
123        }
124
125        $cwd = getcwd();
126
127        if ($cwd === false) {
128            print 'Cannot determine the current working directory.' . PHP_EOL;
129
130            return 1;
131        }
132
133        $workingDirectory = new WorkingDirectory($cwd);
134
135        // ⚠ The two ways a console component ends a command. Uncaught,
136        // Ctrl-C at a prompt printed a stack trace over the half-drawn
137        // frame it had just cancelled.
138        try {
139            return $this->dispatch($options, $argv, $workingDirectory);
140        } catch (CancelledException) {
141            print 'Cancelled.' . PHP_EOL;
142
143            return 130;
144        } catch (NonInteractiveException $e) {
145            print $e->getMessage() . PHP_EOL;
146
147            return 1;
148        }
149    }

151    /**
152     * @param list<string> $argv
153     */
154    private function dispatch(CliOptions $options, array $argv, WorkingDirectory $workingDirectory): int
155    {
156        // Parsing the source scope for method structure is the cost the
157        // Crap4J, XML, and per-method HTML reports pay. Warming it is
158        // paying that cost deliberately, before the run that needs it.
159        if ($options->warmCoverageCache) {
160            return (new WarmCoverageCacheCommand())->execute($options, $workingDirectory);
161        }
162
163        if ($options->validateConfiguration) {
164            return (new ValidateConfigCommand())->execute($options, $workingDirectory);
165        }
166
167        if ($options->init) {
168            return (new InitCommand())->execute($workingDirectory);
169        }
170
171        if ($options->migrateConfig) {
172            return (new MigrateConfigCommand())->execute($workingDirectory);
173        }
174
175        if ($options->phpstanInit) {
176            return (new PhpstanInitCommand())->execute($options, $workingDirectory);
177        }
178
179        if ($options->lintInline) {
180            return (new LintInlineCommand())->execute($options, $workingDirectory);
181        }
182
183        if ($options->mutationIndex) {
184            return (new MutationCommand())->index($options, $workingDirectory);
185        }
186
187        if ($options->mutate) {
188            return (new MutationCommand())->mutate($options, $argv, $workingDirectory);
189        }
190
191        if ($options->flakes) {
192            return (new FlakesCommand())->execute($options, $argv, $workingDirectory);
193        }
194
195        if ($options->compatCheck) {
196            return (new CompatCheckCommand())->execute($options, $argv, $workingDirectory);
197        }
198
199        if ($options->extensions) {
200            return (new ExtensionsCommand())->execute($options, $argv, $workingDirectory);
201        }
202
203        if ($options->completion) {
204            return (new CompletionCommand())->execute($options->argument);
205        }
206
207        if ($options->manual) {
208            return (new ManualCommand())->execute($options->out);
209        }
210
211        if ($options->watch) {
212            return (new WatchCommand())->execute($options, $argv, $workingDirectory);
213        }
214
215        return (new RunTestsCommand())->execute($options, $argv, $workingDirectory);
216    }

218    /**
219     * @return int<0, 1>
220     */
221    private function printPhpConfiguration(): int
222    {
223        $results = PhpConfigurationCheck::results();
224        $width   = 0;
225
226        foreach ($results as $result) {
227            $width = max($width, strlen($result['name'] . ' = ' . $result['expected']));
228        }
229
230        print 'Checking whether PHP is configured for development.' . PHP_EOL . PHP_EOL;
231
232        $exit = 0;
233
234        foreach ($results as $result) {
235            if (!$result['ok']) {
236                $exit = 1;
237            }
238
239            printf(
240                '%-' . $width . "s ... %s" . PHP_EOL,
241                $result['name'] . ' = ' . $result['expected'],
242                $result['ok'] ? 'ok' : sprintf('not ok (%s)', $result['actual'] === '' ? '(empty)' : $result['actual']),
243            );
244        }
245
246        return $exit;
247    }

249    /**
250     * The banner, branded where a human is looking and plain where a log is.
251     *
252     * The colourless form is byte-identical to what this printed before —
253     * it goes to pipes, CI and captured output, and a banner is not worth
254     * changing those. Colour is decided by Capabilities, which already
255     * honours NO_COLOR, FORCE_COLOR and the --colors override, so this adds
256     * no new rule about when to use it.
257     *
258     * The mark is the project's own: a diamond and the wordmark swept by
259     * {@see Splash}, the version in the accent, the author dimmed.
260     *
261     * ⚠ The author is `dim`, not a pinned grey. It was Brand::slate()
262     * (#4A4A4A), which is the unlit state of an animation against a known
263     * background — as a byline on a dark terminal it is unreadable. `dim`
264     * is the terminal's own attenuation of its own foreground, so it stays
265     * legible whatever the theme.
266     *
267     * ⚠ The sweep is forked, not slept through. `--version` is the one
268     * case with nothing behind it to overlap, so there it runs
269     * synchronously and IS the command; everywhere else it sweeps while
270     * the configuration loads and settles when the caller is ready to
271     * print. A banner that costs a real run half a second is a banner
272     * someone turns off.
273     */
274    /**
275     * Whether this run gets a banner line of its own.
276     *
277     * ⚠ Not when the map is coming. Its top border already carries the
278     * mark, the version and the byline, so printing them again above it
279     * is the same identity twice — and on the alternate screen the line
280     * above scrolls away unread anyway.
281     */
282    private function wantsBanner(CliOptions $options): bool
283    {
284        if ($options->noOutput) {
285            return false;
286        }
287
288        // ⚠ A completion script is evaluated by the shell, not read by a
289        // person: `eval "$(crucible completion bash)"` would try to run
290        // the banner as shell and fail on the first word.
291        if ($options->completion) {
292            return false;
293        }
294
295        // --version is the banner, whatever view a run would have used.
296        if ($options->version) {
297            return true;
298        }
299
300        return !MapView::isDefaultFor(
301            $options->view,
302            $options->testdox === true,
303            $options->teamcity === true,
304            Runtime::terminal(),
305        );
306    }

308    private function startBanner(bool $versionOnly): ?Splash
309    {
310        $terminal = Runtime::terminal();
311
312        if (!Capabilities::color($terminal)) {
313            printf('crucible %s by %s.' . PHP_EOL, Version::NUMBER, Version::AUTHOR);
314
315            return null;
316        }
317
318        $splash           = new Splash($terminal, '◆ crucible');
319        $splash->interval = self::BANNER_FRAME_MS;
320        $splash->suffix   = sprintf(
321            ' %s%s%s %sby %s.%s',
322            $this->paint(Theme::accent()),
323            Version::NUMBER,
324            self::RESET,
325            Style::none()->dim()->toAnsi(),
326            Version::AUTHOR,
327            self::RESET,
328        );
329
330        if ($versionOnly) {
331            $splash->sweeps = 1;
332            $splash->render();
333
334            return null;
335        }
336
337        $splash->begin();
338
339        return $splash;
340    }
