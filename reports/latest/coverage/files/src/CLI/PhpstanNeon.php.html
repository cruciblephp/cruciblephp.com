28/**
29 * Pure neon wiring for `crucible phpstan-init` (D-049): build a fresh
30 * phpstan.neon from the project's own configuration, or insert the
31 * one includes line into an existing file. Editing is deliberately
32 * conservative — neon is indentation-sensitive and full of user
33 * comments, so a full parse-and-rewrite would need a dependency and
34 * would destroy formatting on round-trip. A minimal line insertion
35 * covers the real shapes; anything structurally surprising returns
36 * null, and the caller prints the line for the user to place — the
37 * migrate-config principle: never silently lossy.
38 */
39final readonly class PhpstanNeon
40{
41    /**
42     * A fresh phpstan.neon: the extension wired, the paths taken
43     * from what the crucible configuration already declares.
44     *
45     * @param ?non-empty-string      $include null = the installer wires it
46     * @param list<non-empty-string> $paths
47     */
48    public static function create(?string $include, array $paths): string
49    {
50        $lines = [
51            ...self::includes($include === null ? [] : [$include]),
52            'parameters:',
53            '    # Crucible\'s own position: max, honestly earned. Lower it to',
54            '    # get started on an existing suite, raise it back as the',
55            '    # annotations catch up.',
56            '    level: max',
57            '    paths:',
58        ];
59
60        foreach ($paths as $path) {
61            $lines[] = '        - ' . $path;
62        }
63
64        return implode("\n", $lines) . "\n";
65    }
66
67    /**
68     * A configuration Crucible writes for its own analysis — type tests
69     * (D-130), lint-inline's shadow files (D-052) — never for a person to
70     * edit: the extension include, then each parameter, a list parameter
71     * as a neon list. An empty list is left out; PHPStan reads an empty
72     * `paths:` as null and refuses it.
73     *
74     * @param list<string>                                  $includes   none when phpstan/extension-installer loads the extension
75     * @param array<non-empty-string, string|list<string>> $parameters
76     * @param list<class-string>                            $rules      rules registered on their own, outside any include
77     */
78    public static function analysis(array $includes, array $parameters, array $rules = []): string
79    {
80        $lines = [...self::includes($includes), 'parameters:'];
81
82        foreach ($parameters as $name => $value) {
83            if (!is_array($value)) {
84                $lines[] = '    ' . $name . ': ' . $value;
85
86                continue;
87            }
88
89            if ($value !== []) {
90                $lines[] = '    ' . $name . ':';
91
92                foreach ($value as $item) {
93                    $lines[] = '        - ' . $item;
94                }
95            }
96        }
97
98        if ($rules !== []) {
99            $lines = [...$lines, '', 'rules:'];
100
101            foreach ($rules as $rule) {
102                $lines[] = '    - ' . $rule;
103            }
104        }
105
106        return implode("\n", $lines) . "\n";
107    }

109    /**
110     * The include block. Crucible's extension is left out when
111     * phpstan/extension-installer loads it (D-127) — an include written
112     * as well loads it twice.
113     *
114     * @param list<string> $includes
115     *
116     * @return list<string>
117     */
118    private static function includes(array $includes): array
119    {
120        if ($includes === []) {
121            return [];
122        }
123
124        $lines = ['includes:'];
125
126        foreach ($includes as $include) {
127            $lines[] = '    - ' . $include;
128        }
129
130        return [...$lines, ''];
131    }

136    public static function alreadyWired(string $existing, string $include): bool
137    {

141    /**
142     * The existing file with the includes entry inserted, or null
143     * when the file's shape cannot be edited safely.
144     *
145     * @param non-empty-string $include
146     */
147    public static function wire(string $existing, string $include): ?string
148    {
149        $lines = explode("\n", $existing);
150
151        foreach ($lines as $index => $line) {
152            // The inline list form `includes: [...]` — editable only
153            // by rewriting the expression; hands off.
154            if (preg_match('/^includes:\s*\[/', $line) === 1) {
155                return null;
156            }
157
158            if (preg_match('/^includes:\s*(#.*)?$/', $line) !== 1) {
159                continue;
160            }
161
162            // Match the indentation of the first existing entry so
163            // the insertion reads as if it was always there.
164            $indent = '    ';
165
166            if (isset($lines[$index + 1]) && preg_match('/^(\s+)-\s/', $lines[$index + 1], $matches) === 1) {
167                $indent = $matches[1];
168            }
169
170            array_splice($lines, $index + 1, 0, [$indent . '- ' . $include]);
171
172            return implode("\n", $lines);
173        }
174
175        // No includes section: prepend one. Top-level key order is
176        // free in neon, and a leading section above existing comments
177        // parses fine.
178        $prefix = "includes:\n    - " . $include . "\n\n";
179
180        return $prefix . $existing . (str_ends_with($existing, "\n") || $existing === '' ? '' : "\n");
181    }

183    /**
184     * Whether phpstan/extension-installer is installed and has not been
185     * told to skip Crucible (its `ignore` list in the project's
186     * composer.json). When it has, the installer loads extension.neon
187     * (D-127), and a configuration that includes it as well is refused
188     * by PHPStan: a file included twice.
189     *
190     * The working directory's vendor/, not Composer\InstalledVersions:
191     * that answers for the autoloader this process started from, which
192     * is the project's only when Crucible runs from the project's own
193     * vendor/bin. The question here is about the project being analysed.
194     */
195    public static function installerLoadsExtension(WorkingDirectory $workingDirectory): bool
196    {
197        if (!is_dir($workingDirectory->path . '/vendor/phpstan/extension-installer')) {
198            return false;
199        }
200
201        $manifest = is_file($workingDirectory->path . '/composer.json')
202            ? json_decode((string) file_get_contents($workingDirectory->path . '/composer.json'), true)
203            : null;
204        $extra     = is_array($manifest) && is_array($manifest['extra'] ?? null) ? $manifest['extra'] : [];
205        $installer = is_array($extra['phpstan/extension-installer'] ?? null) ? $extra['phpstan/extension-installer'] : [];
206        $ignored   = is_array($installer['ignore'] ?? null) ? $installer['ignore'] : [];
207
208        return !in_array('cruciblephp/crucible', $ignored, true);
209    }
