26/**
27 * Multibyte-aware string helpers tuned for terminal rendering.
28 *
29 * "Width" throughout this class refers to the number of visible columns a
30 * string occupies once ANSI escape sequences are removed.
31 */
32final class Str
33{
34    private const string ANSI_PATTERN = '/\e\[[0-9;?]*[ -\/]*[@-~]/';
35
36    /**
37     * Distinct short strings and graphemes whose width is remembered.
38     *
39     * ⚠ Bounded rather than unbounded: width is a pure function of its
40     * input, so a stale entry is impossible, but a run that measured a
41     * million distinct strings would otherwise keep every one of them.
42     */
43    private const int MEMO = 1024;
44
45    /** Bytes a string may have and still be worth remembering whole. */
46    private const int MEMO_BYTES = 8;
47
48    /** @var array<string, int> */
49    private static array $widths = [];
50
51    /** @var array<string, int> */
52    private static array $charWidths = [];
53
54    private function __construct() {}
55
56    public static function length(string $value): int
57    {
58        return mb_strlen($value);
59    }
60
61    /** Strip ANSI escape sequences from a string. */
62    public static function stripAnsi(string $value): string
63    {
64        return (string) preg_replace(self::ANSI_PATTERN, '', $value);
65    }
66
67    /**
68     * The visible column width of a string, accounting for wide characters.
69     *
70     * ⚠ Memoized because a frame asks this once per cell: an 80x20 map
71     * is 1600 calls, each of which otherwise runs a `preg_replace` over
72     * the ANSI pattern and splits a one-character string.
73     */
74    public static function width(string $value): int
75    {
76        if (isset(self::$widths[$value])) {
77            return self::$widths[$value];
78        }
79
80        $width = 0;
81
82        foreach (self::graphemes(self::stripAnsi($value)) as $grapheme) {
83            $width += self::charWidth($grapheme);
84        }
85
86        // Only the short ones: a cell, a glyph, a padded number. A full
87        // line is measured once and never asked for again.
88        if (strlen($value) <= self::MEMO_BYTES && count(self::$widths) < self::MEMO) {
89            self::$widths[$value] = $width;
90        }
91
92        return $width;
93    }
94
95    /** @return list<string> */
96    public static function graphemes(string $value): array
97    {
98        return mb_str_split($value);
99    }
100
101    /**
102     * The number of terminal columns a single grapheme occupies: 0 for
103     * combining/zero-width marks, 2 for East-Asian-wide and emoji, else 1.
104     */
105    public static function charWidth(string $grapheme): int
106    {
107        if ($grapheme === '') {
108            return 0;
109        }
110
111        if (isset(self::$charWidths[$grapheme])) {
112            return self::$charWidths[$grapheme];
113        }
114
115        $code = mb_ord($grapheme);
116
117        if ($code === false || self::isZeroWidth($code)) {
118            $width = 0;
119        } else {
120            $width = self::isWide($code) ? 2 : 1;
121        }
122
123        if (count(self::$charWidths) < self::MEMO) {
124            self::$charWidths[$grapheme] = $width;
125        }
126
127        return $width;
128    }
129
130    private static function isZeroWidth(int $code): bool
131    {
132        return $code === 0x200D // zero-width joiner
133            || ($code >= 0x0300 && $code <= 0x036F) // combining diacritics
134            || ($code >= 0x200B && $code <= 0x200F) // zero-width spaces/marks
135            || ($code >= 0xFE00 && $code <= 0xFE0F); // variation selectors
136    }
137
138    private static function isWide(int $code): bool
139    {
140        return ($code >= 0x1100 && $code <= 0x115F)   // Hangul Jamo
141            || ($code >= 0x2E80 && $code <= 0x303E)   // CJK radicals, Kangxi
142            || ($code >= 0x3041 && $code <= 0x33FF)   // Hiragana .. CJK symbols
143            || ($code >= 0x3400 && $code <= 0x4DBF)   // CJK Ext A
144            || ($code >= 0x4E00 && $code <= 0x9FFF)   // CJK Unified
145            || ($code >= 0xA000 && $code <= 0xA4CF)   // Yi
146            || ($code >= 0xAC00 && $code <= 0xD7A3)   // Hangul syllables
147            || ($code >= 0xF900 && $code <= 0xFAFF)   // CJK compatibility
148            || ($code >= 0xFE30 && $code <= 0xFE4F)   // CJK compatibility forms
149            || ($code >= 0xFF00 && $code <= 0xFF60)   // fullwidth forms
150            || ($code >= 0xFFE0 && $code <= 0xFFE6)   // fullwidth signs
151            || ($code >= 0x1F300 && $code <= 0x1FAFF) // emoji & symbols
152            || ($code >= 0x20000 && $code <= 0x3FFFD); // CJK Ext B+
153    }
154
155    public static function substr(string $value, int $start, ?int $length = null): string
156    {
157        return mb_substr($value, $start, $length);
158    }
159
160    /**
161     * Truncate a string to a maximum visible width, appending an ellipsis
162     * when the string is longer than the limit.
163     */
164    public static function truncate(string $value, int $width, string $ellipsis = '…'): string
165    {
166        if ($width <= 0) {
167            return '';
168        }
169
170        if (self::width($value) <= $width) {
171            return $value;
172        }
173
174        $keep = max(0, $width - self::width($ellipsis));
175
176        return self::substr($value, 0, $keep) . $ellipsis;
177    }
178
179    /** Pad a string on the right with spaces up to a visible width. */
180    public static function padRight(string $value, int $width): string
181    {
182        $pad = $width - self::width($value);
183
184        return $pad > 0 ? $value . str_repeat(' ', $pad) : $value;
185    }
186
187    /** Pad a string on the left with spaces up to a visible width. */
188    public static function padLeft(string $value, int $width): string
189    {
190        $pad = $width - self::width($value);
191
192        return $pad > 0 ? str_repeat(' ', $pad) . $value : $value;
193    }
194
195    /** Centre a string within a visible width. */
196    public static function padBoth(string $value, int $width): string
197    {
198        $pad = $width - self::width($value);
199
200        if ($pad <= 0) {
201            return $value;
202        }
203
204        $left = intdiv($pad, 2);
205
206        return str_repeat(' ', $left) . $value . str_repeat(' ', $pad - $left);
207    }
208
209    /**
210     * Wrap text to a given width, breaking on word boundaries where possible
211     * and hard-breaking words that exceed the width.
212     *
213     * @return list<string>
214     */
215    public static function wrap(string $value, int $width): array
216    {
217        if ($width <= 0) {
218            return [$value];
219        }
220
221        $lines = [];
222
223        foreach (explode("\n", $value) as $paragraph) {
224            $lines = array_merge($lines, self::wrapParagraph($paragraph, $width));
225        }
226
227        return $lines;
228    }
229
230    /**
231     * @return list<string>
232     */
233    private static function wrapParagraph(string $paragraph, int $width): array
234    {
235        if ($paragraph === '') {
236            return [''];
237        }
238
239        $lines   = [];
240        $current = '';
241
242        foreach (explode(' ', $paragraph) as $word) {
243            $candidate = $current === '' ? $word : $current . ' ' . $word;
244
245            if (self::width($candidate) <= $width) {
246                $current = $candidate;
247
248                continue;
249            }
250
251            if ($current !== '') {
252                $lines[] = $current;
253                $current = '';
254            }
255
256            while (self::width($word) > $width) {
257                $lines[] = self::substr($word, 0, $width);
258                $word    = self::substr($word, $width);
259            }
260
261            $current = $word;
262        }
263
264        $lines[] = $current;
265
266        return $lines;
267    }
