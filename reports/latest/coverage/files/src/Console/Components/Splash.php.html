30/**
31 * The wordmark, lit one character at a time.
32 *
33 * A bright head sweeps the text and bounces off each end, with one step
34 * of falloff behind it and the rest unlit — the whole animation is three
35 * colours and a position. Adapted from the project's own draft; the
36 * colours come from {@see Brand}, which reads them off the logo.
37 *
38 * **It animates only where animation is watched.** A run piped to a file
39 * or a CI log gets the wordmark once, in one colour, with no carriage
40 * returns and no cursor control — ✓ the thing that makes an animated
41 * banner unbearable is a log full of `\r` frames, and
42 * {@see Capabilities::animation()} is the same gate {@see Spinner} uses.
43 *
44 * Two ways to run it, because a banner has two jobs:
45 *
46 * - {@see begin()} and {@see settle()} sweep in a forked child while the
47 *   parent gets on with loading the configuration and discovering the
48 *   suite. ⚠ This is the one that belongs in front of real work: the
49 *   animation costs the run nothing, because the run never waits for it.
50 * - {@see render()} sweeps synchronously for a fixed number of frames,
51 *   for `crucible --version`, where the banner IS the command and there
52 *   is nothing to overlap it with.
53 *
54 * The frame is built from graphemes, not bytes, so a wordmark with a
55 * wide or combining character stays aligned: {@see Str::graphemes()}
56 * splits it and each cell is coloured whole.
57 */
58final class Splash
59{
60    /** Milliseconds between frames. */
61    public int $interval = 110;
62
63    private ?ForkedAnimation $animation = null;
64
65    /** Round trips before the sweep settles. */
66    public int $sweeps = 2;
67
68    /** Pre-styled text drawn after the wordmark, on the same line, unswept. */
69    public string $suffix = '';
70
71    public function __construct(
72        private readonly Terminal $terminal,
73        private readonly string $wordmark = 'cruciblephp',
74    ) {}
75
76    /**
77     * Sweep the wordmark, then leave it lit on its own line.
78     *
79     * Returns the number of frames written, which is what a caller — or
80     * a test — can check without measuring time.
81     */
82    public function render(): int
83    {
84        $cells = Str::graphemes($this->wordmark);
85
86        if ($cells === []) {
87            return 0;
88        }
89
90        if (!Capabilities::animation($this->terminal)) {
91            $this->terminal->write($this->paint($cells, -1) . PHP_EOL);
92
93            return 0;
94        }
95
96        $this->terminal->write(Sequence::HideCursor->render());
97
98        $frames = 0;
99        $last   = count($cells) - 1;
100        $head   = 0;
101        $step   = $last === 0 ? 0 : 1;
102        $turns  = 0;
103
104        // Bounded by frames, not only by turns.
105        //
106        // ⚠ Found by `crucible mutate` 2026-09-20: flipping the `||` in
107        // the bounce guard below to `&&` makes it never true, so $step
108        // never reverses and $turns never advances — the loop does not
109        // end. The test's terminal accumulates every frame, so that ran
110        // at ~130 MiB/s until the OOM killer took the whole run. The
111        // mutant is what broke it; an exit that depends on two coupled
112        // variables agreeing is what let it.
113        //
114        // A sweep cannot legitimately exceed cells * turns frames, so
115        // this ceiling is unreachable when the guard is correct and
116        // immediate when it is not.
117        $ceiling = count($cells) * max(1, $this->sweeps) * 2 + 1;
118
119        while ($turns < max(1, $this->sweeps) * 2 && $frames < $ceiling) {
120            $this->terminal->write("\r" . $this->paint($cells, $head));
121            $frames++;
122
123            if ($step === 0) {
124                break;
125            }
126
127            $head += $step;
128
129            if ($head >= $last || $head <= 0) {
130                $step *= -1;
131                $turns++;
132            }
133
134            usleep($this->interval * 1000);
135        }
136
137        $this->terminal->write(
138            "\r" . $this->paint($cells, -1) . Sequence::ShowCursor->render() . PHP_EOL,
139        );
140
141        return $frames;
142    }
143
144    /**
145     * Start sweeping in a forked child and return immediately.
146     *
147     * False means nothing is animating and the settled line has already
148     * been written — a caller still pairs this with {@see settle()},
149     * which then does nothing.
150     */
151    public function begin(): bool
152    {
153        $cells = Str::graphemes($this->wordmark);
154
155        if ($cells === []) {
156            return false;
157        }
158
159        if (!Capabilities::animation($this->terminal)) {
160            $this->terminal->write($this->paint($cells, -1) . PHP_EOL);
161
162            return false;
163        }
164
165        $animation = new ForkedAnimation($this->terminal, $this->frameAt(...), $this->interval);
166
167        if (!$animation->start()) {
168            $this->terminal->write($this->paint($cells, -1) . PHP_EOL);
169
170            return false;
171        }
172
173        $this->animation = $animation;
174
175        return true;
176    }
177
178    /**
179     * Sweep the wordmark while the callback runs, then clear the line.
180     *
181     * ⚠ This is the one that actually animates. {@see begin()} covers the
182     * gap before the first line of output, and on a warm checkout that gap
183     * is tens of milliseconds — nothing to sweep over. Discovering a suite
184     * is seconds, so the mark is put THERE, over work that exists, and the
185     * run still waits exactly as long as the work takes.
186     *
187     * Leaves nothing behind: the banner has already been printed above, so
188     * a second settled wordmark would just be the same line twice.
189     *
190     * @template TReturn
191     *
192     * @param Closure(): TReturn $work
193     *
194     * @return TReturn
195     */
196    public function sweepWhile(Closure $work): mixed
197    {
198        $cells = Str::graphemes($this->wordmark);
199
200        if ($cells === [] || !Capabilities::animation($this->terminal)) {
201            return $work();
202        }
203
204        $animation = new ForkedAnimation($this->terminal, $this->frameAt(...), $this->interval);
205
206        if (!$animation->start()) {
207            return $work();
208        }
209
210        try {
211            return $work();
212        } finally {
213            $animation->stop();
214        }
215    }
216
217    /**
218     * The frame for tick N: what the forked animation asks for each time.
219     *
220     * Public because it is the whole sweep as a pure function of one
221     * integer — a test reads the bounce off it without forking a process
222     * or measuring a clock.
223     */
224    public function frameAt(int $tick): string
225    {
226        $cells = Str::graphemes($this->wordmark);
227
228        return $this->paint($cells, $this->headAt(count($cells), $tick));
229    }
230
231    /** Stop the sweep and leave the wordmark lit on its own line. Idempotent. */
232    public function settle(): void
233    {
234        if (!$this->animation instanceof ForkedAnimation) {
235            return;
236        }
237
238        $this->animation->stop();
239        $this->animation = null;
240
241        $this->terminal->write("\r" . $this->paint(Str::graphemes($this->wordmark), -1) . PHP_EOL);
242    }
243
244    /**
245     * The lit cell at tick N: a triangle wave over the cells.
246     *
247     * The bounce as arithmetic rather than a mutable step and turn count.
248     * ⚠ The stateful form is what a mutant turned into a loop with no
249     * exit; there is no state here to get the two variables disagreeing.
250     */
251    private function headAt(int $cells, int $tick): int
252    {
253        if ($cells <= 1) {
254            return 0;
255        }
256
257        $period = 2 * ($cells - 1);
258        $phase  = $tick % $period;
259
260        return $phase < $cells ? $phase : $period - $phase;
261    }
262
263    /**
264     * One frame: $head is the lit cell, or -1 to light every cell.
265     *
266     * -1 is the resting state rather than a separate renderer, so the
267     * settled line and the animated ones cannot drift apart in spacing
268     * or colour.
269     *
270     * @param list<string> $cells
271     */
272    private function paint(array $cells, int $head): string
273    {
274        $frame = '';
275
276        foreach ($cells as $index => $cell) {
277            $distance = $head < 0 ? 0 : abs($index - $head);
278
279            $frame .= $this->sgr(match (true) {
280                $distance === 0 => Brand::ember(),
281                $distance === 1 => Brand::flame(),
282                default         => Brand::slate(),
283            }) . $cell;
284        }
285
286        return $frame . $this->sgr(null) . $this->suffix;
287    }
288
289    /**
290     * A foreground colour as an SGR sequence, or the reset when null.
291     *
292     * ⚠ Through Style, which is the same idiom {@see Spinner::line()} uses.
293     * This method first emitted its own `\e[...m` from
294     * `Color::forPalette()->foregroundParams()`, and took a Palette
295     * constructor parameter to feed it — reimplementing Style::toAnsi()
296     * and re-deciding what Theme already decides globally. Written while
297     * removing 75 copies of one annotation, which is the joke.
298     */
299    private function sgr(?Color $color): string
300    {
301        if (!$color instanceof Color) {
302            return "\e[0m";
303        }
304
305        return Style::none()->withForeground($color)->toAnsi();
306    }
