26/**
27 * A component drawn as a dialog, centred on a surface of its own.
28 *
29 * ⚠ This exists because a prompt has no business writing into a screen
30 * something else is already drawing. Asked inline, a question lands in
31 * the middle of whatever was being rendered and leaves its own frame
32 * behind afterwards; there is nothing the prompt can do about that,
33 * because the prompt does not know what else is on screen.
34 *
35 * The alternate screen settles it without either side having to know
36 * about the other: the dialog gets an empty surface, and leaving it puts
37 * the terminal back byte for byte. That is the whole reason a question
38 * asked over a running view can "go back to normal" at all.
39 *
40 * Reports {@see wrapsContent()} as true, so a {@see \LucianoPereira\Crucible\Console\Components\Prompt}
41 * renders bare and this draws the only border.
42 */
43final class ModalPresenter implements Presenter
44{
45    /** Columns of empty surface kept around the dialog on each side. */
46    private const int MARGIN = 4;
47
48    /** Rows of content a dialog may have before it stops growing. */
49    private const int MAX_ROWS = 20;
50
51    private readonly Reconciler $reconciler;
52
53    private bool $entered = false;
54
55    public function __construct(bool $decorated = true)
56    {
57        $this->reconciler = new Reconciler($decorated);
58    }
59
60    public function begin(Terminal $terminal): void
61    {
62        if ($this->entered) {
63            return;
64        }
65
66        $this->entered = true;
67
68        $terminal->write(
69            Sequence::AltScreenOn->render()
70            . Sequence::HideCursor->render()
71            . Sequence::EraseScreen->render()
72            . Sequence::CursorHome->render(),
73        );
74    }
75
76    /** The dialog's own width: the screen less its margins, and never vast. */
77    public function viewportWidth(Terminal $terminal): int
78    {
79        return max(1, min(76, max(1, $terminal->columns()) - 2 * self::MARGIN - 4));
80    }
81
82    public function present(Terminal $terminal, Buffer $buffer): void
83    {
84        $screen = max(1, $terminal->columns());
85        $lines  = max(1, $terminal->lines());
86        $inner  = $buffer->width;
87        $height = min(self::MAX_ROWS, $buffer->height);
88
89        $box  = $inner + 4;
90        $left = max(0, intdiv($screen - $box, 2));
91        $top  = max(0, intdiv($lines - ($height + 2), 2));
92
93        $frame = new Buffer($screen, $top + $height + 2);
94
95        $this->border($left, $box, '┌', '┐')->drawInto($frame, 0, $top);
96
97        for ($row = 0; $row < $height; $row++) {
98            $line = (new Line())
99                ->add(str_repeat(' ', $left))
100                ->add('│ ', $this->edge());
101
102            for ($column = 0; $column < $inner; $column++) {
103                $cell = $buffer->cellAt($column, $row);
104                $line->add($cell->char === '' ? ' ' : $cell->char, $cell->style);
105            }
106
107            $line->add(' │', $this->edge())->drawInto($frame, 0, $top + $row + 1);
108        }
109
110        $this->border($left, $box, '└', '┘')->drawInto($frame, 0, $top + $height + 1);
111
112        $terminal->write(Sequence::CursorHome->render() . $this->reconciler->reconcile($frame));
113    }
114
115    public function finish(Terminal $terminal): void
116    {
117        if (!$this->entered) {
118            return;
119        }
120
121        $this->entered = false;
122
123        $terminal->write(Sequence::ShowCursor->render() . Sequence::AltScreenOff->render());
124    }
125
126    /**
127     * A dialog owns the surface, so there is nothing to scroll past it.
128     *
129     * Resetting is still right: the next frame is drawn from scratch
130     * rather than diffed against one the caller has invalidated.
131     */
132    public function interrupt(Terminal $terminal, string $text): void
133    {
134        $this->reconciler->reset();
135    }
136
137    public function wrapsContent(): bool
138    {
139        return true;
140    }
141
142    private function border(int $left, int $width, string $start, string $end): Line
143    {
144        return (new Line())
145            ->add(str_repeat(' ', $left))
146            ->add($start . str_repeat('─', max(0, $width - 2)) . $end, $this->edge());
147    }
148
149    private function edge(): Style
150    {
151        return Style::none()->withForeground(Theme::accent());
152    }
153}
