31/**
32 * Lays out `--help` as terms and descriptions in two columns.
33 *
34 * ⚠ The text itself stays one readable block in the source, because
35 * that is where it is edited and reviewed. What it cannot carry there
36 * is alignment: a description written two spaces after its term lands
37 * at a different column on every row, and 55 of them did. Column
38 * position is a rendering concern, so it is decided here, once, from
39 * the terms actually present.
40 */
41final readonly class Help
42{
43    /**
44     * The column every description starts at.
45     *
46     * One column for the whole document, not one per indent level: a
47     * sub-option's description lining up with its parent's is the
48     * entire point, and it is why this class exists.
49     *
50     * ✓ 36 measured against the real help: of 191 term rows, 44 overran
51     * a column of 28 and only 10 overrun this one. Fitting all 191 takes
52     * 60 columns — `--do-not-warn-when-php-is-not-configured-for-development`
53     * alone is 56 — which leaves 20 for the descriptions on an 80-column
54     * screen. The curve flattens here.
55     */
56    private const int COLUMN = 36;
57
58    /** Share of a narrow screen the term column may take before it starves the text. */
59    private const int SHARE = 45;
60
61    /** Columns a description needs before a wider term column stops being worth it. */
62    private const int READABLE = 30;
63
64    /** The narrowest screen still laid out in two columns. */
65    private const int MIN_WIDTH = 50;
66
67    /** What a term looks like: a flag, or a `crucible <command>`. */
68    private const string TERM_PATTERN = '/^(\s+)((?:--?[a-z0-9-]+(?:,\s*-[a-z])?(?:\[?=?\s?<[^>]+>\]?)?|crucible\s+[a-z-]+(?:\s+\[[^\]]+\])?))\s\s+(\S.*)$/';
69
70    /**
71     * A bare word naming a value rather than a flag — a view key, say.
72     *
73     * ⚠ Only accepted well inside the margin. Flush against it, this
74     * would match the opening word of any sentence that happened to
75     * carry a double space, and lay a paragraph out as a table.
76     */
77    private const string VALUE_PATTERN = '/^(\s{6,})([a-z][a-z0-9-]*)\s\s+(\S.*)$/';
78
79    public function __construct(
80        private int $width = 80,
81        private bool $decorated = true,
82    ) {}
83
84    public function render(string $text): string
85    {
86        $lines   = explode("\n", $text);
87        $columns = $this->columns($lines);
88        $out     = [];
89
90        foreach ($lines as $line) {
91            foreach ($this->line($line, $columns) as $rendered) {
92                $out[] = $rendered;
93            }
94        }
95
96        return implode("\n", $out);
97    }
98
99    /**
100     * The description column for each indent that needs its own.
101     *
102     * ⚠ Only indents starting at or past the shared column get one — a
103     * list nested under an option, whose terms begin where everything
104     * else's descriptions do. Giving every indent its own column is
105     * what produced the ragged three-column output this replaces.
106     *
107     * @param list<string> $lines
108     *
109     * @return array<int, int>
110     */
111    private function columns(array $lines): array
112    {
113        $shared = $this->column();
114        $groups = [];
115
116        foreach ($lines as $line) {
117            $parts = $this->split($line);
118
119            if ($parts === null) {
120                continue;
121            }
122
123            [$indent, $term] = $parts;
124
125            $groups[Str::width($indent)][] = Str::width($indent) + Str::width($term) + 2;
126        }
127
128        $columns = [];
129
130        foreach ($groups as $at => $ends) {
131            $overflowing = count(array_filter($ends, static fn(int $end): bool => $end > $shared));
132
133            // ⚠ A group keeps the shared column while its overflows are
134            // outliers — the main option list has a handful of very long
135            // flags among a hundred ordinary ones, and widening for them
136            // would starve every description. A group where most terms
137            // overflow is not an outlier, it is a list that does not fit,
138            // and it gets a column of its own.
139            $local = max($ends);
140
141            // ⚠ And only while the description still has room to be
142            // read. A group of two whose second term is sixty columns
143            // wide is not a list that needs its own column; it is one
144            // long term, and long terms take their own line.
145            if ($overflowing > 0 && $overflowing * 2 >= count($ends) && $local <= $this->width - self::READABLE) {
146                $columns[$at] = $local;
147            }
148        }
149
150        return $columns;
151    }
152
153    /**
154     * @param array<int, int> $columns
155     *
156     * @return list<string>
157     */
158    private function line(string $line, array $columns): array
159    {
160        if ($line === '') {
161            return [''];
162        }
163
164        // A heading: flush left, ending in a colon, nothing else on it.
165        if (preg_match('/^([A-Z][A-Za-z0-9 ]*):$/', $line, $heading) === 1) {
166            return [$this->paint($heading[1] . ':', Style::none()->bold())];
167        }
168
169        $parts = $this->split($line);
170
171        if ($parts === null) {
172            return [rtrim($line)];
173        }
174
175        [$indent, $term, $description] = $parts;
176
177        $column = $columns[Str::width($indent)] ?? $this->column();
178        $room   = max(20, $this->width - $column);
179
180        if ($this->width < self::MIN_WIDTH) {
181            return [$indent . $this->paint($term, $this->accent()), $indent . '  ' . $description];
182        }
183
184        // ⚠ A description with nowhere to break is left whole and
185        // allowed to overrun. These are option names, and Str::wrap()
186        // splits on width rather than on words: broken across two lines
187        // a flag cannot be copied, read, or searched for, which is
188        // worse than a line running past the margin.
189        $wrapped = str_contains(trim($description), ' ')
190            ? Str::wrap($description, $room)
191            : [$description];
192        $head = $wrapped[0] ?? '';
193        $rest = array_slice($wrapped, 1);
194
195        // A term too wide for its column keeps the column rather than
196        // pushing its own description out of line with every other one.
197        $lines = Str::width($indent . $term) + 1 > $column
198            ? [$indent . $this->paint($term, $this->accent()), str_repeat(' ', $column) . $head]
199            : [$indent . $this->paint($term, $this->accent()) . str_repeat(' ', $column - Str::width($indent . $term)) . $head];
200
201        foreach ($rest as $continuation) {
202            $lines[] = str_repeat(' ', $column) . $continuation;
203        }
204
205        return $lines;
206    }
207
208    /**
209     * The shared column, given back to the text on a narrow screen.
210     *
211     * A fixed 36 on a 60-column terminal leaves 24 for descriptions,
212     * which wraps every one of them to three lines. The column yields
213     * first, because a term can take its own line and prose cannot.
214     */
215    private function column(): int
216    {
217        return min(self::COLUMN, max(20, intdiv($this->width * self::SHARE, 100)));
218    }
219
220    /**
221     * A line's indent, term and description, or null if it is prose.
222     *
223     * @return array{string, string, string}|null
224     */
225    private function split(string $line): ?array
226    {
227        foreach ([self::TERM_PATTERN, self::VALUE_PATTERN] as $pattern) {
228            if (preg_match($pattern, $line, $parts) === 1) {
229                return [$parts[1], $parts[2], $parts[3]];
230            }
231        }
232
233        return null;
234    }
235
236    private function accent(): Style
237    {
238        return Style::none()->withForeground(Brand::flame());
239    }
240
241    private function paint(string $text, Style $style): string
242    {
243        return $this->decorated ? $style->toAnsi() . $text . "\e[0m" : $text;
244    }
