37/**
38 * Renders a practical subset of Markdown to styled {@see Line}s: headings,
39 * bold/italic/code inline spans, bullet lists, block quotes, horizontal rules,
40 * fenced code, and `[text](target)` links (collected as numbered references).
41 */
42final class Markdown

44    private function __construct() {}

46    public static function render(string $markdown, int $width = 80): MarkdownDocument
47    {
48        $lines  = [];
49        $links  = [];
50        $inCode = false;
51
52        // A paragraph or bullet item is source-wrapped across several
53        // physical lines (READMEs hard-wrap at ~80-100 columns); inline
54        // spans like **bold** routinely straddle that break, and a
55        // bullet's continuation lines (indented, no `-`/`*`) would
56        // otherwise fall through as their own unindented paragraph.
57        // Buffer consecutive lines belonging to the same block and join
58        // them before parsing inline markup and wrapping to width.
59        $paragraph = [];
60        $kind      = 'text';
61
62        /** @var list<string> $table raw `|`-delimited rows, rendered as a block once the run ends */
63        $table = [];
64
65        foreach (explode("\n", str_replace("\r\n", "\n", $markdown)) as $raw) {
66            $text = rtrim($raw);
67
68            if (preg_match('/^\s*```/', $text) === 1) {
69                self::flushParagraph($paragraph, $kind, $lines, $links, $width);
70                self::flushTable($table, $lines, $links);
71                $inCode = ! $inCode;
72
73                continue;
74            }
75
76            if ($inCode) {
77                $lines[] = (new Line())->add($text === '' ? ' ' : $text, self::style('code'));
78
79                continue;
80            }
81
82            if (preg_match('/^\s*\|/', $text) === 1) {
83                self::flushParagraph($paragraph, $kind, $lines, $links, $width);
84                $table[] = \trim($text);
85
86                continue;
87            }
88
89            self::flushTable($table, $lines, $links);
90
91            if ($text === '') {
92                self::flushParagraph($paragraph, $kind, $lines, $links, $width);
93                $lines[] = new Line();
94            } elseif (preg_match('/^(#{1,3})\s+(.*)$/', $text, $m) === 1) {
95                self::flushParagraph($paragraph, $kind, $lines, $links, $width);
96
97                foreach (Str::wrap($m[2], $width) as $piece) {
98                    $lines[] = (new Line())->add($piece, self::heading(strlen($m[1])));
99                }
100            } elseif (preg_match('/^\s*(?:-{3,}|\*{3,})\s*$/', $text) === 1) {
101                self::flushParagraph($paragraph, $kind, $lines, $links, $width);
102                $lines[] = (new Line())->add(str_repeat('─', $width), self::style('dim'));
103            } elseif (preg_match('/^\s*[-*]\s+(.*)$/', $text, $m) === 1) {
104                self::flushParagraph($paragraph, $kind, $lines, $links, $width);
105                $paragraph[] = $m[1];
106                $kind        = 'bullet';
107            } elseif (preg_match('/^>\s?(.*)$/', $text, $m) === 1) {
108                self::flushParagraph($paragraph, $kind, $lines, $links, $width);
109                $paragraph[] = $m[1];
110                $kind        = 'quote';
111            } elseif ($kind === 'bullet' && preg_match('/^\s{2,}\S/', $text) === 1) {
112                $paragraph[] = \trim($text);
113            } else {
114                if ($kind !== 'text') {
115                    self::flushParagraph($paragraph, $kind, $lines, $links, $width);
116                }
117
118                $paragraph[] = $text;
119                $kind        = 'text';
120            }
121        }
122
123        self::flushParagraph($paragraph, $kind, $lines, $links, $width);
124        self::flushTable($table, $lines, $links);
125
126        return new MarkdownDocument($lines, $links);
127    }

129    /**
130     * @param list<string> $paragraph
131     * @param 'text'|'bullet'|'quote' $kind
132     * @param list<Line> $lines
133     * @param list<string> $links
134     */
135    private static function flushParagraph(array &$paragraph, string &$kind, array &$lines, array &$links, int $width): void
136    {
137        if ($paragraph === []) {
138            $kind = 'text';
139
140            return;
141        }
142
143        $joined = implode(' ', $paragraph);
144
145        match ($kind) {
146            'bullet' => self::wrapInline($lines, $links, $joined, $width, '• ', '  ', self::style('accent')),
147            'quote'  => self::wrapInline($lines, $links, $joined, $width, '│ ', '│ ', self::style('dim'), self::style('dim')),
148            'text'   => self::wrapInline($lines, $links, $joined, $width, '', ''),
149        };
150
151        $paragraph = [];
152        $kind      = 'text';
153    }

155    /**
156     * Render a run of buffered `|`-delimited rows as a bordered table. A
157     * second row made only of `-`/`:`/`|`/space is GFM's header separator —
158     * dropped, and the row before it treated as the header.
159     *
160     * @param list<string> $table
161     * @param list<Line> $lines
162     * @param list<string> $links
163     */
164    private static function flushTable(array &$table, array &$lines, array &$links): void
165    {
166        if ($table === []) {
167            return;
168        }
169
170        $rawRows = [];
171
172        foreach ($table as $row) {
173            $rawRows[] = self::tableCells($row);
174        }
175
176        $headers = [];
177
178        if (isset($table[1]) && self::isSeparatorRow($table[1])) {
179            $headers = array_shift($rawRows);
180            array_shift($rawRows);
181        }
182
183        $headers = self::textifyCells($headers, $links);
184        $rows    = [];
185
186        foreach ($rawRows as $row) {
187            $rows[] = self::textifyCells($row, $links);
188        }
189
190        foreach (self::renderTable($headers, $rows) as $line) {
191            $lines[] = $line;
192        }
193
194        $table = [];
195    }
196
197    /** @return list<string> */
198    private static function tableCells(string $row): array
199    {
200        return array_map(trim(...), explode('|', \trim($row, "| \t")));
201    }
202
203    private static function isSeparatorRow(string $row): bool
204    {
205        return preg_match('/^[\s|:-]+$/', $row) === 1 && str_contains($row, '-');
206    }

208    /**
209     * Cell content still allows inline markup (e.g. a `[text](link)`), but
210     * a table cell renders as plain text — styling per cell would need
211     * {@see Line} in the grid itself, more than a single README table
212     * (D-099) needs; link targets still register and keep their `[N]`
213     * marker so they stay followable from a table row.
214     *
215     * @param list<string> $cells
216     * @param list<string> $links
217     *
218     * @return list<string>
219     */
220    private static function textifyCells(array $cells, array &$links): array
221    {
222        $text = [];
223
224        foreach ($cells as $cell) {
225            $plain = '';
226
227            foreach (self::inline($cell, $links, Style::none()) as $token) {
228                $plain .= $token['text'];
229            }
230
231            $text[] = $plain;
232        }
233
234        return $text;
235    }

237    /**
238     * @param list<string> $headers
239     * @param list<list<string>> $rows
240     *
241     * @return list<Line>
242     */
243    private static function renderTable(array $headers, array $rows): array
244    {
245        $columns = max(1, count($headers), ...array_map(count(...), $rows));
246        $widths  = array_fill(0, $columns, 0);
247
248        foreach ([$headers, ...$rows] as $row) {
249            foreach (self::padRow($row, $columns) as $index => $cell) {
250                $widths[$index] = max($widths[$index], Str::width($cell));
251            }
252        }
253
254        $lines = [self::tableBorder($widths, '┌', '┬', '┐')];
255
256        if ($headers !== []) {
257            $lines[] = self::tableRow(self::padRow($headers, $columns), $widths, true);
258            $lines[] = self::tableBorder($widths, '├', '┼', '┤');
259        }
260
261        foreach ($rows as $row) {
262            $lines[] = self::tableRow(self::padRow($row, $columns), $widths, false);
263        }
264
265        $lines[] = self::tableBorder($widths, '└', '┴', '┘');
266
267        return $lines;
268    }

270    /**
271     * @param list<string> $row
272     *
273     * @return list<string>
274     */
275    private static function padRow(array $row, int $columns): array
276    {
277        $out = [];
278
279        for ($i = 0; $i < $columns; ++$i) {
280            $out[] = $row[$i] ?? '';
281        }
282
283        return $out;

287    private static function tableBorder(array $widths, string $left, string $middle, string $right): Line
288    {
289        $line = new Line();
290        $line->add($left, self::style('dim'));
291
292        $last = count($widths) - 1;
293
294        foreach ($widths as $index => $width) {
295            $line->add(str_repeat('─', $width + 2), self::style('dim'));
296            $line->add($index === $last ? $right : $middle, self::style('dim'));
297        }
298
299        return $line;
300    }

303     * @param list<string> $cells
304     * @param array<int, int> $widths
305     */
306    private static function tableRow(array $cells, array $widths, bool $header): Line
307    {
308        $line = new Line();
309        $line->add('│', self::style('dim'));
310
311        foreach ($cells as $index => $cell) {
312            $padded = ' ' . Str::padRight($cell, $widths[$index]) . ' ';

320    /**
321     * Inline markup (bold/italic/code/links) is parsed against the whole
322     * paragraph first, then the resulting styled tokens are word-wrapped —
323     * not the other way around — so a span isn't missed just because the
324     * word-wrap would otherwise have cut it in two (D-099).
325     *
326     * @param list<Line> $lines
327     * @param list<string> $links
328     */
329    private static function wrapInline(
330        array &$lines,
331        array &$links,
332        string $text,
333        int $width,
334        string $firstPrefix,
335        string $contPrefix,
336        ?Style $prefixStyle = null,
337        ?Style $base = null,
338    ): void {
339        $inner  = max(1, $width - Str::width($firstPrefix));
340        $tokens = self::inline($text, $links, $base ?? Style::none());
341        $atoms  = self::atomize($tokens);
342
343        foreach (self::packAtoms($atoms, $inner) as $index => $segments) {
344            $line = new Line();
345            $line->add($index === 0 ? $firstPrefix : $contPrefix, $prefixStyle);
346
347            foreach ($segments as $segment) {
348                $line->add($segment['text'], $segment['style']);
349            }
350
351            $lines[] = $line;
352        }
353    }

355    /**
356     * @param list<string> $links
357     *
358     * @return list<array{text: string, style: Style}>
359     */
360    private static function inline(string $text, array &$links, Style $base): array
361    {
362        $tokens  = [];
363        $pattern = '/(\[[^\]]+\]\([^)]+\))|(`[^`]+`)|(\*\*[^*]+\*\*)|(\*[^*]+\*|_[^_]+_)/';
364        $offset  = 0;
365
366        if (preg_match_all($pattern, $text, $matches, PREG_OFFSET_CAPTURE | PREG_SET_ORDER) !== false) {
367            foreach ($matches as $match) {
368                $whole = $match[0][0];
369                $start = $match[0][1];
370
371                if ($start > $offset) {
372                    $tokens[] = ['text' => substr($text, $offset, $start - $offset), 'style' => $base];
373                }
374
375                foreach (self::inlineToken($match, $whole, $links, $base) as $token) {
376                    $tokens[] = $token;
377                }
378
379                $offset = $start + strlen($whole);
380            }
381        }
382
383        if ($offset < strlen($text)) {
384            $tokens[] = ['text' => substr($text, $offset), 'style' => $base];
385        }
386
387        return $tokens;
388    }

390    /**
391     * @param array<int, array{string, int}> $match
392     * @param list<string> $links
393     *
394     * @return list<array{text: string, style: Style}>
395     */
396    private static function inlineToken(array $match, string $whole, array &$links, Style $base): array
397    {
398        if (($match[1][1] ?? -1) !== -1 && preg_match('/\[([^\]]+)\]\(([^)]+)\)/', $whole, $link) === 1) {
399            $links[] = $link[2];
400
401            return [
402                ['text' => $link[1], 'style' => self::style('link')],
403                ['text' => '[' . count($links) . ']', 'style' => self::style('dim')],
404            ];
405        }
406
407        if (($match[2][1] ?? -1) !== -1) {
408            return [['text' => \trim($whole, '`'), 'style' => self::style('code')]];
409        }
410
411        if (($match[3][1] ?? -1) !== -1) {
412            return [['text' => \trim($whole, '*'), 'style' => $base->bold()]];
413        }
414
415        return [['text' => \trim($whole, '*_'), 'style' => $base->italic()]];
416    }

418    /**
419     * Split styled tokens into word atoms for wrapping. `glue` marks an
420     * atom that must render immediately after the previous one with no
421     * space — e.g. a link label followed by its `[N]` reference marker.
422     *
423     * @param list<array{text: string, style: Style}> $tokens
424     *
425     * @return list<array{text: string, style: Style, glue: bool}>
426     */
427    private static function atomize(array $tokens): array
428    {
429        $atoms    = [];
430        $glueNext = false;
431
432        foreach ($tokens as $token) {
433            $pieces = explode(' ', $token['text']);
434            $last   = count($pieces) - 1;
435
436            foreach ($pieces as $index => $piece) {
437                if ($piece === '') {
438                    $glueNext = false;
439
440                    continue;
441                }
442
443                $atoms[]  = ['text' => $piece, 'style' => $token['style'], 'glue' => $glueNext];
444                $glueNext = $index === $last;
445            }
446        }
447
448        return $atoms;
449    }

451    /**
452     * Greedily pack atoms into lines no wider than `$width`, splitting an
453     * over-wide atom across lines as a last resort.
454     *
455     * @param list<array{text: string, style: Style, glue: bool}> $atoms
456     *
457     * @return list<list<array{text: string, style: Style}>>
458     */
459    private static function packAtoms(array $atoms, int $width): array
460    {
461        $lines        = [];
462        $current      = [];
463        $currentWidth = 0;
464
465        foreach ($atoms as $atom) {
466            $atomText  = $atom['text'];
467            $atomWidth = Str::width($atomText);
468            $withSpace = $current !== [] && ! $atom['glue'];
469
470            if ($current !== [] && $currentWidth + ($withSpace ? 1 : 0) + $atomWidth > $width) {
471                $lines[]      = $current;
472                $current      = [];
473                $currentWidth = 0;
474                $withSpace    = false;
475            }
476
477            if ($withSpace) {
478                $current[] = ['text' => ' ', 'style' => Style::none()];
479                $currentWidth += 1;
480            }
481
482            while ($atomWidth > $width) {
483                if ($current !== []) {
484                    $lines[]      = $current;
485                    $current      = [];
486                    $currentWidth = 0;
487                }
488
489                $lines[]  = [['text' => Str::substr($atomText, 0, $width), 'style' => $atom['style']]];
490                $atomText = Str::substr($atomText, $width);
491
492                $atomWidth = Str::width($atomText);
493            }
494
495            $current[] = ['text' => $atomText, 'style' => $atom['style']];
496            $currentWidth += $atomWidth;
497        }
498
499        if ($current !== []) {
500            $lines[] = $current;
501        }
502
503        return $lines === [] ? [[]] : $lines;
504    }

506    private static function heading(int $level): Style
507    {
508        return match ($level) {
509            1       => Style::none()->withForeground(Theme::accent())->bold(),
510            2       => Style::none()->bold(),
511            default => Style::none()->underline(),
512        };
