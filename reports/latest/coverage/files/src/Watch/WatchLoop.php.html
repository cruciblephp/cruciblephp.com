58/**
59 * The watch loop (growth G3 slice 2): poll the suite and source
60 * directories, and run every iteration in a **child process** — PHP
61 * cannot re-require changed files in one process (redeclare fatals),
62 * so a fresh child per run is the only honest reload. The child's
63 * NDJSON event stream is the result API: --log-events-json to a
64 * temp file, test:finish outcomes decide the failed set. Failed
65 * tests come first in every re-run (--order-by defects) unless the
66 * user pinned an order.
67 *
68 * WatchSession decides what runs; this class only does IO.
69 */
70final class WatchLoop
71{
72    private const int POLL_MICROSECONDS = 250_000;
73
74    private string $sttyState = '';

76    /**
77     * @param list<non-empty-string> $childArgv   the crucible invocation to replay, --watch removed, [0] = script path
78     * @param list<non-empty-string> $directories absolute paths to watch
79     * @param list<non-empty-string> $files       absolute paths to watch (the configuration file)
80     */
81    public function __construct(
82        private readonly array $childArgv,
83        private readonly array $directories,
84        private readonly array $files,
85        private readonly FileWatcher $watcher = new FileWatcher(),
86        private readonly WatchSession $session = new WatchSession(),
87        private readonly ?RetriggerListener $retrigger = null,
88    ) {}

90    public function run(): int
91    {
92        $pid         = getmypid();
93        $eventFile   = sys_get_temp_dir() . '/crucible-watch-' . ($pid === false ? 0 : $pid) . '.ndjson';
94        $interactive = $this->enterRawTty();
95
96        $this->cleanUpOnSignal($eventFile);
97
98        $snapshot = $this->watcher->snapshot($this->directories, $this->files);
99        $last     = WatchRun::full('the initial run');
100
101        $this->execute($last, $eventFile);
102        $this->status($interactive);
103
104        try {
105            while (true) {
106                $key = $this->waitForKey();
107
108                if ($key === 'q') {
109                    $this->out(PHP_EOL);
110
111                    return 0;
112                }
113
114                $run = match ($key) {
115                    'a'        => WatchRun::full('requested with the a key'),
116                    'u'        => WatchRun::full('recording snapshots (D-042)', ['--update-snapshots']),
117                    'f'        => $this->session->failedRun(),
118                    "\n", "\r" => $last,
119                    default    => null,
120                };
121
122                if ($key === 'f' && !$run instanceof WatchRun) {
123                    $this->out(PHP_EOL . '[watch] Nothing is failing.' . PHP_EOL);
124                    $this->status($interactive);
125
126                    continue;
127                }
128
129                // A pushed change set (D-083) is authoritative: the
130                // producer already resolved what changed, so it needs
131                // no debounce. Absorbing the disk state keeps the same
132                // edit from running a second time when the watcher
133                // notices it a tick later.
134                if (!$run instanceof WatchRun) {
135                    $pushed = $this->retrigger?->take() ?? [];
136
137                    if ($pushed !== []) {
138                        $snapshot = $this->watcher->snapshot($this->directories, $this->files);
139                        $run      = $this->session->onChange($this->resolveAll($pushed), false);
140                    }
141                }
142
143                if (!$run instanceof WatchRun) {
144                    $current = $this->watcher->snapshot($this->directories, $this->files);
145                    $diff    = $this->watcher->diff($snapshot, $current);
146
147                    if ($diff['changed'] === [] && $diff['deleted'] === []) {
148                        continue;
149                    }
150
151                    // Debounce: editors write in bursts; wait for a
152                    // quiet interval before running.
153                    do {
154                        usleep(150_000);
155
156                        $settled = $this->watcher->snapshot($this->directories, $this->files);
157                        $more    = $this->watcher->diff($current, $settled);
158                        $current = $settled;
159                    } while ($more['changed'] !== [] || $more['deleted'] !== []);
160
161                    $diff     = $this->watcher->diff($snapshot, $current);
162                    $snapshot = $current;
163
164                    /** @var list<non-empty-string> $changed */
165                    $changed = $diff['changed'];
166
167                    $run = $this->session->onChange($changed, $diff['deleted'] !== []);
168                }
169                // Key-triggered runs leave the snapshot untouched: an
170                // edit racing the keypress still triggers its own run
171                // on the next poll rather than being swallowed.
172
173                $last = $run;
174
175                $this->execute($run, $eventFile);
176                $this->status($interactive);
177            }
178        } finally {
179            $this->restoreTty();
180
181            if (is_file($eventFile)) {
182                unlink($eventFile);
183            }
184        }
185    }

187    /**
188     * Runs one child, feeds the outcome to the session, and runs the
189     * follow-up it demands (the green-again full confirmation).
190     */
191    private function execute(WatchRun $run, string $eventFile): void
192    {
193        $this->out(sprintf(PHP_EOL . '[watch] Running: %s.' . PHP_EOL . PHP_EOL, $run->reason));
194
195        $command = [PHP_BINARY, ...$this->childArgv];
196
197        foreach ($run->related ?? [] as $file) {
198            $command[] = '--related';
199            $command[] = $file;
200        }
201
202        foreach ($run->extraArguments as $argument) {
203            $command[] = $argument;
204        }
205
206        if (!$this->orderPinned()) {
207            $command[] = '--order-by';
208            $command[] = 'defects';
209        }
210
211        // Last occurrence wins in the child's CLI parsing, so this
212        // owns the event stream even if the user passed their own.
213        $command[] = '--log-events-json';
214        $command[] = $eventFile;
215
216        // The child sees a pipe, not the terminal — keep its colors on.
217        if (stream_isatty(STDOUT) && !array_any($this->childArgv, static fn(string $argument): bool => str_starts_with($argument, '--colors'))) {
218            $command[] = '--colors=always';
219        }
220
221        // Child output is copied through the parent rather than
222        // sharing the descriptor: proc_open'd children do not share
223        // the parent's file offset, so direct fd sharing scrambles
224        // (and overwrites) redirected output.
225        $process = proc_open($command, [0 => ['file', '/dev/null', 'r'], 1 => ['pipe', 'w'], 2 => ['pipe', 'w']], $pipes);
226
227        if (!is_resource($process)) {
228            $this->out('[watch] Could not start the test run.' . PHP_EOL);
229
230            return;
231        }
232
233        $this->forward($pipes);
234
235        proc_close($process);
236
237        $failed   = $this->failedFilesIn($eventFile);
238        $followUp = $this->session->onResult($run, $failed);
239
240        if ($followUp instanceof WatchRun) {
241            $this->execute($followUp, $eventFile);
242        }
243    }

245    /**
246     * The child's event stream is the result API: files with a
247     * failed or errored test:finish.
248     *
249     * @return list<non-empty-string>
250     */
251    private function failedFilesIn(string $eventFile): array
252    {
253        /** @var array<non-empty-string, true> $failed */
254        $failed = [];
255
256        $lines = is_file($eventFile) ? file($eventFile) : false;
257
258        foreach ($lines === false ? [] : $lines as $line) {
259            $event = json_decode($line, true);
260
261            if (!is_array($event)
262                || ($event['event'] ?? null) !== 'test:finish'
263                || !in_array($event['outcome'] ?? null, ['fail', 'error'], true)
264                || !is_string($event['id'] ?? null)
265            ) {
266                continue;
267            }
268
269            $id = TestId::fromString($event['id']);
270
271            if ($id instanceof TestId) {
272                $failed[$id->file] = true;
273            }
274        }
275
276        return array_keys($failed);
277    }

281        return array_any($this->childArgv, static fn(string $argument): bool => str_starts_with($argument, '--order-by'));
282    }

284    private function status(bool $interactive): void
285    {
286        $state = $this->session->isRed()
287            ? sprintf('%d file(s) failing', count($this->session->failedRun()->related ?? []))
288            : 'all green';
289
290        $this->out(sprintf(
291            PHP_EOL . '[watch] %s — waiting for changes%s' . PHP_EOL,
292            $state,
293            $interactive ? '  (enter rerun · a all · f failed · u snapshots · q quit)' : '',
294        ));
295    }

297    /**
298     * Copies the child's stdout/stderr onto the parent's until both
299     * close — one writer, one offset, deterministic ordering.
300     *
301     * @param array<int, resource> $pipes
302     */
303    private function forward(array $pipes): void
304    {
305        stream_set_blocking($pipes[1], false);
306        stream_set_blocking($pipes[2], false);
307
308        $open = [1 => $pipes[1], 2 => $pipes[2]];
309
310        while ($open !== []) {
311            $read   = array_values($open);
312            $write  = null;
313            $except = null;
314
315            if (stream_select($read, $write, $except, 0, 200_000) === false) {
316                break;
317            }
318
319            foreach ($open as $channel => $pipe) {
320                $chunk = fread($pipe, 65536);
321
322                if ($chunk !== false && $chunk !== '') {
323                    fwrite($channel === 1 ? STDOUT : STDERR, $chunk);
324                }
325
326                if (feof($pipe)) {
327                    fclose($pipe);
328                    unset($open[$channel]);
329                }
330            }
331        }
332    }

343    /**
344     * @return ?string one key, or null when the poll interval elapsed
345     */
346    /**
347     * A producer may push either shape; the graphs downstream compare
348     * absolute paths, so resolve what resolves and pass the rest through
349     * untouched (a deleted file has no realpath, and `--related` will
350     * resolve it against the working directory anyway).
351     *
352     * @param list<non-empty-string> $paths
353     *
354     * @return list<non-empty-string>
355     */
356    private function resolveAll(array $paths): array
357    {
358        $resolved = [];
359
360        foreach ($paths as $path) {
361            $real       = realpath($path);
362            $resolved[] = $real === false ? $path : $real;
363        }
364
365        return $resolved;
366    }

368    private function waitForKey(): ?string
369    {
370        $pushed = $this->retrigger?->watchStreams() ?? [];
371
372        // Without a tty there is no key to wait for, but a push still
373        // has to wake the loop — so the endpoint's streams are selected
374        // on either way, and only the STDIN half is conditional.
375        if ($this->sttyState === '' && $pushed === []) {
376            usleep(self::POLL_MICROSECONDS);
377
378            return null;
379        }
380
381        $read   = $this->sttyState === '' ? $pushed : [STDIN, ...$pushed];
382        $write  = null;
383        $except = null;
384
385        $ready = stream_select($read, $write, $except, 0, self::POLL_MICROSECONDS);
386
387        if ($ready === false || $ready < 1) {
388            return null;
389        }
390
391        // Serve whatever arrived before reading a key: the pump only
392        // buffers, so the push is collected on this same tick.
393        $this->retrigger?->pump();
394
395        if ($this->sttyState === '' || !in_array(STDIN, $read, true)) {
396            return null;
397        }
398
399        $key = fread(STDIN, 1);
400
401        return $key === false || $key === '' ? null : $key;
402    }

404    /**
405     * @return bool whether interactive keys are live
406     */
407    /**
408     * `finally` does not run when the process is signalled, and a watch
409     * session is usually ended with Ctrl-C rather than `q` — which used
410     * to leave the terminal with echo off, and would now also leave a
411     * hot file claiming a dead session is still listening.
412     *
413     * Signals are the only way to hear about it, so this is the one
414     * place outside mutation that touches pcntl — guarded, never
415     * required: without the extension `q` still cleans up, exactly as
416     * before.
417     */
418    private function cleanUpOnSignal(string $eventFile): void
419    {
420        if (!function_exists('pcntl_async_signals') || !function_exists('pcntl_signal')) {
421            return;
422        }
423
424        pcntl_async_signals(true);
425
426        $cleanUp = function (int $signal) use ($eventFile): never {
427            $this->restoreTty();
428            $this->retrigger?->close();
429
430            if (is_file($eventFile)) {
431                unlink($eventFile);
432            }
433
434            exit(128 + $signal);
435        };
436
437        pcntl_signal(SIGINT, $cleanUp);
438        pcntl_signal(SIGTERM, $cleanUp);
439        pcntl_signal(SIGHUP, $cleanUp);
440    }
441
442    private function enterRawTty(): bool
443    {
444        if (PHP_OS_FAMILY === 'Windows' || !stream_isatty(STDIN)) {
445            return false;
446        }
447
448        $state = shell_exec('stty -g 2>/dev/null');
449
450        if (!is_string($state) || trim($state) === '') {
451            return false;
452        }
453
454        $this->sttyState = trim($state);
455
456        shell_exec('stty -icanon -echo 2>/dev/null');
457
458        return true;
459    }
460
461    private function restoreTty(): void
462    {
463        if ($this->sttyState !== '') {
464            shell_exec('stty ' . $this->sttyState . ' 2>/dev/null');
465
466            $this->sttyState = '';
467        }
468    }
