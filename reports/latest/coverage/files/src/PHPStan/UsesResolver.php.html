41/**
42 * Static reading of a dialect file's `uses(...)` arguments (D-050):
43 * the class and trait names a test file binds its closures to, taken
44 * from the source text alone — the analysis-time mirror of what
45 * PestScopes resolves at runtime by executing the file. Token-based
46 * like every other scanner in this project, name resolution follows
47 * PHP's own rules (leading backslash wins, then the alias map, then
48 * the namespace prefix), and imports are read by PHP's grammar: lists,
49 * groups, aliases (D-137). File-local `uses(...)` reads through
50 * classRefs(); the directory-scoped `pest()/uses()->in()` chains a
51 * Pest.php declares read through scopedRegistrations() (D-067). Both
52 * are one walk over the file, differing only in what they keep.
53 */
54final readonly class UsesResolver
55{
56    /** A name as the tokenizer spells it: bare, qualified, or fully qualified. */
57    private const array NAME = [T_STRING, T_NAME_QUALIFIED, T_NAME_FULLY_QUALIFIED];
58
59    /**
60     * Fully-qualified names passed to top-level uses() calls, in
61     * argument order — `Foo::class` references and FQ string
62     * literals both count, exactly the runtime surface.
63     *
64     * @return list<non-empty-string>
65     */
66    public static function classRefs(string $source): array
67    {
68        $names = [];
69
70        foreach (self::chains($source, ['uses']) as $chain) {
71            $names = [...$names, ...$chain['root']];
72        }
73
74        return $names;
75    }

77    /**
78     * The scoped registrations a Pest.php declares (D-067, closing
79     * the D-050 leftover): every `pest()`/`uses()` chain carrying an
80     * `->in(...)`, read as the class/trait names its extend()/use()/
81     * assign()/initial arguments collect plus the in() glob strings —
82     * the analysis-time mirror of ScopeRegistration. Chains without
83     * an in() are bare (hooks only, classes inert — the D-033 rule)
84     * and are not returned.
85     *
86     * @return list<array{names: list<non-empty-string>, globs: non-empty-list<non-empty-string>}>
87     */
88    public static function scopedRegistrations(string $source): array
89    {
90        $registrations = [];
91
92        foreach (self::chains($source, ['pest', 'uses']) as $chain) {
93            if ($chain['globs'] !== []) {
94                $registrations[] = ['names' => array_merge($chain['root'], $chain['chained']), 'globs' => $chain['globs']];
95            }
96        }
97
98        return $registrations;
99    }

101    /**
102     * Every top-level call to one of $roots and its fluent chain: the
103     * names its own arguments pass, the names `->extend()`, `->use()` and
104     * `->assign()` pass, and the `->in()` globs — resolved against the
105     * namespace and imports in force where the call stands.
106     *
107     * @param non-empty-list<non-empty-string> $roots
108     *
109     * @return list<array{root: list<non-empty-string>, chained: list<non-empty-string>, globs: list<non-empty-string>}>
110     */
111    private static function chains(string $source, array $roots): array
112    {
113        $tokens    = PhpToken::tokenize($source);
114        $total     = count($tokens);
115        $namespace = '';
116
117        /** @var array<string, non-empty-string> $aliases alias => fully-qualified */
118        $aliases = [];
119        $chains  = [];
120
121        for ($i = 0; $i < $total; $i++) {
122            $token = $tokens[$i];
123
124            if ($token->is(T_NAMESPACE)) {
125                $next      = self::nextSignificant($tokens, $i + 1);
126                $namespace = $next !== null && $tokens[$next]->is([T_STRING, T_NAME_QUALIFIED]) ? $tokens[$next]->text : $namespace;
127
128                continue;
129            }
130
131            if ($token->is(T_USE)) {
132                self::collectImport($tokens, $i + 1, $aliases);
133
134                continue;
135            }
136
137            $open = self::rootCall($tokens, $i, $roots);
138
139            if ($open === null) {
140                continue;
141            }
142
143            $chain = ['root' => self::resolveAll(self::arguments($tokens, $open + 1), $namespace, $aliases), 'chained' => [], 'globs' => []];
144            $index = self::closingParen($tokens, $open);
145
146            // Walk the fluent chain: `-> name ( args )` until it ends.
147            while ($index !== null) {
148                $arrow  = self::nextSignificant($tokens, $index + 1);
149                $method = $arrow === null || !$tokens[$arrow]->is(T_OBJECT_OPERATOR) ? null : self::nextSignificant($tokens, $arrow + 1);
150                $args   = $method === null ? null : self::nextSignificant($tokens, $method + 1);
151
152                if ($method === null || $args === null || $tokens[$args]->text !== '(') {
153                    break;
154                }
155
156                if (in_array($tokens[$method]->text, ['extend', 'use', 'assign'], true)) {
157                    $chain['chained'] = [...$chain['chained'], ...self::resolveAll(self::arguments($tokens, $args + 1), $namespace, $aliases)];
158                } elseif ($tokens[$method]->text === 'in') {
159                    $chain['globs'] = [...$chain['globs'], ...self::stringArguments($tokens, $args + 1)];
160                }
161
162                $index = self::closingParen($tokens, $args);
163            }
164
165            $chains[] = $chain;
166            $i        = $index ?? $i;
167        }
168
169        return $chains;
170    }

172    /**
173     * The index of the `(` when the token at $i calls one of $roots as
174     * the dialect global: `uses(` or `\uses(`, never `->uses(`,
175     * `::uses(` or the `function uses(` that declares somebody else's.
176     *
177     * @param array<PhpToken>                  $tokens
178     * @param non-empty-list<non-empty-string> $roots
179     */
180    private static function rootCall(array $tokens, int $i, array $roots): ?int
181    {
182        $token = $tokens[$i];
183        $name  = $token->is(T_NAME_FULLY_QUALIFIED) ? ltrim($token->text, '\\') : ($token->is(T_STRING) ? $token->text : null);
184
185        if ($name === null || !in_array($name, $roots, true)) {
186            return null;
187        }
188
189        $previous = self::previousSignificant($tokens, $i - 1);
190
191        if ($previous !== null && $tokens[$previous]->is([T_OBJECT_OPERATOR, T_DOUBLE_COLON, T_FUNCTION])) {
192            return null;
193        }
194
195        $open = self::nextSignificant($tokens, $i + 1);
196
197        return $open !== null && $tokens[$open]->text === '(' ? $open : null;
198    }

200    /**
201     * Every string literal inside one balanced argument list.
202     *
203     * @param array<PhpToken> $tokens
204     *
205     * @return list<non-empty-string>
206     */
207    private static function stringArguments(array $tokens, int $start): array
208    {
209        $strings = [];
210
211        foreach (self::argumentTokens($tokens, $start) as $token) {
212            $literal = $token->is(T_CONSTANT_ENCAPSED_STRING) ? trim($token->text, "'\"") : '';
213
214            if ($literal !== '') {
215                $strings[] = $literal;
216            }
217        }
218
219        return $strings;
220    }

222    /**
223     * The `Name::class` and FQ-string arguments inside one balanced
224     * argument list, unresolved.
225     *
226     * @param array<PhpToken> $tokens
227     *
228     * @return list<string>
229     */
230    private static function arguments(array $tokens, int $start): array
231    {
232        $names = [];
233
234        foreach (self::argumentTokens($tokens, $start) as $index => $token) {
235            if ($token->is(T_CONSTANT_ENCAPSED_STRING)) {
236                $literal = trim($token->text, "'\"");
237
238                if (str_contains($literal, '\\')) {
239                    $names[] = '\\' . ltrim($literal, '\\');
240                }
241
242                continue;
243            }
244
245            $colons = $token->is(self::NAME) ? self::nextSignificant($tokens, $index + 1) : null;
246            $class  = $colons !== null && $tokens[$colons]->is(T_DOUBLE_COLON) ? self::nextSignificant($tokens, $colons + 1) : null;
247
248            if ($class !== null && $tokens[$class]->is(T_CLASS)) {
249                $names[] = $token->text;
250            }
251        }
252
253        return $names;
254    }

256    /**
257     * The tokens of one balanced argument list, from $start (just past
258     * its `(`) to its `)`, keyed by index.
259     *
260     * @param array<PhpToken> $tokens
261     *
262     * @return array<int, PhpToken>
263     */
264    private static function argumentTokens(array $tokens, int $start): array
265    {
266        $end = self::closingParen($tokens, $start - 1) ?? count($tokens);
267        $in  = [];
268
269        for ($i = $start; $i < $end; $i++) {
270            $in[$i] = $tokens[$i];
271        }
272
273        return $in;
274    }

276    /**
277     * The index of the parenthesis closing the one open at $open.
278     *
279     * @param array<PhpToken> $tokens
280     */
281    private static function closingParen(array $tokens, int $open): ?int
282    {
283        $depth = 0;
284
285        for ($i = $open, $total = count($tokens); $i < $total; $i++) {
286            if ($tokens[$i]->text === '(') {
287                $depth++;
288            } elseif ($tokens[$i]->text === ')' && --$depth === 0) {
289                return $i;
290            }
291        }
292
293        return null;
294    }

296    /**
297     * Each name resolved; one that resolves to nothing is dropped.
298     *
299     * @param list<string>                    $names
300     * @param array<string, non-empty-string> $aliases
301     *
302     * @return list<non-empty-string>
303     */
304    private static function resolveAll(array $names, string $namespace, array $aliases): array
305    {
306        $resolved = [];
307
308        foreach ($names as $name) {
309            $name = self::resolve($name, $namespace, $aliases);
310
311            if ($name !== '') {
312                $resolved[] = $name;
313            }
314        }
315
316        return $resolved;
317    }

319    /**
320     * PHP's own resolution rules: a leading backslash wins, then the
321     * alias map rewrites the first segment, then the namespace
322     * prefixes bare names.
323     *
324     * @param array<string, non-empty-string> $aliases
325     */
326    private static function resolve(string $name, string $namespace, array $aliases): string
327    {
328        if (str_starts_with($name, '\\')) {
329            return ltrim($name, '\\');
330        }
331
332        $segments = explode('\\', $name);
333
334        if (array_key_exists($segments[0], $aliases)) {
335            $segments[0] = $aliases[$segments[0]];
336
337            return implode('\\', $segments);
338        }
339
340        return $namespace === '' ? $name : $namespace . '\\' . $name;
341    }

343    /**
344     * One `use` statement into the alias map, by PHP's grammar: a comma
345     * list (`use A, B as C;`) and group imports (`use A\{B, C as D};`),
346     * each entry optionally aliased. Function and const imports never name
347     * classes and are skipped, whole or inside a group. Trait-use inside
348     * class bodies is not routed here: the dialect files this reads are
349     * top-level scripts.
350     *
351     * @param array<PhpToken>                 $tokens
352     * @param array<string, non-empty-string> $aliases
353     */
354    private static function collectImport(array $tokens, int $start, array &$aliases): void
355    {
356        $index = self::nextSignificant($tokens, $start);
357
358        if ($index === null || $tokens[$index]->is([T_FUNCTION, T_CONST])) {
359            return;
360        }
361
362        while ($index !== null && $tokens[$index]->is(self::NAME)) {
363            $after = self::nextSignificant($tokens, $index + 1);
364            $index = $after !== null && $tokens[$after]->is(T_NS_SEPARATOR)
365                ? self::collectGroup($tokens, $after, ltrim($tokens[$index]->text, '\\') . '\\', $aliases)
366                : self::collectEntry($tokens, $index, '', $aliases);
367
368            if ($index === null || $tokens[$index]->text !== ',') {
369                return;
370            }
371
372            $index = self::nextSignificant($tokens, $index + 1);
373        }
374    }

376    /**
377     * The `{…}` of a group import after its `Prefix\`: every class entry,
378     * prefixed. The index after the closing brace, or null.
379     *
380     * @param array<PhpToken>                 $tokens
381     * @param array<string, non-empty-string> $aliases
382     */
383    private static function collectGroup(array $tokens, int $separator, string $prefix, array &$aliases): ?int
384    {
385        $open = self::nextSignificant($tokens, $separator + 1);
386
387        if ($open === null || $tokens[$open]->text !== '{') {
388            return null;
389        }
390
391        $index = self::nextSignificant($tokens, $open + 1);
392
393        while ($index !== null && $tokens[$index]->text !== '}') {
394            // `use A\{function f, const C, B}`: only B names a class.
395            $class = !$tokens[$index]->is([T_FUNCTION, T_CONST]);
396            $entry = $class ? $index : self::nextSignificant($tokens, $index + 1);
397            $index = $entry === null ? null : self::collectEntry($tokens, $entry, $prefix, $aliases, $class);
398
399            if ($index !== null && $tokens[$index]->text === ',') {
400                $index = self::nextSignificant($tokens, $index + 1);
401            }
402        }
403
404        return $index === null ? null : self::nextSignificant($tokens, $index + 1);
405    }

407    /**
408     * One `Name` or `Name as Alias` entry. The index after it, or null
409     * when the token at $index is no name.
410     *
411     * @param array<PhpToken>                 $tokens
412     * @param array<string, non-empty-string> $aliases
413     */
414    private static function collectEntry(array $tokens, int $index, string $prefix, array &$aliases, bool $record = true): ?int
415    {
416        if (!$tokens[$index]->is(self::NAME)) {
417            return null;
418        }
419
420        $qualified = $prefix . ltrim($tokens[$index]->text, '\\');
421        $alias     = array_last(explode('\\', $qualified));
422        $next      = self::nextSignificant($tokens, $index + 1);
423
424        if ($next !== null && $tokens[$next]->is(T_AS)) {
425            $aliasToken = self::nextSignificant($tokens, $next + 1);
426
427            if ($aliasToken !== null && $tokens[$aliasToken]->is(T_STRING)) {
428                $alias = $tokens[$aliasToken]->text;
429                $next  = self::nextSignificant($tokens, $aliasToken + 1);
430            }
431        }
432
433        if ($record && $qualified !== '' && $alias !== '') {
434            $aliases[$alias] = $qualified;
435        }
436
437        return $next;
438    }

440    /**
441     * @param array<PhpToken> $tokens
442     */
443    private static function nextSignificant(array $tokens, int $start): ?int
444    {
445        for ($i = $start, $total = count($tokens); $i < $total; $i++) {
446            if (!$tokens[$i]->isIgnorable()) {
447                return $i;
448            }
449        }
450
451        return null;
452    }

454    /**
455     * @param array<PhpToken> $tokens
456     */
457    private static function previousSignificant(array $tokens, int $start): ?int
458    {
459        for ($i = $start; $i >= 0; $i--) {
460            if (!$tokens[$i]->isIgnorable()) {
461                return $i;
462            }
463        }
464
465        return null;
466    }
