33/**
34 * What `$this` is inside a dialect file's bound closures (D-050,
35 * completed in D-067): the file's own `uses(...)` wins; without one,
36 * the ancestor Pest.php files' scoped `pest()->extend()->in()`
37 * registrations decide — outer configuration first, the last
38 * matching class wins, glob semantics mirroring PestScopes (plain
39 * names are subtree prefixes, wildcards go through fnmatch, both
40 * relative to the declaring directory). The default stays
41 * PestTestCase. Trait arguments still cannot become part of a PHP
42 * type — the documented limit, erring toward the base type, never a
43 * wrong one.
44 */
45final class DialectThisResolver
46{
47    private const int WALK_CAP = 12;
48
49    /** @var array<string, Type> */
50    private array $byFile = [];
51
52    /** @var array<string, list<array{names: list<non-empty-string>, globs: non-empty-list<string>}>> */
53    private array $configurations = [];
54
55    public function forFile(string $file): Type
56    {
57        return $this->byFile[$file] ??= $this->resolve($file);
58    }
59
60    private function resolve(string $file): Type
61    {
62        $source = @file_get_contents($file);
63
64        if ($source !== false) {
65            foreach (UsesResolver::classRefs($source) as $name) {
66                // The project's autoloader is loaded in the analysis
67                // process, so existence answers are real. Traits
68                // compose at runtime but are not types — skipped.
69                if (class_exists($name) && !trait_exists($name)) {
70                    return new ObjectType($name);
71                }
72            }
73        }
74
75        $scoped = $this->scopedClass($file);
76
77        return new ObjectType($scoped ?? PestTestCase::class);
78    }
79
80    /**
81     * The class the ancestor Pest.php configurations assign to this
82     * file — the analysis-time PestBuilder::effectiveState walk.
83     *
84     * @return ?class-string
85     */
86    private function scopedClass(string $file): ?string
87    {
88        // Ancestors from the file upward, then applied outer-first —
89        // the runtime loads configurations root-down, and the last
90        // matching class wins ($class = $scope->class ?? $class).
91        $directories = [];
92        $directory   = dirname($file);
93
94        for ($depth = 0; $depth < self::WALK_CAP; $depth++) {
95            $directories[] = $directory;
96
97            $isRoot = is_file($directory . DIRECTORY_SEPARATOR . 'crucible.php')
98                || is_file($directory . DIRECTORY_SEPARATOR . 'crucible.dist.php')
99                || is_file($directory . DIRECTORY_SEPARATOR . 'composer.json');
100
101            $parent = dirname($directory);
102
103            if ($isRoot || $parent === $directory) {
104                break;
105            }
106
107            $directory = $parent;
108        }
109
110        $class = null;
111
112        foreach (array_reverse($directories) as $base) {
113            foreach ($this->registrationsIn($base) as $registration) {
114                if (!str_starts_with($file, $base . DIRECTORY_SEPARATOR)) {
115                    continue;
116                }
117
118                // Globs are written with '/' whatever the OS, and
119                // FNM_PATHNAME only treats '/' as a separator.
120                $relative = str_replace(DIRECTORY_SEPARATOR, '/', substr($file, strlen($base) + 1));
121                $matched  = false;
122
123                foreach ($registration['globs'] as $glob) {
124                    if ($this->matches($glob, $relative)) {
125                        $matched = true;
126
127                        break;
128                    }
129                }
130
131                if (!$matched) {
132                    continue;
133                }
134
135                foreach ($registration['names'] as $name) {
136                    if (class_exists($name) && !trait_exists($name)) {
137                        $class = $name;
138
139                        break;
140                    }
141                }
142            }
143        }
144
145        return $class;
146    }
147
148    /**
149     * @return list<array{names: list<non-empty-string>, globs: non-empty-list<string>}>
150     */
151    private function registrationsIn(string $directory): array
152    {
153        if (isset($this->configurations[$directory])) {
154            return $this->configurations[$directory];
155        }
156
157        $config = $directory . DIRECTORY_SEPARATOR . 'Pest.php';
158        $source = is_file($config) ? @file_get_contents($config) : false;
159
160        return $this->configurations[$directory] = $source === false
161            ? []
162            : UsesResolver::scopedRegistrations($source);
163    }
164
165    /** The PestScopes glob rule, mirrored. */
166    private function matches(string $glob, string $relative): bool
167    {
168        if (!str_contains($glob, '*') && !str_contains($glob, '?') && !str_contains($glob, '[')) {
169            return $relative === $glob || str_starts_with($relative, $glob . '/');
170        }
171
172        return fnmatch($glob, $relative, FNM_PATHNAME);
173    }
174}
