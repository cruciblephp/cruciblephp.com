36/**
37 * The data a method's attributes feed it, checked against its signature
38 * (D-129): `#[TestWith]` and `#[TestWithJson]` rows against the test
39 * method's parameters, and `#[Check(args:, returns:)]` claims against the
40 * method they sit on — the arguments against its parameters, the claimed
41 * return against its declared return type. Attribute arguments are
42 * constant expressions, so the analyser knows every value exactly; a row
43 * that cannot be passed is a mistake visible before anything runs.
44 *
45 * What is reported is only what fails at run time (DataRows): a missing
46 * required argument, a value the parameter definitely cannot accept, and
47 * for `#[Check]` a `returns:` the declared return type definitely cannot
48 * produce.
49 *
50 * @implements Rule<FunctionLike>
51 */
52final readonly class DataAttributeRule implements Rule
53{
54    private const array CHECK = [\LucianoPereira\Crucible\Attributes\Check::class];
55
56    private const array TEST_WITH = [
57        \LucianoPereira\Crucible\Attributes\TestWith::class,
58        'PHPUnit\\Framework\\Attributes\\TestWith',
59    ];
60
61    private const array TEST_WITH_JSON = [
62        \LucianoPereira\Crucible\Attributes\TestWithJson::class,
63        'PHPUnit\\Framework\\Attributes\\TestWithJson',
64    ];
65
66    #[Override]
67    public function getNodeType(): string
68    {
69        return FunctionLike::class;
70    }
71
72    #[Override]
73    public function processNode(Node $node, Scope $scope): array
74    {
75        if (!$node instanceof ClassMethod && !$node instanceof Function_) {
76            return [];
77        }
78
79        $subject = $node instanceof ClassMethod
80            ? sprintf('%s::%s()', $scope->getClassReflection()?->getDisplayName() ?? 'class', $node->name->toString())
81            : sprintf('%s()', $node->name->toString());
82
83        $errors = [];
84        $count  = ['check' => 0, 'with' => 0];
85
86        foreach ($node->attrGroups as $group) {
87            foreach ($group->attrs as $attribute) {
88                $name = $attribute->name->toString();
89
90                if (in_array($name, self::CHECK, true)) {
91                    $count['check']++;
92                    $errors = [...$errors, ...$this->checkClaim($node, $attribute, $subject, sprintf('#[Check] #%d', $count['check']), $scope)];
93                } elseif (in_array($name, self::TEST_WITH, true)) {
94                    $count['with']++;
95                    $errors = [...$errors, ...$this->testWith($node, $attribute, $subject, sprintf('#[TestWith] #%d', $count['with']), $scope)];
96                } elseif (in_array($name, self::TEST_WITH_JSON, true)) {
97                    $count['with']++;
98                    $errors = [...$errors, ...$this->testWithJson($node, $attribute, $subject, sprintf('#[TestWithJson] #%d', $count['with']), $scope)];
99                }
100            }
101        }
102
103        return $errors;
104    }

106    /**
107     * @param non-empty-string $subject
108     * @param non-empty-string $row
109     *
110     * @return list<IdentifierRuleError>
111     */
112    private function checkClaim(ClassMethod|Function_ $function, Attribute $attribute, string $subject, string $row, Scope $scope): array
113    {
114        $args    = $this->argument($attribute, 'args', 0);
115        $returns = $this->argument($attribute, 'returns', 1);
116        $throws  = $this->argument($attribute, 'throws', 2);
117        $errors  = [];
118
119        if (!$args instanceof \PhpParser\Node\Arg || $args->value instanceof Array_) {
120            $values = [];
121            $next   = 0;
122
123            foreach ($args?->value instanceof Array_ ? $args->value->items : [] as $item) {
124                if ($item->unpack) {
125                    return [];
126                }
127
128                $key = match (true) {
129                    $item->key instanceof String_ => $item->key->value,
130                    $item->key instanceof Int_    => $item->key->value,
131                    default                       => $next,
132                };
133
134                if (!$item->key instanceof String_) {
135                    $next = (int) $key + 1;
136                }
137
138                $values[$key] = [$scope->getType($item->value), $item->value->getStartLine()];
139            }
140
141            $errors = DataRows::check(DataRows::fromNodes($function->getParams(), $scope), $values, $subject, $row, $attribute->getStartLine(), $scope, 'crucible.check');
142        }
143
144        // A claim that expects a throw makes no claim about the return.
145        if (!$returns instanceof \PhpParser\Node\Arg || $throws instanceof \PhpParser\Node\Arg || !$function->returnType instanceof \PhpParser\Node) {
146            return $errors;
147        }
148
149        $declared = $scope->getFunctionType($function->returnType, false, false);
150        $claimed  = $scope->getType($returns->value);
151
152        if ($declared->accepts($claimed, $scope->isDeclareStrictTypes())->no()) {
153            $errors[] = RuleErrorBuilder::message(sprintf(
154                '%s declares it returns %s, but %s claims it returns %s.',
155                $subject,
156                $declared->describe(VerbosityLevel::typeOnly()),
157                $row,
158                $claimed->describe(VerbosityLevel::precise()),
159            ))->identifier('crucible.checkReturns')->line($attribute->getStartLine())->build();
160        }
161
162        return $errors;
163    }

165    /**
166     * @param non-empty-string $subject
167     * @param non-empty-string $row
168     *
169     * @return list<IdentifierRuleError>
170     */
171    private function testWith(ClassMethod|Function_ $function, Attribute $attribute, string $subject, string $row, Scope $scope): array
172    {
173        $data = $this->argument($attribute, 'data', 0);
174
175        if (!$data instanceof \PhpParser\Node\Arg || !$data->value instanceof Array_) {
176            return [];
177        }
178
179        $values = [];
180
181        foreach ($data->value->items as $position => $item) {
182            if ($item->unpack || $item->key !== null) {
183                return [];
184            }
185
186            $values[$position] = [$scope->getType($item->value), $item->value->getStartLine()];
187        }
188
189        return DataRows::check(DataRows::fromNodes($function->getParams(), $scope), $values, $subject, $row, $attribute->getStartLine(), $scope, 'crucible.testWith');
190    }
191
192    /**
193     * @param non-empty-string $subject
194     * @param non-empty-string $row
195     *
196     * @return list<IdentifierRuleError>
197     */
198    private function testWithJson(ClassMethod|Function_ $function, Attribute $attribute, string $subject, string $row, Scope $scope): array
199    {
200        $json = $this->argument($attribute, 'json', 0);
201
202        if (!$json instanceof \PhpParser\Node\Arg || !$json->value instanceof String_) {
203            return [];
204        }
205
206        $decoded = json_decode($json->value->value, true);
207
208        // Malformed JSON is the attribute's own error at run time; not this rule's.
209        if (!is_array($decoded) || !array_is_list($decoded)) {
210            return [];
211        }
212
213        $values = [];
214
215        foreach ($decoded as $position => $value) {
216            $values[$position] = [ConstantTypeHelper::getTypeFromValue($value), $attribute->getStartLine()];
217        }
218
219        return DataRows::check(DataRows::fromNodes($function->getParams(), $scope), $values, $subject, $row, $attribute->getStartLine(), $scope, 'crucible.testWith');
220    }
221
222    /** An attribute argument by its name, or by its position when unnamed. */
223    private function argument(Attribute $attribute, string $name, int $position): ?Arg
224    {
225        foreach ($attribute->args as $index => $arg) {
226            if ($arg->name?->toString() === $name || ($arg->name === null && $index === $position)) {
227                return $arg;
228            }
229        }
230
231        return null;
232    }
