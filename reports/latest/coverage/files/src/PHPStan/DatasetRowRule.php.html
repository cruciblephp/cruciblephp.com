36/**
37 * A dataset row the test cannot take is wrong at the keyboard, not at
38 * the first run (D-129) — the principle PropertyParameterRule applies to
39 * generators (D-067), applied to the rows of an inline Pest dataset:
40 *
41 *     it('adds', function (int $a, int $b) { ... })->with([
42 *         'strings' => ['1', '2'],        // reported: $a declares int
43 *     ]);
44 *
45 * Only what would fail at run time is reported: a row with fewer values
46 * than the closure requires (an ArgumentCountError), and a value its
47 * declared parameter definitely cannot accept (a TypeError). A value the
48 * parameter might accept is not reported, and neither is an extra value
49 * (PHP passes it and the closure ignores it). Out of reach for now, and
50 * left alone: named datasets, a closure producing the rows, several
51 * ->with() calls on one test, and closure values (Pest calls them
52 * lazily, so their result is what the parameter receives).
53 *
54 * Read from the whole statement: a dataset is known only once every
55 * ->with() on the test is seen, and one call cannot see the call it sits
56 * inside.
57 *
58 * @implements Rule<Expression>
59 */
60final readonly class DatasetRowRule implements Rule
61{
62    #[Override]
63    public function getNodeType(): string
64    {
65        return Expression::class;
66    }
67
68    #[Override]
69    public function processNode(Node $node, Scope $scope): array
70    {
71        $withs = [];
72        $chain = $node->expr;
73
74        while ($chain instanceof MethodCall) {
75            if ($chain->name instanceof Identifier && $chain->name->toString() === 'with') {
76                $withs[] = $chain;
77            }
78
79            $chain = $chain->var;
80        }
81
82        // One dataset, or none to read: several ->with() calls form a
83        // Cartesian product whose rows split the parameters between them.
84        if (count($withs) !== 1) {
85            return [];
86        }
87
88        $with = $withs[0];
89
90        if (!(new ObjectType(TestCall::class))->isSuperTypeOf($scope->getType($with->var))->yes()) {
91            return [];
92        }
93
94        $args = $with->getArgs();
95
96        if (count($args) !== 1 || !$args[0]->value instanceof Array_) {
97            return [];
98        }
99
100        $closure = $this->testClosure($chain);
101
102        if ($closure === null) {
103            return [];
104        }
105
106        $errors = [];
107
108        foreach ($args[0]->value->items as $index => $row) {
109            if ($row->unpack) {
110                continue;
111            }
112
113            $values = $this->rowValues($row->value);
114
115            if ($values === null) {
116                continue;
117            }
118
119            $typed = [];
120
121            foreach ($values as $position => $value) {
122                // Pest calls a closure value and passes its result.
123                $typed[$position] = [
124                    $value instanceof Closure || $value instanceof ArrowFunction ? null : $scope->getType($value),
125                    $value->getStartLine(),
126                ];
127            }
128
129            foreach (DataRows::check(DataRows::fromNodes($closure->getParams(), $scope), $typed, 'The test closure', $this->rowName($row, $index), $row->getStartLine(), $scope, 'crucible.dataset') as $error) {
130                $errors[] = $error;
131            }
132        }
133
134        return $errors;
135    }
136
137    /** The test closure a chain was opened with: test()/it()'s second argument. */
138    private function testClosure(Expr $chain): Closure|ArrowFunction|null
139    {
140        if (!$chain instanceof FuncCall || !$chain->name instanceof Name || !in_array($chain->name->toLowerString(), ['test', 'it'], true)) {
141            return null;
142        }
143
144        $args    = $chain->getArgs();
145        $closure = isset($args[1]) ? $args[1]->value : null;
146
147        return $closure instanceof Closure || $closure instanceof ArrowFunction ? $closure : null;
148    }
149
150    /**
151     * A row's values in order: an array literal's items, or one value
152     * that is not an array literal. Null when the row cannot be read
153     * item by item (string keys, spreads).
154     *
155     * @return ?list<Expr>
156     */
157    private function rowValues(Expr $row): ?array
158    {
159        if (!$row instanceof Array_) {
160            return [$row];
161        }
162
163        $values = [];
164
165        foreach ($row->items as $item) {
166            if ($item->unpack || $item->key instanceof String_) {
167                return null;
168            }
169
170            $values[] = $item->value;
171        }
172
173        return $values;
174    }
175
176    /**
177     * `dataset "name"` for a keyed row, `row #N` otherwise.
178     *
179     * @return non-empty-string
180     */
181    private function rowName(ArrayItem $row, int $index): string
182    {
183        if ($row->key instanceof String_) {
184            return sprintf('dataset "%s"', $row->key->value);
185        }
186
187        if ($row->key instanceof Int_) {
188            return sprintf('row #%d', $row->key->value);
189        }
190
191        return sprintf('row #%d', $index);
192    }
193}
