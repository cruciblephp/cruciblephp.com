39/**
40 * One architecture rule (D-088): a set of targeted classes and the
41 * expectations they must satisfy.
42 *
43 * Built fluently and evaluated once, when the test that `arch()`
44 * registered runs — so a rule is an ordinary test with an ordinary
45 * failure, counted and reported like every other assertion rather than
46 * living in a parallel reporting world.
47 *
48 * The targeting layer is the load-bearing part. Expectations are cheap
49 * to add once "which classes does this rule mean?" has a single answer,
50 * so that question is settled first and every expectation reads the one
51 * resolved set.
52 */
53final class ArchRule
54{
55    /** @var list<string> */
56    private array $targets = [];
57
58    /** @var list<string> */
59    private array $ignored = [];
60
61    /** @var list<Closure(): void> */
62    private array $expectations = [];
63
64    /** @var list<string> the targets, resolved once by assert() */
65    private array $resolved = [];
66
67    /**
68     * The universe is resolved **when the rule runs**, not when it is
69     * built: `arch()` is called while a file is being collected, and the
70     * run configures the ambient afterwards. Binding eagerly gave every
71     * rule an empty universe — caught by the "matches nothing" guard
72     * below rather than passing silently, which is the whole reason that
73     * guard exists.
74     */
75    public function __construct(private ?ArchitectureUniverse $universe = null) {}

77    /**
78     * The classes this rule is about, by namespace or exact name.
79     * `*` matches within a namespace segment, `**` across segments, and
80     * a pattern with no wildcard is a prefix — naming a namespace is
81     * what a rule almost always means.
82     */
83    public function expect(string ...$targets): self
84    {
85        foreach ($targets as $target) {
86            $this->targets[] = $target;
87        }
88
89        return $this;
90    }

93    public function ignoring(string ...$patterns): self
94    {
95        foreach ($patterns as $pattern) {
96            $this->ignored[] = $pattern;
97        }
98
99        return $this;

104    /** Every targeted class references at least one of these. */
105    public function toUse(string ...$prefixes): self
106    {
107        $prefixes = array_values($prefixes);
108
109        return $this->each(
110            fn(string $class): bool => array_any(
111                $this->universe()->referencesOf($class),
112                static fn(string $reference): bool => self::matchesAny($reference, $prefixes),
113            ),
114            sprintf('use %s', implode(' or ', $prefixes)),
115        );
116    }

118    /**
119     * Every reference a targeted class makes **into the project's own
120     * source** is one of these — the layering rule. It says what a
121     * module may depend on, and names anything else, which is the part
122     * that makes it useful.
123     *
124     * References outside the configured source — PHP built-ins like
125     * `PhpToken`, and vendor packages — are not considered. This is a
126     * rule about your own layering, and counting `Closure` as a
127     * dependency would force every rule to carry a whitelist of the
128     * standard library before it said anything at all.
129     */
130    public function toOnlyUse(string ...$prefixes): self
131    {
132        $prefixes = array_values($prefixes);
133
134        $this->expectations[] = function () use ($prefixes): void {
135            $violations = [];
136
137            foreach ($this->ownReferences() as $pair) {
138                if (!self::matchesAny($pair[1], $prefixes)) {
139                    $violations[] = sprintf('%s uses %s', $pair[0], $pair[1]);
140                }
141            }
142
143            Assert::assertSame([], $violations, sprintf(
144                'Expected the targeted classes to only use %s.',
145                implode(' or ', $prefixes),
146            ));
147        };
148
149        return $this;
150    }

152    /** No targeted class references anything from the project's source. */
153    public function toUseNothing(): self
154    {
155        $this->expectations[] = function (): void {
156            $violations = [];
157
158            foreach ($this->ownReferences() as $pair) {
159                $violations[] = sprintf('%s uses %s', $pair[0], $pair[1]);
160            }
161
162            Assert::assertSame([], $violations, 'Expected the targeted classes to use nothing from the source.');
163        };

168    /**
169     * Only these may reference the targeted classes — the inverse
170     * layering rule, and the one that catches a boundary crossed from
171     * the far side, where the offending code is not the code the rule
172     * is about.
173     */
174    public function toOnlyBeUsedIn(string ...$prefixes): self
175    {
176        $prefixes = array_values($prefixes);
177
178        $this->expectations[] = function () use ($prefixes): void {
179            $violations = [];
180
181            foreach (ArchPredicates::inboundPairs($this->universe(), $this->resolved) as [$user, $class]) {
182                if (!self::matchesAny($user, $prefixes)) {
183                    $violations[] = sprintf('%s uses %s', $user, $class);
184                }
185            }
186
187            Assert::assertSame([], $violations, sprintf(
188                'Expected the targeted classes to only be used in %s.',
189                implode(' or ', $prefixes),
190            ));
191        };
192
193        return $this;
194    }

274    /**
275     * Nothing outside the target references a targeted class —
276     * `toOnlyBeUsedIn()` with an empty allowlist, which is how it is
277     * implemented rather than as a second traversal. A reference from
278     * one targeted class to another does not count: ✓ measured against
279     * the incumbent, which reads the target as one unit.
280     */
281    public function toBeUsedInNothing(): self
282    {
283        // Reads usedByNothing(), NOT toOnlyBeUsedIn(): the two questions
284        // differ on a sibling. "Nothing uses it" counts a user inside
285        // the target set — ✓ measured against pest 5.1.1 on 2026-09-06,
286        // which fails `expect('…\Deps\Pure')->toBeUsedInNothing()`
287        // naming two classes both INSIDE the target — while "only these
288        // may use it" is a question about outside users and keeps
289        // inboundPairs().
290        //
291        // Delegating to toOnlyBeUsedIn() is what made the two spellings
292        // of one dialect answer differently, which ArchDependencyTest
293        // exists to catch.
294        $this->expectations[] = function (): void {
295            $used = ArchPredicates::usedByNothing($this->universe(), $this->resolved);
296
297            $violations = [];
298
299            foreach ($this->resolved as $class) {
300                if (!$used($class)) {
301                    $violations[] = $class;
302                }
303            }
304
305            Assert::assertSame([], $violations, 'Expected the targeted classes to be used by nothing.');
306        };
307
308        return $this;
309    }

366    /**
367     * @param 'int'|'string' $type
368     */
369    private function backedBy(string $type): self
370    {
371        return $this->reflected(
372            static function (ReflectionClass $class) use ($type): bool {
373                $name = $class->getName();
374
375                // is_a() rather than isEnum() alone: it is what narrows
376                // the name to a UnitEnum class-string, which is what
377                // ReflectionEnum requires.
378                if (!$class->isEnum() || !is_a($name, UnitEnum::class, true)) {
379                    return false;
380                }
381
382                $backing = (new ReflectionEnum($name))->getBackingType();
383
384                // Casting a ReflectionType to string is deprecated; a
385                // backed enum's type is always a named one.
386                return $backing instanceof ReflectionNamedType && $backing->getName() === $type;
387            },
388            sprintf('be %s-backed enums', $type),
389        );
390    }

418    /**
419     * Runs every expectation against the resolved targets.
420     *
421     * An `arch()` that matches nothing is a **failure**, not a silent
422     * pass: a rule whose namespace was renamed would otherwise stay
423     * green forever while enforcing nothing, which is the worst outcome
424     * on offer.
425     */
426    public function assert(): void
427    {
428        $this->resolved = $this->resolve();
429
430        Assert::assertNotSame([], $this->resolved, sprintf(
431            'No class matches %s — an architecture rule that targets nothing enforces nothing.',
432            implode(', ', $this->targets),
433        ));
434
435        Assert::assertNotSame([], $this->expectations, 'This architecture rule states no expectation.');
436
437        foreach ($this->expectations as $expectation) {
438            $expectation();
439        }
440    }

442    /**
443     * Resolved when the rule RUNS, never when it is built.
444     *
445     * `arch()` chains are assembled while a file is being collected,
446     * and `Architecture::configure()` has not run yet at that point —
447     * so a matcher that reaches for the universe eagerly memoizes the
448     * empty pre-configuration one and the rule then matches nothing.
449     * ✓ Measured: doing exactly that made every rule in
450     * tests/unit/Architecture/rules.pest.php resolve to nothing, in
451     * 0.002s instead of 0.064s. The predicates below are therefore
452     * built per call rather than per chain.
453     */
454    private function universe(): ArchitectureUniverse
455    {
456        return $this->universe ??= Architecture::universe();
457    }
458
459    /**
460     * Every (class, reference) pair where the reference is another class
461     * of the project's own source — the shape both dependency rules read.
462     *
463     * @return list<array{0: string, 1: string}>
464     */
465    private function ownReferences(): array
466    {
467        $pairs = [];
468
469        foreach ($this->resolved as $class) {
470            foreach (ArchPredicates::crossingsOf($this->universe(), $this->resolved, $class) as $reference) {
471                $pairs[] = [$class, $reference];
472            }
473        }
474
475        return $pairs;
476    }
477
478    /**
479     * @return list<string>
480     */
481    private function resolve(): array
482    {
483        $classes = [];
484
485        foreach (array_keys($this->universe()->classes()) as $class) {
486            if (self::matchesAny($class, $this->targets) && !self::matchesAny($class, $this->ignored)) {
487                $classes[] = $class;
488            }
489        }
490
491        return $classes;
492    }

494    /**
495     * @param Closure(string): bool $predicate
496     */
497    private function each(Closure $predicate, string $description): self
498    {
499        $this->expectations[] = function () use ($predicate, $description): void {
500            $violations = [];
501
502            foreach ($this->resolved as $class) {
503                if (!$predicate($class)) {
504                    $violations[] = $class;
505                }
506            }
507
508            Assert::assertSame([], $violations, sprintf('Expected every targeted class to %s.', $description));
509        };
510
511        return $this;
512    }
513
514    /**
515     * @param Closure(ReflectionClass<object>): bool $predicate
516     */
517    private function reflected(Closure $predicate, string $description): self
518    {
519        return $this->each(
520            static function (string $class) use ($predicate): bool {
521                if (!class_exists($class) && !interface_exists($class) && !trait_exists($class)) {
522                    return false;
523                }
524
525                return $predicate(new ReflectionClass($class));
526            },
527            $description,
528        );
529    }

531    /**
532     * @param list<string> $patterns
533     */
534    private static function matchesAny(string $class, array $patterns): bool
535    {
536        return array_any($patterns, fn($pattern) => self::matches($class, $pattern));
537    }
538
539    private static function matches(string $class, string $pattern): bool
540    {
541        if (!str_contains($pattern, '*')) {
542            return $class === $pattern || str_starts_with($class, rtrim($pattern, '\\') . '\\');
543        }
544
545        $regex = preg_quote($pattern, '/');
546        $regex = str_replace('\*\*', '.*', $regex);
547        $regex = str_replace('\*', '[^\\\\\\\\]*', $regex);
548
549        return preg_match('/^' . $regex . '$/', $class) === 1;
550    }
