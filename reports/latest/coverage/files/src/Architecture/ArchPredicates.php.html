40/**
41 * What an architecture question asks of ONE class, in one place.
42 *
43 * Two spellings reach these now — `arch()->expect(...)` and the pest
44 * `expect(...)` arch matchers (D1) — and the answer has to be the same
45 * from both. Holding the predicates here is what makes that structural
46 * rather than a promise: there is one implementation to be right or
47 * wrong, so the two spellings cannot drift into disagreeing.
48 *
49 * That is not a general preference for sharing. It is the specific
50 * lesson of D-108, where one verdict classification lived in five
51 * copies and the copies produced a false green — and of the guards
52 * beside it, which are deliberately NOT shared because they answer
53 * different questions that merely share a position. These answer the
54 * same question, so they are one.
55 */
56final readonly class ArchPredicates

163    /**
164     * The class a file declares is the class its path implies, judged
165     * against composer's PSR-4 map — the project's own autoload
166     * configuration answers, not a convention assumed here.
167     *
168     * @return Closure(string): bool
169     */
170    public static function casedCorrectly(ArchitectureUniverse $universe): Closure
171    {
172        return static function (string $class) use ($universe): bool {
173            $file = $universe->fileOf($class);
174            $real = $file === null ? false : realpath($file);
175
176            if ($real === false) {
177                return false;
178            }
179
180            foreach (self::psr4Prefixes() as $namespace => $directories) {
181                foreach ($directories as $directory) {
182                    $root = realpath($directory);
183
184                    if ($root === false || !str_starts_with($real, $root . DIRECTORY_SEPARATOR)) {
185                        continue;
186                    }
187
188                    $relative = explode('.', substr($real, strlen($root) + 1))[0];
189
190                    return rtrim($namespace, '\\') . '\\' . str_replace(DIRECTORY_SEPARATOR, '\\', $relative) === $class;
191                }
192            }
193
194            return false;
195        };
196    }

254    /**
255     * The references a class makes that LEAVE the target set and land in
256     * the project's own source — the one definition of "uses" both
257     * spellings read, in the outbound direction.
258     *
259     * Two things do not count. References outside the configured source
260     * — PHP built-ins, vendor packages — because this is a question
261     * about your own layering, and counting `Closure` as a dependency
262     * would force every rule to carry a whitelist of the standard
263     * library first. And references to another class of the SAME
264     * target, because they do not cross the boundary the rule is about:
265     * ✓ measured against pest 5.1.1, `expect('App\Models')->toUseNothing()`
266     * passes while one model uses a sibling model, and fails the moment
267     * one reaches out of `App\Models` — with a control proving the
268     * matcher can still fail.
269     *
270     * @param list<string> $targets every class the rule targets
271     *
272     * @return list<string>
273     */
274    public static function crossingsOf(ArchitectureUniverse $universe, array $targets, string $class): array
275    {
276        $known    = $universe->classes();
277        $inside   = array_flip($targets);
278        $crossing = [];
279
280        foreach ($universe->referencesOf($class) as $reference) {
281            if ($reference !== $class && !isset($inside[$reference]) && isset($known[$reference])) {
282                $crossing[] = $reference;
283            }
284        }
285
286        return $crossing;
287    }

289    /**
290     * The inbound direction of the same boundary rule: every pair where
291     * a class OUTSIDE the target set references one inside it.
292     *
293     * One walk of the universe for the whole target set, not one walk
294     * per targeted class — the reverse direction has to read every
295     * file, so asking it per class turns a rule over 200 classes into
296     * 200 full traversals.
297     *
298     * @param list<string> $targets every class the rule targets
299     *
300     * @return list<array{0: string, 1: string}> the user, and the targeted class it reached
301     */
302    public static function inboundPairs(ArchitectureUniverse $universe, array $targets): array
303    {
304        $inside = array_flip($targets);
305        $pairs  = [];
306
307        foreach (array_keys($universe->classes()) as $candidate) {
308            if (isset($inside[$candidate])) {
309                continue;
310            }
311
312            foreach ($universe->referencesOf($candidate) as $reference) {
313                if (isset($inside[$reference])) {
314                    $pairs[] = [$candidate, $reference];
315                }
316            }
317        }
318
319        return $pairs;
320    }

335    /**
336     * Nothing outside the target set references the class.
337     *
338     * @param list<string> $targets
339     *
340     * @return Closure(string): bool
341     */
342    public static function usedByNothing(ArchitectureUniverse $universe, array $targets): Closure
343    {
344        $inside = array_flip($targets);
345        $used   = [];
346
347        // NOT inboundPairs(): this walk counts a SIBLING user too, and
348        // that asymmetry is the incumbent's, measured rather than
349        // reasoned about. ✓ pest 5.1.1, 2026-09-06:
350        // `expect('ArchFixture\Deps\Pure')->toBeUsedInNothing()` fails
351        // with "Expecting 'Deps\Pure\Beta' not to be used on
352        // 'Deps\Pure\Alpha'" — both classes INSIDE the target — while
353        // `toUseNothing()` passes over the same two files. Outbound
354        // ignores the boundary's inside; inbound does not.
355        //
356        // ⚠ This corrects 2026-09-02, which recorded the crossing rule
357        // as measured for BOTH directions and made them share one
358        // implementation. The control run that day had no sibling
359        // reference in the inbound direction, so it proved the outbound
360        // half twice. A control only earns the name when it could have
361        // come out wrong.
362        //
363        // toOnlyBeUsedIn() keeps inboundPairs(): "only these may use it"
364        // is a question about outside users, and it takes arguments, so
365        // no grid has measured it. Changing it here would be reasoning,
366        // not measuring.
367        foreach (array_keys($universe->classes()) as $candidate) {
368            foreach ($universe->referencesOf($candidate) as $reference) {
369                if ($candidate !== $reference && isset($inside[$reference])) {
370                    $used[$reference] = true;
371                }
372            }
373        }
374
375        return static fn(string $class): bool => !isset($used[$class]);
376    }

378    /**
379     * Every own member is in the state $documented names.
380     *
381     * A parameter, not a fixed `true`: the incumbent's negated forms ask
382     * that every member be UNdocumented, which differs from "not every
383     * member is documented" whenever there are no members.
384     *
385     * @param list<ReflectionMethod|ReflectionProperty> $members
386     * @param list<non-empty-string>                    $generated names an enum declares without source
387     */
388    private static function documented(ArchitectureUniverse $universe, string $class, array $members, array $generated, bool $documented): bool
389    {
390        $file = $universe->fileOf($class);
391        $real = $file === null ? false : realpath($file);
392        $enum = enum_exists($class);
393
394        foreach ($members as $member) {
395            if ($enum && in_array($member->getName(), $generated, true)) {
396                continue;
397            }
398
399            $declaredIn = $member->getDeclaringClass()->getFileName();
400
401            if ($declaredIn === false || realpath($declaredIn) !== $real) {
402                continue;
403            }
404
405            if (($member->getDocComment() !== false) !== $documented) {
406                return false;
407            }
408        }
409
410        return true;
411    }

413    /**
414     * Promoted properties excluded: they are constructor parameters.
415     *
416     * @param class-string $class
417     *
418     * @return list<ReflectionProperty>
419     */
420    private static function ownProperties(string $class): array
421    {
422        return array_values(array_filter(
423            (new ReflectionClass($class))->getProperties(),
424            static fn(ReflectionProperty $property): bool => !$property->isPromoted(),
425        ));

428    private static function sourceOf(ArchitectureUniverse $universe, string $class): ?string
429    {
430        $file   = $universe->fileOf($class);
431        $source = $file === null ? false : file_get_contents($file);
432
433        return $source === false ? null : $source;

436    /**
437     * @return array<string, list<string>>
438     */
439    private static function psr4Prefixes(): array
440    {
441        $prefixes = [];
442
443        foreach (spl_autoload_functions() as $function) {
444            if (is_array($function) && $function[0] instanceof ClassLoader) {
445                foreach ($function[0]->getPrefixesPsr4() as $namespace => $directories) {
446                    $prefixes[$namespace] = $directories;
447                }
448            }
449        }
450
451        return $prefixes;
452    }
