11/*
12 * A thin proxy layer for spatie/pest-plugin-snapshots' function
13 * surface (growth G2 follow-up). That package hard-requires
14 * pestphp/pest itself, which D-019's own PestBuilder::ensureUsable()
15 * guard already refuses to coexist with — so it can never actually be
16 * installed alongside Crucible's pest dialect, same reasoning as the
17 * Pest\Laravel/Pest\Livewire bridges (src/Bridge/PestLaravel,
18 * src/Bridge/PestLivewire). Every exported function is, in the real
19 * package, a one-line proxy to a method on
20 * Spatie\Snapshots\MatchesSnapshots — a trait, not a native TestCase
21 * method, so PestBuilder auto-mixes it into every uses() class when
22 * spatie/phpunit-snapshot-assertions (the standalone package that
23 * actually provides the trait — no pestphp/pest dependency at all)
24 * is installed, mirroring real Pest's own
25 * Plugin::uses(MatchesSnapshots::class) (see AutoMixedTrait, in this
26 * same directory, and its call site in PestBuilder::build()).
27 *
28 * Most of these methods are public on the real trait, so a plain
29 * ->method() call would work here too — routed through
30 * CurrentTest::call() anyway, for uniformity with the Laravel bridge
31 * and because that's how real Pest's own HigherOrderMessage::call()
32 * reaches every proxied method regardless of visibility.
33 *
34 * Scoped to the functions real-world test suites actually exercise
35 * (assertMatchesSnapshot, assertMatchesJsonSnapshot — verified
36 * against spatie/laravel-data) plus the rest of the trivial,
37 * same-shape family. The expect()->extend() custom matchers
38 * (toMatchSnapshot etc.) are included too — Crucible's own
39 * Expectation::extend() already exists (built for user-authored
40 * matchers), so these are wiring, not new capability; unlike the
41 * functions above, laravel-data's own suite never calls them, so
42 * they're unverified against a real benchmark, only unit-tested.
43 * One exception: toMatchSnapshot() itself is deliberately NOT
44 * registered — it collides with Crucible's own, unrelated, native
45 * toMatchSnapshot() (D-076) — see the comment at the registrations
46 * below.
47 *
48 * Not verified: first-time snapshot creation. The real trait marks
49 * that case incomplete via a #[PostCondition]-attributed method
50 * (PHPUnit\Framework\Attributes\PostCondition) — whether Crucible's
51 * engine honors that real PHPUnit attribute (as opposed to its own,
52 * src/Attributes/PostCondition.php) on a trait mixed in this way is
53 * untested; the real spatie/laravel-data benchmark this was verified
54 * against ships its snapshots pre-created, so only the
55 * compare-against-existing-snapshot path was exercised.
56 */

71/**
72 * AutoMixedTrait::forClass() (this bridge's own build-time half —
73 * see that file) skips mixing MatchesSnapshots in for a final uses()
74 * class, or one that already declares a method with the same name as
75 * one of the trait's own (Crucible's own TestCase/Assert hierarchy
76 * declares assertMatchesSnapshot() itself, for its own unrelated
77 * D-076 snapshot feature — found via uses(PestTestCase::class), the
78 * default uses() class for pest-dialect tests, with
79 * spatie/phpunit-snapshot-assertions installed). Without this check,
80 * CurrentTest::call() would silently fall through to reflecting
81 * whatever unrelated method of that name the class actually has —
82 * calling Crucible's own, differently-behaved implementation instead
83 * of Spatie's, with no error at all. This turns that into a clear,
84 * actionable one instead.
85 *
86 * Walks the class's own ancestry, not just class_uses($instance::class)
87 * alone — the running instance is usually one more subclass down from
88 * where MatchesSnapshots was actually mixed in
89 * (SnapshotIdentityWrapper's own generated wrapper, which class_uses()
90 * alone can't see past since it only reports a class's own directly
91 * `use`d traits, not an ancestor's).
92 *
93 * @param list<mixed> $arguments
94 */
95function dispatch(string $method, array $arguments): mixed
96{
97    $instance = CurrentTest::get();
98
99    if (!usesMatchesSnapshots($instance::class)) {
100        throw new ConfigurationException(sprintf(
101            "Spatie\\Snapshots\\%s() is not available on %s: either spatie/phpunit-snapshot-assertions "
102                . 'is not installed, or %s already declares a method with the same name as one of '
103                . "MatchesSnapshots's own — use a uses() class without that conflict (a real "
104                . 'PHPUnit-descended TestCase, as spatie/laravel-data itself does), not one from '
105                . "Crucible's own TestCase/Assert hierarchy (it declares assertMatchesSnapshot() itself, "
106                . 'for its own unrelated D-076 snapshot feature).',
107            $method,
108            $instance::class,
109            $instance::class,
110        ));
111    }
112
113    return CurrentTest::call($method, $arguments);
114}

116/** @param class-string $class */
117function usesMatchesSnapshots(string $class): bool
118{
119    do {
120        $uses = class_uses($class);
121
122        if (in_array(MatchesSnapshots::class, $uses === false ? [] : $uses, true)) {
123            return true;
124        }
125
126        $class = get_parent_class($class);
127    } while ($class !== false);
128
129    return false;
130}

202// expect()->extend() registrations mirror the real plugin's own
203// Functions.php exactly: top-level calls, run once (guarded by this
204// whole file's function_exists guard in PestBuilder::ensureUsable()).
205// expect() requires an argument here (unlike real Pest's own
206// zero-arg form), but extend() never reads $this->value at
207// registration time — only when the matcher is later invoked on a
208// real expectation — so the value passed in is inert.
209//
210// No toMatchSnapshot() registration here, deliberately: Crucible
211// already has its own, unrelated, native toMatchSnapshot()
212// (Expectation.php, D-076's file-based snapshot feature). A real
213// PHP method always resolves before __call()'s extension lookup, so
214// registering one under this exact name would be permanently
215// unreachable dead code — and worse, silently misleading, since a
216// migrated Pest test calling expect()->toMatchSnapshot() would
217// silently get Crucible's own different storage format instead of
218// Spatie's, rather than either working correctly or failing loudly.
219// The plain function form, \Spatie\Snapshots\assertMatchesSnapshot(),
220// has no such collision and works as expected — that's the spelling
221// to use for Spatie's specific file format.
222expect(null)->extend('toMatchFileHashSnapshot', function (): void {
223    dispatch('assertMatchesFileHashSnapshot', [$this->value]);
224});
