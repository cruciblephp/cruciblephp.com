65/**
66 * The supervisor of the worker pool (nextest architecture): plans
67 * which tests run in-process and which run in spawned workers, feeds
68 * workers manifests, multiplexes their NDJSON streams, and re-emits
69 * every test-scoped event onto the one supervisor-owned stream —
70 * listeners (console, NDJSON log, result cache) cannot tell where a
71 * test ran.
72 *
73 * Workers are consumed from a queue in LPT order (Graham 1969):
74 * costliest unit first, next unit to the first free slot — cached
75 * durations make the estimate. A worker that reaches EOF without its
76 * run:finish handshake crashed: every expected-but-unfinished test is
77 * reported as errored, with the worker's stderr tail as the reason.
78 * The crash is contained; the run continues.
79 */
80final class Supervisor
81{

129    /**
130     * Workers do not inherit the parent's -d flags, so whichever
131     * driver the parent is running with is re-created on the child
132     * command line (D-041). A driver the child's php.ini already
133     * loads produces a harmless already-loaded notice.
134     *
135     * Without coverage the child gets the opposite instruction: an
136     * xdebug hooking every call inflates every duration the worker
137     * reports, and the durations feed the result cache the scheduler
138     * plans from. Step debugging is left alone — a mode carrying
139     * `debug` is an operator with a debugger attached, not a default.
140     *
141     * @return list<non-empty-string>
142     */
143    private function phpArguments(): array
144    {
145        if ($this->coverageDirectory === null) {
146            return extension_loaded('xdebug') && !str_contains(TimingOverhead::mode(), 'debug')
147                ? ['-d', 'xdebug.mode=off']
148                : [];
149        }
150
151        // Branch analysis is xdebug territory (D-062) — a pcov child
152        // could collect lines but never branches, so the supervisor
153        // recreates the driver its own detection chose.
154        if (extension_loaded('pcov') && !$this->coverageBranch) {
155            return ['-d', 'extension=pcov.so', '-d', 'pcov.enabled=1'];
156        }
157
158        if (extension_loaded('xdebug')) {
159            return ['-d', 'zend_extension=xdebug.so', '-d', 'xdebug.mode=coverage'];
160        }
161
162        return [];
163    }

165    /**
166     * @param list<TestGroup>  $groups     already scheduled
167     * @param ?callable(): ?RunSummary $afterTests run-scoped work emitted inside the run (checks, and the Vitest fold-in returning its count delta)
168     *                                      bracket — after the workers, before RunFinished
169     */
170    public function run(array $groups, ?callable $afterTests = null): RunSummary
171    {
172        $started = hrtime(true);
173
174        $planned = 0;
175
176        foreach ($groups as $group) {
177            $planned += count($group->tests);
178        }
179
180        // Same plan size the sequential runner announces: a reader
181        // showing progress against a total must not lose it because the
182        // run happened to be parallel.
183        $this->emitter->emit(new RunStarted(tests: $planned));
184
185        [$inProcess, $units] = $this->plan($groups);
186
187        // #[Depends] across the process boundary: a unit that needs a
188        // value some other unit produces makes that producer record it.
189        // Asked for before anything runs, so the in-process portion —
190        // which goes first — keeps what the workers will need.
191        $needed = [];
192
193        foreach ($units as $unit) {
194            foreach ($unit->dependsOn as $name) {
195                if (!in_array($name, $needed, true)) {
196                    $needed[] = $name;
197                }
198            }
199        }
200
201        $this->dependencyValues->want($needed);
202
203        $units = array_map(
204            static function (PlannedUnit $planned) use ($needed): PlannedUnit {
205                $provides = [];
206
207                foreach ($planned->expected as $id) {
208                    if (in_array($id->name, $needed, true) && !in_array($id->name, $provides, true)) {
209                        $provides[] = $id->name;
210                    }
211                }
212
213                return $planned->providing($provides);
214            },
215            $units,
216        );
217
218        if ($inProcess !== []) {
219            $this->tallySummary($this->inProcessRunner->execute($inProcess));
220        }
221
222        $this->runUnits($units);
223
224        $delta = $afterTests !== null ? $afterTests() : null;
225
226        $summary = new RunSummary(
227            passed: $this->counts['pass'],
228            failed: $this->counts['fail'],
229            errored: $this->counts['error'],
230            skipped: $this->counts['skip'],
231            incomplete: $this->counts['incomplete'],
232            risky: $this->counts['risky'],
233            deprecations: $this->issueCounts['deprecations'],
234            notices: $this->issueCounts['notices'],
235            warnings: $this->issueCounts['warnings'],
236            untested: $this->counts['untested'],
237        );
238
239        // Same D-071 completeness rule as the sequential runner: lost
240        // workers report their tests as errored, so the tally only falls
241        // short of the plan when something halted the run. Computed on
242        // the PHP plan before external results (Vitest) fold in.
243        $complete = $this->fullSuite && $summary->total() === $planned;
244
245        if ($delta instanceof RunSummary) {
246            $summary = $summary->plus($delta);
247        }
248
249        $this->emitter->emit(new RunFinished(
250            $summary,
251            (hrtime(true) - $started) / 1e9,
252            $complete,
253        ));
254
255        return $summary;
256    }

258    /**
259     * Splits the scheduled groups into the in-process remainder and
260     * the worker units, honoring the spec's three isolation levels:
261     * #[RunClassInSeparateProcess] sends the whole group to one
262     * worker; #[RunTestsInSeparateProcesses] (class) and
263     * #[RunInSeparateProcess] (method) send each marked test to its
264     * own worker. --process-isolation is the run-wide form of the
265     * last one: every test is marked. With --parallel above 1 there
266     * is no in-process remainder: every group becomes a unit.
267     *
268     * @param list<TestGroup> $groups
269     *
270     * @return array{list<TestGroup>, list<PlannedUnit>}
271     */
272    private function plan(array $groups): array
273    {
274        $inProcess = [];
275        $units     = [];
276
277        foreach ($groups as $group) {
278            if ($group->tests === []) {
279                continue;
280            }
281
282            $file = $group->tests[0]->id->file;
283
284            $wholeClass = false;
285
286            foreach ($group->tests as $test) {
287                if ($test->metadata->has(RunClassInSeparateProcess::class)) {
288                    $wholeClass = true;
289
290                    break;
291                }
292            }
293
294            if ($wholeClass) {
295                $units[] = $this->unitFor($file, $group->tests, isolated: true);
296
297                continue;
298            }
299
300            $remaining     = [];
301            $isolatedNames = [];
302
303            foreach ($group->tests as $test) {
304                $isolated = $this->processIsolation
305                    || $test->metadata->has(RunInSeparateProcess::class)
306                    || $test->metadata->has(RunTestsInSeparateProcesses::class);
307
308                if (!$isolated) {
309                    $remaining[] = $test;
310
311                    continue;
312                }
313
314                // Dataset rows share a declared name and one worker.
315                if (!in_array($test->id->name, $isolatedNames, true)) {
316                    $isolatedNames[] = $test->id->name;
317                }
318            }
319
320            foreach ($isolatedNames as $name) {
321                $rows = [];
322
323                foreach ($group->tests as $test) {
324                    if ($test->id->name === $name) {
325                        $rows[] = $test;
326                    }
327                }
328
329                $units[] = $this->unitFor($file, $rows, isolated: true);
330            }
331
332            if ($remaining === []) {
333                continue;
334            }
335
336            if ($this->parallel > 1) {
337                $units[] = $this->unitFor($file, $remaining);
338
339                continue;
340            }
341
342            $inProcess[] = count($remaining) === count($group->tests)
343                ? $group
344                : new TestGroup($group->name, $remaining, $group->beforeAll, $group->afterAll);
345        }
346
347        return [$inProcess, $units];
348    }

350    /**
351     * Units always carry exact id strings, never "the whole file" —
352     * whatever selection (--filter, --group) removed on the
353     * supervisor side must stay removed on the worker side.
354     *
355     * @param non-empty-string     $file
356     * @param list<TestDefinition> $tests
357     * @param bool                 $isolated the tests asked for their own process (see {@see attributeStderr()})
358     */
359    private function unitFor(string $file, array $tests, bool $isolated = false): PlannedUnit
360    {
361        $cost     = 0.0;
362        $ids      = [];
363        $expected = [];
364        $own      = [];
365
366        foreach ($tests as $test) {
367            $expected[] = $test->id;
368            $ids[]      = $test->id->toString();
369            $own[]      = $test->id->name;
370            $cost += $this->history->duration($test->id->toString()) ?? 0.0;
371        }
372
373        // A prerequisite this unit runs itself resolves in-process; only
374        // what it cannot produce has to arrive from an earlier unit.
375        $dependsOn = [];
376
377        foreach ($tests as $test) {
378            foreach ($test->dependencies as $dependency) {
379                if (!in_array($dependency, $own, true) && !in_array($dependency, $dependsOn, true)) {
380                    $dependsOn[] = $dependency;
381                }
382            }
383        }
384
385        // Method-before-class precedence is already settled in the
386        // collection, so the first match is the effective one.
387        $preserve = false;
388
389        foreach ($tests as $test) {
390            if ($test->metadata->first(PreserveGlobalState::class)?->enabled === true) {
391                $preserve = true;
392
393                break;
394            }
395        }
396
397        return new PlannedUnit(new WorkUnit($file, $ids, $cost), $expected, $dependsOn, [], $preserve, $isolated);
398    }

400    /**
401     * @param list<PlannedUnit> $units
402     */
403    private function runUnits(array $units): void
404    {
405        // LPT: costliest unit first, next unit to the first free slot.
406        // The sort is stable, so unknown costs keep scheduled order.
407        usort($units, static fn(PlannedUnit $a, PlannedUnit $b): int => $b->unit->estimatedCost <=> $a->unit->estimatedCost);
408
409        /** @var array<int, WorkerProcess> $active */
410        $active = [];
411
412        /** @var array<int, array<string, true>> $finished finished test ids per worker */
413        $finished = [];
414
415        $slots = max($this->parallel, 1);
416
417        // ParaTest's parallel contract: each worker slot carries a
418        // stable TEST_TOKEN (1..N, reused as slots free up, so
419        // token-keyed databases and caches are reused, not
420        // multiplied) plus a run-unique UNIQUE_TEST_TOKEN. Only a
421        // parallel run sets them.
422        /** @var list<int<1, max>> $freeSlots */
423        $freeSlots = range(1, $slots);
424
425        /** @var array<int, int<1, max>> $slotOf worker object id => slot */
426        $slotOf = [];
427
428        $runToken = bin2hex(random_bytes(8));
429
430        // What earlier units (and the in-process portion) produced.
431        $available = $this->dependencyValues->provided();
432
433        $exported = null;
434
435        /** @var array<int, ?non-empty-string> $artifactOf worker object id => where it writes its values */
436        $artifactOf = [];
437
438        /** @var array<int, bool> $isolatedOf worker object id => its unit asked for its own process */
439        $isolatedOf = [];
440
441        while ($units !== [] || $active !== []) {
442            while ($units !== [] && count($active) < $slots) {
443                // A unit whose prerequisites have not been produced yet
444                // waits for them, so long as something is still running
445                // that might produce them. When nothing is active the
446                // head goes anyway: an unsatisfiable dependency must
447                // report untested, not stall the run.
448                $index = $this->nextReady($units, $available, $active === []);
449
450                if ($index === null) {
451                    break;
452                }
453
454                $planned = $units[$index];
455                unset($units[$index]);
456                $units = array_values($units);
457                $slot  = array_shift($freeSlots) ?? 1;
458
459                $artifact = $planned->provides === []
460                    ? null
461                    : sys_get_temp_dir() . '/crucible-depends-' . bin2hex(random_bytes(6)) . '.json';
462
463                $incoming = [];
464
465                foreach ($planned->dependsOn as $name) {
466                    if (isset($available[$name])) {
467                        $incoming[$name] = $available[$name];
468                    }
469                }
470
471                // #[PreserveGlobalState(true)]: captured at dispatch, so
472                // the worker sees the state as it stands now rather than
473                // as it stood when the run was planned. Captured once —
474                // every later unit that asks gets the same snapshot,
475                // which is also what makes the skip warnings print once.
476                $globalState = null;
477
478                if ($planned->preserveGlobalState) {
479                    if (!$exported instanceof GlobalStateExport) {
480                        $exported = GlobalStateExport::capture();
481
482                        // On stderr: stdout is the progress view's, and a
483                        // note about the run is not a test event — it has
484                        // no place in the NDJSON contract.
485                        foreach ($exported->skipped as $name) {
486                            fwrite(STDERR, sprintf(
487                                'Global state was not preserved for %s: no serializer can carry it.' . PHP_EOL,
488                                $name,
489                            ));
490                        }
491                    }
492
493                    $globalState = $exported->toJson();
494                }
495
496                $worker = WorkerProcess::spawn(
497                    $this->crucibleBinary,
498                    $this->workingDirectory,
499                    new WorkerManifest(
500                        $this->configurationPath,
501                        [$planned->unit],
502                        $this->order,
503                        $this->seed,
504                        $this->parallel > 1 ? $slot : null,
505                        $this->parallel > 1 ? $runToken . '_' . $slot : null,
506                        $this->retries,
507                        $this->coverageDirectory !== null
508                            ? $this->coverageDirectory . '/worker-' . bin2hex(random_bytes(6)) . '.json'
509                            : null,
510                        $this->updateSnapshots,
511                        $this->coverageBranch,
512                        $this->strictCoverage,
513                        $this->browserOverride,
514                        $this->debugOverride,
515                        $this->overrides,
516                        $this->includePaths,
517                        $incoming,
518                        $planned->provides,
519                        $artifact,
520                        $this->pathCoverage,
521                        $globalState,
522                    ),
523                    $planned->expected,
524                    $this->phpArguments(),
525                );
526
527                if (!$worker instanceof WorkerProcess) {
528                    $freeSlots[] = $slot;
529                    $this->reportLost($planned->expected, [], 'The worker process for this test could not be started.');
530
531                    continue;
532                }
533
534                $active[spl_object_id($worker)]     = $worker;
535                $finished[spl_object_id($worker)]   = [];
536                $slotOf[spl_object_id($worker)]     = $slot;
537                $artifactOf[spl_object_id($worker)] = $artifact;
538                $isolatedOf[spl_object_id($worker)] = $planned->isolated;
539            }
540
541            if ($active === []) {
542                continue;
543            }
544
545            $this->await($active);
546
547            foreach ($active as $key => $worker) {
548                foreach ($worker->readLines() as $line) {
549                    if ($this->parser->isRunFinished($line)) {
550                        $worker->markFinished();
551
552                        continue;
553                    }
554
555                    $event = $this->parser->parse($line);
556
557                    if (!$event instanceof \LucianoPereira\Crucible\Event\Event) {
558                        continue;
559                    }
560
561                    // What the worker wrote before a test started (a
562                    // startup notice, the bootstrap) belongs to no test;
563                    // what it wrote while one ran belongs to that test.
564                    if ($event instanceof TestStarted) {
565                        $this->forwardStderr($worker->takeStderr());
566                    }
567
568                    if ($event instanceof TestFinished && $isolatedOf[$key]) {
569                        $event = $this->attributeStderr($event, $worker->takeStderr());
570                    } elseif ($event instanceof TestFinished) {
571                        $this->forwardStderr($worker->takeStderr());
572                    }
573
574                    $this->emitter->emit($event);
575
576                    if ($event instanceof TestFinished) {
577                        $finished[$key][$event->test->toString()] = true;
578                        $this->tally($event->outcome);
579
580                        if ($event->blocked) {
581                            $this->counts['untested']++;
582                        }
583
584                        $this->tallyIssues($event);
585                    }
586                }
587
588                $worker->drainStderr();
589
590                if (!$worker->atEof()) {
591                    continue;
592                }
593
594                $worker->close();
595
596                // A crash carries the tail in its reason already; a clean
597                // exit hands over whatever came after the last test.
598                if ($worker->hasFinished()) {
599                    $this->forwardStderr($worker->takeStderr());
600                }
601
602                if (!$worker->hasFinished()) {
603                    $this->reportLost(
604                        $worker->expected,
605                        $finished[$key],
606                        'The worker process running this test exited unexpectedly.'
607                            . ($worker->stderrTail() !== '' ? ' Worker stderr: ' . $worker->stderrTail() : ''),
608                    );
609                }
610
611                $freeSlots[] = $slotOf[$key];
612
613                // Whatever this unit was asked to produce is now
614                // available to the units that were waiting for it.
615                $artifact = $artifactOf[$key] ?? null;
616
617                if ($artifact !== null) {
618                    $available = [...$available, ...$this->readValues($artifact)];
619                }
620
621                unset($active[$key], $finished[$key], $slotOf[$key], $artifactOf[$key], $isolatedOf[$key]);
622            }
623        }
624    }

626    /**
627     * A test that asked for its own process and wrote to STDERR there (D-124)
628     * errors, with that text as the message — PHPUnit's rule for a
629     * child process (ChildProcessResultProcessor), so a suite moving
630     * across keeps its verdicts. The output is never dropped: dropping
631     * it hid a debugging probe with nothing saying it was lost.
632     */
633    private function attributeStderr(TestFinished $event, string $stderr): TestFinished
634    {
635        $text = trim($stderr);
636
637        if ($text === '') {
638            return $event;
639        }
640
641        $message = 'The test wrote to STDERR in its separate process:' . PHP_EOL . $text;
642
643        if ($event->failure instanceof Failure) {
644            $message .= PHP_EOL . PHP_EOL . 'The test itself also reported: ' . $event->failure->message;
645        }
646
647        return new TestFinished(
648            $event->test,
649            Outcome::Errored,
650            $event->duration,
651            new Failure($message),
652            $event->attempt,
653            null,
654            $event->issues,
655            $event->quarantined,
656            $event->property,
657            $event->retried,
658            $event->snapshots,
659            $event->propertyClean,
660            $event->snapshotKeys,
661            false,
662            $event->assertions,
663        );
664    }

666    /**
667     * Output that belongs to no isolated test reaches the console as it
668     * would have in process: on STDERR, verbatim. A --parallel worker is
669     * only a share of the run, so its tests keep the in-process rule.

671    private function forwardStderr(string $stderr): void
672    {
673        if ($stderr !== '') {
674            fwrite(STDERR, $stderr);
675        }

678    /**
679     * The first unit whose prerequisites are all available, or — when
680     * nothing is running that could still satisfy one — the first unit
681     * regardless, so an unproducible dependency ends as a reported
682     * untested rather than a stalled run.
683     *
684     * @param list<PlannedUnit>     $units
685     * @param array<string, string> $available
686     */
687    private function nextReady(array $units, array $available, bool $nothingActive): ?int
688    {
689        foreach ($units as $index => $planned) {
690            $ready = true;
691
692            foreach ($planned->dependsOn as $name) {
693                if (!isset($available[$name])) {
694                    $ready = false;
695
696                    break;
697                }
698            }
699
700            if ($ready) {
701                return $index;
702            }
703        }
704
705        return $nothingActive && $units !== [] ? 0 : null;
706    }
707
708    /**
709     * @return array<string, string>
710     */
711    private function readValues(string $artifact): array
712    {
713        if (!is_file($artifact)) {
714            return [];
715        }
716
717        $decoded = json_decode((string) file_get_contents($artifact), true);
718        @unlink($artifact);
719
720        return is_array($decoded) ? DependencyValues::fromArray($decoded) : [];
721    }

723    /**
724     * Blocks until any active worker has data (or the timeout passes).
725     *
726     * @param array<int, WorkerProcess> $active
727     */
728    private function await(array $active): void
729    {
730        $read = [];
731
732        foreach ($active as $worker) {
733            $read[] = $worker->stdout();
734            $read[] = $worker->stderr();
735        }
736
737        $write  = null;
738        $except = null;
739
740        @stream_select($read, $write, $except, 0, (int) (self::SELECT_TIMEOUT_SECONDS * 1_000_000));
741    }
742
743    /**
744     * Every expected test the dead worker never finished is errored —
745     * visibly, on the stream, with the crash diagnostics. A lost
746     * worker must never mean silently lost tests.
747     *
748     * @param list<TestId>        $expected
749     * @param array<string, true> $finished
750     * @param non-empty-string    $reason
751     */
752    private function reportLost(array $expected, array $finished, string $reason): void
753    {
754        foreach ($expected as $id) {
755            if (isset($finished[$id->toString()])) {
756                continue;
757            }
758
759            $this->emitter->emit(new TestStarted($id));
760            $this->emitter->emit(new TestFinished($id, Outcome::Errored, 0.0, reason: $reason));
761            $this->tally(Outcome::Errored);
762        }
763    }
764
765    private function tally(Outcome $outcome): void
766    {
767        $this->counts[$outcome->value]++;
768    }
769
770    private function tallyIssues(TestFinished $event): void
771    {
772        foreach ($event->issues as $issue) {
773            $this->issueCounts[match ($issue->kind) {
774                IssueKind::Deprecation => 'deprecations',
775                IssueKind::Notice      => 'notices',
776                IssueKind::Warning     => 'warnings',
777            }]++;
778        }
779    }
780
781    private function tallySummary(RunSummary $summary): void
782    {
783        $this->counts['pass'] += $summary->passed;
784        $this->counts['fail'] += $summary->failed;
785        $this->counts['error'] += $summary->errored;
786        $this->counts['skip'] += $summary->skipped;
787        $this->counts['incomplete'] += $summary->incomplete;
788        $this->counts['risky'] += $summary->risky;
789        $this->counts['untested'] += $summary->untested;
790        $this->issueCounts['deprecations'] += $summary->deprecations;
791        $this->issueCounts['notices'] += $summary->notices;
792        $this->issueCounts['warnings'] += $summary->warnings;
793    }
