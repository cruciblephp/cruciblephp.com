33/**
34 * One spawned worker: `php crucible --worker`, manifest written to stdin
35 * (then closed — the worker reads to EOF), NDJSON events read from
36 * stdout line by line, stderr drained for crash diagnostics. All
37 * reads are non-blocking; the Supervisor multiplexes many workers
38 * over stream_select.
39 */
40final class WorkerProcess
41{
42    private string $stdoutBuffer = '';
43
44    private string $stderrTail = '';
45
46    private string $stderrPending = '';
47
48    private bool $finished = false;
49
50    /**
51     * @param resource     $process
52     * @param resource     $stdout
53     * @param resource     $stderr
54     * @param list<TestId> $expected the test ids this worker must finish
55     */
56    private function __construct(private $process, private $stdout, private $stderr, public readonly array $expected) {}
57
58    /**
59     * @param non-empty-string        $crucibleBinary path of the crucible entry script
60     * @param list<TestId>            $expected
61     * @param list<non-empty-string>  $phpArguments extra php CLI arguments (e.g. the -d flags that
62     *                                              load the coverage driver — children do not inherit
63     *                                              the parent's -d flags)
64     * @param array<string, string>   $env          extra environment for the child, merged over the
65     *                                              inherited environment (the mutation worker injects
66     *                                              its mutant this way); empty = plain inheritance
67     */
68    public static function spawn(
69        string $crucibleBinary,
70        WorkingDirectory $workingDirectory,
71        WorkerManifest $manifest,
72        array $expected,
73        array $phpArguments = [],
74        array $env = [],
75    ): ?self {
76        $process = proc_open(
77            [PHP_BINARY, ...$phpArguments, $crucibleBinary, '--worker'],
78            [0 => ['pipe', 'r'], 1 => ['pipe', 'w'], 2 => ['pipe', 'w']],
79            $pipes,
80            $workingDirectory->path,
81            $env === [] ? null : [...getenv(), ...$env],
82        );
83
84        if (!is_resource($process)) {
85            return null;
86        }
87
88        fwrite($pipes[0], $manifest->toJson());
89        fclose($pipes[0]);
90
91        stream_set_blocking($pipes[1], false);
92        stream_set_blocking($pipes[2], false);
93
94        return new self($process, $pipes[1], $pipes[2], $expected);
95    }
96
97    /**
98     * @return resource
99     */
100    public function stdout()
101    {
102        return $this->stdout;
103    }
104
105    /**
106     * @return resource
107     */
108    public function stderr()
109    {
110        return $this->stderr;
111    }
112
113    /**
114     * Complete NDJSON lines currently available, without blocking.
115     * Partial lines stay buffered until their newline arrives.
116     *
117     * @return list<string>
118     */
119    public function readLines(): array
120    {
121        $lines = [];
122
123        while (($chunk = fgets($this->stdout)) !== false) {
124            $this->stdoutBuffer .= $chunk;
125
126            if (!str_ends_with($this->stdoutBuffer, "\n")) {
127                continue;
128            }
129
130            $line               = rtrim($this->stdoutBuffer, "\n");
131            $this->stdoutBuffer = '';
132
133            if ($line !== '') {
134                $lines[] = $line;
135            }
136        }
137
138        return $lines;
139    }
140
141    /**
142     * Drains stderr into a bounded tail kept for crash diagnostics, and
143     * into the pending text {@see takeStderr()} hands over — what a test
144     * wrote is the test's output, not only a crash's last words.
145     */
146    public function drainStderr(): void
147    {
148        // close() drained it one last time; what it held stays pending.
149        if (!is_resource($this->stderr)) {
150            return;
151        }
152
153        while (($chunk = fgets($this->stderr)) !== false) {
154            $this->stderrTail = substr($this->stderrTail . $chunk, -2048);
155            $this->stderrPending .= $chunk;
156        }
157    }
158
159    /**
160     * Everything drained since the last call. A pipe is written in
161     * order, so draining when a test:finish line is read collects all
162     * the test wrote before it finished.
163     */
164    public function takeStderr(): string
165    {
166        $this->drainStderr();
167
168        $pending             = $this->stderrPending;
169        $this->stderrPending = '';
170
171        return $pending;
172    }
173
174    public function stderrTail(): string
175    {
176        return trim($this->stderrTail);
177    }
178
179    /**
180     * The worker emitted its run:finish — the completion handshake.
181     */
182    public function markFinished(): void
183    {
184        $this->finished = true;
185    }
186
187    public function hasFinished(): bool
188    {
189        return $this->finished;
190    }
191
192    public function atEof(): bool
193    {
194        return feof($this->stdout);
195    }
196
197    /**
198     * Hard-kill the worker (signal 9 by number, so no pcntl constant is
199     * needed — the cold mutation path runs where pcntl is absent). Used to
200     * enforce a mutant's timeout; close() still reaps it afterwards.
201     */
202    public function terminate(): void
203    {
204        proc_terminate($this->process, 9);
205    }
206
207    public function close(): int
208    {
209        $this->drainStderr();
210
211        fclose($this->stdout);
212        fclose($this->stderr);
213
214        return proc_close($this->process);
215    }
216}
