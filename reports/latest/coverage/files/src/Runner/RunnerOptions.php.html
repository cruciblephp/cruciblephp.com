28     * @param bool                   $fullSuite           no CLI selection narrowed the plan (D-071); with every planned test finished, run:finish reports complete
29     * @param bool                   $disallowTestOutput  the spec's --disallow-test-output: a test that prints is risky
30     * @param bool                   $enforceTimeLimit    the spec's --enforce-time-limit: a test slower than its size allows is risky
31     * @param ?int                   $defaultTimeLimit    the spec's --default-time-limit, in seconds, for a test that declares no size
32     * @param bool                   $coverageTargeting   honour the Covers and Uses metadata when deciding what a test contributes; false is the spec's --disable-coverage-targeting
33     */

65    /**
66     * @param ?bool                  $stopOnDefect        CLI override; null falls back to the configuration
67     * @param ?bool                  $stopOnError         CLI override; null falls back to the configuration
68     * @param ?bool                  $stopOnFailure       CLI override; null falls back to the configuration
69     * @param ?bool                  $stopOnRisky         CLI override; null falls back to the configuration
70     * @param ?bool                  $stopOnSkipped       CLI override; null falls back to the configuration
71     * @param ?bool                  $stopOnIncomplete    CLI override; null falls back to the configuration
72     * @param ?bool                  $stopOnDeprecation   CLI override; null falls back to the configuration
73     * @param ?bool                  $stopOnNotice        CLI override; null falls back to the configuration
74     * @param ?bool                  $stopOnWarning       CLI override; null falls back to the configuration
75     * @param list<non-empty-string> $projectDirectories
76     * @param list<non-empty-string> $deprecationBaseline
77     * @param ?int<0, max>           $retries             CLI count override; the configured backoff shape survives it
78     * @param array<string, list<list<int>>> $propertyFailures
79     * @param ?bool                  $strictCoverage      CLI --strict-coverage override (D-063); null falls back to the configuration
80     * @param bool                   $fullSuite           no CLI selection narrowed the plan (D-071)
81     * @param Overrides              $overrides           CLI values that win over the configuration's, carried across the worker boundary
82     */
83    public static function fromConfiguration(
84        Configuration $configuration,
85        ?bool $stopOnDefect = null,
86        ?bool $stopOnError = null,
87        ?bool $stopOnFailure = null,
88        ?bool $stopOnRisky = null,
89        ?bool $stopOnSkipped = null,
90        ?bool $stopOnIncomplete = null,
91        ?bool $stopOnDeprecation = null,
92        ?bool $stopOnNotice = null,
93        ?bool $stopOnWarning = null,
94        array $projectDirectories = [],
95        WorkingDirectory $workingDirectory = new WorkingDirectory('/'),
96        array $deprecationBaseline = [],
97        ?int $retries = null,
98        array $propertyFailures = [],
99        bool $updateSnapshots = false,
100        ?bool $strictCoverage = null,
101        bool $fullSuite = false,
102        bool $inlineSnapshotRewrite = false,
103        Overrides $overrides = new Overrides(),
104    ): self {
105        return new self(
106            backupGlobals: $overrides->backupGlobals ?? $configuration->backupGlobals,
107            backupStaticProperties: $overrides->backupStaticProperties ?? $configuration->backupStaticProperties,
108            beStrictAboutChangesToGlobalState: $overrides->beStrictAboutChangesToGlobalState ?? $configuration->beStrictAboutChangesToGlobalState,
109            requireCoverageMetadata: $overrides->requireCoverageMetadata ?? $configuration->requireCoverageMetadata,
110            beStrictAboutCoverageMetadata: $strictCoverage ?? $configuration->beStrictAboutCoverageMetadata,
111            coverageTargeting: $overrides->coverageTargeting ?? true,
112            disallowTestOutput: $overrides->disallowTestOutput ?? false,
113            enforceTimeLimit: $overrides->enforceTimeLimit ?? false,
114            defaultTimeLimit: $overrides->defaultTimeLimit,
115            stopOnDefect: $stopOnDefect ?? $configuration->stopOnDefect,
116            stopOnError: $stopOnError ?? $configuration->stopOnError,
117            stopOnFailure: $stopOnFailure ?? $configuration->stopOnFailure,
118            stopOnRisky: $stopOnRisky ?? $configuration->stopOnRisky,
119            stopOnSkipped: $stopOnSkipped ?? $configuration->stopOnSkipped,
120            stopOnIncomplete: $stopOnIncomplete ?? $configuration->stopOnIncomplete,
121            stopOnDeprecation: $stopOnDeprecation ?? $configuration->stopOnDeprecation,
122            stopOnNotice: $stopOnNotice ?? $configuration->stopOnNotice,
123            stopOnWarning: $stopOnWarning ?? $configuration->stopOnWarning,
124            projectDirectories: $projectDirectories,
125            workingDirectory: $workingDirectory,
126            deprecationBaseline: $deprecationBaseline,
127            retries: $retries !== null ? $configuration->retries->withCount($retries) : $configuration->retries,
128            quarantine: $configuration->quarantine,
129            propertyFailures: $propertyFailures,
130            updateSnapshots: $updateSnapshots,
131            fullSuite: $fullSuite,
132            inlineSnapshotRewrite: $inlineSnapshotRewrite,
133            reportUselessTests: $overrides->reportUselessTests ?? true,
134        );
135    }
