35/**
36 * Captures the PHP-level issues a test triggers — deprecations,
37 * notices, warnings — without changing its outcome. Installed around
38 * each test, drained after it; the drained issues ride the
39 * test:finish event.
40 *
41 * Deprecations are attributed on capture (DeprecationScope): a
42 * trigger inside the project's own directories is `self`; a trigger
43 * in a dependency is `direct` when the nearest calling frame is
44 * project code and `indirect` otherwise. Baselined deprecations
45 * (acknowledged legacy, keyed file|message) are suppressed here, at
46 * the source — they appear nowhere downstream, which keeps every
47 * consumer of the stream consistent.
48 */
49final class IssueCollector
50{
51    private const int CAPTURED = E_DEPRECATED | E_USER_DEPRECATED
52        | E_NOTICE | E_USER_NOTICE
53        | E_WARNING | E_USER_WARNING;
54
55    /** @var list<Issue> */
56    private array $issues = [];
57
58    private bool $installed = false;
59
60    private int $deprecations = 0;
61
62    private int $notices = 0;
63
64    private int $warnings = 0;
65
66    /**
67     * @param list<non-empty-string> $projectDirectories absolute prefixes counting as "own code"
68     * @param list<non-empty-string> $baseline           suppressed deprecation keys, "file|message"
69     */
70    public function __construct(
71        private readonly array $projectDirectories,
72        private readonly WorkingDirectory $workingDirectory,
73        private readonly array $baseline = [],
74    ) {}
75
76    public function install(): void
77    {
78        set_error_handler($this->capture(...), self::CAPTURED);
79        $this->installed = true;
80    }
81
82    /**
83     * Uninstalls the handler and returns what the test triggered.
84     *
85     * @return list<Issue>
86     */
87    public function drain(): array
88    {
89        if ($this->installed) {
90            restore_error_handler();
91            $this->installed = false;
92        }
93
94        $issues       = $this->issues;
95        $this->issues = [];
96
97        return $issues;
98    }
99
100    /**
101     * Run-level tallies, accumulated across drains.
102     *
103     * @return array{deprecations: int, notices: int, warnings: int}
104     */
105    public function tallies(): array
106    {
107        return [
108            'deprecations' => $this->deprecations,
109            'notices'      => $this->notices,
110            'warnings'     => $this->warnings,
111        ];
112    }
113
114    public function resetTallies(): void
115    {
116        $this->deprecations = 0;
117        $this->notices      = 0;
118        $this->warnings     = 0;
119    }
120
121    private function capture(int $level, string $message, string $file = '', int $line = 0): bool
122    {
123        // Respect @-suppression and the configured reporting level:
124        // what PHP would not report, Crucible does not count.
125        if ((error_reporting() & $level) === 0) {
126            return false;
127        }
128
129        $kind = IssueKind::fromErrorLevel($level);
130
131        if (!$kind instanceof \LucianoPereira\Crucible\Event\IssueKind || $message === '' || $file === '') {
132            return false;
133        }
134
135        $relative = $this->relative($file);
136
137        $scope = $kind === IssueKind::Deprecation ? $this->scopeOf($file) : null;
138
139        if ($kind === IssueKind::Deprecation && in_array($relative . '|' . $message, $this->baseline, true)) {
140            return true; // acknowledged legacy: suppressed at the source
141        }
142
143        $this->issues[] = new Issue($kind, $message, $relative, max(1, $line), $scope);
144
145        match ($kind) {
146            IssueKind::Deprecation => $this->deprecations++,
147            IssueKind::Notice      => $this->notices++,
148            IssueKind::Warning     => $this->warnings++,
149        };
150
151        return true;
152    }
153
154    private function scopeOf(string $file): DeprecationScope
155    {
156        if ($this->isProjectFile($file)) {
157            return DeprecationScope::Self_;
158        }
159
160        // Triggered in a dependency: the nearest calling frame that
161        // is not the trigger file (and not this collector) decides
162        // direct vs indirect.
163        foreach (debug_backtrace(DEBUG_BACKTRACE_IGNORE_ARGS, 25) as $frame) {
164            $frameFile = $frame['file'] ?? '';
165
166            if (in_array($frameFile, ['', $file, __FILE__], true)) {
167                continue;
168            }
169
170            return $this->isProjectFile($frameFile) ? DeprecationScope::Direct : DeprecationScope::Indirect;
171        }
172
173        return DeprecationScope::Indirect;
174    }
175
176    private function isProjectFile(string $file): bool
177    {
178        // PHP reports the OS's own separator; the configured prefixes
179        // may be joined with '/'.
180        $file = WorkingDirectory::native($file);
181
182        return array_any(
183            $this->projectDirectories,
184            static fn(string $directory): bool => str_starts_with($file, WorkingDirectory::native($directory)),
185        );
186    }
187
188    /**
189     * @param non-empty-string $file
190     *
191     * @return non-empty-string
192     */
193    private function relative(string $file): string
194    {
195        $relative = $this->workingDirectory->relative($file);
196
197        return $relative === '' ? $file : $relative;
198    }
199}
