33/**
34 * Deterministic test ordering (Rothermel/Elbaum prioritization line).
35 * Pure: groups in, groups out; the same inputs — order, seed, history —
36 * always produce the same schedule, per the determinism principle.
37 *
38 * Ordering is two-level (groups, then tests within each group), and
39 * every ordering ends with a dependency repair pass so #[Depends]
40 * prerequisites always run before their dependents — reordering can
41 * never turn a passing suite into skips. --ignore-dependencies drops
42 * that pass and leaves the declared positions, which is the only thing
43 * the spec's switch turns off: an unmet #[Depends] still skips.
44 *
45 * `defects` is history-aware beyond the spec: outcomes are scored by
46 * defect severity and *recency* (each run further back counts half),
47 * so what broke yesterday outranks what flaked last week; ties break
48 * toward shorter tests for the fastest time-to-first-failure.
49 * `duration` runs fastest-first (the spec's semantics); LPT
50 * partitioning of the same cached durations arrives with the worker
51 * pool.
52 */
53final readonly class Scheduler
54{
55    /**
56     * @param bool $resolveDependencies run the repair pass; false is the spec's --ignore-dependencies
57     */
58    public function __construct(
59        private ExecutionOrder $order = ExecutionOrder::Declared,
60        private int $seed = 0,
61        private ResultCache $history = new ResultCache(),
62        private bool $resolveDependencies = true,
63    ) {}
64
65    /**
66     * @param list<TestGroup> $groups
67     *
68     * @return list<TestGroup>
69     */
70    public function schedule(array $groups): array
71    {
72        $groups = match ($this->order) {
73            ExecutionOrder::Declared      => $groups,
74            ExecutionOrder::Reversed      => $this->reversed($groups),
75            ExecutionOrder::Random        => $this->shuffled($groups),
76            ExecutionOrder::DefectsFirst  => $this->defectsFirst($groups),
77            ExecutionOrder::Duration      => $this->fastestFirst($groups),
78            ExecutionOrder::SizeAscending => $this->sizeAscending($groups),
79        };
80
81        return $this->resolveDependencies ? array_map($this->repaired(...), $groups) : $groups;
82    }
83
84    /**
85     * @param list<TestGroup> $groups
86     *
87     * @return list<TestGroup>
88     */
89    private function reversed(array $groups): array
90    {
91        return array_map(
92            static fn(TestGroup $group): TestGroup => new TestGroup(
93                $group->name,
94                array_reverse($group->tests),
95                $group->beforeAll,
96                $group->afterAll,
97            ),
98            array_reverse($groups),
99        );
100    }

102    /**
103     * Seeded and replayable: the same seed reproduces the same order,
104     * which is what turns "fails when shuffled" into a bug report.
105     *
106     * @param list<TestGroup> $groups
107     *
108     * @return list<TestGroup>
109     */
110    private function shuffled(array $groups): array
111    {
112        $randomizer = new Randomizer(new Mt19937($this->seed));
113
114        return array_map(
115            fn(TestGroup $group): TestGroup => new TestGroup(
116                $group->name,
117                $this->permuted($randomizer, $group->tests),
118                $group->beforeAll,
119                $group->afterAll,
120            ),
121            $this->permuted($randomizer, $groups),
122        );
123    }
124
125    /**
126     * Fisher–Yates over getInt(): the permutation a seed produces is
127     * defined by Crucible itself, not by an engine's shuffle internals —
128     * replayability is a documented guarantee, so the algorithm behind
129     * it must be owned code.
130     *
131     * @template T
132     *
133     * @param list<T> $items
134     *
135     * @return list<T>
136     */
137    private function permuted(Randomizer $randomizer, array $items): array
138    {
139        for ($i = count($items) - 1; $i > 0; $i--) {
140            $j = $randomizer->getInt(0, $i);
141
142            [$items[$i], $items[$j]] = [$items[$j], $items[$i]];
143        }
144
145        return array_values($items);
146    }

148    /**
149     * @param list<TestGroup> $groups
150     *
151     * @return list<TestGroup>
152     */
153    private function defectsFirst(array $groups): array
154    {
155        $score    = fn(TestDefinition $test): float => $this->defectScore($test);
156        $duration = fn(TestDefinition $test): float => $this->history->duration($test->id->toString()) ?? 0.0;
157
158        $groups = $this->sortedGroups($groups, static function (TestGroup $a, TestGroup $b) use ($score): int {
159            $best = static fn(TestGroup $group): float => max([0.0, ...array_map($score, $group->tests)]);
160
161            return $best($b) <=> $best($a);
162        });
163
164        return $this->sortedTests($groups, static fn(TestDefinition $a, TestDefinition $b): int => [$score($b), $duration($a)] <=> [$score($a), $duration($b)]);
165    }
166
167    /**
168     * @param list<TestGroup> $groups
169     *
170     * @return list<TestGroup>
171     */
172    private function fastestFirst(array $groups): array
173    {
174        $duration = fn(TestDefinition $test): float => $this->history->duration($test->id->toString()) ?? 0.0;
175
176        $groups = $this->sortedGroups($groups, static function (TestGroup $a, TestGroup $b) use ($duration): int {
177            $total = static fn(TestGroup $group): float => array_sum(array_map($duration, $group->tests));
178
179            return $total($a) <=> $total($b);
180        });
181
182        return $this->sortedTests($groups, static fn(TestDefinition $a, TestDefinition $b): int => $duration($a) <=> $duration($b));
183    }
184
185    /**
186     * @param list<TestGroup> $groups
187     *
188     * @return list<TestGroup>
189     */
190    private function sizeAscending(array $groups): array
191    {
192        $weight = $this->sizeWeight(...);
193
194        $groups = $this->sortedGroups($groups, static function (TestGroup $a, TestGroup $b) use ($weight): int {
195            $smallest = static fn(TestGroup $group): int => min([4, ...array_map($weight, $group->tests)]);
196
197            return $smallest($a) <=> $smallest($b);
198        });
199
200        return $this->sortedTests($groups, static fn(TestDefinition $a, TestDefinition $b): int => $weight($a) <=> $weight($b));
201    }
202
203    /**
204     * Defect severity, recency-weighted: each run further back in the
205     * history counts half as much as the one before it.
206     */
207    private function defectScore(TestDefinition $test): float
208    {
209        $score  = 0.0;
210        $factor = 1.0;
211
212        foreach ($this->history->outcomes($test->id->toString()) as $outcome) {
213            $score += $factor * match ($outcome) {
214                Outcome::Errored    => 6.0,
215                Outcome::Failed     => 5.0,
216                Outcome::Incomplete => 3.0,
217                Outcome::Risky      => 2.0,
218                Outcome::Skipped    => 1.0,
219                Outcome::Passed     => 0.0,
220            };
221            $factor /= 2.0;
222        }
223
224        return $score;
225    }
226
227    private function sizeWeight(TestDefinition $test): int
228    {
229        return match (true) {
230            $test->metadata->has(Small::class)  => 1,
231            $test->metadata->has(Medium::class) => 2,
232            $test->metadata->has(Large::class)  => 3,
233            default                             => 4,
234        };
235    }

237    /**
238     * @param list<TestGroup>                       $groups
239     * @param callable(TestGroup, TestGroup): int   $comparator
240     *
241     * @return list<TestGroup>
242     */
243    private function sortedGroups(array $groups, callable $comparator): array
244    {
245        usort($groups, $comparator);
246
247        return $groups;
248    }
249
250    /**
251     * @param list<TestGroup>                                   $groups
252     * @param callable(TestDefinition, TestDefinition): int     $comparator
253     *
254     * @return list<TestGroup>
255     */
256    private function sortedTests(array $groups, callable $comparator): array
257    {
258        return array_map(
259            static function (TestGroup $group) use ($comparator): TestGroup {
260                $tests = $group->tests;
261                usort($tests, $comparator);
262
263                return new TestGroup($group->name, $tests, $group->beforeAll, $group->afterAll);
264            },
265            $groups,
266        );
267    }

269    /**
270     * Keeps #[Depends] prerequisites ahead of their dependents by
271     * *deferring* dependents — the spec's observable behavior: a test
272     * whose dependencies have not run yet moves later, everything else
273     * keeps its scheduled position. Dataset rows share a name and are
274     * emitted as one block. A cyclic dependency can never become
275     * ready, so its block is appended in scheduled order and left to
276     * the runner's skip semantics.
277     */
278    private function repaired(TestGroup $group): TestGroup
279    {
280        /** @var array<string, list<TestDefinition>> $blocks name → dataset rows, in scheduled order */
281        $blocks = [];
282
283        foreach ($group->tests as $test) {
284            $blocks[$test->id->name][] = $test;
285        }
286
287        /** @var array<string, true> $emitted */
288        $emitted = [];
289
290        /** @var list<TestDefinition> $ordered */
291        $ordered = [];
292
293        do {
294            $progressed = false;
295
296            foreach ($blocks as $name => $block) {
297                if (isset($emitted[$name])) {
298                    continue;
299                }
300
301                $ready = array_all(
302                    $block[0]->dependencies,
303                    static fn(string $dependency): bool => isset($emitted[$dependency]) || !isset($blocks[$dependency]),
304                );
305
306                if (!$ready) {
307                    continue;
308                }
309
310                $emitted[$name] = true;
311                $progressed     = true;
312
313                foreach ($block as $test) {
314                    $ordered[] = $test;
315                }
316            }
317        } while ($progressed);
318
319        // Whatever never became ready is cyclic: emit in scheduled order.
320        foreach ($blocks as $name => $block) {
321            if (!isset($emitted[$name])) {
322                foreach ($block as $test) {
323                    $ordered[] = $test;
324                }
325            }
326        }
327
328        return new TestGroup($group->name, $ordered, $group->beforeAll, $group->afterAll);
329    }
