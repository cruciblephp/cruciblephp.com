71/**
72 * Walking-skeleton runner: sequential, in-process, single-threaded.
73 * The supervisor/worker pool with isolation and scheduling replaces

99    /**
100     * @param list<TestGroup> $groups
101     * @param ?callable(): ?RunSummary $afterTests run-scoped work emitted inside the run (checks, and the Vitest fold-in returning its count delta)
102     *                                      bracket — after the suite, before RunFinished, so their
103     *                                      events precede the last event of the stream
104     */
105    public function run(array $groups, ?callable $afterTests = null): RunSummary
106    {
107        $started = hrtime(true);
108
109        $planned = 0;
110
111        foreach ($groups as $group) {
112            $planned += count($group->tests);
113        }
114
115        $this->emitter->emit(new RunStarted(tests: $planned));
116
117        $summary = $this->execute($groups);
118
119        // Complete = nothing narrowed the plan (CLI knowledge, carried
120        // in the options) and nothing halted it (stop-on leaves
121        // planned tests unfinished) — the D-071 pruning gate. Computed on
122        // the PHP plan BEFORE any external results (Vitest) fold in, so a
123        // folded-in suite never touches completeness.
124        $complete = $this->options->fullSuite && $summary->total() === $planned;
125
126        if ($afterTests !== null) {
127            $delta = $afterTests();
128
129            if ($delta instanceof RunSummary) {
130                $summary = $summary->plus($delta);
131            }
132        }
133
134        $this->emitter->emit(new RunFinished(
135            $summary,
136            (hrtime(true) - $started) / 1e9,
137            $complete,
138        ));
139
140        return $summary;
141    }

143    /**
144     * Runs the groups without opening or closing a stream — run:start
145     * and run:finish belong to whoever owns the whole run, which is
146     * this method's caller when executions are composed (the process
147     * pool mixing in-process and worker execution on one stream).
148     *
149     * @param list<TestGroup> $groups
150     */
151    public function execute(array $groups): RunSummary
152    {
153        $counts = ['pass' => 0, 'fail' => 0, 'error' => 0, 'skip' => 0, 'incomplete' => 0, 'risky' => 0, 'untested' => 0];
154
155        $this->issues->resetTallies();
156
157        foreach ($groups as $group) {
158            if ($this->runGroup($group, $counts)) {
159                break;
160            }
161        }
162
163        $issues = $this->issues->tallies();
164
165        return new RunSummary(
166            passed: $counts['pass'],
167            failed: $counts['fail'],
168            errored: $counts['error'],
169            skipped: $counts['skip'],
170            incomplete: $counts['incomplete'],
171            risky: $counts['risky'],
172            deprecations: $issues['deprecations'],
173            notices: $issues['notices'],
174            warnings: $issues['warnings'],
175            untested: $counts['untested'],
176        );
177    }

179    /**
180     * Returns true when a stop-on option asks the whole run to halt.
181     * The group's after-all cleanup still runs before halting.
182     *
183     * @param array<string, int> $counts
184     */
185    private function runGroup(TestGroup $group, array &$counts): bool
186    {
187        try {
188            if ($group->beforeAll instanceof Closure) {
189                ($group->beforeAll)();
190            }
191        } catch (Throwable $throwable) {
192            // A failing before-all errors every test in the group.
193            foreach ($group->tests as $definition) {
194                $this->emitter->emit(new TestStarted($definition->id));
195                $this->emitter->emit(new TestFinished(
196                    $definition->id,
197                    Outcome::Errored,
198                    0.0,
199                    $this->failureFrom($throwable),
200                ));
201                $counts['error']++;
202            }
203
204            return $this->shouldStop(Outcome::Errored);
205        }
206
207        // Seeded, not empty: under process isolation a prerequisite ran
208        // in an earlier worker, and its return value came back with it.
209        /** @var array<string, array{passed: bool, value: mixed}> $results */
210        $results = $this->dependencies->seed();
211
212        $halt = false;
213
214        foreach ($group->tests as $definition) {
215            [$outcome, $blocked, $kinds] = $this->runTest($definition, $results);
216            $counts[$outcome->value]++;
217
218            if ($blocked) {
219                $counts['untested']++;
220            }
221
222            if ($this->shouldStop($outcome, $kinds)) {
223                $halt = true;
224
225                break;
226            }
227        }
228
229        try {
230            if ($group->afterAll instanceof Closure) {
231                ($group->afterAll)();
232            }
233        } catch (Throwable) {
234            // After-all failures do not change test outcomes.
235        }
236
237        return $halt;
238    }

240    /**
241     * The spec's stop-on semantics: failure and error each have their
242     * own switch; defect covers both plus risky. The issue-triggered
243     * switches read what the test emitted rather than how it ended — a
244     * passing test that deprecates something still stops the run under
245     * --stop-on-deprecation, which is the whole point of having it.
246     *
247     * @param list<IssueKind> $issues the issues this test triggered
248     */
249    private function shouldStop(Outcome $outcome, array $issues = []): bool
250    {
251        $byOutcome = match ($outcome) {
252            Outcome::Errored    => $this->options->stopOnError || $this->options->stopOnDefect,
253            Outcome::Failed     => $this->options->stopOnFailure || $this->options->stopOnDefect,
254            Outcome::Risky      => $this->options->stopOnRisky || $this->options->stopOnDefect,
255            Outcome::Skipped    => $this->options->stopOnSkipped,
256            Outcome::Incomplete => $this->options->stopOnIncomplete,
257            default             => false,
258        };
259
260        if ($byOutcome) {
261            return true;
262        }
263
264        return array_any($issues, fn(IssueKind $kind): bool => match ($kind) {
265            IssueKind::Deprecation => $this->options->stopOnDeprecation,
266            IssueKind::Notice      => $this->options->stopOnNotice,
267            IssueKind::Warning     => $this->options->stopOnWarning,
268        });
269    }

271    /**
272     * @param array<string, array{passed: bool, value: mixed}> $results per-group results, by test name
273     *
274     * @return array{0: Outcome, 1: bool, 2: list<IssueKind>} the terminal outcome, whether it went untested (a gated blocked skip), and the issues it triggered
275     */
276    private function runTest(TestDefinition $definition, array &$results): array
277    {
278        $this->emitter->emit(new TestStarted($definition->id));
279
280        // A todo marker blocks execution before anything else is even
281        // considered (D-045): the placeholder reports incomplete with
282        // its label — requirements and dependencies of a body that will
283        // not run are irrelevant. wip and done run their bodies (Pest
284        // parity); a body-less wip/done is folded to a todo upstream.
285        $todo = $definition->metadata->first(Todo::class);
286
287        if ($todo instanceof Todo && $todo->blocksExecution()) {
288            return $this->finish($definition, Outcome::Incomplete, 0.0, reason: $todo->label());
289        }
290
291        $unmet = $this->requirements->unmet($definition->metadata);
292
293        if ($unmet !== []) {
294            return $this->finish($definition, Outcome::Skipped, 0.0, reason: implode(' ', $unmet), blocked: true);
295        }
296
297        $dependencyValues = [];
298
299        foreach ($definition->dependencies as $dependency) {
300            if (!isset($results[$dependency]) || !$results[$dependency]['passed']) {
301                return $this->finish($definition, Outcome::Skipped, 0.0, reason: sprintf(
302                    'This test depends on "%s", which did not pass.',
303                    $dependency,
304                ), blocked: true);
305            }
306
307            $dependencyValues[] = $results[$dependency]['value'];
308        }
309
310        // Retry-with-classification (G4, nextest semantics): a fail or
311        // error re-runs up to the retry budget; a pass on a retry is
312        // FLAKY, never silently green — the attempt count travels on
313        // the event, the reason names the earlier failure, and every
314        // failed attempt's Failure rides along for the JUnit flaky
315        // markup (D-043). Between attempts the policy's backoff pauses
316        // (fixed/exponential/jittered).
317        $policy  = $definition->metadata->first(Retry::class)?->policy() ?? $this->options->retries;
318        $attempt = 0;
319        $earlier = null;
320
321        /** @var list<Failure> $retried */
322        $retried = [];
323
324        do {
325            $attempt++;
326            $run = $this->attempt($definition, $dependencyValues);
327
328            if ($run['outcome'] !== Outcome::Failed && $run['outcome'] !== Outcome::Errored) {
329                break;
330            }
331
332            $earlier = $run['failure'];
333
334            if ($attempt <= $policy->count) {
335                if ($earlier instanceof Failure) {
336                    $retried[] = $earlier;
337                }
338
339                $delay = $policy->delayFor($attempt);
340
341                if ($delay > 0.0) {
342                    usleep((int) ($delay * 1_000_000));
343                }
344            }
345        } while ($attempt <= $policy->count);
346
347        $outcome = $run['outcome'];
348        $reason  = $run['reason'];
349        $name    = $definition->id->name;
350
351        if ($outcome === Outcome::Passed) {
352            // Every dataset row must pass for the method to count as a
353            // passed dependency; the last row's value is what
354            // dependents receive.
355            $results[$name] = [
356                'passed' => !isset($results[$name]) || $results[$name]['passed'],
357                'value'  => $run['value'],
358            ];
359
360            $this->dependencies->capture($name, $run['value']);
361
362            if ($attempt > 1) {
363                $reason = sprintf(
364                    'Flaky: passed on attempt %d of %d after: %s',
365                    $attempt,
366                    $policy->count + 1,
367                    $earlier instanceof Failure ? $earlier->message : 'an earlier failure',
368                );
369            }
370        } elseif ($run['thrown']) {
371            $results[$name] = ['passed' => false, 'value' => null];
372        }
373
374        return $this->finish(
375            $definition,
376            $outcome,
377            $run['duration'],
378            $run['failure'],
379            $attempt,
380            $reason,
381            $run['issues'],
382            $run['property'],
383            $retried,
384            $run['snapshots'],
385            // Clean only counts on a first-attempt pass (D-071): a
386            // flaky pass must not prune a sequence that still catches
387            // the bug some of the time.
388            $outcome === Outcome::Passed && $attempt === 1 ? $run['propertyClean'] : [],
389            $run['snapshotKeys'],
390        );
391    }

393    /**
394     * One full execution of the test — lifecycle, classification,
395     * state settlement — with no event emitted: runTest owns the
396     * one test:finish, whatever the attempt count was.
397     *
398     * @param list<mixed> $dependencyValues
399     *
400     * @return array{outcome: Outcome, duration: float, failure: ?Failure, reason: ?string, issues: list<\LucianoPereira\Crucible\Event\Issue>, value: mixed, thrown: bool, property: ?array{key: non-empty-string, choices: list<int>}, snapshots: ?array{created: int, updated: int}, propertyClean: list<non-empty-string>, snapshotKeys: list<non-empty-string>}
401     */
402    private function attempt(TestDefinition $definition, array $dependencyValues): array
403    {
404        // Per-test assertion accounting: the runner owns the counter's
405        // window; user code may reset it inside a test without
406        // corrupting risky detection.
407        Assert::resetAssertionCount();
408
409        $snapshot = $this->snapshotFor($definition->metadata);
410
411        $this->issues->install();
412
413        // The property bridge (D-040): opened per attempt so property
414        // keys count identically on retries. The snapshot context
415        // (D-042) follows the same rule — snapshot counters restart
416        // per attempt.
417        PropertyContext::begin($definition->id, $this->options->propertyFailures);
418        Snapshots::begin($definition->id, $this->options->updateSnapshots, $this->snapshots, $this->options->workingDirectory, $this->options->inlineSnapshotRewrite);
419        Browsing::begin();
420        MockeryContainer::reset();
421
422        // Per-test coverage window (D-041): the definition closure is
423        // the whole lifecycle, so hooks are attributed to their test.
424        $this->coverage?->begin();
425
426        $started = hrtime(true);
427        $value   = null;
428        $thrown  = null;
429        $window  = null;
430        $output  = '';
431
432        // What the test prints is the test's, not the runner's: buffered
433        // so it can be reported as an event and policed by
434        // --disallow-test-output, then written back out so capturing it
435        // never means losing it.
436        ob_start();
437
438        try {
439            $value = ($definition->test)($dependencyValues);
440        } catch (Throwable $throwable) {
441            $thrown = $throwable;
442        } finally {
443            $buffered = ob_get_clean();
444            $output   = $buffered === false ? '' : $buffered;
445
446            PropertyContext::end();
447            Snapshots::end();
448            // Recording waits: what this window may contribute
449            // depends on the outcome and the covers claim (D-063).
450            $window = $this->coverage?->end();
451        }
452
453        if ($output !== '') {
454            $this->emitter->emit(new TestOutputWritten($definition->id, OutputChannel::Stdout, $output));
455
456            if (!$this->options->disallowTestOutput) {
457                print $output;
458            }
459        }
460
461        $duration  = $this->since($started);
462        $issues    = $this->issues->drain();
463        $pollution = $this->settleGlobalState($definition->metadata, $snapshot);
464
465        // Browser failure artifact: captured before the test's
466        // contexts close, only when the test actually visited a page
467        // and genuinely failed (skips and incompletes are not
468        // failures). The screenshot path joins the failure message —
469        // the incumbent's exact courtesy.
470        $browserScreenshot = null;
471
472        if ($thrown instanceof Throwable
473            && !$thrown instanceof SkippedTestError
474            && !$thrown instanceof IncompleteTestError
475            && !RealPhpUnitOutcomes::isSkipped($thrown)
476            && !RealPhpUnitOutcomes::isIncomplete($thrown)) {
477            $browserScreenshot = Browsing::failureScreenshot($definition->id);
478        }
479
480        Browsing::end();
481
482        $result = [
483            'outcome'  => Outcome::Passed,
484            'duration' => $duration,
485            'failure'  => null,
486            'reason'   => null,
487            'issues'   => $issues,
488            'value'    => $value,
489            'thrown'   => $thrown instanceof Throwable,
490            'property' => $thrown instanceof PropertyFailedError
491                ? ['key' => $thrown->key, 'choices' => $thrown->choices]
492                : null,
493            'snapshots'     => Snapshots::recorded(),
494            'propertyClean' => PropertyContext::replayedClean(),
495            'snapshotKeys'  => Snapshots::visited(),
496        ];
497
498        if ($thrown instanceof Throwable) {
499            $result = match (true) {
500                $thrown instanceof SkippedTestError || RealPhpUnitOutcomes::isSkipped($thrown) => [...$result, 'outcome' => Outcome::Skipped, 'reason' => $thrown->getMessage()],
501                // Incomplete carries its trace where skipped does not:
502                // "not finished yet" is a note about a place in the
503                // code, and the reader wants to jump to it. A skip is a
504                // statement about the environment, and points nowhere.
505                $thrown instanceof IncompleteTestError || RealPhpUnitOutcomes::isIncomplete($thrown) => [...$result, 'outcome' => Outcome::Incomplete, 'reason' => $thrown->getMessage(), 'failure' => $this->failureFrom($thrown, $browserScreenshot)],
506                $thrown instanceof AssertionFailedError || RealPhpUnitOutcomes::isFailure($thrown)   => [...$result, 'outcome' => Outcome::Failed, 'failure' => $this->failureFrom($thrown, $browserScreenshot)],
507                default                                                                              => [...$result, 'outcome' => Outcome::Errored, 'failure' => $this->failureFrom($thrown, $browserScreenshot)],
508            };
509
510            return $this->settleCoverage($definition, $result, $window);
511        }
512
513        if ($pollution !== []) {
514            return $this->settleCoverage($definition, [...$result, 'outcome' => Outcome::Risky, 'reason' => 'This test modified global state: ' . implode(', ', $pollution) . '.'], $window);
515        }
516
517        // A test slower than its declared size allows (D-101). Judged on
518        // the duration already measured, so the budget covers exactly
519        // what ran and the reason states the measurement.
520        $limit = $this->options->enforceTimeLimit
521            ? TimeLimit::forTest($definition->metadata, $this->options->defaultTimeLimit)
522            : null;
523
524        // --disallow-test-output: a test that prints is a test whose
525        // result depends on something the assertions do not describe.
526        if ($this->options->disallowTestOutput && $output !== '') {
527            return $this->settleCoverage($definition, [
528                ...$result,
529                'outcome' => Outcome::Risky,
530                'reason'  => 'This test printed unexpected output: ' . trim($output),
531            ], $window);
532        }
533
534        if ($limit !== null && $duration > $limit) {
535            // The budget is judged on a measured duration, so a verdict
536            // reached under a call-hooking xdebug says so rather than
537            // blaming the test for the profiler.
538            $notice = TimingOverhead::notice();
539
540            return $this->settleCoverage($definition, [
541                ...$result,
542                'outcome' => Outcome::Risky,
543                'reason'  => sprintf('This test took %.3f seconds, longer than the %d second(s) its size allows.', $duration, $limit)
544                    . ($notice === null ? '' : ' ' . $notice),
545            ], $window);
546        }
547
548        $performedAssertions = Assert::assertionCount() > 0;
549
550        // --do-not-report-useless-tests turns the whole classification off,
551        // for a suite whose tests assert through something Crucible cannot
552        // see. #[DoesNotPerformAssertions] stays the per-test way to say it.
553        if ($this->options->reportUselessTests && !$performedAssertions && !$definition->metadata->has(DoesNotPerformAssertions::class)) {
554            return $this->settleCoverage($definition, [...$result, 'outcome' => Outcome::Risky, 'reason' => 'This test did not perform any assertions.'], $window);
555        }
556
557        return $this->settleCoverage($definition, $result, $window);
558    }

560    /**
561     * The coverage-metadata settlement (D-063), oracle-pinned:
562     * requireCoverageMetadata reclassifies a passing test without a
563     * covers claim as risky (with or without a coverage run);
564     * beStrictAboutCoverageMetadata reclassifies a passing test that
565     * executed source code outside its covered/used targets, naming
566     * the stray units. Then the window records — the per-test map
567     * keeps observed truth, the aggregate takes the covers-filtered
568     * contribution, and a risky test's hits are demoted (the oracle
569     * discards the coverage but keeps the denominators).
570     *
571     * @param array{outcome: Outcome, duration: float, failure: ?Failure, reason: ?string, issues: list<\LucianoPereira\Crucible\Event\Issue>, value: mixed, thrown: bool, property: ?array{key: non-empty-string, choices: list<int>}, snapshots: ?array{created: int, updated: int}, propertyClean: list<non-empty-string>, snapshotKeys: list<non-empty-string>} $result
572     *
573     * @return array{outcome: Outcome, duration: float, failure: ?Failure, reason: ?string, issues: list<\LucianoPereira\Crucible\Event\Issue>, value: mixed, thrown: bool, property: ?array{key: non-empty-string, choices: list<int>}, snapshots: ?array{created: int, updated: int}, propertyClean: list<non-empty-string>, snapshotKeys: list<non-empty-string>}
574     */
575    private function settleCoverage(TestDefinition $definition, array $result, ?CoverageWindow $window): array
576    {
577        $enforcing = $this->options->requireCoverageMetadata || $this->options->beStrictAboutCoverageMetadata;
578
579        if (!$enforcing && !$window instanceof CoverageWindow) {
580            return $result;
581        }
582
583        $targets = CoversTargets::from($definition->metadata);
584
585        if ($result['outcome'] === Outcome::Passed
586            && $this->options->requireCoverageMetadata
587            && !$targets->declared) {
588            $result = [...$result, 'outcome' => Outcome::Risky, 'reason' => 'This test does not define a code coverage target but is expected to do so'];
589        }
590
591        if ($result['outcome'] === Outcome::Passed
592            && $this->options->beStrictAboutCoverageMetadata
593            && $window instanceof CoverageWindow) {
594            $strays = $targets->strays($window);
595
596            if ($strays !== []) {
597                $result = [...$result, 'outcome' => Outcome::Risky, 'reason' => "This test executed code that is not listed as code to be covered or used:\n- " . implode("\n- ", $strays)];
598            }
599        }
600
601        if ($window instanceof CoverageWindow) {
602            // --disable-coverage-targeting: the aggregate takes everything
603            // the test executed, rather than only what its Covers* claims
604            // let it contribute.
605            $contribution = $this->options->coverageTargeting ? $targets->contribution($window) : $window;
606
607            // A test that did not finish did not cover what it
608            // touched. A skip runs markTestSkipped() and stops; an
609            // error threw something nobody expected partway through.
610            // Those lines executed in the literal sense and say nothing
611            // about the code under test — the incumbent lists them as
612            // executable with no hits, which is what withoutHits()
613            // already said for risky. A *failure* is different and
614            // still counts: the test ran to its assertion and disagreed
615            // with it, which is a complete measurement.
616            $settled = in_array($result['outcome'], [Outcome::Risky, Outcome::Skipped, Outcome::Incomplete, Outcome::Errored], true);
617
618            $this->coverage?->record(
619                $definition->id,
620                $window,
621                $settled ? $window->withoutHits() : $contribution,
622            );
623        }
624
625        return $result;
626    }

628    /**
629     * Takes the pre-test snapshot when backup or strict-global-state
630     * checking asks for one. Method metadata overrides configuration
631     * (method-before-class precedence is already in the collection).
632     */
633    private function snapshotFor(MetadataCollection $metadata): ?GlobalStateSnapshot
634    {
635        $backupGlobals = $metadata->first(BackupGlobals::class)->enabled ?? $this->options->backupGlobals;
636        $backupStatics = $metadata->first(BackupStaticProperties::class)->enabled ?? $this->options->backupStaticProperties;
637
638        if (!$backupGlobals && !$backupStatics && !$this->options->beStrictAboutChangesToGlobalState) {
639            return null;
640        }
641
642        $excludedGlobals = [];
643
644        foreach ($metadata->ofType(ExcludeGlobalVariableFromBackup::class) as $exclusion) {
645            $excludedGlobals[] = $exclusion->globalVariableName;
646        }
647
648        $excludedStatics = [];
649
650        foreach ($metadata->ofType(ExcludeStaticPropertyFromBackup::class) as $exclusion) {
651            $excludedStatics[] = [$exclusion->className, $exclusion->propertyName];
652        }
653
654        return new GlobalStateSnapshot(
655            $this->statics,
656            $excludedGlobals,
657            $excludedStatics,
658            includeStatics: $backupStatics || $this->options->beStrictAboutChangesToGlobalState,
659        );
660    }

662    /**
663     * Detects pollution (strict mode) and restores backed-up state.
664     * Returns the change list when the run is strict, empty otherwise.
665     *
666     * @return list<non-empty-string>
667     */
668    private function settleGlobalState(MetadataCollection $metadata, ?GlobalStateSnapshot $snapshot): array
669    {
670        if (!$snapshot instanceof \LucianoPereira\Crucible\Isolation\GlobalStateSnapshot) {
671            return [];
672        }
673
674        $changes = $this->options->beStrictAboutChangesToGlobalState ? $snapshot->changes() : [];
675
676        $backupGlobals = $metadata->first(BackupGlobals::class)->enabled ?? $this->options->backupGlobals;
677        $backupStatics = $metadata->first(BackupStaticProperties::class)->enabled ?? $this->options->backupStaticProperties;
678
679        if ($backupGlobals || $backupStatics) {
680            $snapshot->restore();
681        }
682
683        return $changes;
684    }

686    /**
687     * @param positive-int                                      $attempt
688     * @param list<\LucianoPereira\Crucible\Event\Issue>           $issues
689     * @param ?array{key: non-empty-string, choices: list<int>} $property
690     * @param list<Failure>                                     $retried
691     * @param ?array{created: int, updated: int}                $snapshots
692     * @param list<non-empty-string>                            $propertyClean
693     * @param list<non-empty-string>                            $snapshotKeys
694     *
695     * @return array{0: Outcome, 1: bool, 2: list<IssueKind>} the terminal outcome, whether it went untested (a gated blocked skip), and the kinds of issue it triggered — what the stop-on-issue switches read
696     */
697    private function finish(TestDefinition $definition, Outcome $outcome, float $duration, ?Failure $failure = null, int $attempt = 1, ?string $reason = null, array $issues = [], ?array $property = null, array $retried = [], ?array $snapshots = null, array $propertyClean = [], array $snapshotKeys = [], bool $blocked = false): array
698    {
699        // Expected-outcome reconciliation (D-073): a test that declared
700        // how it expects to end is judged against the outcome computed
701        // above. This is the single choke point every terminal outcome
702        // passes through — a declared ->skip(), a body markTestSkipped,
703        // an unmet requirement, and a todo block alike — so the
704        // inversion needs no cooperation from any of the callers.
705        $expected = $definition->metadata->first(ExpectedOutcome::class);
706
707        if ($expected instanceof ExpectedOutcome) {
708            [$outcome, $failure, $reason] = $this->reconcileExpectedOutcome($expected, $outcome, $reason);
709        }
710
711        // The blocked flag (D-075) only rides a skip that stayed a skip:
712        // an ->expectsSkip() that reconciled a requirement-skip to Passed
713        // asserted its way out of "untested" and must not carry it.
714        $blocked = $blocked && $outcome === Outcome::Skipped;
715
716        // Read and clear here rather than at the measurement: finish()
717        // is the one path every outcome takes, including the skips that
718        // never reach attempt() and so never reset it themselves. A
719        // test that did not run reports 0 because the previous test's
720        // finish() left the counter there.
721        $assertions = Assert::assertionCount();
722        Assert::resetAssertionCount();
723
724        $this->emitter->emit(new TestFinished(
725            $definition->id,
726            $outcome,
727            $duration,
728            $failure,
729            $attempt,
730            $reason !== null && $reason !== '' ? $reason : null,
731            $issues,
732            $this->quarantined($definition),
733            $property,
734            $retried,
735            $snapshots,
736            $propertyClean,
737            $snapshotKeys,
738            $blocked,
739            $assertions,
740        ));
741
742        return [$outcome, $blocked, array_map(static fn(Issue $issue): IssueKind => $issue->kind, $issues)];
743    }

745    /**
746     * Reconcile the actual terminal outcome against a declared
747     * expectation (D-073): a match is the pass, a mismatch the failure.
748     * When a reason fragment was declared it must appear in the actual
749     * reason (the ->fails('fragment') symmetry). The returned outcome
750     * is what finish() emits and returns, so stop-on-failure and the
751     * exit code see the reconciled result, not the raw one.
752     *
753     * @return array{0: Outcome, 1: ?Failure, 2: ?string}
754     */
755    private function reconcileExpectedOutcome(ExpectedOutcome $expected, Outcome $actual, ?string $reason): array
756    {
757        if ($actual !== $expected->expected) {
758            return [
759                Outcome::Failed,
760                new Failure(sprintf(
761                    'Expected the test to %s, but it %s.',
762                    $expected->verb(),
763                    $this->describeOutcome($actual),
764                )),
765                null,
766            ];
767        }
768
769        if ($expected->reasonFragment !== null && !str_contains((string) $reason, $expected->reasonFragment)) {
770            return [
771                Outcome::Failed,
772                new Failure(sprintf(
773                    'Expected the %s reason to contain "%s", but it was "%s".',
774                    $expected->verb(),
775                    $expected->reasonFragment,
776                    (string) $reason,
777                )),
778                null,
779            ];
780        }
781
782        return [
783            Outcome::Passed,
784            null,
785            sprintf('Expected outcome met: %s.', $actual->value),
786        ];
787    }

789    /**
790     * The actual outcome read as a past-tense clause, for the
791     * "but it <clause>" half of a mismatch message.
792     *
793     * @return non-empty-string
794     */
795    private function describeOutcome(Outcome $outcome): string
796    {
797        return match ($outcome) {
798            Outcome::Passed     => 'passed',
799            Outcome::Failed     => 'failed',
800            Outcome::Errored    => 'errored',
801            Outcome::Skipped    => 'was skipped',
802            Outcome::Incomplete => 'was incomplete',
803            Outcome::Risky      => 'was risky',
804        };
805    }

807    /**
808     * Whether the test is quarantined (G4): marked #[Quarantined] in
809     * code, or listed in the configuration by id — the full
810     * `file::name#dataset` string, or `file::name` covering every row.
811     */
812    private function quarantined(TestDefinition $definition): bool
813    {
814        if ($definition->metadata->has(Quarantined::class)) {
815            return true;
816        }
817
818        if ($this->options->quarantine === []) {
819            return false;
820        }
821
822        $id = $definition->id;
823
824        return in_array($id->toString(), $this->options->quarantine, true)
825            || in_array($id->file . '::' . $id->name, $this->options->quarantine, true);
826    }

833    private function failureFrom(Throwable $throwable, ?string $browserScreenshot = null): Failure
834    {
835        $frames = [];
836
837        // The throw site first, because getTrace() does not contain it:
838        // a trace is the call stack, and the line that threw is only on
839        // the throwable itself. Without it a `throw` written directly in
840        // a test body reports the frame that called the test — engine
841        // code — and the formats that need a real file and line (SARIF,
842        // JSON) point somewhere the author never wrote.
843        if ($throwable->getFile() !== '' && $throwable->getLine() > 0) {
844            $frames[] = new Frame($throwable->getFile(), $throwable->getLine());
845        }
846
847        foreach ($throwable->getTrace() as $frame) {
848            $file = $frame['file'] ?? '';
849            $line = $frame['line'] ?? 0;
850
851            if ($file === '' || $line < 1) {
852                continue;
853            }
854
855            $frames[] = new Frame(
856                $file,
857                $line,
858                $frame['function'] !== '' ? $frame['function'] : null,
859            );
860        }
861
862        $message = $throwable->getMessage() !== '' ? $throwable->getMessage() : $throwable::class;
863
864        if ($browserScreenshot !== null) {
865            $message .= sprintf("\nA screenshot of the page has been saved to [%s].", $browserScreenshot);
866        }
867
868        $comparison = $throwable instanceof AssertionFailedError ? $throwable->comparison : null;
869        $expected   = $comparison?->expected;
870        $actual     = $comparison?->actual;
871
872        return new Failure(
873            $message,
874            $throwable::class,
875            $frames,
876            $expected === '' ? null : $expected,
877            $actual === '' ? null : $actual,
878        );
879    }
