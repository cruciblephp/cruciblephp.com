28/**
29 * The fluent configuration of one stubbed/expected method — the
30 * spec's InvocationMocker surface: with(), willReturn*(),
31 * willThrowException(), backed by an optional invocation-count
32 * expectation.
33 */
34final class MethodConfigurator
35{
36    /** @var ?non-empty-string */
37    private ?string $method = null;
38
39    /** @var ?list<mixed> */
40    private ?array $argumentMatchers = null;
41
42    private ?Constraint $argumentListMatcher = null;
43
44    /** @var ?Closure(list<mixed>, object): mixed */
45    private ?Closure $behavior = null;
46
47    private int $invocations = 0;

49    /**
50     * @param ?list<non-empty-string> $configurable           methods the double's
51     *                                                        specification doubled; null = all
52     * @param bool                    $prefixArgumentMatching with() constraints match as a
53     *                                                        positional prefix, not exact arity
54     *                                                        (off = Mockery's oracle-pinned exact
55     *                                                        match; see DoubleSpecification)
56     */
57    public function __construct(
58        private ?InvocationCount $count = null,
59        private readonly ?array $configurable = null,
60        private readonly bool $prefixArgumentMatching = false,
61    ) {}

63    /**
64     * The Mockery grammar attaches counts fluently after creation
65     * (`->once()`, `->between(2, 3)`); the PHPUnit grammar fixes them
66     * at construction. Latest declaration wins, like every other
67     * fluent reconfiguration.
68     */
69    public function expectCount(InvocationCount $count): void
70    {
71        $this->count = $count;
72    }
73
74    /** Count window fully consumed — the Mockery fall-through signal. */
75    public function exhausted(): bool
76    {
77        return $this->count instanceof InvocationCount
78            && $this->count->max !== null
79            && $this->invocations >= $this->count->max;
80    }

82    /**
83     * @param non-empty-string $method
84     */
85    public function method(string $method): self
86    {
87        // A method the specification left real never reaches dispatch —
88        // configuring it would be a silent no-op, so it is a named
89        // error instead (the spec errors here too).
90        if ($this->configurable !== null && !in_array($method, $this->configurable, true)) {
91            throw new DoubleConfigurationException(sprintf(
92                '%s() cannot be configured: it is not among the doubled methods.',
93                $method,
94            ));
95        }
96
97        $this->method = $method;
98
99        return $this;
100    }

102    /**
103     * Argument expectations: plain values compare by the equality
104     * spec; Constraint instances match directly.
105     */
106    public function with(mixed ...$arguments): self
107    {
108        $this->argumentMatchers    = array_values($arguments);
109        $this->argumentListMatcher = null;

111        return $this;
112    }

114    public function withAnyParameters(): self
115    {
116        $this->argumentMatchers    = null;
117        $this->argumentListMatcher = null;

122    /**
123     * One constraint over the WHOLE argument list — the Mockery

167    /**
168     * @param list<array<int, mixed>> $map rows of arguments with the return value last
169     */
170    public function willReturnMap(array $map): self
171    {
172        $this->behavior = static function (array $args, object $double) use ($map): mixed {
173            foreach ($map as $row) {
174                $expected = array_slice($row, 0, -1);
175
176                if ((new IsEqual($expected))->matches($args)) {
177                    return array_last($row);
178                }
179            }
180
181            return null;
182        };
183
184        return $this;
185    }
186
187    public function willReturnOnConsecutiveCalls(mixed ...$values): self
188    {
189        $call = 0;
190
191        $this->behavior = static function (array $args, object $double) use (&$call, $values): mixed {
192            $value = $values[$call] ?? null;
193            $call++;
194
195            return $value;
196        };
197
198        return $this;
199    }

201    public function willThrowException(Throwable $throwable): self
202    {
203        $this->behavior = static fn(array $args, object $double): mixed => throw $throwable;
204
205        return $this;
206    }

238    /**
239     * @param non-empty-string $method
240     * @param list<mixed>      $arguments
241     */
242    public function appliesTo(string $method, array $arguments): bool
243    {
244        if ($this->method !== $method) {
245            return false;
246        }
247
248        if ($this->argumentListMatcher instanceof Constraint) {
249            // evaluate(..., returnResult: true), not matches() directly:
250            // $this->argumentListMatcher may be a user-supplied,
251            // real-PHPUnit-descended constraint (withArgumentList()'s
252            // own public contract accepts any Constraint) whose own
253            // matches() override is protected, matching real PHPUnit's
254            // convention — calling it here, from outside the Constraint
255            // hierarchy, would fatal (PHP forbids protected access
256            // across unrelated classes) even though matches() itself is
257            // declared validly. evaluate() is Constraint's own public
258            // entry point for exactly this.
259            return $this->argumentListMatcher->evaluate($arguments, '', true) === true;
260        }
261
262        if ($this->argumentMatchers === null) {
263            return true;
264        }
265
266        $matcherCount = count($this->argumentMatchers);
267
268        // The PHPUnit grammar's with() constraints are a positional
269        // prefix, not a full arity match (oracle-verified against
270        // PHPUnit's own Parameters rule: it only rejects a call with
271        // FEWER arguments than constraints — trailing, unconstrained
272        // real arguments are never checked). The Mockery grammar keeps
273        // exact arity: with(any()) matching a call with an extra,
274        // unmatched argument is oracle-pinned as a non-match
275        // (spec/mockery-api.md).
276        $arityMismatch = $this->prefixArgumentMatching
277            ? $matcherCount > count($arguments)
278            : $matcherCount !== count($arguments);
279
280        if ($arityMismatch) {
281            return false;
282        }
283
284        foreach ($this->argumentMatchers as $index => $matcher) {
285            $constraint = $matcher instanceof Constraint ? $matcher : new IsEqual($matcher);
286
287            // Same reasoning as appliesTo()'s argumentListMatcher branch
288            // above: $matcher may be a user-supplied, real-PHPUnit-descended
289            // constraint with a protected matches() override.
290            if ($constraint->evaluate($arguments[$index], '', true) !== true) {
291                return false;
292            }
293        }
294
295        return true;
296    }

298    /**
299     * @param bool $failFast the PHPUnit-spec posture: an exceeded maximum
300     *                       fails at call time. The Mockery grammar passes
301     *                       false — every count violation settles at close
302     *                       (oracle-pinned, spec/mockery-api.md §5).
303     *
304     * @throws AssertionFailedError when the call exceeds the expected maximum
305     */
306    public function registerInvocation(bool $failFast = true): void
307    {
308        $this->invocations++;
309        $this->count?->recordInvocation();
310
311        if ($failFast && $this->count instanceof \LucianoPereira\Crucible\Double\InvocationCount && $this->count->exceededBy($this->invocations)) {
312            throw new AssertionFailedError(sprintf(
313                '%s() was not expected to be called more than %s.',
314                $this->method ?? '{method}',
315                $this->count->description,
316            ));
317        }
318    }

326    /**
327     * @param list<mixed>       $arguments
328     * @param Closure(): mixed  $default
329     */
330    public function invoke(object $double, array $arguments, Closure $default): mixed
331    {
332        return $this->behavior instanceof Closure ? ($this->behavior)($arguments, $double) : $default();
333    }

335    /**
336     * @throws AssertionFailedError when the expectation was not met
337     */
338    public function verify(): void
339    {
340        if (!$this->count instanceof \LucianoPereira\Crucible\Double\InvocationCount) {
341            return;
342        }
343
344        if (!$this->count->satisfiedBy($this->invocations)) {
345            throw new AssertionFailedError(sprintf(
346                '%s() was expected to be called %s, but was called %d time(s).',
347                $this->method ?? '{method}',
348                $this->count->description,
349                $this->invocations,
350            ));
351        }
352
353        // A met expectation is a verified assertion.
354        Assert::assertThat(true, new IsTrue());
355    }
