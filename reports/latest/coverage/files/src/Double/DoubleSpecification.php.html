11namespace LucianoPereira\Crucible\Double;
12
13use function implode;
14use function sort;
15
16/**
17 * Everything that distinguishes one double from another (the spec's
18 * MockBuilder knobs, D-046): the doubled type(s), which methods are
19 * doubled, how the instance is constructed, and how unconfigured
20 * calls behave. The generation identity — what makes two doubles
21 * share a generated class — is a strict subset: constructor and
22 * auto-return settings are instantiation state, not code.
23 */
24final readonly class DoubleSpecification
25{
26    /**
27     * @param non-empty-list<class-string>  $types                   one class or interface, or
28     *                                                               several interfaces (intersection)
29     * @param list<class-string>            $traits                  mixed into the double itself, so their
30     *                                                               own methods stay REAL — ✓ measured, this
31     *                                                               is what the real mockery does for
32     *                                                               Mockery::mock(SomeTrait::class)
33     * @param ?list<non-empty-string>       $onlyMethods             null = every doubleable method,
34     *                                                               [] = none (everything stays real)
35     * @param ?list<mixed>                  $constructorArguments
36     * @param ?non-empty-string             $className               explicit generated-class name
37     * @param bool                          $mockerySurface          the generated class carries the
38     *                                                               Mockery verbs and a catch-all
39     *                                                               __call instead of the PHPUnit
40     *                                                               configuration API — the third
41     *                                                               spec's grammar over the same
42     *                                                               brain (D-060)
43     * @param bool                          $mergeConfigurations     dispatch falls back through
44     *                                                               older matching configurators
45     *                                                               for a return value when the
46     *                                                               newest one carries none, instead
47     *                                                               of fabricating a default —
48     *                                                               additive, off by default;
49     *                                                               the PHPUnit-spec construction
50     *                                                               path opts in explicitly
51     * @param bool                          $prefixArgumentMatching  with() constraints match as a
52     *                                                               positional prefix — extra real
53     *                                                               arguments beyond the given
54     *                                                               constraints are unconstrained,
55     *                                                               not a mismatch (PHPUnit's own
56     *                                                               Parameters rule, oracle-verified) —
57     *                                                               off by default: the Mockery
58     *                                                               grammar's with()/any() is
59     *                                                               oracle-pinned to exact arity
60     *                                                               (spec/mockery-api.md); the
61     *                                                               PHPUnit-spec construction path
62     *                                                               opts in explicitly
63     */
64    public function __construct(
65        public array $types,
66        public ?array $onlyMethods = null,
67        public bool $callOriginalConstructor = false,
68        public ?array $constructorArguments = null,
69        public bool $callOriginalClone = true,
70        public bool $autoReturnValues = true,
71        public ?string $className = null,
72        public bool $mockerySurface = false,
73        public bool $mergeConfigurations = false,
74        public bool $prefixArgumentMatching = false,
75        public array $traits = [],
76    ) {}
77
78    /**
79     * @return non-empty-string cache key for the generated class
80     */
81    public function classIdentity(): string
82    {
83        $methods = $this->onlyMethods;
84
85        if ($methods !== null) {
86            sort($methods);
87        }
88
89        return implode('&', $this->types)
90            . '@' . implode(',', $this->traits)
91            . '#' . ($methods === null ? '*' : implode(',', $methods))
92            . '#' . ($this->callOriginalClone ? 'clone' : 'noclone')
93            . '#' . ($this->className ?? '')
94            . '#' . ($this->mockerySurface ? 'mockery' : 'phpunit');
95    }
96}
