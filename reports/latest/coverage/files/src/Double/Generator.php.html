39/**
40 * Builds the doubled class for an interface, a non-final class, or an
41 * intersection of interfaces: every selected method forwards to
42 * DoubleState::dispatch(); unselected methods keep their real
43 * implementation (partial doubles, D-046). Code generation is
44 * unavoidable for runtime type doubling in PHP; it is done from
45 * reflection with typed rendering, evaluated once per specification
46 * identity, and cached.
47 *
48 * Documented limits (the honest-boundary rule): final classes, enums,
49 * and readonly classes cannot be doubled — use the real value.
50 * Final methods on doubled classes keep their real implementation.
51 * By-reference parameters are forwarded by value. State attaches
52 * after construction, so a constructor that calls its own doubled
53 * methods is out of contract.
54 */
55final class Generator
56{
57    /**
58     * Generated classes are process-global and stateless (state is
59     * injected per instance), so the cache is static: one generated
60     * class per specification identity for the whole run.
61     *
62     * @var array<string, class-string>
63     */
64    private static array $cache = [];

76    /**
77     * @return class-string
78     */
79    public function classFor(DoubleSpecification $specification): string
80    {
81        $identity = $specification->classIdentity();
82
83        if (isset(self::$cache[$identity])) {
84            return self::$cache[$identity];
85        }
86
87        $reflections = $this->validated($specification);
88        $name        = $specification->className;
89
90        // trait_exists is the third check because it is the one PHP
91        // cannot be asked to survive: for a trait both class_exists and
92        // interface_exists are false, and declaring over the name is an
93        // uncatchable fatal — ✓ exit 255, catch (Throwable) never fires.
94        // An enum needs no clause of its own; class_exists is already
95        // true for one. Same lesson MockeryContainer.php:213-219 records.
96        if ($name !== null && (class_exists($name) || interface_exists($name) || trait_exists($name))) {
97            throw new DoubleCreationException(sprintf(
98                'Cannot generate a double named %s: the name is already in use.',
99                $name,
100            ));
101        }
102
103        // Named by identity, not by call order: GeneratedCode::record()
104        // files generated sources under md5($source) and calls the
105        // directory "a set of distinct shapes rather than a log of
106        // calls", which a counter made false — the same shape got a new
107        // name whenever an unrelated double was created first.
108        $name ??= 'CrucibleDouble_' . $reflections[0]->getShortName() . '_' . md5($identity);
109
110        GeneratedCode::evaluate(
111            $this->classCode($name, $reflections, $specification),
112            'the double ' . $name . ' of ' . implode(', ', $specification->types),
113        );
114
115        /** @var class-string $name */
116        return self::$cache[$identity] = $name;
117    }

119    /**
120     * @return non-empty-list<ReflectionClass<object>>
121     */
122    private function validated(DoubleSpecification $specification): array
123    {
124        $reflections = [];
125
126        foreach ($specification->types as $type) {
127            $reflection = new ReflectionClass($type);
128
129            if ($reflection->isFinal()) {
130                throw new DoubleCreationException(sprintf('Cannot double final class %s.', $type));
131            }
132
133            if ($reflection->isEnum()) {
134                throw new DoubleCreationException(sprintf('Cannot double enum %s; use a real case.', $type));
135            }
136
137            // Reserved names differ per API surface; the incumbent
138            // FATALS on Mockery-verb collisions — a named error is the
139            // better behavior (D-017 rule, extended in D-060).
140            $reserved = $specification->mockerySurface
141                ? ['shouldReceive', 'shouldNotReceive', 'allows', 'expects', 'mockery_verbs']
142                : ['method', 'expects'];
143
144            foreach ($reserved as $name) {
145                if ($reflection->hasMethod($name)) {
146                    throw new DoubleCreationException(sprintf(
147                        'Cannot double %s: it declares %s(), which collides with the double configuration API.',
148                        $type,
149                        $name,
150                    ));
151                }
152            }
153
154            $reflections[] = $reflection;
155        }
156
157        if (count($reflections) > 1) {
158            foreach ($reflections as $reflection) {
159                if (!$reflection->isInterface()) {
160                    throw new DoubleCreationException(sprintf(
161                        'An intersection double takes interfaces only; %s is a class.',
162                        $reflection->getName(),
163                    ));
164                }
165            }
166        }
167
168        if ($specification->onlyMethods !== null) {
169            $this->validateSelection($reflections[0], $specification->onlyMethods);
170        }
171
172        return $reflections;
173    }

175    /**
176     * @param ReflectionClass<object> $reflection
177     * @param list<non-empty-string>  $methods
178     */
179    private function validateSelection(ReflectionClass $reflection, array $methods): void
180    {
181        // An interface double must implement every method — leaving
182        // some out of onlyMethods() would leave them abstract. (The
183        // spec's runner fatals here; a named error is the better
184        // behavior, recorded in D-046.)
185        if ($reflection->isInterface()) {
186            throw new DoubleCreationException(sprintf(
187                'onlyMethods() cannot be used when doubling interface %s: every method must be implemented. Double the full interface.',
188                $reflection->getName(),
189            ));
190        }
191
192        foreach ($methods as $method) {
193            if (!$reflection->hasMethod($method)) {
194                throw new DoubleCreationException(sprintf(
195                    'onlyMethods(): %s has no method named %s().',
196                    $reflection->getName(),
197                    $method,
198                ));
199            }
200
201            $candidate = $reflection->getMethod($method);
202
203            if ($candidate->isPrivate() || $candidate->isFinal() || $candidate->isStatic()) {
204                throw new DoubleCreationException(sprintf(
205                    'onlyMethods(): %s::%s() cannot be doubled — %s methods keep their real implementation.',
206                    $reflection->getName(),
207                    $method,
208                    $candidate->isPrivate() ? 'private' : ($candidate->isFinal() ? 'final' : 'static'),
209                ));
210            }
211        }
212    }

214    /**
215     * @param non-empty-list<ReflectionClass<object>> $reflections
216     *
217     * @return non-empty-string a class declaration, always
218     */
219    private function classCode(string $name, array $reflections, DoubleSpecification $specification): string
220    {
221        $base = $reflections[0];
222
223        // The marker differs per API surface: the two grammars declare
224        // incompatible expects() signatures (D-060).
225        $marker = $specification->mockerySurface
226            ? \LucianoPereira\Crucible\Double\Mockery\MockeryMock::class
227            : \LucianoPereira\Crucible\Double\Mocked::class;
228
229        if ($base->isInterface()) {
230            $interfaces = [];
231
232            foreach ($reflections as $reflection) {
233                $interfaces[] = '\\' . $reflection->getName();
234            }
235
236            $relation = 'implements ' . implode(', ', $interfaces) . ', ' . $marker;
237        } else {
238            $relation = 'extends \\' . $base->getName() . ' implements ' . $marker;
239        }
240
241        $methods = [];
242
243        foreach ($this->selectedMethods($reflections, $specification) as $method) {
244            $methods[] = $this->methodCode($method, $method->getDeclaringClass());
245        }
246
247        // Suppressing the original __clone (spec: disableOriginalClone)
248        // is generated code, not state — a clone must not run the
249        // parent's logic at all.
250        if (!$specification->callOriginalClone && !$base->isInterface()) {
251            $methods[] = "    public function __clone(): void\n    {\n    }";
252        }
253
254        $api = $specification->mockerySurface ? <<<'PHP'
255            public ?\LucianoPereira\Crucible\Double\DoubleState $__crucibleState = null;
256
257            public function shouldReceive(string ...$methods): \LucianoPereira\Crucible\Double\Mockery\MockeryExpectation
258            {
259                return \LucianoPereira\Crucible\Double\Mockery\MockeryExpectation::receiving(\LucianoPereira\Crucible\Double\DoubleState::of($this->__crucibleState, $this, __FUNCTION__), \array_values($methods));
260            }
261
262            public function shouldNotReceive(string ...$methods): \LucianoPereira\Crucible\Double\Mockery\MockeryExpectation
263            {
264                return \LucianoPereira\Crucible\Double\Mockery\MockeryExpectation::receiving(\LucianoPereira\Crucible\Double\DoubleState::of($this->__crucibleState, $this, __FUNCTION__), \array_values($methods))->never();
265            }
266
267            /** @param array<string, mixed>|string ...$methods */
268            public function allows(array|string ...$methods): \LucianoPereira\Crucible\Double\Mockery\MockeryExpectation
269            {
270                return \LucianoPereira\Crucible\Double\Mockery\MockeryExpectation::allowing(\LucianoPereira\Crucible\Double\DoubleState::of($this->__crucibleState, $this, __FUNCTION__), \array_values($methods));
271            }
272
273            public function expects(string ...$methods): \LucianoPereira\Crucible\Double\Mockery\MockeryExpectation
274            {
275                return \LucianoPereira\Crucible\Double\Mockery\MockeryExpectation::receiving(\LucianoPereira\Crucible\Double\DoubleState::of($this->__crucibleState, $this, __FUNCTION__), \array_values($methods))->once();
276            }
277
278            public function __call(string $method, array $arguments): mixed
279            {
280                // $obj->{''}() is legal syntax and reaches here; the state keeps call names non-empty.
281                if ($method === '') {
282                    throw new \LucianoPereira\Crucible\Double\Mockery\MockeryBadMethodCallException('A double cannot answer a call to a method with an empty name.');
283                }
284
285                return \LucianoPereira\Crucible\Double\DoubleState::of($this->__crucibleState, $this, $method)->dispatch($this, $method, $arguments);
286            }
287            PHP : <<<'PHP'
288            public ?\LucianoPereira\Crucible\Double\DoubleState $__crucibleState = null;
289
290            public function method(string $method): \LucianoPereira\Crucible\Double\MethodConfigurator
291            {
292                return \LucianoPereira\Crucible\Double\DoubleState::of($this->__crucibleState, $this, __FUNCTION__)->configureMethod($method);
293            }
294
295            public function expects(\LucianoPereira\Crucible\Double\InvocationCount $count): \LucianoPereira\Crucible\Double\MethodConfigurator
296            {
297                return \LucianoPereira\Crucible\Double\DoubleState::of($this->__crucibleState, $this, __FUNCTION__)->expectInvocation($count);
298            }
299            PHP;
300
301        // The traits are mixed into the DOUBLE, not into a parent it
302        // extends, because that is what keeps their own methods real —
303        // ✓ measured against the mockery-main oracle, whose trait mock
304        // answers from the trait itself and reports the trait in
305        // class_uses(). selectedMethods() leaves those names alone for
306        // the same reason.
307        $uses = $specification->traits === []
308            ? ''
309            : '    use \\' . implode(', \\', $specification->traits) . ";\n\n";
310
311        return sprintf(
312            "class %s %s {\n%s%s\n\n%s\n}",
313            $name,
314            $relation,
315            $uses,
316            $api,
317            implode("\n\n", $methods),
318        );
319    }

321    /**
322     * The methods that forward to dispatch: every doubleable method,
323     * or — under onlyMethods() — the listed ones. Abstract methods are
324     * always generated whatever the list says: they cannot be left
325     * unimplemented.
326     *
327     * @param non-empty-list<ReflectionClass<object>> $reflections
328     *
329     * @return list<ReflectionMethod>
330     */
331    private function selectedMethods(array $reflections, DoubleSpecification $specification): array
332    {
333        /** @var array<non-empty-string, ReflectionMethod> $methods */
334        $methods = [];
335
336        // A method the double gets FROM a mixed-in trait is left
337        // alone: overriding it would replace the trait's own code with
338        // a dispatch, which is exactly what the incumbent does not do.
339        $real = [];
340
341        foreach ($specification->traits as $trait) {
342            foreach ((new ReflectionClass($trait))->getMethods() as $method) {
343                if (!$method->isAbstract()) {
344                    $real[strtolower($method->getName())] = true;
345                }
346            }
347        }
348
349        foreach ($reflections as $reflection) {
350            foreach ($this->doubleableMethods($reflection, $specification->onlyMethods) as $method) {
351                if (isset($real[strtolower($method->getName())])) {
352                    continue;
353                }
354
355                $methods[$method->getName()] ??= $method;
356            }
357        }
358
359        if ($specification->onlyMethods === null) {
360            return array_values($methods);
361        }
362
363        $selected = [];
364
365        foreach ($methods as $name => $method) {
366            if ($method->isAbstract() || in_array($name, $specification->onlyMethods, true)) {
367                $selected[] = $method;
368            }
369        }
370
371        return $selected;
372    }

374    /**
375     * @param ReflectionClass<object> $reflection
376     * @param ?list<non-empty-string> $onlyMethods
377     *
378     * @return list<ReflectionMethod>
379     */
380    private function doubleableMethods(ReflectionClass $reflection, ?array $onlyMethods): array
381    {
382        $methods = [];
383
384        foreach ($reflection->getMethods() as $method) {
385            if ($method->isConstructor() || $method->isDestructor() || $method->isFinal()) {
386                continue;
387            }
388
389            if ($method->isPrivate()) {
390                continue;
391            }
392
393            if (in_array($method->getName(), ['__clone', '__get', '__set', '__isset', '__unset', '__sleep', '__wakeup'], true)) {
394                continue;
395            }
396
397            // Concrete protected methods on classes keep their real
398            // implementation by default — but onlyMethods() can still
399            // name one explicitly: real-world classes (Monolog's
400            // SocketHandler, for one) expose protected wrapper methods
401            // specifically so tests can override them ("Wrapper to
402            // allow mocking"), and the spec's onlyMethods() honors
403            // that explicit request even for a protected method.
404            $requested = $onlyMethods !== null && in_array($method->getName(), $onlyMethods, true);
405
406            if (!$reflection->isInterface() && $method->isProtected() && !$method->isAbstract() && !$requested) {
407                continue;
408            }
409
410            $methods[] = $method;
411        }
412
413        return $methods;
414    }

416    /**
417     * @param ReflectionClass<object> $reflection
418     */
419    private function methodCode(ReflectionMethod $method, ReflectionClass $reflection): string
420    {
421        $parameters = [];
422
423        foreach ($method->getParameters() as $parameter) {
424            $parameters[] = $this->parameterCode($parameter, $reflection);
425        }
426
427        // Built-in interfaces declare TENTATIVE return types
428        // (Countable::count(): int) — invisible to getReturnType();
429        // omitting them makes every generated override a deprecation.
430        $returnType = $method->getReturnType() ?? $method->getTentativeReturnType();
431        $returns    = $returnType !== null ? ': ' . $this->typeCode($returnType, $reflection) : '';
432
433        $signature = sprintf(
434            '    public %sfunction %s(%s)%s',
435            $method->isStatic() ? 'static ' : '',
436            $method->getName(),
437            implode(', ', $parameters),
438            $returns,
439        );
440
441        if ($method->isStatic()) {
442            return $signature . "\n    {\n        throw new \\LogicException('Static methods are not doubled.');\n    }";
443        }
444
445        // Through DoubleState::of(), not the nullable property: the
446        // state is attached after construction, so every generated
447        // method dereferences a nullable — 39 true reports from the
448        // generated-code type tier. A double built outside TestDoubles
449        // now says so instead of dying on a null.
450        $dispatch = '\\LucianoPereira\\Crucible\\Double\\DoubleState::of($this->__crucibleState, $this, __FUNCTION__)'
451            . '->dispatch($this, __FUNCTION__, \\func_get_args())';
452
453        $body = match (true) {
454            $returnType instanceof ReflectionNamedType && $returnType->getName() === 'void'
455                => sprintf("        %s;", $dispatch),
456            $returnType instanceof ReflectionNamedType && $returnType->getName() === 'never'
457                    => sprintf("        %s;\n        throw new \\LogicException('A never-returning double method must be configured to throw.');", $dispatch),
458            default => sprintf('        return %s;', $dispatch),
459        };
460
461        return $signature . "\n    {\n" . $body . "\n    }";
462    }

464    /**
465     * A parameter default, as a constant expression.
466     *
467     * var_export() is right for everything that is not an object, and
468     * wrong for objects in a way that cannot be recovered from: it
469     * renders one as `\Cls::__set_state(array(...))`, a static call,
470     * which is not a constant expression. PHP 8.1 made `new` legal in a
471     * parameter initializer, so an interface declaring
472     * `find(Config $c = new Config())` is ordinary code — and ✓ measured,
473     * emitting the var_export form for it is not a catchable failure:
474     * `try { eval(...) } catch (Throwable)` does not catch "Constant
475     * expression contains invalid operations". The process dies, naming
476     * neither the type being doubled nor the reason.
477     *
478     * PHP renders the initializer itself, and it is the text PHP
479     * compiled the original from — no reconstruction, no guessing.
480     * ✓ Measured: class constants come back fully qualified
481     * (`new \App\Cfg(\App\Cfg::LIM)`) and a namespaced constant comes
482     * back namespace-relative (`App\LIMIT`), which resolves correctly
483     * because doubles are generated in the global namespace.
484     *
485     * Enum cases stay with var_export: they are objects, and it already
486     * renders them as the constant expression `\Suit::Hearts`.
487     */
488    private function defaultCode(ReflectionParameter $parameter): string
489    {
490        $default = $parameter->getDefaultValue();
491
492        if (!is_object($default) || $default instanceof UnitEnum) {
493            return var_export($default, true);
494        }
495
496        $rendered = (string) $parameter;
497        $opens    = strpos($rendered, ' = ');
498        $closes   = strrpos($rendered, ' ]');
499
500        if ($opens === false || $closes === false || $closes <= $opens) {
501            throw new DoubleCreationException(sprintf(
502                'Cannot double a method whose parameter $%s defaults to a %s: PHP did not render the '
503                    . 'initializer, and there is no constant expression to emit in its place.',
504                $parameter->getName(),
505                $default::class,
506            ));
507        }
508
509        return substr($rendered, $opens + 3, $closes - $opens - 3);
510    }

512    /**
513     * @param ReflectionClass<object> $reflection
514     */
515    private function parameterCode(ReflectionParameter $parameter, ReflectionClass $reflection): string
516    {
517        $type = $parameter->getType();
518
519        $code = ($type !== null ? $this->typeCode($type, $reflection) . ' ' : '')
520            . ($parameter->isPassedByReference() ? '&' : '')
521            . ($parameter->isVariadic() ? '...' : '')
522            . '$' . $parameter->getName();
523
524        if ($parameter->isDefaultValueAvailable()) {
525            $code .= ' = ' . $this->defaultCode($parameter);
526        } elseif ($parameter->allowsNull() && $parameter->isOptional() && !$parameter->isVariadic()) {
527            $code .= ' = null';
528        }
529
530        return $code;
531    }

533    /**
534     * @param ReflectionClass<object> $reflection
535     */
536    private function typeCode(ReflectionType $type, ReflectionClass $reflection): string
537    {
538        if ($type instanceof ReflectionUnionType) {
539            $members = [];
540
541            foreach ($type->getTypes() as $member) {
542                $members[] = $this->typeCode($member, $reflection);
543            }
544
545            return implode('|', $members);
546        }
547
548        if ($type instanceof ReflectionIntersectionType) {
549            $members = [];
550
551            foreach ($type->getTypes() as $member) {
552                $members[] = $this->typeCode($member, $reflection);
553            }
554
555            return implode('&', $members);
556        }
557
558        if (!$type instanceof ReflectionNamedType) {
559            return 'mixed';
560        }
561
562        $name = $type->getName();
563
564        if ($name === 'self') {
565            $name = '\\' . $reflection->getName();
566        } elseif ($name === 'parent') {
567            $parent = $reflection->getParentClass();
568            $name   = $parent !== false ? '\\' . $parent->getName() : 'parent';
569        } elseif (!$type->isBuiltin() && $name !== 'static') {
570            $name = '\\' . $name;
571        }
572
573        if ($type->allowsNull() && !in_array($name, ['mixed', 'null'], true)) {
574            return '?' . $name;
575        }
576
577        return $name;
578    }
