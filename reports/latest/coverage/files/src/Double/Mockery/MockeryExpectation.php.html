29/**
30 * The fluent surface behind shouldReceive/allows/expects — the
31 * Mockery verbs as spellings over the one state brain (D-060): every
32 * verb registers ordinary MethodConfigurators; the return sugar maps
33 * onto their willReturn* internals; counts attach fluently through
34 * expectCount. Dispatch semantics live in DoubleState's
35 * FirstDeclared strategy, not here.
36 */
37final readonly class MockeryExpectation
38{

40     * @param non-empty-list<MethodConfigurator> $configurators
41     */
42    private function __construct(private array $configurators) {}

44    /**
45     * shouldReceive(...): zero-or-more by default; array entries are
46     * the method => return sugar. The mock verb (expects()) chains
47     * ->once() on the result; the negative (shouldNotReceive) chains
48     * ->never().
49     *
50     * @param list<array<string, mixed>|string> $methods
51     */
52    public static function receiving(DoubleState $state, array $methods): self
53    {
54        if ($methods === []) {
55            throw new MockeryException('shouldReceive() needs at least one method name (the fluent no-argument form is not part of this tier).');
56        }
57
58        $configurators = [];
59
60        foreach ($methods as $method) {
61            if (is_array($method)) {
62                foreach ($method as $name => $return) {
63                    if ($name === '') {
64                        continue;
65                    }
66
67                    $configurator = $state->configureMethod($name);
68                    $configurator->willReturn($return);
69                    MockeryContainer::guardMockableMethod($state, $name);
70                    $configurators[] = $configurator;
71                }
72
73                continue;
74            }
75
76            if ($method === '') {
77                continue;
78            }
79
80            MockeryContainer::guardMockableMethod($state, $method);
81            $configurators[] = $state->configureMethod($method);
82        }
83
84        if ($configurators === []) {
85            throw new MockeryException('shouldReceive() needs at least one method name.');
86        }
87
88        return new self($configurators);
89    }
90
91    /**
92     * allows(...): the stub verb — same registration, never verified.

226    /**
227     * Closure form: all args to one predicate, whose result must be
228     * === true (truthy is refused — §4 battery 8; an arity underflow
229     * is the user's own ArgumentCountError, propagated). Array form:
230     * identical to with(...$args). Anything else: the oracle's own
231     * InvalidArgumentException — hence the mixed parameter; the
232     * validation IS the contract.
233     */
234    public function withArgs(mixed $argumentsOrPredicate): self
235    {
236        if (is_array($argumentsOrPredicate)) {
237            return $this->with(...$argumentsOrPredicate);
238        }
239
240        if (!$argumentsOrPredicate instanceof Closure) {
241            // The oracle's own exception, in its own words (§4 battery 8).
242            throw new InvalidArgumentException(sprintf(
243                'Call to Mockery\Expectation::withArgs with an invalid argument (%s), only array and closure are allowed',
244                get_debug_type($argumentsOrPredicate),
245            ));
246        }
247
248        foreach ($this->configurators as $configurator) {
249            $configurator->withArgumentList(new Callback(
250                static fn(mixed $arguments): bool => is_array($arguments) && $argumentsOrPredicate(...$arguments) === true,
251                'matches the withArgs() predicate',
252            ));
253        }
254
255        return $this;
256    }
257
258    /**
259     * Every given value present among the args, any position, STRICT
260     * (§4 battery 8); the zero-argument form matches any call.
261     */
262    public function withSomeOfArgs(mixed ...$values): self
263    {
264        if ($values === []) {
265            return $this->withAnyArgs();
266        }
267
268        foreach ($this->configurators as $configurator) {
269            $configurator->withArgumentList(new ContainsValues(array_values($values), strict: true));
270        }
271
272        return $this;
273    }

329    /**
330     * @param array<mixed> $values variadics may carry named keys
331     */
332    private function returnSequence(array $values): self
333    {
334        foreach ($this->configurators as $configurator) {
335            if (count($values) <= 1) {
336                $configurator->willReturn($values[0] ?? null);
337
338                continue;
339            }
340
341            $call = 0;
342            $configurator->willReturnCallback(function () use (&$call, $values): mixed {
343                $value = $values[min($call, count($values) - 1)];
344                $call++;
345
346                return $value;
347            });
348        }
349
350        return $this;
351    }
