142    /**
143     * The state of a double, or a refusal naming what was called.
144     *
145     * Generated code holds its state in a nullable property, because
146     * the double is instantiated first and given its state after. Every
147     * generated method and every method of the configuration surface
148     * therefore dereferences a nullable, which is exactly what the
149     * generated-code type tier reported — 39 times, all true. Left
150     * alone, a double built outside TestDoubles answers a method call
151     * with "Call to a member function dispatch() on null", which names
152     * neither the double nor the method.
153     */
154    public static function of(?self $state, object $double, string $method): self
155    {
156        return $state ?? throw new DoubleConfigurationException(sprintf(
157            '%s::%s() was called on a double with no state attached. Doubles are created through '
158                . 'TestDoubles, which attaches it; one built any other way cannot answer anything.',
159            $double::class,
160            $method,
161        ));
162    }

164    /**
165     * @param non-empty-string $method
166     * @param list<mixed>      $arguments
167     */
168    public function dispatch(object $double, string $method, array $arguments): mixed
169    {
170        $this->calls[] = [$method, $arguments];
171
172        if ($this->strategy === DispatchStrategy::FirstDeclared) {
173            return $this->dispatchFirstDeclared($double, $method, $arguments);
174        }
175
176        // Latest configuration wins, so the walk is newest-first.
177        //
178        // Without merging — the default — only the newest match is ever
179        // used, so the loop stops there. It used to array_reverse() the
180        // whole list and collect every match into a second array, two
181        // allocations per call to read one element, and every later
182        // configurator was asked appliesTo() for an answer nobody read.
183        if (!$this->mergeConfigurations) {
184            for ($index = count($this->configurators) - 1; $index >= 0; $index--) {
185                $configurator = $this->configurators[$index];
186
187                if (!$configurator->appliesTo($method, $arguments)) {
188                    continue;
189                }
190
191                $configurator->registerInvocation();
192
193                return $configurator->invoke(
194                    $double,
195                    $arguments,
196                    $this->fallbackFor($method),
197                );
198            }
199
200            return $this->defaultReturn($double, $method);
201        }
202
203        // Merging reads every match, so it still collects them all,
204        // newest-first. ✓ Measured: an index loop over the same list is
205        // slower here than the reverse-and-walk, so this branch keeps
206        // what it had.
207        $matching = [];
208
209        foreach (array_reverse($this->configurators) as $configurator) {
210            if ($configurator->appliesTo($method, $arguments)) {
211                $matching[] = $configurator;
212            }
213        }
214
215        if ($matching === []) {
216            return $this->defaultReturn($double, $method);
217        }
218
219        // Real-world code commonly layers a bare expects()->method()
220        // invocation-count expectation over a return value configured
221        // earlier via a shared helper. Every matching configurator still
222        // tracks its own count (each is independent, spec-accurate), but
223        // the return value comes from the newest one that actually
224        // carries a behavior — not from an emptier configurator that
225        // would otherwise shadow it.
226        foreach ($matching as $configurator) {
227            $configurator->registerInvocation();
228        }
229
230        foreach ($matching as $configurator) {
231            if ($configurator->hasBehavior()) {
232                return $configurator->invoke(
233                    $double,
234                    $arguments,
235                    $this->fallbackFor($method),
236                );
237            }
238        }
239
240        return $this->defaultReturn($double, $method);
241    }

243    /**
244     * The Mockery grammar's dispatch, oracle-pinned (D-060 /
245     * spec/mockery-api.md §3): first-declared expectation wins, a
246     * count-exhausted one falls through to the next match, an
247     * exhausted one with no successor still handles the call (its
248     * violation settles at close, never at call time), and an
249     * unmatched call is a NAMED error — unknown method vs
250     * non-matching arguments are different mistakes.
251     *
252     * @param non-empty-string $method
253     * @param list<mixed>      $arguments
254     */
255    private function dispatchFirstDeclared(object $double, string $method, array $arguments): mixed
256    {
257        $fallback = null;
258
259        foreach ($this->configurators as $configurator) {
260            if (!$configurator->appliesTo($method, $arguments)) {
261                continue;
262            }
263
264            if (!$configurator->exhausted()) {
265                $configurator->registerInvocation(failFast: false);
266
267                return $configurator->invoke($double, $arguments, $this->fallbackFor($method));
268            }
269
270            $fallback ??= $configurator;
271        }
272
273        if ($fallback instanceof MethodConfigurator) {
274            $fallback->registerInvocation(failFast: false);
275
276            return $fallback->invoke($double, $arguments, $this->fallbackFor($method));
277        }
278
279        foreach ($this->configurators as $configurator) {
280            if ($configurator->methodName() === $method) {
281                throw new Mockery\NoMatchingExpectationException(sprintf(
282                    'No matching handler found for %s::%s(). Either the method was unexpected or its arguments matched no expected argument list for this method.',
283                    $this->targets[0],
284                    $method,
285                ));
286            }
287        }
288
289        throw new Mockery\MockeryBadMethodCallException(sprintf(
290            'Method %s::%s() does not exist on this mock object.',
291            $this->targets[0],
292            $method,
293        ));
294    }

307     * @param non-empty-string $method
308     *
309     * @return list<array{non-empty-string, list<mixed>}>
310     */
311    public function callsTo(string $method): array
312    {
313        $matching = [];
314
315        foreach ($this->calls as $call) {
316            if ($call[0] === $method) {
317                $matching[] = $call;
318            }
319        }
320
321        return $matching;
322    }

324    /**
325     * @param non-empty-string $method
326     */
327    private function defaultReturn(object $double, string $method): mixed
328    {
329        if (!$this->autoReturnValues) {
330            throw new DoubleConfigurationException(sprintf(
331                '%s::%s() has no configured return value and automatic return value generation is disabled.',
332                $this->targets[0],
333                $method,
334            ));
335        }
336
337        // Memoised per method, and the TYPE rather than the
338        // ReflectionMethod, because the type is all this reads. A fresh
339        // ReflectionMethod per dispatch is why a double with ZERO
340        // configurators already cost many times a plain object.
341        // array_key_exists, not ??=: a method with no declared return
342        // type resolves to null and must not be resolved again.
343        if (!array_key_exists($method, $this->returnTypes)) {
344            $reflection = $this->reflectionOf($method);
345
346            // Tentative return types are invisible to getReturnType() —
347            // Countable::count(): int reads as null there. The generated
348            // SIGNATURE already falls back (Generator::methodCode), so
349            // without the same fallback here the double promises int and
350            // answers null: ✓ measured, doubling Countable, ArrayAccess or
351            // Iterator produced a TypeError on the first call.
352            $this->returnTypes[$method] = $reflection?->getReturnType() ?? $reflection?->getTentativeReturnType();
353        }
354
355        return $this->defaults->forType($this->returnTypes[$method], $double);
356    }

358    /**
359     * @param non-empty-string $method
360     *
361     * @return Closure(): mixed
362     */
363    private function fallbackFor(string $method): Closure
364    {
365        return $this->fallbacks[$method] ??= fn(): mixed => $this->defaultReturn($this->double, $method);
366    }

368    /**
369     * @param non-empty-string $method
370     */
371    private function reflectionOf(string $method): ?ReflectionMethod
372    {
373        foreach ($this->targets as $target) {
374            if (method_exists($target, $method)) {
375                return new ReflectionMethod($target, $method);
376            }
377        }
378
379        // Mockery-surface doubles accept methods no target declares
380        // (anonymous mocks, allowMockingNonExistentMethods) — untyped,
381        // so the default is null.
382        return null;
383    }
