141    {
142        $this->source  = new Source();
143        $this->retries = new RetryPolicy();
144        $this->browser = new BrowserConfiguration();
145
146        // Crucible's own PDF/Markdown/JSON/SARIF reports, pre-seeded
147        // through the exact same report-format registration path a

157        // project's crucible.php overrides it by calling ->subscriber()

160
161        // The documentation view's three file targets, registered the

164        $this->subscribers['testdox-text']    = ['class' => TestDoxTextSubscriber::class, 'params' => []];
165        $this->subscribers['testdox-html']    = ['class' => TestDoxHtmlSubscriber::class, 'params' => []];

171        // Same non-privileged pre-seeding for the three built-in
172        // progress views — a project's crucible.php overrides one by
173        // calling ->progressView() again with the same key, which also

199    /**
200     * phpunit.xml: <testsuite name="..."><directory>...</directory></testsuite>
201     *
202     * @param non-empty-string                          $name
203     * @param non-empty-string|array<non-empty-string>  $directories accepts a single path or a list of paths
204     * @param list<non-empty-string>                    $files
205     * @param non-empty-string                          $suffix
206     */
207    public function testSuite(string $name, string|array $directories, array $files = [], string $suffix = 'Test.php'): self
208    {
209        foreach ($this->testSuites as $existing) {
210            if ($existing->name === $name) {
211                throw new ConfigurationException(
212                    sprintf('Test suite "%s" is defined more than once.', $name),
213                );
214            }
215        }
216
217        $directories = is_string($directories) ? [$directories] : $directories;
218
219        if (!array_is_list($directories)) {
220            throw new ConfigurationException(
221                sprintf('Directories of test suite "%s" must be a list of paths.', $name),
222            );
223        }
224
225        $this->testSuites[] = new TestSuite($name, $directories, $files, $suffix);
226
227        return $this;
228    }

661    /**
662     * Crucible-native (extension surface, Tier 0): orchestrate an external
663     * command as part of the run — a static analyser, a JS test runner,
664     * a linter, any tool with a meaningful exit code. The command runs
665     * after the PHP suite, its output streams through to the console
666     * verbatim, and its **exit code is the verdict**: zero passes,
667     * non-zero fails the run with a named reason. Crucible never parses the
668     * command's output — exit code is the one contract that does not
669     * break across tool versions. Tools wanting their results folded into
670     * the test tree write a PHP Extension instead; this is the robust,
671     * no-plugin floor.
672     *
673     *     ->command('larastan', ['vendor/bin/phpstan', 'analyse'])
674     *     ->command('js',       ['npx', 'vitest', 'run'])
675     *
676     * @param non-empty-string       $label            names the check in the summary, the reason, and the report
677     * @param list<non-empty-string> $argv             argv form — no shell, no quoting, no injection
678     * @param ?WorkingDirectory      $workingDirectory null = the run's working directory
679     * @param ?positive-int          $timeout          seconds before Crucible kills the command; null = no limit
680     */
681    public function command(string $label, array $argv, ?WorkingDirectory $workingDirectory = null, ?int $timeout = null): self
682    {
683        foreach ($this->commandGates as $existing) {
684            if ($existing->label === $label) {
685                throw new ConfigurationException(
686                    sprintf('Command gate "%s" is defined more than once.', $label),
687                );
688            }
689        }
690
691        if ($argv === []) {
692            throw new ConfigurationException(
693                sprintf('Command gate "%s" needs a non-empty list of argv parts.', $label),
694            );
695        }
696
697        $this->commandGates[] = new CommandGate($label, $argv, $workingDirectory, $timeout);
698
699        return $this;
700    }

761    /**
762     * Declare that changing these files puts these test groups in doubt
763     * (D-083). The dependency graph resolves *class references*, so a
764     * file reached by convention or at runtime — an asset, a Blade
765     * template, a translation, a fixture — is invisible to it whatever
766     * its extension. Only the author knows what such a change endangers.
767     *
768     *     ->impactRule('resources/js', ['browser'])
769     *     ->impactRule('resources/lang', ['i18n'])
770     *     ->impactRule('resources/views/mail/*.blade.php', ['mail'])
771     *
772     * Rules are **additive**: a match adds groups under `--changed` and
773     * in watch mode, and can never remove one, so a missing rule is
774     * never worse than declaring none.
775     *
776     * @param non-empty-string       $pattern project-relative path; a directory covers everything
777     *                                        beneath it, and `*`/`?` glob (matching across `/`)
778     * @param list<non-empty-string> $groups
779     */
780    public function impactRule(string $pattern, array $groups): self
781    {
782        $this->impactRules[] = new ImpactRule($pattern, $groups);
783
784        return $this;
785    }

857    /**
858     * Let whoever already knows what changed push a change set into
859     * `crucible --watch` (D-083), instead of Crucible reconstructing one from
860     * artifacts it would have to understand. A bundler plugin, a build
861     * step, a git hook — see `integration/` for a ready-made producer.
862     *
863     *     ->retrigger()
864     *
865     * **Off by default**: a test tool must not open a socket unasked.
866     * When enabled, watch binds 127.0.0.1 on an ephemeral port and
867     * writes the full URL (token included) to the hot file, whose very
868     * presence tells a producer that Crucible is listening. It is removed
869     * on exit, and a failure to bind is never fatal — the session just
870     * runs without it.
871     *
872     * @param non-empty-string $hotFile where the address is published; relative to the working directory
873     * @param int              $port    0 = ephemeral, which never collides between projects
874     */
875    public function retrigger(string $hotFile = '.crucible.hot', int $port = 0): self
876    {
877        $this->retriggerHotFile = $hotFile;
878        $this->retriggerPort    = $port;
879
880        return $this;
881    }
