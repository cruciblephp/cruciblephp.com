27/**
28 * The one sanctioned read of phpunit.xml (ROADMAP: out-of-scope
29 * declines XML *support*; a one-time conversion is how the support
30 * debt is avoided). Reads a phpunit.xml document, emits the
31 * equivalent crucible.php — and refuses to be silently lossy: whatever
32 * it cannot map is named in the generated file's header comment, so
33 * the migration's gaps are part of its output, not a surprise.
34 *
35 * Pure: XML string in, generated code + notes out. File IO belongs
36 * to the CLI.
37 */
38final readonly class XmlMigrator
39{
40    /**
41     * Root attributes that map 1:1 onto a boolean Builder method,
42     * keyed by XML attribute name. Emitted only when they differ from
43     * Crucible's (= the spec's) defaults.
44     */
45    private const array BOOLEAN_ATTRIBUTES = [
46        'colors'                            => ['colors', false],
47        'cacheResult'                       => ['cacheResult', true],
48        'failOnDeprecation'                 => ['failOnDeprecation', false],
49        'failOnIncomplete'                  => ['failOnIncomplete', false],
50        'failOnNotice'                      => ['failOnNotice', false],
51        'failOnRisky'                       => ['failOnRisky', false],
52        'failOnSkipped'                     => ['failOnSkipped', false],
53        'failOnWarning'                     => ['failOnWarning', false],
54        'stopOnDefect'                      => ['stopOnDefect', false],
55        'stopOnError'                       => ['stopOnError', false],
56        'stopOnFailure'                     => ['stopOnFailure', false],
57        'testdox'                           => ['testdox', false],
58        'backupGlobals'                     => ['backupGlobals', false],
59        'backupStaticProperties'            => ['backupStaticProperties', false],
60        'beStrictAboutChangesToGlobalState' => ['beStrictAboutChangesToGlobalState', false],
61        'requireCoverageMetadata'           => ['requireCoverageMetadata', false],
62        'beStrictAboutCoverageMetadata'     => ['beStrictAboutCoverageMetadata', false],
63    ];

65    /**
66     * @throws ConfigurationException when the document is not parseable XML
67     */
68    public function migrate(string $xml): MigrationResult
69    {
70        libxml_use_internal_errors(true);
71
72        $document = simplexml_load_string($xml);
73
74        if ($document === false) {
75            throw new ConfigurationException('The phpunit.xml document cannot be parsed.');
76        }
77
78        $notes = [];
79        $calls = [];
80        $uses  = [\LucianoPereira\Crucible\Configuration\Crucible::class];
81
82        $this->rootAttributes($document, $calls, $notes, $uses);
83        $this->testSuites($document, $calls, $notes);
84        $this->source($document, $calls, $notes);
85        $this->php($document, $calls, $notes);
86
87        foreach ($document->children() as $child) {
88            $name = $child->getName();
89
90            if (!in_array($name, ['testsuites', 'source', 'php'], true)) {
91                $notes[] = sprintf('element <%s>', $name);
92            }
93        }
94
95        return new MigrationResult($this->render($uses, $calls, $notes), $notes);
96    }

98    /**
99     * @param list<non-empty-string> $calls
100     * @param list<non-empty-string> $notes
101     * @param list<non-empty-string> $uses
102     */
103    private function rootAttributes(SimpleXMLElement $document, array &$calls, array &$notes, array &$uses): void
104    {
105        foreach ($document->attributes() ?? [] as $name => $value) {
106            $value = (string) $value;
107
108            if ($name === 'bootstrap') {
109                if ($value !== '') {
110                    $calls[] = sprintf('->bootstrap(%s)', var_export($value, true));
111                }
112
113                continue;
114            }
115
116            if ($name === 'cacheDirectory') {
117                if ($value !== '') {
118                    $calls[] = sprintf('->cacheDirectory(%s)', var_export($value, true));
119                }
120
121                continue;
122            }
123
124            if ($name === 'executionOrder') {
125                $order = ExecutionOrder::tryFrom($value);
126
127                if ($order instanceof ExecutionOrder) {
128                    if ($order !== ExecutionOrder::Declared) {
129                        $uses[]  = \LucianoPereira\Crucible\Configuration\ExecutionOrder::class;
130                        $calls[] = sprintf('->executionOrder(ExecutionOrder::%s)', $order->name);
131                    }
132                } else {
133                    $notes[] = sprintf('attribute executionOrder="%s" (combined orders are implicit in Crucible: dependencies always resolve)', $value);
134                }
135
136                continue;
137            }
138
139            if (isset(self::BOOLEAN_ATTRIBUTES[$name])) {
140                [$method, $default] = self::BOOLEAN_ATTRIBUTES[$name];
141                $enabled            = $value === 'true';
142
143                if ($enabled !== $default) {
144                    $calls[] = sprintf('->%s(%s)', $method, $enabled ? 'true' : 'false');
145                }
146
147                continue;
148            }
149
150            $notes[] = sprintf('attribute %s="%s"', $name, $value);
151        }
152    }

154    /**
155     * @param list<non-empty-string> $calls
156     * @param list<non-empty-string> $notes
157     */
158    private function testSuites(SimpleXMLElement $document, array &$calls, array &$notes): void
159    {
160        foreach ($document->testsuites->testsuite ?? [] as $suite) {
161            $name        = (string) ($suite['name'] ?? '');
162            $directories = [];
163            $files       = [];
164            $suffixes    = [];
165
166            foreach ($suite->directory ?? [] as $directory) {
167                $directories[] = (string) $directory;
168
169                if (isset($directory['suffix'])) {
170                    $suffixes[] = (string) $directory['suffix'];
171                }
172
173                foreach (['prefix', 'phpVersion', 'phpVersionOperator'] as $attribute) {
174                    if (isset($directory[$attribute])) {
175                        $notes[] = sprintf('testsuite "%s" <directory %s="%s">', $name, $attribute, (string) $directory[$attribute]);
176                    }
177                }
178            }
179
180            foreach ($suite->file ?? [] as $file) {
181                $files[] = (string) $file;
182            }
183
184            foreach ($suite->exclude ?? [] as $excluded) {
185                $notes[] = sprintf('testsuite "%s" <exclude>%s</exclude>', $name, (string) $excluded);
186            }
187
188            if ($name === '' || ($directories === [] && $files === [])) {
189                $notes[] = sprintf('testsuite "%s" (empty or unnamed)', $name);
190
191                continue;
192            }
193
194            $arguments = [var_export($name, true), $this->exportList($directories)];
195
196            if ($files !== []) {
197                $arguments[] = 'files: ' . $this->exportList($files);
198            }
199
200            if ($suffixes !== [] && $suffixes[0] !== 'Test.php') {
201                $arguments[] = 'suffix: ' . var_export($suffixes[0], true);
202            }
203
204            $calls[] = sprintf('->testSuite(%s)', implode(', ', $arguments));
205        }
206    }

208    /**
209     * @param list<non-empty-string> $calls
210     * @param list<non-empty-string> $notes
211     */
212    private function source(SimpleXMLElement $document, array &$calls, array &$notes): void
213    {
214        if (!property_exists($document, 'source') || $document->source === null) {
215            return;
216        }
217
218        $source       = $document->source;
219        $include      = [];
220        $exclude      = [];
221        $includeFiles = [];
222        $excludeFiles = [];
223
224        foreach ($source->include->directory ?? [] as $directory) {
225            $include[] = (string) $directory;
226
227            if (isset($directory['suffix']) && (string) $directory['suffix'] !== '.php') {
228                $notes[] = sprintf('<source> <directory suffix="%s"> (Crucible sources are .php)', (string) $directory['suffix']);
229            }
230        }
231
232        foreach ($source->include->file ?? [] as $file) {
233            $includeFiles[] = (string) $file;
234        }
235
236        foreach ($source->exclude->directory ?? [] as $directory) {
237            $exclude[] = (string) $directory;
238        }
239
240        foreach ($source->exclude->file ?? [] as $file) {
241            $excludeFiles[] = (string) $file;
242        }
243
244        foreach ($source->attributes() ?? [] as $name => $value) {
245            $notes[] = sprintf('<source> attribute %s="%s"', $name, (string) $value);
246        }
247
248        if ($include === [] && $exclude === [] && $includeFiles === [] && $excludeFiles === []) {
249            return;
250        }
251
252        $arguments = [];
253
254        if ($include !== []) {
255            $arguments[] = 'include: ' . $this->exportList($include);
256        }
257
258        if ($exclude !== []) {
259            $arguments[] = 'exclude: ' . $this->exportList($exclude);
260        }
261
262        if ($includeFiles !== []) {
263            $arguments[] = 'includeFiles: ' . $this->exportList($includeFiles);
264        }
265
266        if ($excludeFiles !== []) {
267            $arguments[] = 'excludeFiles: ' . $this->exportList($excludeFiles);
268        }
269
270        $calls[] = sprintf('->source(%s)', implode(', ', $arguments));
271    }

273    /**
274     * @param list<non-empty-string> $calls
275     * @param list<non-empty-string> $notes
276     */
277    private function php(SimpleXMLElement $document, array &$calls, array &$notes): void
278    {
279        if (!property_exists($document, 'php') || $document->php === null) {
280            return;
281        }
282
283        $php = $document->php;
284
285        foreach ($php->ini ?? [] as $ini) {
286            $calls[] = sprintf(
287                '->ini(%s, %s)',
288                var_export((string) $ini['name'], true),
289                var_export((string) $ini['value'], true),
290            );
291        }
292
293        foreach ($php->env ?? [] as $env) {
294            $calls[] = sprintf(
295                '->env(%s, %s)',
296                var_export((string) $env['name'], true),
297                var_export((string) $env['value'], true),
298            );
299
300            if ((string) ($env['force'] ?? '') === 'true') {
301                $notes[] = sprintf('<env name="%s" force="true"> (Crucible env values always apply)', (string) $env['name']);
302            }
303        }
304
305        foreach ($php->const ?? [] as $constant) {
306            $calls[] = sprintf(
307                '->constant(%s, %s)',
308                var_export((string) $constant['name'], true),
309                $this->exportConstant((string) $constant['value']),
310            );
311        }
312
313        foreach ($php->children() as $child) {
314            $name = $child->getName();
315
316            if (!in_array($name, ['ini', 'env', 'const'], true)) {
317                $notes[] = sprintf('<php> element <%s name="%s">', $name, (string) $child['name']);
318            }
319        }
320    }

322    /**
323     * The spec's <const> value coercion: "true"/"false" become
324     * booleans, numeric strings become numbers, the rest stay strings.
325     */
326    private function exportConstant(string $value): string
327    {
328        if ($value === 'true' || $value === 'false') {
329            return $value;
330        }
331
332        // A trimmed numeric string is already a valid PHP literal.
333        if (is_numeric($value) && $value === trim($value)) {
334            return $value;
335        }
336
337        return var_export($value, true);
338    }

340    /**
341     * @param list<string> $items
342     */
343    private function exportList(array $items): string
344    {
345        $exported = [];
346
347        foreach ($items as $item) {
348            $exported[] = var_export($item, true);
349        }
350
351        return '[' . implode(', ', $exported) . ']';
352    }

354    /**
355     * @param list<non-empty-string> $uses
356     * @param list<non-empty-string> $calls
357     * @param list<non-empty-string> $notes
358     *
359     * @return non-empty-string
360     */
361    private function render(array $uses, array $calls, array $notes): string
362    {
363        $header = "<?php\n\ndeclare(strict_types=1);\n\n/*\n * Migrated from phpunit.xml by `crucible migrate-config`.\n";
364
365        if ($notes !== []) {
366            $header .= " *\n * Not migrated (had no Crucible equivalent):\n";
367
368            foreach ($notes as $note) {
369                $header .= ' *   - ' . $note . "\n";
370            }
371        }
372
373        $header .= " */\n\n";
374
375        sort($uses);
376
377        foreach ($uses as $use) {
378            $header .= 'use ' . $use . ";\n";
379        }
380
381        $body = "\nreturn Crucible::configure()";
382
383        foreach ($calls as $call) {
384            $body .= "\n    " . $call;
385        }
386
387        return $header . $body . ";\n";
388    }
