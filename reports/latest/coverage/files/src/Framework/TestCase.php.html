91    /**
92     * Runs one test method through the per-test lifecycle. Internal:
93     * called by dialect frontends, not by user code. Returns the test
94     * method's return value (consumed by #[Depends] injection).
95     *
96     * The hook lists carry setUp(), the condition templates and
97     * tearDown() at their priority-0 place (HookPlan). A failure in the
98     * before phase ends the test without the after phase, as a failing
99     * setUp() always has; once the test body is entered, the after
100     * phase always runs.
101     *
102     * @param non-empty-string  $methodName
103     * @param list<mixed>       $arguments
104     * @param ?non-empty-string $dataset
105     */
106    final public function invokeTest(string $methodName, array $arguments = [], ?HookPlan $hooks = null, ?string $dataset = null): mixed
107    {
108        $hooks ??= new HookPlan();
109        $this->nameTest($methodName, $dataset);
110
111        try {
112            foreach ($hooks->before as $hook) {
113                $this->{$hook}();
114            }
115
116            foreach ($hooks->preConditions as $hook) {
117                $this->{$hook}();
118            }
119        } catch (Throwable $throwable) {
120            throw $this->transformed($throwable);
121        }
122
123        try {
124            $result = $this->{$methodName}(...$arguments);
125            $this->verifyNoExpectedExceptionWasMissed();
126            $this->verifyTestDoubles($this->allowsExpectationlessMocks($methodName));
127
128            foreach ($hooks->postConditions as $hook) {
129                $this->{$hook}();
130            }
131
132            return $result;
133        } catch (AssertionFailedError|SkippedTestError|IncompleteTestError $outcome) {
134            // An expected exception may itself be an assertion failure
135            // (testing a test framework does exactly this).
136            if ($outcome instanceof AssertionFailedError && $this->expectsException()) {
137                $this->verifyThrowableMatchesExpectation($outcome);
138
139                return null;
140            }
141
142            throw $outcome;
143        } catch (Throwable $throwable) {
144            if (!$this->expectsException()) {
145                throw $this->transformed($throwable);
146            }
147
148            $this->verifyThrowableMatchesExpectation($throwable);
149
150            return null;
151        } finally {
152            foreach ($hooks->after as $hook) {
153                $this->{$hook}();
154            }
155        }
156    }

158    /**
159     * An error goes through transformException(); a failure, a skip or
160     * an incomplete is an outcome, not an error, and passes unchanged.
161     */
162    private function transformed(Throwable $throwable): Throwable
163    {
164        if ($throwable instanceof AssertionFailedError
165            || $throwable instanceof AssertionError
166            || $throwable instanceof SkippedTestError
167            || $throwable instanceof IncompleteTestError) {
168            return $throwable;
169        }
170
171        return $this->transformException($throwable);
172    }

394    /**
395     * Internal: end-of-test double settlement — expectation
396     * verification plus the expectation-less-mock advisory (a notice,
397     * never an outcome). Called by invokeTest and by dialect frontends
398     * that orchestrate the lifecycle themselves (D-046).
399     */
400    final public function verifyTestDoubles(bool $expectationlessMocksAllowed = false): void
401    {
402        $this->doubles?->verify();
403
404        // Mockery-grammar doubles settle in the same window — their
405        // Mockery::close() is a spelling for this call (D-060).
406        MockeryContainer::settle();
407
408        if ($expectationlessMocksAllowed || !$this->doubles instanceof TestDoubles) {
409            return;
410        }
411
412        foreach ($this->doubles->expectationlessMocks() as $target) {
413            trigger_error(sprintf(
414                'The mock for %s ended the test without any expectation; create a stub when none are needed, or add #[AllowMockObjectsWithoutExpectations].',
415                $target,
416            ), E_USER_NOTICE);
417        }
418    }

480    private function expectsException(): bool
481    {
482        return $this->expectedException !== null
483            || $this->expectedExceptionMessage !== null
484            || $this->expectedExceptionMessageRegularExpression !== null
485            || $this->expectedExceptionCode !== null;
486    }
487
488    private function verifyNoExpectedExceptionWasMissed(): void
489    {
490        if (!$this->expectsException()) {
491            return;
492        }
493
494        $expected = $this->expectedException ?? Throwable::class;
495
496        // Disarm before failing: the missed-expectation failure itself
497        // must never re-enter expectation matching (AssertionFailedError
498        // extends RuntimeException — expecting RuntimeException would
499        // otherwise swallow this failure; found by the conformance suite).
500        $this->expectedException                         = null;
501        $this->expectedExceptionMessage                  = null;
502        $this->expectedExceptionMessageRegularExpression = null;
503        $this->expectedExceptionCode                     = null;
504
505        self::fail(sprintf(
506            'Failed asserting that exception of type "%s" is thrown.',
507            $expected,
508        ));
509    }
510
511    private function verifyThrowableMatchesExpectation(Throwable $throwable): void
512    {
513        if (!$this->expectsException()) {
514            throw $throwable;
515        }
516
517        if ($this->expectedException !== null && !$throwable instanceof $this->expectedException) {
518            throw new AssertionFailedError(sprintf(
519                'Failed asserting that exception of type "%s" matches expected exception "%s". Message was: "%s".',
520                $throwable::class,
521                $this->expectedException,
522                $throwable->getMessage(),
523            ));
524        }
525
526        if ($this->expectedExceptionMessage !== null && !str_contains($throwable->getMessage(), $this->expectedExceptionMessage)) {
527            throw new AssertionFailedError(sprintf(
528                'Failed asserting that exception message "%s" contains "%s".',
529                $throwable->getMessage(),
530                $this->expectedExceptionMessage,
531            ));
532        }
533
534        if ($this->expectedExceptionMessageRegularExpression !== null && preg_match($this->expectedExceptionMessageRegularExpression, $throwable->getMessage()) !== 1) {
535            throw new AssertionFailedError(sprintf(
536                'Failed asserting that exception message "%s" matches "%s".',
537                $throwable->getMessage(),
538                $this->expectedExceptionMessageRegularExpression,
539            ));
540        }
541
542        if ($this->expectedExceptionCode !== null && (string) $throwable->getCode() !== (string) $this->expectedExceptionCode) {
543            throw new AssertionFailedError(sprintf(
544                'Failed asserting that exception code "%s" is equal to "%s".',
545                (string) $throwable->getCode(),
546                (string) $this->expectedExceptionCode,
547            ));
548        }
549
550        // The expectation itself counts as a verified assertion.
551        self::assertThat(true, new \LucianoPereira\Crucible\Assert\Constraint\IsTrue());
552    }
