32/**
33 * A point-in-time capture of PHP's mutable global surface: $GLOBALS,
34 * eligible static properties, environment variables, ini settings,
35 * error_reporting, and the working directory.
36 *
37 * restore() puts the world back (the VMVM model: reset state
38 * in-process instead of paying for a process per test); changes()
39 * names what a test polluted — the diagnostic behind
40 * beStrictAboutChangesToGlobalState's risky outcome.
41 *
42 * Object values are held by reference (spec behavior): mutations
43 * inside retained objects are neither restored nor detected.
44 */
45final readonly class GlobalStateSnapshot
46{
47    /** @var array<string, mixed> */
48    private array $globals;
49
50    /** @var array<int, mixed> indexed like $statics */
51    private array $staticValues;
52
53    /** @var list<array{class-string, non-empty-string}> */
54    private array $statics;
55
56    /** @var array<string, string> */
57    private array $environment;
58
59    /** @var array<string, string> */
60    private array $ini;
61
62    private int $errorReporting;
63
64    private WorkingDirectory $workingDirectory;
65
66    /**
67     * @param list<non-empty-string> $excludedGlobals
68     * @param list<array{class-string, non-empty-string}> $excludedStatics
69     */
70    public function __construct(
71        StaticRegistry $registry,
72        private array $excludedGlobals = [],
73        private array $excludedStatics = [],
74        private bool $includeStatics = true,
75    ) {
76        $globals = [];
77
78        foreach ($GLOBALS as $name => $value) {
79            if ($name === 'GLOBALS' || in_array($name, $this->excludedGlobals, true)) {
80                continue;
81            }
82
83            $globals[(string) $name] = $value;
84        }
85
86        $this->globals = $globals;
87
88        $statics      = [];
89        $staticValues = [];
90
91        if ($this->includeStatics) {
92            foreach ($registry->eligibleProperties() as $pair) {
93                if (in_array($pair, $this->excludedStatics, true)) {
94                    continue;
95                }
96
97                $property = new ReflectionProperty($pair[0], $pair[1]);
98
99                if (!$property->isInitialized()) {
100                    continue;
101                }
102
103                $statics[]      = $pair;
104                $staticValues[] = $property->getValue();
105            }
106        }
107
108        $this->statics      = $statics;
109        $this->staticValues = $staticValues;
110
111        /** @var array<string, string> $environment */
112        $environment       = getenv();
113        $this->environment = $environment;
114
115        $ini      = [];
116        $iniTable = ini_get_all(null, false);
117
118        foreach ($iniTable === false ? [] : $iniTable as $name => $value) {
119            if (is_string($value)) {
120                $ini[(string) $name] = $value;
121            }
122        }
123
124        $this->ini              = $ini;
125        $this->errorReporting   = error_reporting();
126        $this->workingDirectory = WorkingDirectory::current();
127    }
128
129    public function restore(): void
130    {
131        // Globals: reinstate snapshot values, drop additions.
132        foreach ($this->globals as $name => $value) {
133            $GLOBALS[$name] = $value;
134        }
135
136        foreach (array_keys($GLOBALS) as $name) {
137            if ($name === 'GLOBALS' || in_array($name, $this->excludedGlobals, true)) {
138                continue;
139            }
140
141            if (!array_key_exists($name, $this->globals)) {
142                unset($GLOBALS[$name]);
143            }
144        }
145
146        foreach ($this->statics as $index => [$class, $property]) {
147            (new ReflectionProperty($class, $property))->setValue(null, $this->staticValues[$index]);
148        }
149
150        // Environment: reinstate and drop additions.
151        foreach ($this->environment as $name => $value) {
152            if (getenv($name) !== $value) {
153                putenv(sprintf('%s=%s', $name, $value));
154            }
155        }
156
157        /** @var array<string, string> $currentEnvironment */
158        $currentEnvironment = getenv();
159
160        foreach (array_keys($currentEnvironment) as $name) {
161            if (!array_key_exists($name, $this->environment)) {
162                putenv($name);
163            }
164        }
165
166        foreach ($this->ini as $name => $value) {
167            if (ini_get($name) !== $value) {
168                @ini_set($name, $value);
169            }
170        }
171
172        error_reporting($this->errorReporting);
173
174        // The non-empty half of this guard was the annotation's job and is
175        // now the type's, so only the comparison survives.
176        if (getcwd() !== $this->workingDirectory->path) {
177            @chdir($this->workingDirectory->path);
178        }
179    }
180
181    /**
182     * Human-readable names of everything that differs from the
183     * snapshot, capped for readable risky reasons.
184     *
185     * @return list<non-empty-string>
186     */
187    public function changes(int $limit = 5): array
188    {
189        $changes = [];
190
191        foreach ($this->globals as $name => $value) {
192            if (!array_key_exists($name, $GLOBALS) || $GLOBALS[$name] !== $value) {
193                $changes[] = sprintf('global $%s', $name);
194            }
195        }
196
197        foreach (array_keys($GLOBALS) as $name) {
198            if ($name !== 'GLOBALS' && !array_key_exists($name, $this->globals) && !in_array($name, $this->excludedGlobals, true)) {
199                $changes[] = sprintf('global $%s (added)', (string) $name);
200            }
201        }
202
203        foreach ($this->statics as $index => [$class, $property]) {
204            if ((new ReflectionProperty($class, $property))->getValue() !== $this->staticValues[$index]) {
205                $changes[] = sprintf('%s::$%s', $class, $property);
206            }
207        }
208
209        foreach ($this->environment as $name => $value) {
210            if (getenv($name) !== $value) {
211                $changes[] = sprintf('env %s', $name);
212            }
213        }
214
215        if (error_reporting() !== $this->errorReporting) {
216            $changes[] = 'error_reporting';
217        }
218
219        if (count($changes) > $limit) {
220            $changes   = array_slice($changes, 0, $limit);
221            $changes[] = '…';
222        }
223
224        return $changes;
225    }
