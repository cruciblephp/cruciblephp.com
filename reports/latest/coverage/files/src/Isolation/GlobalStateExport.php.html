27/**
28 * The parent's runtime global state, carried into an isolated worker —
29 * what `#[PreserveGlobalState(true)]` asks for.
30 *
31 * Only the state a worker cannot already reconstruct for itself:
32 * user-defined constants and globals set while the run was going. A
33 * worker re-runs the configuration's bootstrap and re-applies its ini
34 * settings on its own, so those are preserved by construction and
35 * carrying them again would only let a stale copy win.
36 *
37 * Each value is serialized on its own, so a global holding a closure or
38 * a resource is skipped by name rather than taking the whole export
39 * down with it — the spec degrades the same way, and reports the same
40 * way, because a silently missing global is worse than a named one.
41 */
42final readonly class GlobalStateExport
43{
44    /**
45     * PHP owns these: a worker builds its own, and overwriting them
46     * with the parent's would misdescribe the process they run in.
47     */
48    private const array SUPERGLOBALS = [
49        'GLOBALS', '_ENV', '_POST', '_GET', '_COOKIE', '_SERVER', '_FILES', '_REQUEST',
50    ];
51
52    /**
53     * @param array<string, string> $constants name => base64 of the serialized value
54     * @param array<string, string> $globals   name => base64 of the serialized value
55     * @param list<string>          $skipped   names no serializer could carry
56     */
57    private function __construct(
58        public array $constants,
59        public array $globals,
60        public array $skipped,
61    ) {}
62
63    public static function capture(): self
64    {
65        $constants = [];
66        $globals   = [];
67        $skipped   = [];
68
69        /** @var array<string, mixed> $defined */
70        $defined = get_defined_constants(true)['user'] ?? [];
71
72        foreach ($defined as $name => $value) {
73            $encoded = self::encode($value);
74
75            if ($encoded === null) {
76                $skipped[] = 'constant ' . $name;
77
78                continue;
79            }
80
81            $constants[$name] = $encoded;
82        }
83
84        foreach ($GLOBALS as $name => $value) {
85            if (!is_string($name) || in_array($name, self::SUPERGLOBALS, true)) {
86                continue;
87            }
88
89            $encoded = self::encode($value);
90
91            if ($encoded === null) {
92                $skipped[] = '$' . $name;
93
94                continue;
95            }
96
97            $globals[$name] = $encoded;
98        }
99
100        return new self($constants, $globals, $skipped);
101    }
102
103    /**
104     * Applied before the worker's bootstrap, so a bootstrap that sets
105     * the same name still wins — the spec's ordering, and the one that
106     * keeps a worker's own setup authoritative.
107     */
108    public function restore(): void
109    {
110        foreach ($this->constants as $name => $encoded) {
111            $value = $this->decode($encoded);
112
113            if (!defined($name) && $value !== null) {
114                define($name, $value['value']);
115            }
116        }
117
118        foreach ($this->globals as $name => $encoded) {
119            $value = $this->decode($encoded);
120
121            if ($value !== null) {
122                $GLOBALS[$name] = $value['value'];
123            }
124        }
125    }
126
127    public function toJson(): string
128    {
129        return base64_encode(serialize(['constants' => $this->constants, 'globals' => $this->globals]));
130    }
131
132    public static function fromJson(?string $encoded): ?self
133    {
134        if ($encoded === null || $encoded === '') {
135            return null;
136        }
137
138        $decoded = base64_decode($encoded, true);
139
140        if ($decoded === false) {
141            return null;
142        }
143
144        try {
145            // Suppressed: malformed input is an EXPECTED path here and
146            // unserialize() warns rather than throws, so the catch below
147            // never sees it. The try stays for a throwing __wakeup().
148            $payload = @unserialize($decoded);
149        } catch (Throwable) {
150            return null;
151        }
152
153        if (!is_array($payload)) {
154            return null;
155        }
156
157        return new self(
158            self::stringMap($payload['constants'] ?? null),
159            self::stringMap($payload['globals'] ?? null),
160            [],
161        );
162    }
163
164    /**
165     * @return ?string base64 of the serialized value, or null when nothing can carry it
166     */
167    private static function encode(mixed $value): ?string
168    {
169        try {
170            // Wrapped, so a legitimately-null value round-trips as a
171            // value rather than as "could not decode".
172            return base64_encode(serialize(['value' => $value]));
173        } catch (Throwable) {
174            return null;
175        }
176    }
177
178    /**
179     * @return ?array{value: mixed}
180     */
181    private function decode(string $encoded): ?array
182    {
183        $decoded = base64_decode($encoded, true);
184
185        if ($decoded === false) {
186            return null;
187        }
188
189        try {
190            $value = unserialize($decoded);
191        } catch (Throwable) {
192            return null;
193        }
194
195        return is_array($value) && array_key_exists('value', $value) ? ['value' => $value['value']] : null;
196    }
197
198    /**
199     * @return array<string, string>
200     */
201    private static function stringMap(mixed $decoded): array
202    {
203        $map = [];
204
205        foreach (is_array($decoded) ? $decoded : [] as $name => $value) {
206            if (is_string($name) && $name !== '' && is_string($value)) {
207                $map[$name] = $value;
208            }
209        }
210
211        return $map;
212    }
