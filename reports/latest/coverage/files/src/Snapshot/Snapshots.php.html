33/**
34 * The snapshot engine (D-042), dialect-neutral like everything else:
35 * one ambient context per test attempt (the PropertyContext pattern —
36 * a test body cannot be handed parameters), one canonical serializer
37 * (the Exporter, which D-012 built with snapshots already in mind),
38 * every dialect one thin call away — assertMatchesSnapshot() in
39 * TestCase, ->toMatchSnapshot() in pest/crucible, both callable from
40 * doctests.
41 *
42 * The behavior is deterministic and explicit, by principle: a missing
43 * snapshot FAILS and names the flag — snapshots are never invented as
44 * a side effect of a normal run — and `--update-snapshots` is the one
45 * way values get recorded or replaced.
46 */
47final class Snapshots

98    /**
99     * What the just-ended attempt recorded under --update-snapshots —
100     * the console counts (D-066). Null outside update mode / when
101     * nothing changed, so the event payload stays default-omitted.
102     * Stashed at end() by the ENDING context: a nested harness run
103     * (tests that run their own inner runner, the self-hosted suite's
104     * own pattern) cannot leak its counts onto the outer attempt —
105     * the outer end() overwrites the stash from its own state.
106     *
107     * @return ?array{created: int, updated: int}
108     */
109    public static function recorded(): ?array
110    {
111        return self::$lastRecorded;
112    }

114    public static function end(): void
115    {
116        self::$repository?->flush();
117
118        self::$lastRecorded = self::$update && (self::$created > 0 || self::$updated > 0)
119            ? ['created' => self::$created, 'updated' => self::$updated]
120            : null;
121
122        self::$lastVisited = self::$visited;
123
124        self::$test          = null;
125        self::$update        = false;
126        self::$inlineRewrite = false;
127        self::$repository    = null;
128        self::$counter       = 0;
129        self::$created       = 0;
130        self::$updated       = 0;
131        self::$visited       = [];
132    }

148    /**
149     * The one snapshot assertion. Compares the exported value with
150     * the stored entry; in update mode it records instead. Multiple
151     * calls in one test number themselves; $name pins a stable key.
152     *
153     * @param ?non-empty-string $name
154     */
155    public static function match(mixed $value, ?string $name = null): void
156    {
157        [$test, $repository, $key] = self::context($name);
158
159        $rendered = Exporter::export($value);
160        $file     = SnapshotRepository::pathFor(self::workingDirectory(), $test->file);
161        $stored   = $repository->get($file, $key);
162
163        if (self::$update) {
164            if ($stored === null) {
165                self::$created++;
166                $repository->put($file, $key, $rendered);
167            } elseif ($stored !== $rendered) {
168                self::$updated++;
169                $repository->put($file, $key, $rendered);
170            }
171
172            Assert::countSatisfiedAssertion(); // recording is the assertion
173
174            return;
175        }
176
177        if ($stored === null) {
178            Assert::fail(sprintf(
179                'No snapshot recorded for "%s". Run with --update-snapshots to record the current value.',
180                $key,
181            ));
182        }
183
184        Assert::assertSame($stored, $rendered, sprintf(
185            'Snapshot mismatch for "%s" (run --update-snapshots to accept the new value).',
186            $key,
187        ));
188    }

190    /**
191     * The inline snapshot assertion (D-076): the expected value lives in
192     * the test source, passed as $expected (null the first time). Same
193     * explicit-recording principle as the file flavor — a missing value
194     * FAILS and names the flag; `--update-snapshots` rewrites the source
195     * to record or replace it, buffered and applied at run end. Recording
196     * is sequential-run only; a worker cannot rewrite a shared file, so
197     * it fails naming the constraint instead of racing.
198     *
199     * @param string $file the call site, captured by the dialect entry point ('' when unknown)
200     */
201    public static function matchInline(mixed $value, ?string $expected, string $file, int $line): void
202    {
203        if (!self::$test instanceof TestId) {
204            throw new ConfigurationException('Snapshot assertions need the crucible runner — there is no snapshot context here.');
205        }
206
207        $rendered = Exporter::export($value);
208
209        // A value already recorded in the source: compare, or replace
210        // under --update-snapshots.
211        if ($expected !== null) {
212            if ($rendered === $expected) {
213                Assert::countSatisfiedAssertion();
214
215                return;
216            }
217
218            if (self::$update) {
219                self::recordInline($file, $line, $rendered);
220                self::$updated++;
221                Assert::countSatisfiedAssertion();
222
223                return;
224            }
225
226            Assert::assertSame($expected, $rendered, 'Inline snapshot mismatch (run --update-snapshots to accept the new value).');
227
228            return;
229        }
230
231        // Nothing recorded yet.
232        if (self::$update) {
233            self::recordInline($file, $line, $rendered);
234            self::$created++;
235            Assert::countSatisfiedAssertion();
236
237            return;
238        }
239
240        Assert::fail('No inline snapshot recorded. Run with --update-snapshots to record the current value.');
241    }

243    /**
244     * Queue a source rewrite, or fail naming the constraint when this
245     * process may not rewrite (a worker / a parallel run) or the call
246     * site could not be located.
247     */
248    private static function recordInline(string $file, int $line, string $rendered): void
249    {
250        if (!self::$inlineRewrite) {
251            Assert::fail('Inline snapshot recording needs a sequential run — rerun without --parallel (or process isolation) to record it.');
252        }
253
254        if ($file === '' || $line <= 0) {
255            Assert::fail('Inline snapshot recording could not locate its call site in the source.');
256        }
257
258        InlineSnapshotWriter::record($file, $line, $rendered);
259    }

261    /**
262     * The screenshot flavor (D-064): the .snap entry stays the
263     * content hash (byte-exact, collision-proof, diffable as text),
264     * while the update run also stores the reference PNG beside the
265     * snapshot file — so a later mismatch can render a VISUAL diff:
266     * reference, actual, and a blend-mode overlay (black where the
267     * pixels agree), all in one dependency-free HTML file named by
268     * the failure message. Recording stays explicit — the reference
269     * image is only ever written under --update-snapshots.
270     *
271     * @param ?non-empty-string $name
272     */
273    public static function matchScreenshot(string $png, ?string $name = null): void
274    {
275        [$test, $repository, $key] = self::context($name);
276
277        $rendered = 'png-sha256:' . hash('sha256', $png);
278        $file     = SnapshotRepository::pathFor(self::workingDirectory(), $test->file);
279        $stored   = $repository->get($file, $key);
280
281        $imageDirectory = substr($file, 0, -5); // …/__snapshots__/<basename> (the .snap dropped)
282        $imageBase      = $imageDirectory . '/' . self::fileSafe($key);
283
284        if (self::$update) {
285            if ($stored === null) {
286                self::$created++;
287                $repository->put($file, $key, $rendered);
288            } elseif ($stored !== $rendered) {
289                self::$updated++;
290                $repository->put($file, $key, $rendered);
291            }
292
293            if (!is_dir($imageDirectory)) {
294                mkdir($imageDirectory, 0o777, true);
295            }
296
297            file_put_contents($imageBase . '.png', $png);
298
299            Assert::countSatisfiedAssertion();
300
301            return;
302        }
303
304        if ($stored === null) {
305            Assert::fail(sprintf(
306                'No screenshot snapshot recorded for "%s". Run with --update-snapshots to record the current rendering.',
307                $key,
308            ));
309        }
310
311        if ($stored === $rendered) {
312            Assert::countSatisfiedAssertion();
313
314            return;
315        }
316
317        $diff = self::writeVisualDiff($imageBase, $png, $key);
318
319        Assert::fail(sprintf(
320            "Screenshot mismatch for \"%s\" (run --update-snapshots to accept the new rendering).%s",
321            $key,
322            $diff !== null ? "\nVisual diff: " . $diff : '',
323        ));
324    }

326    /**
327     * The side-by-side + difference-overlay view. Dependency-free:
328     * both PNGs embed as data URIs, the overlay uses CSS
329     * mix-blend-mode difference — black where the renderings agree.
330     * Null when the reference image is missing (hash recorded before
331     * the screenshot flavor existed).
332     */
333    private static function writeVisualDiff(string $imageBase, string $actual, string $key): ?string
334    {
335        $referenceFile = $imageBase . '.png';
336
337        if (!is_file($referenceFile)) {
338            return null;
339        }
340
341        file_put_contents($imageBase . '.actual.png', $actual);
342
343        $reference = base64_encode((string) file_get_contents($referenceFile));
344        $current   = base64_encode($actual);
345
346        $html = sprintf(
347            <<<'HTML'
348                <!DOCTYPE html>
349                <html lang="en">
350                <head><meta charset="utf-8"><title>%1$s — screenshot diff</title><style>
351                body { font-family: ui-monospace, monospace; margin: 2rem; background: #fff; color: #1a1a1a; }
352                h1 { font-size: 1rem; } h2 { font-size: .85rem; margin: 1.5rem 0 .5rem; }
353                img { max-width: 100%%; border: 1px solid #ddd; display: block; }
354                .side { display: flex; gap: 1rem; } .side > div { flex: 1; }
355                .overlay { position: relative; } .overlay img { position: absolute; inset: 0; }
356                .overlay img + img { mix-blend-mode: difference; }
357                .overlay { aspect-ratio: auto; } .overlay::after { content: ""; display: block; padding-top: 60%%; }
358                </style></head>
359                <body>
360                <h1>%1$s</h1>
361                <div class="side">
362                <div><h2>Reference</h2><img src="data:image/png;base64,%2$s" alt="reference"></div>
363                <div><h2>Actual</h2><img src="data:image/png;base64,%3$s" alt="actual"></div>
364                </div>
365                <h2>Difference (black = identical)</h2>
366                <div class="overlay"><img src="data:image/png;base64,%2$s" alt=""><img src="data:image/png;base64,%3$s" alt=""></div>
367                </body>
368                </html>
369                HTML,
370            htmlspecialchars($key, ENT_QUOTES),
371            $reference,
372            $current,
373        );
374
375        $diffFile = $imageBase . '.diff.html';
376
377        file_put_contents($diffFile, $html);
378
379        return $diffFile;
380    }

382    /**
383     * @param ?non-empty-string $name
384     *
385     * @return array{TestId, SnapshotRepository, non-empty-string}
386     */
387    private static function context(?string $name): array
388    {
389        $test       = self::$test;
390        $repository = self::$repository;
391
392        if (!$test instanceof TestId || !$repository instanceof SnapshotRepository) {
393            throw new ConfigurationException('Snapshot assertions need the crucible runner — there is no snapshot context here.');
394        }
395
396        $key = $test->name
397            . ($test->dataset !== null ? '#' . $test->dataset : '')
398            . ' » '
399            . ($name ?? '#' . ++self::$counter);
400
401        self::$visited[] = $key;
402
403        return [$test, $repository, $key];
404    }

406    /**
407     * @return non-empty-string
408     */
409    public static function fileSafe(string $key): string
410    {
411        $safe = (string) preg_replace('/[^A-Za-z0-9._-]+/', '_', $key);
412
413        return $safe === '' ? '_' : $safe;
414    }
