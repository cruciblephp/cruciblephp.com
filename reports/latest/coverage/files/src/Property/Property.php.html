37/**
38 * The property runner (growth G5, Hypothesis-style): N cases drawn
39 * from a seeded choice stream; the first falsifying case is shrunk
40 * by editing its recorded choices — delete spans, zero spans, walk
41 * single choices toward 0 — and the minimal counterexample is
42 * reported with the seed, so the failure replays exactly. Engine-
43 * owned and dialect-neutral: it runs inside any test body, whatever
44 * frontend declared it.
45 *
46 *     Property::forAll(Gen::int(), Gen::int())
47 *         ->check(fn (int $a, int $b) => Assert::assertSame($a + $b, $b + $a));
48 */
49final class Property
50{
51    private const int SHRINK_BUDGET = 500;
52
53    private int $cases = 100;
54
55    private ?int $seed = null;

102    /**
103     * Runs the property against every generated case. The property
104     * may assert (preferred) or return false to falsify; any other
105     * return value passes. Throws the shrunk counterexample as an
106     * assertion failure.
107     */
108    public function check(callable $property): void
109    {
110        $context = PropertyContext::claim();
111
112        // The failure database (D-040): stored counterexamples replay
113        // before any random case, so a bug that once falsified is
114        // caught on case 1 forever, whatever today's seed is.
115        $sequences = $context['sequences'] ?? [];
116        $clean     = $sequences !== [];
117
118        foreach ($sequences as $stored) {
119            $source = new ChoiceSource(null, $stored);
120
121            try {
122                $arguments = $this->arguments($source);
123            } catch (CannotGenerate) {
124                // The property changed shape; the entry is inert — and
125                // an entry that could not replay was not proven fixed.
126                $clean = false;
127
128                continue;
129            }
130
131            $failure = $this->falsify($property, $arguments);
132
133            if ($failure instanceof Throwable) {
134                [$choices, $arguments, $failure, $steps] = $this->shrink($property, $source->choices(), $arguments, $failure);
135
136                $this->fail('by the failure database', $steps, $arguments, $failure, $context['key'] ?? null, $choices);
137            }
138        }
139
140        if ($clean && isset($context['key'])) {
141            PropertyContext::confirmClean($context['key']);
142        }
143
144        $seed       = $this->seed ?? random_int(0, 2_147_483_647);
145        $randomizer = new Randomizer(new Mt19937($seed));
146        $discards   = 0;
147
148        for ($case = 1; $case <= $this->cases; $case++) {
149            $source = new ChoiceSource($randomizer);
150
151            try {
152                $arguments = $this->arguments($source);
153            } catch (CannotGenerate $exhausted) {
154                if (++$discards > max(100, $this->cases * 10)) {
155                    throw new ConfigurationException($exhausted->getMessage(), $exhausted->getCode(), $exhausted);
156                }
157
158                $case--;
159
160                continue;
161            }
162
163            $failure = $this->falsify($property, $arguments);
164
165            if (!$failure instanceof Throwable) {
166                continue;
167            }
168
169            [$choices, $arguments, $failure, $steps] = $this->shrink($property, $source->choices(), $arguments, $failure);
170
171            $this->fail(
172                sprintf('at case %d of %d (seed %d)', $case, $this->cases, $seed),
173                $steps,
174                $arguments,
175                $failure,
176                $context['key'] ?? null,
177                $choices,
178                $seed,
179            );
180        }
181
182        // The cases are the assertion; without this a pure-generator
183        // property would read as risky.
184        Assert::countSatisfiedAssertion();
185    }

187    /**
188     * @param non-empty-string  $origin
189     * @param list<mixed>       $arguments
190     * @param ?non-empty-string $key       the failure-database key, when a runner context is open
191     * @param list<int>         $choices
192     */
193    private function fail(string $origin, int $steps, array $arguments, Throwable $failure, ?string $key, array $choices, ?int $seed = null): never
194    {
195        $message = sprintf(
196            "Property falsified %s, %d shrink steps.\nCounterexample: %s\n%s\n\n%s",
197            $origin,
198            $steps,
199            implode(', ', array_map(self::compact(...), $arguments)),
200            $seed !== null ? sprintf('Replay with ->seed(%d).', $seed) : 'It replays on every run until fixed.',
201            $failure->getMessage(),
202        );
203
204        throw $key !== null
205            ? new PropertyFailedError($message, $key, $choices)
206            : new AssertionFailedError($message);
207    }

217    /**
218     * Null when the property holds; the failure otherwise. Skip and
219     * incomplete signals pass through — they are verdicts about the
220     * test, not the property.
221     *
222     * @param list<mixed> $arguments
223     */
224    private function falsify(callable $property, array $arguments): ?Throwable
225    {
226        try {
227            $verdict = $property(...$arguments);
228        } catch (SkippedTestError|IncompleteTestError $signal) {
229            throw $signal;
230        } catch (Throwable $thrown) {
231            return $thrown;
232        }
233
234        return $verdict === false
235            ? new AssertionFailedError('The property returned false.')
236            : null;
237    }

239    /**
240     * Choice-sequence shrinking: candidates are edits of the failing
241     * sequence — spans deleted, spans zeroed, single choices halved
242     * and decremented — and a candidate replaces the incumbent when
243     * it still fails and its consumed sequence is simpler (shorter,
244     * or lexicographically smaller at equal length). Runs to a
245     * fixpoint or the budget.
246     *
247     * @param list<int>   $choices
248     * @param list<mixed> $arguments
249     *
250     * @return array{list<int>, list<mixed>, Throwable, int}
251     */
252    private function shrink(callable $property, array $choices, array $arguments, Throwable $failure): array
253    {
254        $budget = self::SHRINK_BUDGET;
255        $steps  = 0;
256
257        $improved = true;
258
259        while ($improved && $budget > 0) {
260            $improved = false;
261
262            foreach ($this->candidates($choices) as $candidate) {
263                if ($budget-- <= 0) {
264                    break;
265                }
266
267                $source = new ChoiceSource(null, $candidate);
268
269                try {
270                    $candidateArguments = $this->arguments($source);
271                } catch (CannotGenerate) {
272                    continue;
273                }
274
275                $candidateFailure = $this->falsify($property, $candidateArguments);
276
277                if (!$candidateFailure instanceof Throwable) {
278                    continue;
279                }
280
281                $consumed = $source->choices();
282
283                if (!$this->simpler($consumed, $choices)) {
284                    continue;
285                }
286
287                $choices   = $consumed;
288                $arguments = $candidateArguments;
289                $failure   = $candidateFailure;
290                $steps++;
291                $improved = true;
292
293                break; // restart the passes from the new incumbent
294            }
295        }
296
297        return [$choices, $arguments, $failure, $steps];
298    }

300    /**
301     * The edit passes, most aggressive first.
302     *
303     * @param list<int> $choices
304     *
305     * @return iterable<array<int, int>> contiguously indexed from 0
306     */
307    private function candidates(array $choices): iterable
308    {
309        $count = count($choices);
310
311        // Delete spans — halves down to single choices.
312        for ($span = max(1, intdiv($count, 2)); $span >= 1; $span = intdiv($span, 2)) {
313            for ($at = 0; $at + $span <= $count; $at += $span) {
314                yield array_merge(array_slice($choices, 0, $at), array_slice($choices, $at + $span));
315            }
316        }
317
318        // Zero spans.
319        for ($span = max(1, intdiv($count, 2)); $span >= 1; $span = intdiv($span, 2)) {
320            for ($at = 0; $at + $span <= $count; $at += $span) {
321                $zeroed = $choices;
322
323                for ($i = $at; $i < $at + $span; $i++) {
324                    $zeroed[$i] = 0;
325                }
326
327                if ($zeroed !== $choices) {
328                    yield $zeroed;
329                }
330            }
331        }
332
333        // Redistribute-and-delete: move choice $at onto a nearby
334        // choice and delete $at's span in the same candidate — the
335        // only edit that can merge two list elements ([10, 90] →
336        // [100]) under shortlex, because raising a choice alone is
337        // never simpler but the merged sequence is strictly shorter.
338        // Both span guesses are structure-blind: ($at-1, $at) covers
339        // continuation-bit + value, ($at) alone covers bare values.
340        foreach ($choices as $at => $choice) {
341            if ($choice <= 0) {
342                continue;
343            }
344
345            for ($to = max(0, $at - 8); $to < min($count, $at + 9); $to++) {
346                if ($to === $at) {
347                    continue;
348                }
349
350                foreach ([[$at - 1, $at], [$at, $at]] as [$from, $till]) {
351                    if ($from < 0 || ($to >= $from && $to <= $till)) {
352                        continue;
353                    }
354
355                    $merged = $choices;
356                    $merged[$to] += $choice;
357
358                    for ($i = $from; $i <= $till; $i++) {
359                        unset($merged[$i]);
360                    }
361
362                    yield array_values($merged);
363                }
364            }
365        }
366
367        // Walk single choices toward zero: halve, then decrement.
368        foreach ($choices as $at => $choice) {
369            if ($choice > 0) {
370                $halved      = $choices;
371                $halved[$at] = intdiv($choice, 2);
372
373                yield $halved;
374
375                $decremented      = $choices;
376                $decremented[$at] = $choice - 1;
377
378                yield $decremented;
379            }
380        }
381    }

383    /**
384     * One-line rendering for the counterexample: arrays inline,
385     * scalars through the canonical exporter.
386     */
387    private static function compact(mixed $value): string
388    {
389        if (!is_array($value)) {
390            return Exporter::describe($value);
391        }
392
393        $parts  = [];
394        $isList = array_is_list($value);
395
396        foreach ($value as $key => $item) {
397            $parts[] = ($isList ? '' : Exporter::describe($key) . ' => ') . self::compact($item);
398        }
399
400        return '[' . implode(', ', $parts) . ']';
401    }

403    /**
404     * @param list<int> $candidate
405     * @param list<int> $incumbent
406     */
407    private function simpler(array $candidate, array $incumbent): bool
408    {
409        if (count($candidate) !== count($incumbent)) {
410            return count($candidate) < count($incumbent);
411        }
412
413        foreach ($candidate as $at => $choice) {
414            if ($choice !== $incumbent[$at]) {
415                return $choice < $incumbent[$at];
416            }
417        }
418
419        return false;
420    }
