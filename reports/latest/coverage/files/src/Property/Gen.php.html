31/**
32 * Value generators for property-based testing (growth G5). A Gen is
33 * one closure from the choice stream to a value; combinators compose
34 * closures, never touch randomness directly, and stay shrinkable for
35 * free — the shrinker edits the underlying choices, and simpler
36 * choices flow through map() and suchThat() into simpler values.
37 *
38 * Generic over what it produces (D-051): Gen<int>, Gen<list<string>>
39 * — map() and suchThat() carry the type through, and the PHPStan
40 * extension hands check() closures their real parameter types.
41 *
42 * @template-covariant T
43 */
44final readonly class Gen
45{

47     * @param Closure(ChoiceSource): T $build
48     */
49    private function __construct(
50        private Closure $build,
51    ) {}

61    /**
62     * Integers in [$min, $max], shrinking toward the value of
63     * smallest magnitude the range allows (0 when it contains 0 —
64     * the Hypothesis ordering, not "toward $min").
65     *
66     * @return self<int>
67     */
68    public static function int(int $min = -1_000_000, int $max = 1_000_000): self
69    {
70        if ($min > $max) {
71            throw new ConfigurationException(sprintf('Gen::int(%d, %d): min exceeds max.', $min, $max));
72        }
73
74        // PHP integer overflow degrades to float, so test the width
75        // before subtracting (rector note: an is_int() check on the
76        // difference gets "simplified" away — its inference does not
77        // model overflow).
78        if ($min < 0 && $max > PHP_INT_MAX + $min) {
79            throw new ConfigurationException('Gen::int(): the range overflows; use a narrower one.');
80        }
81
82        $range = $max - $min;
83
84        return new self(static fn(ChoiceSource $source): int => self::intFromChoice($source->draw($range), $min, $max));
85    }

95    /**
96     * Finite floats in [$min, $max] (D-040, refined by D-069).
97     * Structured for shrinking: a leading choice picks between the
98     * in-range *specials* pool (0.0, 1.0, -1.0, the bounds — earlier
99     * is simpler) and the continuum. A range wider than one unit
100     * encodes as integer part (riding the int ordering, toward zero)
101     * plus a dyadic n/65536 fraction that shrinks toward whole
102     * numbers; anything narrower uses 65,536 evenly spaced points, so
103     * resolution scales with the range instead of collapsing when it
104     * drops under the fraction quantum (the D-069 probe: [0, 1e-6]
105     * produced two distinct values and could never falsify). Non-
106     * finite edge cases (NAN, ±INF) are never produced by a bounded
107     * generator — compose them explicitly: `Gen::oneOf(
108     * Gen::elementOf([NAN, INF, -INF]), Gen::float(...))`.
109     *
110     * @return self<float>
111     */
112    public static function float(float $min = -1_000.0, float $max = 1_000.0): self
113    {
114        if (!is_finite($min) || !is_finite($max) || $min > $max) {
115            throw new ConfigurationException(sprintf('Gen::float(%s, %s): needs a finite range with min <= max.', (string) $min, (string) $max));
116        }
117
118        $specials = [];
119
120        foreach ([0.0, 1.0, -1.0, $min, $max] as $candidate) {
121            if ($candidate >= $min && $candidate <= $max && !in_array($candidate, $specials, true)) {
122                $specials[] = $candidate;
123            }
124        }
125
126        if ($specials === []) {
127            $specials = [$min]; // unreachable ($min always qualifies), but provable
128        }
129
130        // Both whole-part bounds FLOOR (not ceil/floor): the fraction
131        // only adds, so flooring $min keeps the sub-integer interior
132        // below the first whole number reachable ([-0.5, 0.5] used to
133        // have no way to produce an interior negative), and the clamp
134        // still guarantees the bounds.
135        $intMin = (int) floor($min);
136        $intMax = (int) floor($max);
137
138        $continuum = $max - $min > 1.0
139            ? static function (ChoiceSource $source) use ($min, $max, $intMin, $intMax): float {
140                $whole    = self::intFromChoice($source->draw($intMax - $intMin), $intMin, $intMax);
141                $fraction = $source->draw(65_535) / 65_536.0;
142
143                return max($min, min($max, $whole + $fraction));
144            }
145        : static fn(ChoiceSource $source): float => $min + ($max - $min) * ($source->draw(65_535) / 65_535.0);
146
147        return new self(static function (ChoiceSource $source) use ($specials, $continuum): float {
148            // Choice 0 = the specials pool = simplest; shrinking any
149            // continuum float first tries to land on a named value.
150            if ($source->draw(7) === 0) {
151                return $specials[$source->draw(count($specials) - 1)];
152            }
153
154            return $continuum($source);
155        });
156    }

211    /**
212     * Strings over the alphabet, shrinking toward the shortest allowed
213     * and toward the alphabet's first character.
214     *
215     * @param int<0, max>      $maxLength
216     * @param non-empty-string $alphabet
217     * @param int<0, max>      $minLength characters every string has (D-132): `non-empty-string` is 1,
218     *                                    drawn directly rather than filtered, so no draw is wasted
219     *
220     * @return self<string>
221     */
222    public static function string(
223        int $maxLength = 20,
224        string $alphabet = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789 _-',
225        int $minLength = 0,
226    ): self {
227        $characters = mb_str_split($alphabet);
228        $last       = count($characters) - 1;
229
230        // Continuation choices, not a length prefix (the Hypothesis
231        // list encoding): each element is preceded by a "one more?"
232        // choice, so deleting an element's choices from the stream
233        // deletes the element — no stale length drawing phantom
234        // entries. The choice is biased (0 of 8 stops) or lengths
235        // would collapse to a coin-flip average of one.
236        return new self(static function (ChoiceSource $source) use ($maxLength, $characters, $last, $minLength): string {
237            $string = '';
238            $length = 0;
239
240            // The required characters carry no continuation choice.
241            while ($length < $minLength) {
242                $string .= $characters[$source->draw($last)];
243                $length++;
244            }
245
246            while ($length < $maxLength && $source->draw(7) !== 0) {
247                $string .= $characters[$source->draw($last)];
248                $length++;
249            }
250
251            return $string;
252        });
253    }

255    /**
256     * Lists of up to $maxCount elements, shrinking toward [].
257     *
258     * @template TElement
259     *
260     * @param self<TElement> $element
261     * @param int<0, max>    $maxCount
262     * @param int<0, max>    $minCount items every list has (D-132), for `non-empty-list`
263     *
264     * @return self<list<TElement>>
265     */
266    public static function listOf(self $element, int $maxCount = 25, int $minCount = 0): self
267    {
268        // Continuation-choice encoding — see string() for why.
269        return new self(static function (ChoiceSource $source) use ($element, $maxCount, $minCount): array {
270            $items = [];
271
272            while (count($items) < $minCount) {
273                $items[] = $element->generate($source);
274            }
275
276            while (count($items) < $maxCount && $source->draw(7) !== 0) {
277                $items[] = $element->generate($source);
278            }
279
280            return $items;
281        });
282    }

284    /**
285     * Arrays of a fixed shape (D-132): every key its own generator, an
286     * optional key present or not. Presence is a choice drawn before the
287     * value, 0 for absent, so shrinking removes optional keys first — the
288     * smallest counterexample names only the keys the failure needs.
289     *
290     *     Gen::shape(['id' => Gen::int(1, 99), 'nick' => [Gen::string(), true]])
291     *
292     * @param array<array-key, self<mixed>|array{self<mixed>, bool}> $entries key => generator, or [generator, optional]
293     *
294     * @return self<array<array-key, mixed>>
295     */
296    public static function shape(array $entries): self
297    {
298        return new self(static function (ChoiceSource $source) use ($entries): array {
299            $value = [];
300
301            foreach ($entries as $key => $entry) {
302                [$generator, $optional] = $entry instanceof self ? [$entry, false] : $entry;
303
304                if ($optional && $source->draw(1) === 0) {
305                    continue;
306                }
307
308                $value[$key] = $generator->generate($source);
309            }
310
311            return $value;
312        });
313    }

350    /**
351     * Keeps drawing until the predicate holds. A filter that rejects
352     * too much is a configuration problem, reported as such — the
353     * property runner discards the case, it never fails the test.
354     *
355     * @param Closure(T): bool $predicate
356     * @param positive-int     $maxAttempts
357     *
358     * @return self<T>
359     */
360    public function suchThat(Closure $predicate, int $maxAttempts = 50): self
361    {
362        $build = $this->build;
363
364        return new self(static function (ChoiceSource $source) use ($build, $predicate, $maxAttempts): mixed {
365            for ($attempt = 0; $attempt < $maxAttempts; $attempt++) {
366                $value = $build($source);
367
368                if ($predicate($value)) {
369                    return $value;
370                }
371            }
372
373            throw new CannotGenerate('suchThat() rejected every candidate; loosen the filter or shape the generator.');
374        });
375    }

377    /**
378     * The choice → value ordering for integers: choice 0 is the
379     * in-range value closest to zero, and successive choices
380     * alternate outward (0, 1, -1, 2, -2, …) until one side of the
381     * range runs out. A bijection, so every value stays reachable.
382     */
383    private static function intFromChoice(int $choice, int $min, int $max): int
384    {
385        $target = match (true) {
386            $min > 0 => $min,
387            $max < 0 => $max,
388            default  => 0,
389        };
390
391        if ($choice === 0) {
392            return $target;
393        }
394
395        $down  = $target - $min;
396        $up    = $max - $target;
397        $pairs = min($down, $up);
398
399        if ($choice <= 2 * $pairs) {
400            $step = intdiv($choice + 1, 2);
401
402            return $choice % 2 === 1 ? $target + $step : $target - $step;
403        }
404
405        $rest = $choice - 2 * $pairs;
406
407        return $up > $down ? $target + $pairs + $rest : $target - $pairs - $rest;
408    }
