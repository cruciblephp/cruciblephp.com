35/**
36 * The file-level dependency graph for impact selection (growth G3,
37 * Ekstazi's granularity): file → the project files it references,
38 * derived lazily from static analysis (the anti-haste-map lesson —
39 * nothing is scanned until a closure walks into it) and resolved
40 * through the Composer autoloaders, which are the authority on
41 * name → file. Vendor code is not tracked: a dependency bump is an
42 * environment change, handled upstream as a full run.
43 *
44 * A second edge source — per-test covered-files maps — plugs in here
45 * when the coverage phase lands (RESEARCH.md §2).
46 */
47final class DependencyGraph
48{
49    /** @var array<string, list<string>> */
50    private array $dependencies = [];
51
52    /** @var list<ClassLoader> */
53    private readonly array $loaders;
54
55    /** @var non-empty-string */
56    private readonly string $projectRoot;
57
58    /**
59     * @param non-empty-string            $projectRoot
60     * @param array<string, list<string>> $observedEdges coverage-observed dependencies (D-041),
61     *                                                   absolute file => absolute files it executed —
62     *                                                   what static analysis cannot see, execution can
63     */
64    public function __construct(
65        string $projectRoot,
66        private readonly ReferenceScanner $scanner = new ReferenceScanner(),
67        private readonly array $observedEdges = [],
68        private readonly ?DependencyIndex $index = null,
69    ) {
70        $this->index?->load();
71
72        $real = realpath($projectRoot);
73
74        $this->projectRoot = rtrim(WorkingDirectory::native($real === false ? $projectRoot : $real), DIRECTORY_SEPARATOR) . DIRECTORY_SEPARATOR;
75        $this->loaders     = $this->composerLoaders();
76    }
77
78    /**
79     * Every project file reachable from $file, including itself —
80     * the set whose intersection with the changed files decides
81     * whether $file's tests run.
82     *
83     * @param non-empty-string $file absolute path
84     *
85     * @return array<string, true> keyed by absolute path
86     */
87    public function closureOf(string $file): array
88    {
89        $real = realpath($file);
90        $real = $real === false ? WorkingDirectory::native($file) : $real;
91
92        /** @var array<string, true> $visited */
93        $visited = [];
94
95        // Observed edges apply one hop from the root only: coverage
96        // already recorded everything the root's tests executed —
97        // transitively — so expanding them again from inner nodes
98        // would conflate "this test runs file X" with "file X's own
99        // inline tests run Y" (a file can be both production code
100        // and a test carrier) and over-select wildly.
101        $stack = [$real, ...($this->observedEdges[$real] ?? [])];
102
103        while ($stack !== []) {
104            $current = array_pop($stack);
105
106            if (isset($visited[$current])) {
107                continue;
108            }
109
110            $visited[$current] = true;
111
112            foreach ($this->dependenciesOf($current) as $dependency) {
113                if (!isset($visited[$dependency])) {
114                    $stack[] = $dependency;
115                }
116            }
117        }
118
119        return $visited;
120    }
121
122    /**
123     * @return list<string> absolute paths, memoized per file
124     */
125    private function dependenciesOf(string $file): array
126    {
127        if (isset($this->dependencies[$file])) {
128            return $this->dependencies[$file];
129        }
130
131        $reused = $this->index?->reuse($file);
132
133        if ($reused !== null) {
134            return $this->dependencies[$file] = $reused;
135        }
136
137        $source = is_file($file) ? file_get_contents($file) : false;
138
139        /** @var array<string, true> $found */
140        $found = [];
141
142        if ($source !== false) {
143            foreach ($this->scanner->referencesIn($source) as $candidate) {
144                $resolved = $this->fileOf($candidate);
145
146                if ($resolved !== null && $resolved !== $file) {
147                    $found[$resolved] = true;
148                }
149            }
150        }
151
152        // Pest-family files also depend on the suite configuration
153        // above them: Pest.php scopes and shared Datasets/*.php are
154        // loaded per ancestor directory (D-033), so a change there
155        // must re-run the files beneath.
156        if (str_ends_with($file, '.pest.php') || str_ends_with($file, '.crucible.php')) {
157            foreach ($this->suiteConfigurationAbove($file) as $configuration) {
158                if ($configuration !== $file) {
159                    $found[$configuration] = true;
160                }
161            }
162        }
163
164        $resolved = array_keys($found);
165        $this->index?->record($file, $resolved);
166
167        return $this->dependencies[$file] = $resolved;
168    }
169
170    /** Persist what this run resolved, when an index is in use. */
171    public function persist(): void
172    {
173        $this->index?->save();
174    }
175
176    /**
177     * @return ?string absolute path inside the project (vendor excluded)
178     */
179    private function fileOf(string $class): ?string
180    {
181        foreach ($this->loaders as $loader) {
182            $file = $loader->findFile($class);
183
184            if ($file === false) {
185                continue;
186            }
187
188            $real = realpath($file);
189
190            if ($real === false) {
191                continue;
192            }
193
194            // 'vendor' as a path SEGMENT anywhere, not only at the root.
195            // A vendored tool or a nested package carries its own
196            // vendor/, and a class resolved there is still someone
197            // else's code — ✓ found 2026-09-16 when phpcpd-main/ gained
198            // symfony/polyfill-php80, whose PhpToken stub entered the
199            // graph through a root-only check.
200            if (str_starts_with($real, $this->projectRoot)
201                && !in_array('vendor', explode(DIRECTORY_SEPARATOR, substr($real, strlen($this->projectRoot))), true)
202            ) {
203                return $real;
204            }
205        }
206
207        return null;
208    }
209
210    /**
211     * @param non-empty-string $file
212     *
213     * @return list<string> absolute paths
214     */
215    private function suiteConfigurationAbove(string $file): array
216    {
217        $found     = [];
218        $directory = dirname($file);
219        $root      = rtrim($this->projectRoot, DIRECTORY_SEPARATOR);
220
221        while (str_starts_with($directory, $root)) {
222            $pest = $directory . DIRECTORY_SEPARATOR . 'Pest.php';
223
224            if (is_file($pest)) {
225                $found[] = $pest;
226            }
227
228            $datasets = glob($directory . DIRECTORY_SEPARATOR . 'Datasets' . DIRECTORY_SEPARATOR . '*.php');
229
230            foreach ($datasets === false ? [] : $datasets as $dataset) {
231                $found[] = $dataset;
232            }
233
234            if ($directory === $root) {
235                break;
236            }
237
238            $directory = dirname($directory);
239        }
240
241        return $found;
242    }
243
244    /**
245     * @return list<ClassLoader>
246     */
247    private function composerLoaders(): array
248    {
249        $loaders = [];
250
251        foreach (spl_autoload_functions() as $function) {
252            if (is_array($function) && $function[0] instanceof ClassLoader) {
253                $loaders[] = $function[0];
254            }
255        }
256
257        return $loaders;
258    }
