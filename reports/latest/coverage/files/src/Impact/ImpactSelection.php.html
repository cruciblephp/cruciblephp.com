35/**
36 * Ekstazi-style test selection at file granularity (growth G3): a
37 * test group runs when the transitive closure of the file declaring
38 * it intersects the changed files. The safety direction is fixed —
39 * everything that cannot be reasoned about (deletions, environment
40 * files) widens the selection to the full suite.
41 *
42 * Every change that does not select a group is reported under the
43 * structural reason it did not ({@see ImpactReason}), never as one
44 * bucket: `Uncovered` wants a test, `Undeclared` wants an impact rule,
45 * and a category that cannot say which would hide both.
46 *
47 * Declared impact rules (D-083) are the second, additive source: a
48 * changed file the graph cannot reach — an asset, a template, a
49 * translation — adds the groups its rule names. They only ever widen,
50 * so a missing rule leaves the graph's own answer untouched.
51 */
52final readonly class ImpactSelection
53{
54    /**
55     * A dependency bump invalidates the graph wholesale, so a change to
56     * one of these widens the run to everything. The graph cannot see
57     * inside an installed package, so narrowing after one moved would
58     * be guessing.
59     *
60     * The JavaScript lockfiles are here for the same reason as the PHP
61     * ones and not by analogy: Crucible runs JavaScript suites through
62     * the Vitest tier (D-079), so a bumped npm dependency can break a
63     * test in this run — and without this the selection would narrow
64     * past exactly the tests that would have caught it.
65     *
66     * @var list<non-empty-string>
67     */
68    private const array MANIFESTS = [
69        'composer.json', 'composer.lock',
70        'package.json', 'package-lock.json', 'yarn.lock', 'pnpm-lock.yaml',
71    ];
72
73    /**
74     * @param non-empty-string       $projectRoot
75     * @param list<non-empty-string> $environmentFiles absolute paths whose change invalidates every test
76     *                                                 (the configuration file, the bootstrap)
77     * @param list<ImpactRule>       $rules            declared path → groups rules (D-083); additive only
78     * @param list<non-empty-string> $claimedPaths     absolute directories another tier answers for
79     *                                                 (the configured Vitest suites, D-080)
80     */
81    public function __construct(
82        private DependencyGraph $graph,
83        private string $projectRoot,
84        private array $environmentFiles = [],
85        private array $rules = [],
86        private array $claimedPaths = [],
87    ) {}
88
89    /**
90     * @param list<TestGroup> $groups
91     */
92    public function select(array $groups, ChangedFiles $changed): ImpactResult
93    {
94        if ($changed->deleted !== []) {
95            return ImpactResult::everything(sprintf(
96                'Impact: %d deleted file(s) — their dependents are invisible to the graph',
97                count($changed->deleted),
98            ));
99        }
100
101        /** @var array<string, true> $targets */
102        $targets = [];
103        /** @var array<string, true> $ruled the groups matching rules put in doubt */
104        $ruled      = [];
105        $undeclared = 0;
106        $claimed    = 0;
107        $matched    = 0;
108
109        foreach ($changed->files as $file) {
110            $real = realpath($file);
111            $real = $real === false ? WorkingDirectory::native($file) : $real;
112
113            if (in_array($real, array_map(WorkingDirectory::native(...), $this->environmentFiles), true)
114                || in_array(basename($real), self::MANIFESTS, true)
115            ) {
116                return ImpactResult::everything(sprintf('Impact: %s defines the environment', basename($real)));
117            }
118
119            // Rules see every change, whatever its extension: a Blade
120            // template and a translation file both end in .php and are
121            // both unreachable by class reference.
122            $named = false;
123
124            foreach ($this->rules as $rule) {
125                if (!$rule->matches($this->relative($real))) {
126                    continue;
127                }
128
129                $named = true;
130
131                foreach ($rule->groups as $group) {
132                    $ruled[$group] = true;
133                }
134            }
135
136            if ($named) {
137                ++$matched;
138            }
139
140            if (!str_ends_with($real, '.php')) {
141                if ($named) {
142                    continue;
143                }
144
145                if ($this->isClaimed($real)) {
146                    ++$claimed;
147
148                    continue;
149                }
150
151                ++$undeclared;
152
153                continue;
154            }
155
156            $targets[$real] = true;
157        }
158
159        $notes = [];
160
161        if ($claimed > 0) {
162            $notes[] = sprintf(
163                'Impact: %d change(s) %s.',
164                $claimed,
165                ImpactReason::Claimed->heading(),
166            );
167        }
168
169        if ($undeclared > 0) {
170            $notes[] = sprintf(
171                'Impact: %d change(s) %s.',
172                $undeclared,
173                ImpactReason::Undeclared->heading(),
174            );
175        }
176
177        if ($matched > 0) {
178            $notes[] = sprintf(
179                'Impact: %d change(s) matched an impact rule — group(s) %s put in doubt.',
180                $matched,
181                implode(', ', array_keys($ruled)),
182            );
183        }
184
185        $definite = [];
186        $possible = [];
187        $carried  = [];
188        /** @var array<string, true> $reached the changed files some closure actually contains */
189        $reached = [];
190
191        foreach ($groups as $group) {
192            $memberships = $ruled === [] ? [] : $this->membershipsOf($group);
193            $carried += $memberships;
194
195            $hits = $targets === []
196                ? []
197                : array_intersect_key($this->graph->closureOf($this->fileOf($group)), $targets);
198
199            if ($hits !== []) {
200                $reached += $hits;
201                $definite[] = $group;
202
203                continue;
204            }
205
206            if (array_intersect_key($memberships, $ruled) !== []) {
207                $possible[] = $group;
208            }
209        }
210
211        // The changed PHP files no closure contained. Reported apart from
212        // the undeclared ones because the fix differs: a test, not a rule.
213        $uncovered = count(array_diff_key($targets, $reached));
214
215        if ($uncovered > 0) {
216            $notes[] = sprintf(
217                'Impact: %d change(s) %s.',
218                $uncovered,
219                ImpactReason::Uncovered->heading(),
220            );
221        }
222
223        // A rule naming a group no test carries selects nothing, silently
224        // — the one way a declared rule can be wrong. Name it.
225        $orphans = array_keys(array_diff_key($ruled, $carried));
226
227        if ($orphans !== []) {
228            $notes[] = sprintf(
229                'Impact: impact rule group(s) %s are carried by no test.',
230                implode(', ', $orphans),
231            );
232        }
233
234        $tiered = ImpactResult::tiered($definite, $possible, $notes);
235
236        if ($tiered->possibleCount() > 0) {
237            $notes[] = sprintf(
238                'Impact: %d group(s) selected by a declared rule rather than a resolved edge.',
239                $tiered->possibleCount(),
240            );
241        }
242
243        $notes[] = sprintf(
244            'Impact: %d of %d test file(s) affected.',
245            count($tiered->selected()),
246            count($groups),
247        );
248
249        return ImpactResult::tiered($definite, $possible, $notes);
250    }
251
252    /**
253     * The groups every test in this group carries, as a set.
254     *
255     * @return array<string, true>
256     */
257    private function membershipsOf(TestGroup $group): array
258    {
259        $memberships = [];
260
261        foreach ($group->tests as $test) {
262            foreach ($test->metadata->ofType(Group::class) as $membership) {
263                $memberships[$membership->name] = true;
264            }
265        }
266
267        return $memberships;
268    }
269
270    /** Whether another tier of the run answers for this file. */
271    private function isClaimed(string $file): bool
272    {
273        $file = WorkingDirectory::native($file);
274
275        foreach ($this->claimedPaths as $directory) {
276            if (str_starts_with($file, rtrim(WorkingDirectory::native($directory), DIRECTORY_SEPARATOR) . DIRECTORY_SEPARATOR)) {
277                return true;
278            }
279        }
280
281        return false;
282    }
283
284    /**
285     * @return string project-relative, or the path itself when outside the root
286     */
287    private function relative(string $file): string
288    {
289        $prefix   = rtrim(WorkingDirectory::portable($this->projectRoot), '/') . '/';
290        $portable = WorkingDirectory::portable($file);
291
292        return str_starts_with($portable, $prefix) ? substr($portable, strlen($prefix)) : $file;
293    }
294
295    /**
296     * The file a group was declared in — dialect-neutral via the
297     * TestIds (group names are class names in the phpunit dialect,
298     * paths elsewhere; ids always carry the project-relative file).
299     *
300     * @return non-empty-string absolute path
301     */
302    private function fileOf(TestGroup $group): string
303    {
304        $relative = $group->tests[0]->id->file;
305
306        return WorkingDirectory::native(rtrim($this->projectRoot, '/' . DIRECTORY_SEPARATOR) . '/' . $relative);
307    }
