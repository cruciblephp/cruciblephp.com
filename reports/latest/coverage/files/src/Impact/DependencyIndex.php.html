36/**
37 * Per-file invalidation for the dependency graph, keyed on content hash.
38 *
39 * Without it every run re-reads and re-scans every source file to find
40 * its references; the graph memoizes only within one process, so nothing
41 * survives to the next. This persists each file's resolved dependency
42 * list against a hash of its contents, and a file whose bytes have not
43 * changed is replayed instead of re-scanned.
44 *
45 * **The invariant:** the closure a run computes with this index is
46 * byte-for-byte the one it computes without it. Incrementality changes
47 * only how much work is skipped, never the answer. A selection feature
48 * that cannot state that property is not trustworthy — its failure mode
49 * is skipping a test that would have failed, which is invisible.
50 *
51 * **Everything stored is relative to the project root**, keys and
52 * dependency lists alike, and so is everything the fingerprint is taken
53 * over. An index is written on one machine and read on another — the
54 * CI cache being restored into a checkout at a different absolute path
55 * is the normal case, not the exotic one. Keyed on absolute paths the
56 * file would load, match nothing, and silently degrade every run to a
57 * cold one: no wrong answer, just the feature quietly not working
58 * exactly where it was built to work.
59 *
60 * That is also why the index is keyed `{dir}/{fingerprint}.deps.json`,
61 * one file per configuration. A dependency list is *not* a pure function
62 * of the file's bytes: it also depends on the autoload map that resolves
63 * a class name to a path, and on which `Pest.php` / `Datasets/*.php`
64 * sit above the file (D-033). Adding a `Pest.php` changes what a test
65 * file depends on without touching that file, so a cache keyed on
66 * content alone would answer confidently from stale data. The
67 * fingerprint carries that context, so a configuration change lands on
68 * a different index rather than silently reusing the old one.
69 */
70final class DependencyIndex
71{
72    private readonly string $directory;
73
74    /** @var array<string, array{hash: string, dependencies: list<string>}> */
75    private array $stored = [];
76
77    /** @var array<string, array{hash: string, dependencies: list<string>}> */
78    private array $fresh = [];
79
80    private readonly string $root;
81
82    /**
83     * @param non-empty-string $directory
84     * @param non-empty-string $fingerprint everything but file content that the answer depends on
85     * @param string           $projectRoot paths are stored relative to this, so the index survives being moved
86     */
87    public function __construct(string $directory, private readonly string $fingerprint, string $projectRoot = '')
88    {
89        $this->directory = rtrim($directory, '/');
90        $this->root      = $projectRoot === '' ? '' : rtrim($projectRoot, '/') . '/';
91    }
92
93    /**
94     * Project-relative when the path is inside the root, unchanged when
95     * it is not. A dependency in a sibling checkout has no relative form
96     * to give, and inventing one with `../..` would key the index on the
97     * layout around the project rather than the project.
98     */
99    private function relative(string $path): string
100    {
101        return $this->root !== '' && str_starts_with($path, $this->root)
102            ? substr($path, strlen($this->root))
103            : $path;
104    }
105
106    private function absolute(string $path): string
107    {
108        return $this->root !== '' && !str_starts_with($path, '/') ? $this->root . $path : $path;
109    }
110
111    public function load(): void
112    {
113        $this->stored = [];
114        $this->fresh  = [];
115
116        $path = $this->path();
117
118        if (!is_file($path)) {
119            return;
120        }
121
122        $contents = file_get_contents($path);
123
124        if ($contents === false) {
125            return;
126        }
127
128        $decoded = json_decode($contents, true);
129
130        if (!is_array($decoded)) {
131            return;
132        }
133
134        foreach ($decoded as $file => $entry) {
135            if (!is_string($file) || !is_array($entry) || !isset($entry['hash'], $entry['dependencies'])) {
136                continue;
137            }
138
139            $hash         = $entry['hash'];
140            $dependencies = $entry['dependencies'];
141
142            if (!is_string($hash) || !is_array($dependencies)) {
143                continue;
144            }
145
146            $paths = [];
147
148            foreach ($dependencies as $dependency) {
149                if (is_string($dependency)) {
150                    $paths[] = $this->absolute($dependency);
151                }
152            }
153
154            $this->stored[$this->absolute($file)] = ['hash' => $hash, 'dependencies' => $paths];
155        }
156    }
157
158    /**
159     * The stored dependencies for $file when its bytes still hash to what
160     * was recorded, otherwise null — added, changed, or never seen.
161     *
162     * @return ?list<string>
163     */
164    public function reuse(string $file): ?array
165    {
166        $entry = $this->stored[$file] ?? null;
167
168        if ($entry === null) {
169            return null;
170        }
171
172        $hash = $this->hash($file);
173
174        if ($hash === null || $hash !== $entry['hash']) {
175            return null;
176        }
177
178        $this->fresh[$file] = $entry;
179
180        return $entry['dependencies'];
181    }
182
183    /**
184     * @param list<string> $dependencies
185     */
186    public function record(string $file, array $dependencies): void
187    {
188        $hash = $this->hash($file);
189
190        if ($hash === null) {
191            return;
192        }
193
194        $this->fresh[$file] = ['hash' => $hash, 'dependencies' => $dependencies];
195    }
196
197    /**
198     * Persist what this run actually touched. Entries for files the run
199     * never asked about are dropped rather than carried forward, so a
200     * deleted file cannot linger in the index and answer for itself.
201     */
202    public function save(): void
203    {
204        if (!is_dir($this->directory) && !mkdir($this->directory, 0o777, true) && !is_dir($this->directory)) {
205            return;
206        }
207
208        $portable = [];
209
210        foreach ($this->fresh as $file => $entry) {
211            $paths = [$file, ...$entry['dependencies']];
212
213            // An entry naming anything outside the project root cannot be
214            // made portable, so it is not written at all. Storing it with
215            // the absolute path left in would replay THIS machine's path
216            // on a machine that restores the cache, and a changed file
217            // there would never match it — a test silently not selected,
218            // which is the one failure this index must never cause.
219            //
220            // It is not hypothetical: a composer path repository, which
221            // is how monorepos link local packages, resolves outside the
222            // root. Dropping the entry costs a rescan; keeping it costs
223            // correctness.
224            //
225            // With no root configured the index makes no portability
226            // claim, so there is nothing to protect and entries are kept.
227            foreach ($paths as $path) {
228                if ($this->root !== '' && !str_starts_with($path, $this->root)) {
229                    continue 2;
230                }
231            }
232
233            $portable[$this->relative($file)] = [
234                'hash'         => $entry['hash'],
235                'dependencies' => array_map($this->relative(...), $entry['dependencies']),
236            ];
237        }
238
239        file_put_contents(
240            $this->path(),
241            json_encode($portable, JSON_THROW_ON_ERROR),
242            LOCK_EX,
243        );
244    }
245
246    private function hash(string $file): ?string
247    {
248        if (!is_file($file)) {
249            return null;
250        }
251
252        $hash = hash_file('xxh128', $file);
253
254        return $hash === false ? null : $hash;
255    }
256
257    /** @return non-empty-string */
258    private function path(): string
259    {
260        return $this->directory . '/' . $this->fingerprint . '.deps.json';
261    }
262
263    /**
264     * The context a dependency answer depends on beyond file content: the
265     * autoload map that turns a class name into a path, and the suite
266     * configuration files whose mere presence adds edges (D-033).
267     *
268     * @param list<string> $suiteConfiguration absolute paths; recorded relative, so the fingerprint is the same on every machine
269     *
270     * @return non-empty-string
271     */
272    public static function fingerprintOf(string $projectRoot, array $suiteConfiguration): string
273    {
274        $parts = [];
275        $root  = rtrim($projectRoot, '/');
276
277        foreach (['composer.lock', 'composer.json'] as $manifest) {
278            $path = $root . '/' . $manifest;
279
280            if (is_file($path)) {
281                $hash = hash_file('xxh128', $path);
282
283                if ($hash !== false) {
284                    $parts[] = $manifest . ':' . $hash;
285                }
286            }
287        }
288
289        // The paths, not their contents: a Pest.php that CHANGES already
290        // invalidates through its own content hash, because it is itself
291        // a graph node. What content hashing cannot see is one appearing
292        // or disappearing above a file that did not change.
293        $prefix = $root . '/';
294
295        foreach ($suiteConfiguration as $path) {
296            $parts[] = 'suite:' . (str_starts_with($path, $prefix) ? substr($path, strlen($prefix)) : $path);
297        }
298
299        return hash('xxh128', implode("\n", $parts));
300    }
301
302    /**
303     * Every suite-configuration file under the project root, sorted — the
304     * set the fingerprint is taken over.
305     *
306     * @return list<string> absolute paths
307     */
308    public static function suiteConfigurationIn(string $projectRoot): array
309    {
310        $root  = rtrim($projectRoot, '/');
311        $found = [];
312
313        $patterns = [
314            '/Pest.php', '/*/Pest.php', '/*/*/Pest.php', '/*/*/*/Pest.php',
315            '/Datasets/*.php', '/*/Datasets/*.php', '/*/*/Datasets/*.php',
316        ];
317
318        foreach ($patterns as $pattern) {
319            $matches = glob($root . $pattern);
320
321            foreach ($matches === false ? [] : $matches as $path) {
322                $found[] = $path;
323            }
324        }
325
326        sort($found);
327
328        return $found;
329    }
