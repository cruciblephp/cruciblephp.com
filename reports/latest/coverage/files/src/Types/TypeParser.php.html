26/**
27 * Reads a PHPStan type string into a TypeExpression (D-131): a
28 * recursive-descent parser over the subset TypeExpression documents. A
29 * string it cannot read is refused with the position where it stops —
30 * a type assertion that silently meant something else would be worse
31 * than none.
32 */
33final class TypeParser
34{
35    /** @var list<array{string, int}> token text, offset */
36    private array $tokens = [];
37
38    private int $at = 0;
39
40    public function __construct(private readonly string $source)
41    {
42        preg_match_all(
43            '/\s*(\.\.\.|\'(?:[^\'\\\\]|\\\\.)*\'|"(?:[^"\\\\]|\\\\.)*"|-?\d+(?:\.\d+)?|[A-Za-z_\\\\][A-Za-z0-9_\\\\-]*|\[\]|[|&?<>{}(),:]|\S)/',
44            $source,
45            $matches,
46            PREG_SET_ORDER | PREG_OFFSET_CAPTURE,
47        );
48
49        foreach ($matches as $match) {
50            $this->tokens[] = [$match[1][0], $match[1][1]];
51        }
52    }
53
54    public function parse(): TypeExpression
55    {
56        $type = $this->union();
57
58        if ($this->peek() !== null) {
59            $this->refuse('unexpected ' . $this->peek());
60        }
61
62        return $type;
63    }
64
65    private function union(): TypeExpression
66    {
67        $members = [$this->intersection()];
68
69        while ($this->peek() === '|') {
70            $this->at++;
71            $members[] = $this->intersection();
72        }
73
74        return count($members) === 1 ? $members[0] : TypeExpression::union($members);
75    }
76
77    private function intersection(): TypeExpression
78    {
79        $members = [$this->postfix()];
80
81        while ($this->peek() === '&') {
82            $this->at++;
83            $members[] = $this->postfix();
84        }
85
86        return count($members) === 1 ? $members[0] : TypeExpression::intersection($members);
87    }
88
89    private function postfix(): TypeExpression
90    {
91        $type = $this->atom();
92
93        while ($this->peek() === '[]') {
94            $this->at++;
95            $type = TypeExpression::array('array', [$type]);
96        }
97
98        return $type;
99    }
100
101    private function atom(): TypeExpression
102    {
103        $token = $this->next();
104
105        if ($token === '?') {
106            return TypeExpression::union([$this->atom(), TypeExpression::keyword('null')]);
107        }
108
109        if ($token === '(') {
110            $type = $this->union();
111            $this->expect(')');
112
113            return $type;
114        }
115
116        if ($token[0] === "'" || $token[0] === '"') {
117            return TypeExpression::literal(stripcslashes(substr($token, 1, -1)));
118        }
119
120        $integer = TypeExpression::integer($token);
121
122        if ($integer !== null) {
123            return TypeExpression::literal($integer);
124        }
125
126        if (TypeExpression::looksLikeFloat($token)) {
127            return TypeExpression::literal((float) $token);
128        }
129
130        if (preg_match_all('/^[A-Za-z_\\\\]/', $token) !== 1) {
131            $this->refuse('unexpected ' . $token, -1);
132        }
133
134        $name = strtolower($token);
135
136        if ($this->peek() === '{' && in_array($name, ['array', 'list', 'non-empty-array', 'non-empty-list'], true)) {
137            return $this->shape($name);
138        }
139
140        if ($this->peek() === '<') {
141            return $this->generic($name, $token);
142        }
143
144        return TypeExpression::isKeyword($token)
145            ? TypeExpression::keyword($name)
146            : TypeExpression::class(TypeExpression::className($token));
147    }
148
149    private function generic(string $name, string $token): TypeExpression
150    {
151        $this->expect('<');
152
153        if ($name === 'int') {
154            $min = $this->bound('min');
155            $this->expect(',');
156            $max = $this->bound('max');
157            $this->expect('>');
158
159            return TypeExpression::range($min, $max);
160        }
161
162        $arguments = [$this->union()];
163
164        while ($this->peek() === ',') {
165            $this->at++;
166            $arguments[] = $this->union();
167        }
168
169        $this->expect('>');
170
171        if ($name === 'class-string') {
172            return TypeExpression::classString($arguments);
173        }
174
175        if (!in_array($name, ['array', 'list', 'non-empty-array', 'non-empty-list', 'iterable'], true) || count($arguments) > 2) {
176            $this->refuse(sprintf('%s<…> is not a type this reads', $token));
177        }
178
179        return TypeExpression::array($name, $arguments);
180    }
181
182    private function bound(string $open): ?int
183    {
184        $token = $this->next();
185
186        if (strtolower($token) === $open) {
187            return null;
188        }
189
190        $integer = TypeExpression::integer($token);
191
192        if ($integer === null) {
193            $this->refuse('expected an integer or ' . $open, -1);
194        }
195
196        return $integer;
197    }
198
199    private function shape(string $name): TypeExpression
200    {
201        $this->expect('{');
202
203        $keys     = [];
204        $open     = true;
205        $position = 0;
206
207        while ($this->peek() !== '}') {
208            if ($this->peek() === '...') {
209                $this->at++;
210                $open = true;
211
212                // `...<K, V>`: the unnamed keys' types; read and allowed.
213                if ($this->peek() === '<') {
214                    $this->generic('array', 'array');
215                }
216            } elseif ($this->isKeyed()) {
217                $key      = $this->next();
218                $key      = $key[0] === "'" || $key[0] === '"' ? stripcslashes(substr($key, 1, -1)) : (TypeExpression::integer($key) ?? $key);
219                $optional = $this->peek() === '?';
220
221                if ($optional) {
222                    $this->at++;
223                }
224
225                $this->expect(':');
226                $keys[$key] = [$this->union(), $optional];
227            } else {
228                $keys[$position++] = [$this->union(), false];
229            }
230
231            if ($this->peek() !== ',') {
232                break;
233            }
234
235            $this->at++;
236        }
237
238        $this->expect('}');
239
240        return TypeExpression::shape($name, $keys, $open);
241    }
242
243    /** Whether the next entry is `key:` or `key?:` rather than a bare type. */
244    private function isKeyed(): bool
245    {
246        $after = $this->tokens[$this->at + 1][0] ?? null;
247
248        return $after === ':' || ($after === '?' && ($this->tokens[$this->at + 2][0] ?? null) === ':');
249    }
250
251    private function peek(): ?string
252    {
253        return $this->tokens[$this->at][0] ?? null;
254    }
255
256    private function next(): string
257    {
258        $token = $this->peek();
259
260        if ($token === null) {
261            $this->refuse('the type ends too early', 0);
262        }
263
264        $this->at++;
265
266        return $token;
267    }
268
269    private function expect(string $token): void
270    {
271        if ($this->peek() !== $token) {
272            $this->refuse(sprintf('expected %s', $token));
273        }
274
275        $this->at++;
276    }
277
278    private function refuse(string $why, int $shift = 0): never
279    {
280        $offset = $this->tokens[$this->at + $shift][1] ?? null;
281
282        throw new ConfigurationException(sprintf(
283            'Cannot read the type "%s"%s: %s.',
284            $this->source,
285            $offset === null ? ' at its end' : sprintf(' at offset %d', $offset),
286            $why,
287        ));
288    }
289}
