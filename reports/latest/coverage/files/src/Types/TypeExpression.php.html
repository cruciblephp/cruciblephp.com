45/**
46 * A PHPStan type string, read at run time (D-131): the one type a
47 * `toMatchShape('array{id: positive-int, tags: list<string>}')` states for
48 * two readers. PHPStan narrows the value to it; this checks the value
49 * against it and names the first place it does not fit —
50 * `$.tags[2]: expected string, got int (7)`.
51 *
52 * Crucible's own reading of PHPStan's syntax, not a dependency: the
53 * assertion runs in any test run, and a run-time check cannot require a
54 * development tool to be installed. Its agreement with PHPStan's reading
55 * is held by differential tests against the real phpstan binary. Where
56 * PHPStan's reading was measured rather than assumed, it is followed:
57 * shapes are open (a key the shape does not name is allowed, as
58 * PHPStan 2 accepts it), `float` does not accept an int.
59 *
60 * The subset: unions and intersections, `?T`, `T[]`, parentheses;
61 * array shapes `array{key: T, key?: T, ...}`, positional
62 * `array{T, U}` and `list{T, U}`; `array<V>`, `array<K, V>`,
63 * `list<V>`, `non-empty-array<…>`, `non-empty-list<V>`, `iterable<…>`,
64 * `int<min, max>`, `class-string<T>`; the keywords below; literals
65 * (`'on'`, `42`, `1.5`); class and interface names.
66 */
67final readonly class TypeExpression
68{

72        'non-negative-int', 'non-positive-int', 'non-zero-int', 'non-empty-string', 'non-falsy-string', 'truthy-string',
73        'numeric-string', 'lowercase-string', 'non-empty-array', 'non-empty-list', 'array-key', 'class-string',
74        'resource', 'never', 'void', 'empty',

181    /** The first place $value does not fit, or null when it fits. */
182    public function mismatch(mixed $value, string $path = '$'): ?string
183    {
184        return match ($this->kind) {
185            'union'        => $this->unionMismatch($value, $path),
186            'intersection' => $this->intersectionMismatch($value, $path),
187            'literal'      => $value === $this->literal ? null : $this->expected($path, Exporter::export($this->literal), $value),
188            'keyword'      => $this->keywordMismatch($value, $path),
189            'class'        => $value instanceof $this->name ? null : $this->expected($path, $this->name, $value),
190            'class-string' => $this->classStringMismatch($value, $path),
191            'range'        => $this->rangeMismatch($value, $path),
192            'shape'        => $this->shapeMismatch($value, $path),
193            'array'        => $this->arrayMismatch($value, $path),
194            default        => null,
195        };
196    }

198    public function describe(): string
199    {
200        return match ($this->kind) {
201            'union'        => implode('|', $this->descriptions()),
202            'intersection' => implode('&', $this->descriptions()),
203            'literal'      => Exporter::export($this->literal),
204            'range'        => sprintf('int<%s, %s>', $this->min ?? 'min', $this->max ?? 'max'),
205            default        => $this->name !== '' ? $this->name : $this->kind,
206        };
207    }

226    private function unionMismatch(mixed $value, string $path): ?string
227    {
228        foreach ($this->children as $child) {
229            if ($child->mismatch($value, $path) === null) {
230                return null;
231            }
232        }
233
234        // TypeParser builds a union from two members or more, so no one
235        // member's reason is more specific than the union's own.
236        return $this->expected($path, $this->describe(), $value);
237    }

239    private function intersectionMismatch(mixed $value, string $path): ?string
240    {
241        foreach ($this->children as $child) {
242            $reason = $child->mismatch($value, $path);
243
244            if ($reason !== null) {
245                return $reason;
246            }
247        }
248
249        return null;
250    }

252    private function keywordMismatch(mixed $value, string $path): ?string
253    {
254        $fits = match ($this->name) {
255            'mixed'                             => true,
256            'null', 'void'                      => $value === null,
257            'bool', 'boolean'                   => is_bool($value),
258            'true'                              => $value === true,
259            'false'                             => $value === false,
260            'int', 'integer'                    => is_int($value),
261            'float', 'double'                   => is_float($value),
262            'string'                            => is_string($value),
263            'array'                             => is_array($value),
264            'list'                              => is_array($value) && array_is_list($value),
265            'iterable'                          => is_iterable($value),
266            'object'                            => is_object($value),
267            'callable'                          => is_callable($value),
268            'scalar'                            => is_scalar($value),
269            'numeric'                           => is_int($value) || is_float($value) || (is_string($value) && is_numeric($value)),
270            'positive-int'                      => is_int($value) && $value > 0,
271            'negative-int'                      => is_int($value) && $value < 0,
272            'non-negative-int'                  => is_int($value) && $value >= 0,
273            'non-positive-int'                  => is_int($value) && $value <= 0,
274            'non-zero-int'                      => is_int($value) && $value !== 0,
275            'non-empty-string'                  => is_string($value) && $value !== '',
276            'non-falsy-string', 'truthy-string' => is_string($value) && $value !== '' && $value !== '0',
277            'numeric-string'                    => is_string($value) && is_numeric($value),
278            'lowercase-string'                  => is_string($value) && strtolower($value) === $value,
279            'non-empty-array'                   => is_array($value) && $value !== [],
280            'non-empty-list'                    => is_array($value) && $value !== [] && array_is_list($value),
281            'array-key'                         => is_int($value) || is_string($value),
282            'class-string'                      => is_string($value) && (class_exists($value) || interface_exists($value) || enum_exists($value)),
283            'resource'                          => is_resource($value),
284            'empty'                             => empty($value),
285            default                             => false,
286        };
287
288        return $fits ? null : $this->expected($path, $this->name, $value);
289    }

291    private function classStringMismatch(mixed $value, string $path): ?string
292    {
293        $class = $this->children[0] ?? null;
294
295        return is_string($value) && $class?->kind === 'class' && is_a($value, $class->name, true)
296            ? null
297            : $this->expected($path, $this->describe(), $value);
298    }

300    private function rangeMismatch(mixed $value, string $path): ?string
301    {
302        return is_int($value) && ($this->min === null || $value >= $this->min) && ($this->max === null || $value <= $this->max)
303            ? null
304            : $this->expected($path, $this->describe(), $value);
305    }

307    private function shapeMismatch(mixed $value, string $path): ?string
308    {
309        if (!is_array($value)) {
310            return $this->expected($path, 'array', $value);
311        }
312
313        if ($this->name === 'list' && !array_is_list($value)) {
314            return $this->expected($path, 'a list', $value);
315        }
316
317        foreach ($this->keys as $key => [$type, $optional]) {
318            $at = $this->at($path, $key);
319
320            if (!array_key_exists($key, $value)) {
321                if ($optional) {
322                    continue;
323                }
324
325                return sprintf('%s: missing', $at);
326            }
327
328            $reason = $type->mismatch($value[$key], $at);
329
330            if ($reason !== null) {
331                return $reason;
332            }
333        }
334
335        if (!$this->open) {
336            foreach (array_keys($value) as $key) {
337                if (!array_key_exists($key, $this->keys)) {
338                    return sprintf('%s: a key the shape does not allow', $this->at($path, $key));
339                }
340            }
341        }
342
343        return null;
344    }

346    /**
347     * array<V>, array<K, V>, list<V>, non-empty-*, iterable<…>: every
348     * key and value checked, the first misfit named.
349     */
350    private function arrayMismatch(mixed $value, string $path): ?string
351    {
352        $shape = $this->name;
353
354        if ($shape === 'iterable' ? !is_iterable($value) : !is_array($value)) {
355            return $this->expected($path, $shape === 'iterable' ? 'iterable' : 'array', $value);
356        }
357
358        if (in_array($shape, ['list', 'non-empty-list'], true) && is_array($value) && !array_is_list($value)) {
359            return $this->expected($path, 'a list', $value);
360        }
361
362        if (str_contains($shape, 'non-empty') && $value === []) {
363            return $this->expected($path, 'a non-empty ' . (str_contains($shape, 'list') ? 'list' : 'array'), $value);
364        }
365
366        [$keyType, $valueType] = count($this->children) === 2 ? $this->children : [null, $this->children[0] ?? null];
367
368        foreach ($value as $key => $item) {
369            $at = $this->at($path, $key);
370
371            $reason = $keyType?->mismatch($key, $at . ' (key)') ?? $valueType?->mismatch($item, $at);
372
373            if ($reason !== null) {
374                return $reason;
375            }
376        }
377
378        return null;
379    }

381    /** Where a key sits under $path: `$.id`, `$[0]`, and an iterable's non-scalar key exported. */
382    private function at(string $path, mixed $key): string
383    {
384        return match (true) {
385            is_int($key)    => $path . '[' . $key . ']',
386            is_string($key) => $path . '.' . $key,
387            default         => $path . '[' . Exporter::export($key) . ']',
388        };
389    }
