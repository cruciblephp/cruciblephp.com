58/**
59 * Runs a type-test suite (D-130): one PHPStan analysis over its
60 * `*.types.php` files, every assertion a test with its own verdict.
61 *
62 *   - PHPStan's assertion functions — `assertType()`, `assertNativeType()`,
63 *     `assertSuperType()`, `assertVariableCertainty()` — pass when PHPStan
64 *     reports nothing on their line, fail on the identifier the function
65 *     reports a mismatch with (`phpstan.type`, …), and error on anything
66 *     else there: the line did not analyse.
67 *   - A line ending `// crucible-type-error <identifier>` passes when
68 *     PHPStan reports that identifier on it, and fails when the line
69 *     analyses clean or reports something else (both named).
70 *   - Errors on no assertion's line are the file's, one errored entry;
71 *     so is a file with no assertion at all, which tests nothing — the
72 *     rule PHPStan's own TypeInferenceTestCase keeps.
73 *
74 * Never a silent pass: PHPStan that cannot start, writes no report, or
75 * reports an error it places in no file, is one errored test with the
76 * reason.
77 */
78final readonly class TypeTestRunner
79{
80    private const string MARKER = 'crucible-type-error';

84        'assertType'              => 'phpstan.type',
85        'assertNativeType'        => 'phpstan.nativeType',
86        'assertSuperType'         => 'phpstan.superType',
87        'assertVariableCertainty' => 'phpstan.variable',
88    ];

101    /**
102     * @param list<TypeTestSuite> $suites
103     */
104    public function run(array $suites, WorkingDirectory $workingDirectory): RunSummary
105    {
106        $summary = new RunSummary();
107
108        foreach ($suites as $suite) {
109            $summary = $summary->plus($this->runSuite($suite, $workingDirectory));
110        }
111
112        return $summary;
113    }

115    private function runSuite(TypeTestSuite $suite, WorkingDirectory $workingDirectory): RunSummary
116    {
117        $files = $this->files($workingDirectory->absolute($suite->directory));
118
119        if ($files === []) {
120            return new RunSummary();
121        }
122
123        $started = hrtime(true);
124        $report  = $this->analyse($suite, $files, $workingDirectory);
125        $elapsed = (hrtime(true) - $started) / 1e9;
126
127        if ($report instanceof PhpstanReport && $report->general !== []) {
128            $report = 'PHPStan reported an error in no file: ' . $report->general[0];
129        }
130
131        if (!$report instanceof PhpstanReport) {
132            return $this->fold->couldNotRun(new TestId($suite->directory, 'types'), $report);
133        }
134
135        return $this->fold->emit($this->translate($suite, $files, $report, $elapsed, $workingDirectory));
136    }

138    /**
139     * The report as finished tests: one per assertion the filter selects,
140     * plus a file's own entry when something on it is no assertion's.
141     *
142     * @param list<non-empty-string> $files
143     *
144     * @return list<TestFinished>
145     */
146    private function translate(TypeTestSuite $suite, array $files, PhpstanReport $report, float $elapsed, WorkingDirectory $workingDirectory): array
147    {
148        $points = [];
149        $total  = 0;
150
151        foreach ($files as $file) {
152            $points[$file] = $this->points($file);
153            $total += count($points[$file]);
154        }
155
156        // One analysis answered every assertion: its time is shared, so the
157        // tree's totals add up to the time the run actually spent.
158        $each    = $elapsed / max(1, $total);
159        $filter  = $suite->filter === null ? null : new NameFilter($suite->filter);
160        $results = [];
161
162        foreach ($files as $file) {
163            $relative = $workingDirectory->relative($file);
164            $relative = $relative !== '' ? $relative : $file;
165
166            /** @var array<int, list<PhpstanMessage>> $errors line => what PHPStan reported there */
167            $errors = [];
168
169            foreach ($report->forFile($file) as $message) {
170                $errors[$message->line][] = $message;
171            }
172
173            foreach ($points[$file] as $point) {
174                if (!$filter instanceof NameFilter || $filter->matchesName($point['name'])) {
175                    [$outcome, $failure] = $this->verdict($point, $errors[$point['line']] ?? []);
176                    $results[]           = new TestFinished(new TestId($relative, $point['name']), $outcome, $each, $failure);
177                }
178
179                unset($errors[$point['line']]);
180            }
181
182            if ($filter instanceof NameFilter) {
183                continue;
184            }
185
186            $unclaimed = [];
187
188            foreach ($errors as $line => $messages) {
189                foreach ($messages as $message) {
190                    $unclaimed[] = sprintf('line %d: %s', $line, $message->message);
191                }
192            }
193
194            $why = match (true) {
195                $unclaimed !== []     => 'PHPStan reported errors outside any type assertion:' . "\n" . implode("\n", $unclaimed),
196                $points[$file] === [] => 'No type assertion in this file: it tests nothing. Call PHPStan\Testing\assertType(), or mark a line // ' . self::MARKER . ' <identifier>.',
197                default               => null,
198            };
199
200            if ($why !== null) {
201                $results[] = new TestFinished(new TestId($relative, 'the file analyses'), Outcome::Errored, 0.0, new Failure($why));
202            }
203        }
204
205        return $results;
206    }

208    /**
209     * @param array{line: int, name: non-empty-string, identifier: ?string, mismatch: ?string} $point
210     * @param list<PhpstanMessage>                                                               $errors on its line
211     *
212     * @return array{Outcome, ?Failure}
213     */
214    private function verdict(array $point, array $errors): array
215    {
216        if ($point['identifier'] !== null) {
217            if (array_any($errors, static fn(PhpstanMessage $error): bool => $error->identifier === $point['identifier'])) {
218                return [Outcome::Passed, null];
219            }
220
221            return [Outcome::Failed, new Failure($errors === []
222                ? sprintf('Expected a %s error on this line; it analyses clean.', $point['identifier'])
223                : sprintf('Expected a %s error on this line; PHPStan reported: %s', $point['identifier'], $this->describe($errors)))];
224        }
225
226        if ($errors === []) {
227            return [Outcome::Passed, null];
228        }
229
230        $mismatch = array_find($errors, static fn(PhpstanMessage $error): bool => $error->identifier === $point['mismatch']);
231
232        return $mismatch instanceof PhpstanMessage
233            ? [Outcome::Failed, new Failure($mismatch->message)]
234            : [Outcome::Errored, new Failure('The line does not analyse: ' . $this->describe($errors))];
235    }

237    /**
238     * @param list<PhpstanMessage> $errors
239     */
240    private function describe(array $errors): string
241    {
242        $parts = [];
243
244        foreach ($errors as $error) {
245            $parts[] = $error->message . ($error->identifier !== null ? ' [' . $error->identifier . ']' : '');
246        }
247
248        return implode('; ', $parts);
249    }

251    /**
252     * The *.types.php files under a directory, in path order.
253     *
254     * @return list<non-empty-string>
255     */
256    private function files(string $directory): array
257    {
258        if (!is_dir($directory)) {
259            return [];
260        }
261
262        $files = [];
263
264        foreach (new RecursiveIteratorIterator(new RecursiveDirectoryIterator($directory, FilesystemIterator::SKIP_DOTS)) as $file) {
265            $path = $file instanceof SplFileInfo && str_ends_with($file->getFilename(), '.types.php') ? $file->getRealPath() : false;
266
267            if ($path !== false && $path !== '') {
268                $files[] = $path;
269            }
270        }
271
272        sort($files);
273
274        return $files;
275    }

277    /**
278     * The assertions in a file, from its tokens: every call to one of
279     * PHPStan's assertion functions, and every line marked
280     * crucible-type-error.
281     *
282     * @return list<array{line: int, name: non-empty-string, identifier: ?string, mismatch: ?string}>
283     */
284    private function points(string $file): array
285    {
286        $tokens = PhpToken::tokenize((string) file_get_contents($file));
287        $points = [];
288
289        foreach ($tokens as $i => $token) {
290            if ($token->is(T_COMMENT) && preg_match('/' . self::MARKER . '\s+([A-Za-z0-9_.]+)/', $token->text, $match) === 1) {
291                $points[] = ['line' => $token->line, 'name' => sprintf('line %d: type error %s', $token->line, $match[1]), 'identifier' => $match[1], 'mismatch' => null];
292
293                continue;
294            }
295
296            if (!$token->is([T_STRING, T_NAME_QUALIFIED, T_NAME_FULLY_QUALIFIED])) {
297                continue;
298            }
299
300            $function = ltrim((string) strrchr('\\' . $token->text, '\\'), '\\');
301            $next     = $this->significant($tokens, $i + 1);
302
303            if (!array_key_exists($function, self::ASSERTIONS) || $next === null || $tokens[$next]->text !== '(') {
304                continue;
305            }
306
307            $argument = $this->significant($tokens, $next + 1);
308            $expected = $argument !== null && $tokens[$argument]->is(T_CONSTANT_ENCAPSED_STRING)
309                ? substr($tokens[$argument]->text, 1, -1)
310                : '…';
311            $name = match ($function) {
312                'assertNativeType'        => 'native ' . $expected,
313                'assertSuperType'         => 'subtype of ' . $expected,
314                'assertVariableCertainty' => 'variable certainty',
315                default                   => $expected,
316            };
317
318            $points[] = ['line' => $token->line, 'name' => sprintf('line %d: %s', $token->line, $name), 'identifier' => null, 'mismatch' => self::ASSERTIONS[$function]];
319        }
320
321        return $points;
322    }
323
324    /**
325     * @param array<PhpToken> $tokens as PhpToken::tokenize() returns them
326     */
327    private function significant(array $tokens, int $from): ?int
328    {
329        for ($i = $from, $count = count($tokens); $i < $count; $i++) {
330            if (!$tokens[$i]->isIgnorable()) {
331                return $i;
332            }
333        }
334
335        return null;
336    }

338    /**
339     * One analysis over every file.
340     *
341     * @param list<non-empty-string> $files
342     *
343     * @return PhpstanReport|non-empty-string
344     */
345    private function analyse(TypeTestSuite $suite, array $files, WorkingDirectory $workingDirectory): PhpstanReport|string
346    {
347        $arguments = ['--memory-limit=1G'];
348        $neon      = $this->configuration($suite, $workingDirectory);
349
350        if ($neon !== null) {
351            $arguments[] = '--configuration=' . $neon;
352        }
353
354        return Phpstan::analyse($workingDirectory->absolute($suite->phpstan ?? 'vendor/bin/phpstan'), [...$arguments, ...$files], $workingDirectory);
355    }

357    /**
358     * The configuration the analysis reads: the suite's, the project's
359     * phpstan.neon(.dist), or — with neither — a written one that loads
360     * Crucible's extension at level max (unless the extension installer
361     * already does, D-127).
362     *
363     * @return ?non-empty-string
364     */
365    private function configuration(TypeTestSuite $suite, WorkingDirectory $workingDirectory): ?string
366    {
367        $configuration = $suite->configuration ?? Phpstan::projectConfiguration($workingDirectory);
368
369        if ($configuration !== null) {
370            return $workingDirectory->absolute($configuration);
371        }
372
373        $home = ($this->cacheDirectory ?? sys_get_temp_dir() . '/crucible-types') . '/types';
374
375        if (!is_dir($home) && !@mkdir($home, 0o777, true) && !is_dir($home)) {
376            return null;
377        }
378
379        // A stable path and tmpDir: PHPStan's result cache is keyed on
380        // them, and a written file at a new path every run analysed
381        // everything every run.
382        file_put_contents($home . '/phpstan.neon', PhpstanNeon::analysis(Phpstan::extensionIncludes($workingDirectory), [
383            'level'  => 'max',
384            'tmpDir' => $home . '/phpstan',
385        ]));
386
387        return $home . '/phpstan.neon';
388    }
