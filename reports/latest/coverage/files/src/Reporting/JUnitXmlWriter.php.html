32/**
33 * The opt-in JUnit XML emitter — the one place XML survives in Crucible,
34 * because CI systems consume it (ROADMAP: "XML output survives only
35 * as an opt-in JUnit emitter for CI interop"). Plain string building;
36 * no DOM dependency for a write-only format.
37 *
38 * Granularity matches what the spec's own JUnit report shows (D-018):
39 * risky reports as a plain pass, incomplete as skipped. Full fidelity
40 * lives on the NDJSON stream; this file exists for CI dashboards.
41 */
42final class JUnitXmlWriter implements Listener
43{
44    /** @var list<TestFinished> */
45    private array $finished = [];
46
47    /** @var array<string, array{0: string, 1: array<string, int>}> */
48    private array $declarations = [];
49
50    /**
51     * @param resource $stream
52     */
53    public function __construct(private $stream) {}
54
55    public function handle(Envelope $envelope): void
56    {
57        $event = $envelope->event;
58
59        if ($event instanceof TestFinished) {
60            $this->finished[] = $event;
61
62            return;
63        }
64
65        if ($event instanceof RunFinished) {
66            $this->render();
67        }
68    }
69
70    private function render(): void
71    {
72        $model = new RunModel($this->finished);
73
74        $suites     = '';
75        $runCounts  = ['failures' => 0, 'errors' => 0, 'skipped' => 0];
76        $runTime    = 0.0;
77        $runTests   = 0;
78        $runAsserts = 0;
79
80        foreach ($model->sections as $file => $tests) {
81            $counts     = ['failures' => 0, 'errors' => 0, 'skipped' => 0];
82            $time       = 0.0;
83            $assertions = 0;
84            $body       = '';
85
86            foreach ($tests as $test) {
87                $time += $test->duration;
88                $assertions += $test->assertions;
89                $body .= $this->testcase($test, $file, $counts);
90            }
91
92            $suites .= sprintf(
93                "    <testsuite name=\"%s\" file=\"%s\" tests=\"%d\" assertions=\"%d\" failures=\"%d\" errors=\"%d\" skipped=\"%d\" time=\"%.6f\">\n%s    </testsuite>\n",
94                $this->escape($this->declaration($file, '')[0]),
95                $this->escape($file),
96                count($tests),
97                $assertions,
98                $counts['failures'],
99                $counts['errors'],
100                $counts['skipped'],
101                $time,
102                $body,
103            );
104
105            $runTests += count($tests);
106            $runAsserts += $assertions;
107            $runTime += $time;
108
109            foreach ($counts as $key => $value) {
110                $runCounts[$key] += $value;
111            }
112        }
113
114        // The run's own suite wraps the per-file ones: a reader that
115        // shows "the suite" wants one node to hang the totals off, and
116        // the per-file suites are what it drills into.
117        fwrite($this->stream, sprintf(
118            "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<testsuites>\n  <testsuite name=\"%s\" tests=\"%d\" assertions=\"%d\" failures=\"%d\" errors=\"%d\" skipped=\"%d\" time=\"%.6f\">\n%s  </testsuite>\n</testsuites>\n",
119            'Crucible',
120            $runTests,
121            $runAsserts,
122            $runCounts['failures'],
123            $runCounts['errors'],
124            $runCounts['skipped'],
125            $runTime,
126            $suites,
127        ));
128    }

130    /**
131     * @param array<string, int> $counts
132     */
133    private function testcase(TestFinished $test, string $file, array &$counts): string
134    {
135        $name = $test->test->name
136            . ($test->test->dataset !== null ? PrettyName::dataset($test->test->dataset) : '');
137
138        // The declaring class and its method's line come from the same
139        // source analysis the coverage reports parse, so a reader can
140        // jump to the test rather than only name it. A file that
141        // declares no class still reports, keyed by its path.
142        [$class, $line] = $this->declaration($file, $test->test->name);
143
144        $open = sprintf(
145            '      <testcase name="%s" file="%s" line="%d" class="%s" classname="%s" assertions="%d" time="%.6f"',
146            $this->escape($name),
147            $this->escape($file),
148            $line,
149            $this->escape($class),
150            $this->escape(str_replace('\\', '.', $class)),
151            $test->assertions,
152            $test->duration,
153        );
154
155        $message = $test->failure->message ?? $test->reason ?? '';
156
157        switch ($test->outcome) {
158            case Outcome::Failed:
159                $counts['failures']++;
160
161                return $open . sprintf(
162                    ">\n        <failure type=\"%s\">%s</failure>\n%s      </testcase>\n",
163                    $this->escape($test->failure->throwableClass ?? 'AssertionFailedError'),
164                    $this->escape($message),
165                    $this->reruns($test, 'rerunFailure'),
166                );
167
168            case Outcome::Errored:
169                $counts['errors']++;
170
171                return $open . sprintf(
172                    ">\n        <error type=\"%s\">%s</error>\n%s      </testcase>\n",
173                    $this->escape($test->failure->throwableClass ?? 'Error'),
174                    $this->escape($message),
175                    $this->reruns($test, 'rerunError'),
176                );
177
178            case Outcome::Skipped:
179            case Outcome::Incomplete:
180                $counts['skipped']++;
181
182                // The reason rides as text, not an attribute: the
183                // incumbent's <skipped/> carries no attributes at all,
184                // and text content keeps the reason without inventing
185                // a shape its readers do not expect.
186                return $open . sprintf(
187                    ">\n        <skipped>%s</skipped>\n      </testcase>\n",
188                    $this->escape($message),
189                );
190
191            case Outcome::Passed:
192            case Outcome::Risky:
193                // A pass that needed retries is FLAKY (G4): the earlier
194                // attempts' failures ride along in the Surefire flaky
195                // markup, so CI dashboards can track flakiness (D-043).
196                if ($test->retried !== []) {
197                    return $open . sprintf(
198                        ">\n%s    </testcase>\n",
199                        $this->reruns($test, 'flakyFailure'),
200                    );
201                }
202
203                return $open . "/>\n";
204        }
205    }

207    /**
208     * One Surefire rerun element per earlier failed attempt, in
209     * attempt order: flakyFailure under a pass, rerunFailure or
210     * rerunError under a final failure.
211     *
212     * @param non-empty-string $element
213     */
214    private function reruns(TestFinished $test, string $element): string
215    {
216        $markup = '';
217
218        foreach ($test->retried as $failure) {
219            $markup .= sprintf(
220                "      <%s type=\"%s\" message=\"%s\"/>\n",
221                $element,
222                $this->escape($failure->throwableClass ?? 'AssertionFailedError'),
223                $this->escape($failure->message),
224            );
225        }
226
227        return $markup;
228    }

230    /**
231     * tests/unit/Runner/SchedulerTest.php → tests.unit.Runner.SchedulerTest
232     */
233    /**
234     * The class a test file declares and the line its method starts on,
235     * from the analysis the coverage reports already build and cache.
236     * Falls back to the path-derived name when the file declares no
237     * class — a Pest file, or a test defined outside one.
238     *
239     * @return array{0: string, 1: int}
240     */
241    private function declaration(string $file, string $method): array
242    {
243        if (!isset($this->declarations[$file])) {
244            $classes = $file !== '' && is_file($file) ? SourceAnalysis::of($file)->classes : [];
245            $lines   = [];
246            $name    = '';
247
248            foreach ($classes as $qualified => $class) {
249                $name = $name === '' ? $qualified : $name;
250
251                foreach ($class->methods as $candidate) {
252                    $lines[$candidate->name] ??= $candidate->startLine;
253                }
254            }
255
256            $this->declarations[$file] = [$name === '' ? $this->classname($file) : $name, $lines];
257        }
258
259        [$class, $lines] = $this->declarations[$file];
260
261        return [$class, $lines[$method] ?? 0];
262    }
263
264    private function classname(string $file): string
265    {
266        $classname = str_replace('/', '.', $file);
267
268        return str_ends_with($classname, '.php') ? substr($classname, 0, -4) : $classname;
269    }
270
271    private function escape(string $value): string
272    {
273        return htmlspecialchars($value, ENT_XML1 | ENT_QUOTES, 'UTF-8');
274    }
