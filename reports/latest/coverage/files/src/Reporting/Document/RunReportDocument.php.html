36/**
37 * Builds the one comprehensive `Document` every registered report
38 * format renders: title, verdict badge, summary strip, a proportion
39 * bar, problems grouped by severity, the untested-by-reason section
40 * (D-075), the slowest tests, and the full passed/folded directory
41 * tree. Ported from `PdfWriter`'s pre-existing assembly (the richer of
42 * the two legacy writers) and now shared across every format instead
43 * of duplicated per writer — the report-format extension system needs
44 * exactly one canonical Document to hand any registered format, built
45 * once per run regardless of how many formats are selected.
46 */
47final class RunReportDocument
48{
49    /**
50     * @param ?non-empty-string $timingNotice why the durations below cannot be taken at face
51     *                                        value, passed in rather than read from the
52     *                                        environment: a persisted report that varies with
53     *                                        the ini of whoever rendered it is not a report
54     */
55    public static function build(RunModel $model, RunFinished $event, bool $flaky, string $title, ?string $timingNotice = null): Document
56    {
57        $blocks   = [new Heading(1, [new Text($title)])];
58        $blocks[] = Badge::forRun($event->summary, $flaky);
59        $blocks[] = new Paragraph([new Text(sprintf('Crucible %s by %s', Version::NUMBER, Version::AUTHOR))], quiet: true);
60        $blocks[] = new Paragraph([new Text(self::strip($event))]);
61        $blocks[] = self::proportionBar($event);
62
63        foreach (self::problemBlocks($model->problems) as $block) {
64            $blocks[] = $block;
65        }
66
67        foreach (self::untestedBlocks($model->untestedByReason) as $block) {
68            $blocks[] = $block;
69        }
70
71        $slowest = RankedList::build($model->sections, $event->duration);
72
73        if ($slowest->entries !== []) {
74            $blocks[] = new Heading(2, [new Text('Slowest tests')]);
75
76            if ($timingNotice !== null) {
77                $blocks[] = new Paragraph([new Text($timingNotice)]);
78            }
79
80            $blocks[] = $slowest;
81        }
82
83        $blocks[] = new Heading(2, [new Text('Results')]);
84        $blocks[] = FoldingTree::build($model->sections, $event->duration);
85
86        return new Document($blocks);
87    }
88
89    private static function strip(RunFinished $event): string
90    {
91        $summary = $event->summary;
92        $flagged = $summary->total() - $summary->passed - $summary->untested;
93
94        $strip = sprintf('%d tests · %d passed', $summary->total(), $summary->passed);
95
96        if ($flagged > 0) {
97            $strip .= sprintf(' · %d flagged', $flagged);
98        }
99
100        if ($summary->untested > 0) {
101            $strip .= sprintf(' · %d untested', $summary->untested);
102        }
103
104        return $strip . sprintf(' · %.3fs', $event->duration);
105    }
106
107    private static function proportionBar(RunFinished $event): ProportionBar
108    {
109        $summary = $event->summary;
110
111        return new ProportionBar([
112            new ProportionShare($summary->passed, Tone::Success),
113            new ProportionShare($summary->skipped, Tone::Muted),
114            new ProportionShare($summary->incomplete + $summary->risky, Tone::Caution),
115            new ProportionShare($summary->failed + $summary->errored, Tone::Danger),
116        ]);
117    }
118
119    /**
120     * @param  list<TestFinished>  $problems
121     * @return list<Block>
122     */
123    private static function problemBlocks(array $problems): array
124    {
125        if ($problems === []) {
126            return [];
127        }
128
129        $blocks = [new Heading(2, [new Text('Problems')])];
130
131        $order = [
132            [Outcome::Errored, 'Errors', Tone::Danger],
133            [Outcome::Failed, 'Failures', Tone::Danger],
134            [Outcome::Risky, 'Risky', Tone::Caution],
135            [Outcome::Incomplete, 'Incomplete', Tone::Caution],
136            [Outcome::Skipped, 'Skipped', Tone::Muted],
137        ];
138
139        foreach ($order as [$outcome, $groupTitle, $tone]) {
140            $entries = array_values(array_filter(
141                $problems,
142                static fn(TestFinished $problem): bool => $problem->outcome === $outcome,
143            ));
144
145            if ($entries === []) {
146                continue;
147            }
148
149            $blocks[] = new Heading(3, [new Text(sprintf('%s (%d)', $groupTitle, count($entries)))]);
150            $blocks[] = new ProblemList(array_map(
151                static fn(TestFinished $problem): ProblemEntry => new ProblemEntry(
152                    $problem->test->toString(),
153                    $problem->outcome->value,
154                    $problem->failure->message ?? $problem->reason,
155                    $problem->quarantined,
156                ),
157                $entries,
158            ), $tone);
159        }
160
161        return $blocks;
162    }
163
164    /**
165     * @param  array<string, list<TestFinished>>  $untestedByReason
166     * @return list<Block>
167     */
168    private static function untestedBlocks(array $untestedByReason): array
169    {
170        if ($untestedByReason === []) {
171            return [];
172        }
173
174        $blocks   = [];
175        $blocks[] = new Heading(2, [new Text('Untested')]);
176        $blocks[] = new Paragraph([new Text('Tests that could not run — a missing requirement or dependency.')]);
177
178        foreach ($untestedByReason as $reason => $events) {
179            $blocks[] = new Paragraph([new Strong($reason)]);
180            $blocks[] = new BulletList(array_map(
181                static fn(TestFinished $event): array => [new Text(PrettyName::ofTest($event->test))],
182                $events,
183            ));
184        }
185
186        return $blocks;
187    }
