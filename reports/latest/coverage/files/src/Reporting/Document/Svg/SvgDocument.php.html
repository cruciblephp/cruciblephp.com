44/**
45 * Walks an SVG document's DOM tree and emits the raw PDF content-stream
46 * operators (in the SVG's own unscaled coordinate space) that draw it —
47 * `PdfPrimitives::svg()` wraps the result in one outer `cm` mapping that
48 * space into the placed box. No `/Resources`/`XObject` involvement at
49 * all: unlike a JPEG, every shape draws as vector operators directly in
50 * the page's content stream, the same way `box()`/`rule()`/`disc()`
51 * already do.
52 *
53 * Explicitly unsupported (throws `InvalidArgumentException` naming the
54 * feature, never silently mis-renders): gradients/patterns, `<filter>`,
55 * `clip-path`/`mask`, `<text>`, embedded `<image>`, `<style>`/CSS-class
56 * presentation, and non-1 opacity/blend-modes.
57 *
58 * @phpstan-import-type Subpath from SvgPathData
59 * @phpstan-type PaintContext array{fill: ?array{float, float, float}, stroke: ?array{float, float, float}, strokeWidth: float, evenOdd: bool}
60 * @phpstan-type Rendered array{ops: string, minX: float, minY: float, width: float, height: float}
61 */
62final class SvgDocument

72    /** @return Rendered */
73    public function render(string $markup): array
74    {
75        $dom      = new DOMDocument();
76        $previous = libxml_use_internal_errors(true);
77
78        try {
79            if (!$dom->loadXML($markup)) {
80                throw new InvalidArgumentException('Could not parse SVG markup as XML.');
81            }
82        } finally {
83            libxml_use_internal_errors($previous);
84        }
85
86        $root = $dom->documentElement;
87        if (!$root instanceof DOMElement || strtolower($root->localName ?? '') !== 'svg') {
88            throw new InvalidArgumentException('SVG markup has no root <svg> element.');
89        }
90
91        if ($root->getElementsByTagName('style')->count() > 0) {
92            throw new InvalidArgumentException('CSS <style> elements are not supported in SVG images.');
93        }
94
95        [$minX, $minY, $width, $height] = $this->viewBox($root);
96
97        $context = ['fill' => [0.0, 0.0, 0.0], 'stroke' => null, 'strokeWidth' => 1.0, 'evenOdd' => false];
98        $ops     = $this->walkChildren($root, $context);
99
100        return ['ops' => $ops, 'minX' => $minX, 'minY' => $minY, 'width' => $width, 'height' => $height];
101    }

103    /** @return array{float, float, float, float} [minX, minY, width, height] */
104    private function viewBox(DOMElement $svg): array
105    {
106        $viewBox = trim($svg->getAttribute('viewBox'));
107
108        if ($viewBox !== '') {
109            $split = preg_split('/[\s,]+/', $viewBox);
110            $parts = $split !== false ? $split : [];
111            if (count($parts) === 4 && is_numeric($parts[0]) && is_numeric($parts[1]) && is_numeric($parts[2]) && is_numeric($parts[3])) {
112                return [(float) $parts[0], (float) $parts[1], (float) $parts[2], (float) $parts[3]];
113            }
114
115            throw new InvalidArgumentException("Malformed SVG viewBox attribute: \"{$viewBox}\".");
116        }
117
118        $width  = $this->numericAttribute($svg, 'width');
119        $height = $this->numericAttribute($svg, 'height');
120
121        if ($width === null || $height === null) {
122            throw new InvalidArgumentException('SVG has no viewBox and no numeric width/height to establish its coordinate space.');
123        }
124
125        return [0.0, 0.0, $width, $height];
126    }

128    private function numericAttribute(DOMElement $el, string $name): ?float
129    {

145    private function walkChildren(DOMElement $parent, array $context): string
146    {
147        $ops = '';
148
149        foreach ($parent->childNodes as $child) {
150            if (!$child instanceof DOMElement) {
151                continue;
152            }

160    /** @param PaintContext $context */
161    private function walkElement(DOMElement $el, array $context): string
162    {
163        $tag = strtolower($el->localName ?? '');
164
165        // Non-visual containers and definitions — never rendered, and
166        // never recursed into (a <defs> block full of unused shapes
167        // must not draw anything).
168        if (in_array($tag, ['defs', 'symbol', 'marker', 'clippath', 'mask', 'pattern', 'lineargradient', 'radialgradient', 'title', 'desc', 'metadata'], true)) {
169            return '';
170        }
171
172        if ($tag === 'text') {
173            throw new InvalidArgumentException('SVG <text> elements are not supported.');
174        }
175
176        if ($tag === 'image') {
177            throw new InvalidArgumentException('Embedded <image> elements inside SVG are not supported.');
178        }
179
180        $this->rejectUnsupportedAttributes($el);
181
182        $context  = $this->resolvePaint($el, $context);
183        $matrices = $this->parseTransformList($el->getAttribute('transform'));
184
185        $body = match ($tag) {
186            'g', 'svg', 'a' => $this->walkChildren($el, $context),
187            'rect'          => $this->drawRect($el, $context),
188            'circle'        => $this->drawEllipse($this->numericAttr($el, 'cx'), $this->numericAttr($el, 'cy'), $this->numericAttr($el, 'r'), $this->numericAttr($el, 'r'), $context),
189            'ellipse'       => $this->drawEllipse($this->numericAttr($el, 'cx'), $this->numericAttr($el, 'cy'), $this->numericAttr($el, 'rx'), $this->numericAttr($el, 'ry'), $context),
190            'line'          => $this->drawSubpaths([['start' => [$this->numericAttr($el, 'x1'), $this->numericAttr($el, 'y1')], 'segments' => [['l', $this->numericAttr($el, 'x2'), $this->numericAttr($el, 'y2')]], 'closed' => false]], $context),
191            'polyline'      => $this->drawSubpaths([$this->pointsToSubpath($el, false)], $context),
192            'polygon'       => $this->drawSubpaths([$this->pointsToSubpath($el, true)], $context),
193            'path'          => $this->drawSubpaths(SvgPathData::parse($el->getAttribute('d')), $context),
194            default         => '',
195        };
196
197        if ($matrices === []) {
198            return $body;
199        }
200
201        $open  = implode('', array_map(static fn(array $m): string => sprintf("q %s %s %s %s %s %s cm\n", ...array_map(self::number(...), $m)), $matrices));
202        $close = str_repeat("Q\n", count($matrices));
203
204        return $open . $body . $close;
205    }

207    /**
208     * Throws for the presentation-level features this "basic subset"
209     * doesn't attempt: filters, clip-path/mask, and non-1 opacity —
210     * checked once per element regardless of which shape it draws.
211     */

213    {
214        $style = $this->inlineStyle($el);
215
216        foreach (['filter' => 'SVG filters', 'clip-path' => 'SVG clip-paths', 'mask' => 'SVG masks'] as $attr => $label) {
217            $value = $style[$attr] ?? $el->getAttribute($attr);
218            if ($value !== '' && strtolower(trim($value)) !== 'none') {
219                throw new InvalidArgumentException("{$label} are not supported in SVG images.");
220            }

231            }
232        }

236            throw new InvalidArgumentException('Opacity/blend-modes are not supported in SVG images.');

261    /**
262     * A hand-rolled `key:value;key2:value2` splitter for the inline
263     * `style=""` attribute — not real CSS (no selectors, no cascade,
264     * no `<style>`/class support, which are explicitly out of scope),
265     * just the same shorthand real-world tools (Inkscape, some
266     * Illustrator exports) commonly use for the handful of presentation
267     * properties this renderer already understands via plain attributes.
268     *
269     * @return array<string, string>
270     */
271    private function inlineStyle(DOMElement $el): array
272    {
273        $raw = trim($el->getAttribute('style'));
274        if ($raw === '') {
275            return [];
276        }
277
278        $out = [];
279        foreach (explode(';', $raw) as $declaration) {
280            $parts = explode(':', $declaration, 2);
281            if (count($parts) !== 2) {
282                continue;
283            }
284
285            $out[strtolower(trim($parts[0]))] = trim($parts[1]);
286        }
287
288        return $out;
289    }

291    /** @return ?array{float, float, float} null means "none" */
292    private function parseColor(string $value): ?array
293    {
294        $value = trim($value);
295
296        if ($value === '' || strtolower($value) === 'none' || strtolower($value) === 'transparent') {
297            return null;
298        }
299
300        if (str_starts_with($value, 'url(')) {
301            throw new InvalidArgumentException('Gradient/pattern fills and strokes are not supported in SVG images.');
302        }
303
304        if (preg_match('/^#([0-9a-fA-F]{3})$/', $value, $m) === 1) {
305            $r = hexdec($m[1][0] . $m[1][0]);
306            $g = hexdec($m[1][1] . $m[1][1]);
307            $b = hexdec($m[1][2] . $m[1][2]);
308
309            return [$r / 255.0, $g / 255.0, $b / 255.0];
310        }
311
312        if (preg_match('/^#([0-9a-fA-F]{6})$/', $value, $m) === 1) {
313            $r = hexdec(substr($m[1], 0, 2));
314            $g = hexdec(substr($m[1], 2, 2));
315            $b = hexdec(substr($m[1], 4, 2));
316
317            return [$r / 255.0, $g / 255.0, $b / 255.0];
318        }
319
320        if (preg_match('/^rgb\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)\s*\)$/i', $value, $m) === 1) {
321            return [(float) $m[1] / 255.0, (float) $m[2] / 255.0, (float) $m[3] / 255.0];
322        }
323
324        $named = self::NAMED_COLORS[strtolower($value)] ?? null;
325        if ($named !== null) {
326            return [$named[0] / 255.0, $named[1] / 255.0, $named[2] / 255.0];
327        }
328
329        throw new InvalidArgumentException("Unrecognized SVG color value: \"{$value}\".");
330    }

332    /**
333     * Splits a `transform="fn(...) fn(...)"` attribute into its
334     * individual function matrices, left-to-right in source order —
335     * per spec, a multi-function transform attribute is exactly
336     * equivalent to nesting one `<g transform="fn(...)">` per function,
337     * so callers nest one `q ... cm`/`Q` pair per matrix returned here
338     * rather than pre-multiplying them by hand.
339     *
340     * @return list<array{float, float, float, float, float, float}>
341     */
342    private function parseTransformList(string $value): array
343    {
344        $value = trim($value);
345        if ($value === '') {
346            return [];
347        }
348
349        preg_match_all('/([a-zA-Z]+)\s*\(([^)]*)\)/', $value, $matches, PREG_SET_ORDER);
350
351        $result = [];
352        foreach ($matches as $match) {
353            $name     = strtolower($match[1]);
354            $split    = preg_split('/[\s,]+/', trim($match[2]));
355            $rawArgs  = $split !== false ? $split : [];
356            $nonEmpty = array_values(array_filter($rawArgs, static fn(string $s): bool => $s !== ''));
357            $args     = array_map(static fn(string $s): float => (float) trim($s), $nonEmpty);
358
359            $result[] = $this->transformMatrix($name, $args);
360        }
361
362        return $result;
363    }

365    /**
366     * @param  list<float>  $args
367     * @return array{float, float, float, float, float, float}
368     */
369    private function transformMatrix(string $name, array $args): array
370    {
371        return match ($name) {
372            'translate' => [1.0, 0.0, 0.0, 1.0, $args[0] ?? 0.0, $args[1] ?? 0.0],
373            'scale'     => [$args[0] ?? 1.0, 0.0, 0.0, $args[1] ?? ($args[0] ?? 1.0), 0.0, 0.0],
374            'matrix'    => [$args[0] ?? 1.0, $args[1] ?? 0.0, $args[2] ?? 0.0, $args[3] ?? 1.0, $args[4] ?? 0.0, $args[5] ?? 0.0],
375            'rotate'    => $this->rotateMatrix($args),
376            'skewx'     => [1.0, 0.0, tan(($args[0] ?? 0.0) * M_PI / 180.0), 1.0, 0.0, 0.0],
377            'skewy'     => [1.0, tan(($args[0] ?? 0.0) * M_PI / 180.0), 0.0, 1.0, 0.0, 0.0],
378            default     => throw new InvalidArgumentException("Unsupported SVG transform function: \"{$name}\"."),
379        };
380    }

382    /**
383     * @param  list<float>  $args
384     * @return array{float, float, float, float, float, float}
385     */
386    private function rotateMatrix(array $args): array
387    {
388        $angle = ($args[0] ?? 0.0) * M_PI / 180.0;
389        $cos   = cos($angle);
390        $sin   = sin($angle);
391
392        if (!isset($args[1], $args[2])) {
393            return [$cos, $sin, -$sin, $cos, 0.0, 0.0];
394        }
395
396        // rotate(a, cx, cy) === translate(cx,cy) rotate(a) translate(-cx,-cy),
397        // combined here into one matrix since it's the same 3-function
398        // composition either way (the caller still gets one nested q/cm).
399        $cx = $args[1];
400        $cy = $args[2];
401
402        return [$cos, $sin, -$sin, $cos, $cx - $cos * $cx + $sin * $cy, $cy - $sin * $cx - $cos * $cy];
403    }

405    /** @param PaintContext $context */
406    private function drawRect(DOMElement $el, array $context): string
407    {
408        $x  = $this->numericAttr($el, 'x');
409        $y  = $this->numericAttr($el, 'y');
410        $w  = $this->numericAttr($el, 'width');
411        $h  = $this->numericAttr($el, 'height');
412        $rx = $this->numericAttr($el, 'rx');
413        $ry = $el->hasAttribute('ry') ? $this->numericAttr($el, 'ry') : $rx;
414        $rx = $el->hasAttribute('rx') ? $rx : $ry;
415
416        if ($rx <= 0.0 || $ry <= 0.0) {
417            $paint = $this->paintOperator($context);
418            if ($paint === null) {
419                return '';
420            }
421
422            $strokeWidthOp = $context['strokeWidth'] !== 1.0 ? sprintf("%s w\n", self::number($context['strokeWidth'])) : '';
423
424            return sprintf("%s%s %s %s %s re %s\n", $strokeWidthOp, self::number($x), self::number($y), self::number($w), self::number($h), $paint);
425        }
426
427        // Rounded corners: 4 short cubic Bézier arcs (the same
428        // kappa-derived corner every rounded-rect renderer uses),
429        // built as one closed subpath.
430        $k       = 0.5522847498 ;
431        $subpath = [
432            'start'    => [$x + $rx, $y],
433            'segments' => [
434                ['l', $x + $w - $rx, $y],
435                ['c', $x + $w - $rx + $k * $rx, $y, $x + $w, $y + $ry - $k * $ry, $x + $w, $y + $ry],
436                ['l', $x + $w, $y + $h - $ry],
437                ['c', $x + $w, $y + $h - $ry + $k * $ry, $x + $w - $rx + $k * $rx, $y + $h, $x + $w - $rx, $y + $h],
438                ['l', $x + $rx, $y + $h],
439                ['c', $x + $rx - $k * $rx, $y + $h, $x, $y + $h - $ry + $k * $ry, $x, $y + $h - $ry],
440                ['l', $x, $y + $ry],
441                ['c', $x, $y + $ry - $k * $ry, $x + $rx - $k * $rx, $y, $x + $rx, $y],
442            ],
443            'closed' => true,
444        ];
445
446        return $this->drawSubpaths([$subpath], $context);
447    }

468    /** @return array{start: array{float, float}, segments: list<array{'l', float, float}>, closed: bool} */
469    private function pointsToSubpath(DOMElement $el, bool $closed): array
470    {
471        $raw = trim($el->getAttribute('points'));
472        preg_match_all('/-?[\d.]+(?:[eE][-+]?\d+)?/', $raw, $matches);
473        $numbers = array_map(static fn(string $s): float => (float) $s, $matches[0]);
474
475        if (count($numbers) < 2) {
476            throw new InvalidArgumentException('SVG polyline/polygon requires at least one point.');
477        }
478
479        $segments = [];
480        $counter  = count($numbers);
481        for ($i = 2; $i + 1 < $counter; $i += 2) {
482            $segments[] = ['l', $numbers[$i], $numbers[$i + 1]];
483        }
484
485        return ['start' => [$numbers[0], $numbers[1]], 'segments' => $segments, 'closed' => $closed];
486    }

488    /**
489     * @param  list<Subpath>  $subpaths
490     * @param  PaintContext  $context
491     */
492    private function drawSubpaths(array $subpaths, array $context): string
493    {
494        $paint = $this->paintOperator($context);
495        if ($paint === null) {
496            return '';
497        }
498
499        $ops = '';
500
501        if ($context['strokeWidth'] !== 1.0) {
502            $ops .= sprintf("%s w\n", self::number($context['strokeWidth']));
503        }
504
505        foreach ($subpaths as $subpath) {
506            $ops .= sprintf("%s %s m\n", self::number($subpath['start'][0]), self::number($subpath['start'][1]));
507
508            foreach ($subpath['segments'] as $segment) {
509                $ops .= $segment[0] === 'l'
510                    ? sprintf("%s %s l\n", self::number($segment[1]), self::number($segment[2]))
511                    : sprintf("%s %s %s %s %s %s c\n", self::number($segment[1]), self::number($segment[2]), self::number($segment[3]), self::number($segment[4]), self::number($segment[5]), self::number($segment[6]));
512            }
513
514            if ($subpath['closed']) {
515                $ops .= "h\n";
516            }
517        }
518
519        return $ops . $paint . "\n";
520    }

522    /** @param PaintContext $context */
523    private function paintOperator(array $context): ?string
524    {
525        $fill   = $context['fill'];
526        $stroke = $context['stroke'];
527
528        if ($fill === null && $stroke === null) {
529            return null;
530        }
531
532        $colorOps = '';
533        if ($fill !== null) {
534            $colorOps .= sprintf('%s %s %s rg ', self::number($fill[0]), self::number($fill[1]), self::number($fill[2]));
535        }
536        if ($stroke !== null) {
537            $colorOps .= sprintf('%s %s %s RG ', self::number($stroke[0]), self::number($stroke[1]), self::number($stroke[2]));
538        }
539
540        $op = match (true) {
541            $fill !== null && $stroke !== null => $context['evenOdd'] ? 'B*' : 'B',
542            $fill !== null                     => $context['evenOdd'] ? 'f*' : 'f',
543            default                            => 'S',
544        };
545
546        return rtrim($colorOps) . "\n" . $op;
547    }

553        return is_numeric($value) ? (float) $value : 0.0;
554    }
