33/**
34 * Parses an SVG `<path d="...">` attribute into a normalized list of
35 * subpaths, each a moveto followed by line/curve segments in absolute
36 * coordinates — quadratics upgraded to cubics (PDF's `c` operator is the
37 * only curve primitive it has) and arcs converted via the standard
38 * endpoint-to-center parameterization (SVG spec Appendix F.6), the same
39 * technique every SVG-to-PDF/Canvas converter uses since PDF has no arc
40 * operator of its own.
41 *
42 * @phpstan-type Segment array{0: 'l', 1: float, 2: float}|array{0: 'c', 1: float, 2: float, 3: float, 4: float, 5: float, 6: float}
43 * @phpstan-type Subpath array{start: array{float, float}, segments: list<Segment>, closed: bool}
44 */
45final class SvgPathData
46{
47    /**
48     * @return list<Subpath>
49     */
50    public static function parse(string $d): array
51    {
52        $tokens = self::tokenize($d);
53        $pos    = 0;
54        $count  = count($tokens);
55
56        /** @var list<Subpath> $subpaths */
57        $subpaths = [];
58
59        $currentX = 0.0;
60        $currentY = 0.0;
61        $startX   = 0.0;
62        $startY   = 0.0;
63
64        /** @var list<Segment> $segments */
65        $segments = [];
66        $hasOpen  = false;
67
68        // The absolute position of the previous cubic/quadratic control
69        // point, used to reflect S/s and T/t's implicit control point —
70        // null whenever the previous command wasn't the matching curve
71        // type, per spec ("equivalent to specifying the current point
72        // as the reflection").
73        $prevCubicControl = null;
74        $prevQuadControl  = null;
75
76        while ($pos < $count) {
77            $command = $tokens[$pos];
78            if (!self::isCommandLetter($command)) {
79                throw new InvalidArgumentException("Expected an SVG path command letter, found \"{$command}\".");
80            }
81
82            $pos++;
83            $isRelative = $command === strtolower($command);
84            $upper      = strtoupper($command);
85
86            if ($upper !== 'S' && $upper !== 'T') {
87                $prevCubicControl = $upper === 'C' ? $prevCubicControl : null;
88                $prevQuadControl  = $upper === 'Q' ? $prevQuadControl : null;
89            }
90
91            switch ($upper) {
92                case 'M':
93                    [$x, $y, $pos] = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
94
95                    if ($hasOpen) {
96                        $subpaths[] = ['start' => [$startX, $startY], 'segments' => $segments, 'closed' => false];
97                    }
98
99                    $currentX = $startX = $x;
100                    $currentY = $startY = $y;
101                    $segments = [];
102                    $hasOpen  = true;
103
104                    // Subsequent coordinate pairs after the initial
105                    // moveto are implicit linetos — 0 more is valid here.
106                    while ($pos < $count && !self::isCommandLetter($tokens[$pos])) {
107                        [$x, $y, $pos] = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
108                        $segments[]    = ['l', $x, $y];
109                        $currentX      = $x;
110                        $currentY      = $y;
111                    }
112
113                    break;
114
115                case 'L':
116                    self::requireArgument($tokens, $pos, $count, $command);
117
118                    while ($pos < $count && !self::isCommandLetter($tokens[$pos])) {
119                        [$x, $y, $pos] = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
120                        $segments[]    = ['l', $x, $y];
121                        $currentX      = $x;
122                        $currentY      = $y;
123                    }
124
125                    break;
126
127                case 'H':
128                    self::requireArgument($tokens, $pos, $count, $command);
129
130                    while ($pos < $count && !self::isCommandLetter($tokens[$pos])) {
131                        $value = self::readNumber($tokens, $pos);
132                        $pos++;
133                        $currentX   = $isRelative ? $currentX + $value : $value;
134                        $segments[] = ['l', $currentX, $currentY];
135                    }
136
137                    break;
138
139                case 'V':
140                    self::requireArgument($tokens, $pos, $count, $command);
141
142                    while ($pos < $count && !self::isCommandLetter($tokens[$pos])) {
143                        $value = self::readNumber($tokens, $pos);
144                        $pos++;
145                        $currentY   = $isRelative ? $currentY + $value : $value;
146                        $segments[] = ['l', $currentX, $currentY];
147                    }
148
149                    break;
150
151                case 'C':
152                    self::requireArgument($tokens, $pos, $count, $command);
153
154                    while ($pos < $count && !self::isCommandLetter($tokens[$pos])) {
155                        [$x1, $y1, $pos] = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
156                        [$x2, $y2, $pos] = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
157                        [$x, $y, $pos]   = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
158
159                        $segments[]       = ['c', $x1, $y1, $x2, $y2, $x, $y];
160                        $prevCubicControl = [$x2, $y2];
161                        $currentX         = $x;
162                        $currentY         = $y;
163                    }
164
165                    break;
166
167                case 'S':
168                    self::requireArgument($tokens, $pos, $count, $command);
169
170                    while ($pos < $count && !self::isCommandLetter($tokens[$pos])) {
171                        [$x2, $y2, $pos] = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
172                        [$x, $y, $pos]   = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
173
174                        $x1 = $prevCubicControl !== null ? 2 * $currentX - $prevCubicControl[0] : $currentX;
175                        $y1 = $prevCubicControl !== null ? 2 * $currentY - $prevCubicControl[1] : $currentY;
176
177                        $segments[]       = ['c', $x1, $y1, $x2, $y2, $x, $y];
178                        $prevCubicControl = [$x2, $y2];
179                        $currentX         = $x;
180                        $currentY         = $y;
181                    }
182
183                    break;
184
185                case 'Q':
186                    self::requireArgument($tokens, $pos, $count, $command);
187
188                    while ($pos < $count && !self::isCommandLetter($tokens[$pos])) {
189                        [$qx, $qy, $pos] = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
190                        [$x, $y, $pos]   = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
191
192                        $segments[]      = self::quadToCubic($currentX, $currentY, $qx, $qy, $x, $y);
193                        $prevQuadControl = [$qx, $qy];
194                        $currentX        = $x;
195                        $currentY        = $y;
196                    }
197
198                    break;
199
200                case 'T':
201                    self::requireArgument($tokens, $pos, $count, $command);
202
203                    while ($pos < $count && !self::isCommandLetter($tokens[$pos])) {
204                        [$x, $y, $pos] = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
205
206                        $qx = $prevQuadControl !== null ? 2 * $currentX - $prevQuadControl[0] : $currentX;
207                        $qy = $prevQuadControl !== null ? 2 * $currentY - $prevQuadControl[1] : $currentY;
208
209                        $segments[]      = self::quadToCubic($currentX, $currentY, $qx, $qy, $x, $y);
210                        $prevQuadControl = [$qx, $qy];
211                        $currentX        = $x;
212                        $currentY        = $y;
213                    }
214
215                    break;
216
217                case 'A':
218                    self::requireArgument($tokens, $pos, $count, $command);
219
220                    while ($pos < $count && !self::isCommandLetter($tokens[$pos])) {
221                        $rx            = self::readNumber($tokens, $pos++);
222                        $ry            = self::readNumber($tokens, $pos++);
223                        $xRotation     = self::readNumber($tokens, $pos++);
224                        $largeArc      = self::readNumber($tokens, $pos++) !== 0.0;
225                        $sweep         = self::readNumber($tokens, $pos++) !== 0.0;
226                        [$x, $y, $pos] = self::readPoint($tokens, $pos, $isRelative, $currentX, $currentY);
227
228                        if ($currentX === $x && $currentY === $y) {
229                            // Per spec: identical endpoints omit the arc
230                            // entirely, not even as a zero-length line.
231                        } elseif ($rx === 0.0 || $ry === 0.0) {
232                            $segments[] = ['l', $x, $y];
233                        } else {
234                            foreach (self::arcToBeziers($currentX, $currentY, $rx, $ry, $xRotation, $largeArc, $sweep, $x, $y) as $bezier) {
235                                $segments[] = ['c', ...$bezier];
236                            }
237                        }
238
239                        $currentX = $x;
240                        $currentY = $y;
241                    }
242
243                    break;
244
245                case 'Z':
246                    $subpaths[] = ['start' => [$startX, $startY], 'segments' => $segments, 'closed' => true];
247                    $currentX   = $startX;
248                    $currentY   = $startY;
249                    $segments   = [];
250                    $hasOpen    = false;
251
252                    break;
253            }
254        }
255
256        if ($hasOpen) {
257            $subpaths[] = ['start' => [$startX, $startY], 'segments' => $segments, 'closed' => false];
258        }
259
260        return $subpaths;
261    }

263    /**
264     * Converts an SVG elliptical-arc endpoint segment into a list of
265     * absolute cubic Bézier curves — endpoint-to-center parameterization
266     * (SVG spec F.6.5/F.6.6) followed by splitting the resulting arc
267     * into ≤90°-sweep segments, each approximated with the standard
268     * kappa-derived control-point formula.
269     *
270     * @return list<array{float, float, float, float, float, float}> [cp1x, cp1y, cp2x, cp2y, x, y] segments
271     */
272    public static function arcToBeziers(
273        float $x1,
274        float $y1,
275        float $rx,
276        float $ry,
277        float $xAxisRotationDeg,
278        bool $largeArc,
279        bool $sweep,
280        float $x2,
281        float $y2,
282    ): array {
283        $rx  = abs($rx);
284        $ry  = abs($ry);
285        $phi = $xAxisRotationDeg * M_PI / 180.0;
286
287        $cosPhi = cos($phi);
288        $sinPhi = sin($phi);
289
290        // F.6.5 step 1: move to the rotated coordinate system centered
291        // between the two endpoints.
292        $dx2 = ($x1 - $x2) / 2.0;
293        $dy2 = ($y1 - $y2) / 2.0;
294        $x1p = $cosPhi * $dx2 + $sinPhi * $dy2;
295        $y1p = -$sinPhi * $dx2 + $cosPhi * $dy2;
296
297        // F.6.6: scale up rx/ry if the endpoints are further apart than
298        // the given radii could ever reach.
299        $lambda = ($x1p * $x1p) / ($rx * $rx) + ($y1p * $y1p) / ($ry * $ry);
300        if ($lambda > 1.0) {
301            $scale = sqrt($lambda);
302            $rx *= $scale;
303            $ry *= $scale;
304        }
305
306        // F.6.5 step 2: the center in the rotated coordinate system.
307        $rxSq  = $rx * $rx;
308        $rySq  = $ry * $ry;
309        $x1pSq = $x1p * $x1p;
310        $y1pSq = $y1p * $y1p;
311
312        $num = max(0.0, $rxSq * $rySq - $rxSq * $y1pSq - $rySq * $x1pSq);
313        $den = $rxSq * $y1pSq + $rySq * $x1pSq;
314        $co  = $den > 0.0 ? sqrt($num / $den) : 0.0;
315        $co  = $largeArc === $sweep ? -$co : $co;
316
317        $cxp = $co * ($rx * $y1p / $ry);
318        $cyp = $co * -($ry * $x1p / $rx);
319
320        // F.6.5 step 3: the center in the original coordinate system.
321        $cx = $cosPhi * $cxp - $sinPhi * $cyp + ($x1 + $x2) / 2.0;
322        $cy = $sinPhi * $cxp + $cosPhi * $cyp + ($y1 + $y2) / 2.0;
323
324        // F.6.5 step 4: start/sweep angles.
325        $theta1 = self::angleBetween(1.0, 0.0, ($x1p - $cxp) / $rx, ($y1p - $cyp) / $ry);
326        $delta  = self::angleBetween(
327            ($x1p - $cxp) / $rx,
328            ($y1p - $cyp) / $ry,
329            (-$x1p - $cxp) / $rx,
330            (-$y1p - $cyp) / $ry,
331        );
332
333        if (!$sweep && $delta > 0.0) {
334            $delta -= 2.0 * M_PI;
335        } elseif ($sweep && $delta < 0.0) {
336            $delta += 2.0 * M_PI;
337        }
338
339        $segmentCount = (int) ceil(abs($delta) / (M_PI / 2.0));
340        $segmentCount = max($segmentCount, 1);
341        $segmentDelta = $delta / $segmentCount;
342
343        $beziers = [];
344
345        for ($i = 0; $i < $segmentCount; $i++) {
346            $start     = $theta1 + $i * $segmentDelta;
347            $end       = $start + $segmentDelta;
348            $beziers[] = self::unitArcToBezier($start, $end, $rx, $ry, $cx, $cy, $cosPhi, $sinPhi);
349        }
350
351        return $beziers;
352    }

354    /**
355     * One ≤90° arc segment (in the ellipse's own parameter space, from
356     * angle $start to $end) as a single cubic Bézier, mapped through the
357     * ellipse's radii/rotation/center back into path coordinates. The
358     * `4/3 * tan(delta/4)` control-point-distance formula is the
359     * standard circular/elliptical-arc Bézier approximation.
360     *
361     * @return array{float, float, float, float, float, float}
362     */
363    private static function unitArcToBezier(float $start, float $end, float $rx, float $ry, float $cx, float $cy, float $cosPhi, float $sinPhi): array
364    {
365        $alpha = 4.0 / 3.0 * tan(($end - $start) / 4.0);
366
367        $cosStart = cos($start);
368        $sinStart = sin($start);
369        $cosEnd   = cos($end);
370        $sinEnd   = sin($end);
371        $p2x      = $cosStart - $alpha * $sinStart;
372        $p2y      = $sinStart + $alpha * $cosStart;
373        $p3x      = $cosEnd + $alpha * $sinEnd;
374        $p3y      = $sinEnd - $alpha * $cosEnd;
375        $p4x      = $cosEnd;
376        $p4y      = $sinEnd;
377
378        return [
379            ...self::ellipsePoint($p2x, $p2y, $rx, $ry, $cx, $cy, $cosPhi, $sinPhi),
380            ...self::ellipsePoint($p3x, $p3y, $rx, $ry, $cx, $cy, $cosPhi, $sinPhi),
381            ...self::ellipsePoint($p4x, $p4y, $rx, $ry, $cx, $cy, $cosPhi, $sinPhi),
382        ];
383    }

385    /** @return array{float, float} */
386    private static function ellipsePoint(float $unitX, float $unitY, float $rx, float $ry, float $cx, float $cy, float $cosPhi, float $sinPhi): array
387    {
388        $x = $rx * $unitX;
389        $y = $ry * $unitY;
390
391        return [
392            $cosPhi * $x - $sinPhi * $y + $cx,
393            $sinPhi * $x + $cosPhi * $y + $cy,
394        ];
395    }

397    /** The signed angle (radians) from vector (ux,uy) to vector (vx,vy). */
398    private static function angleBetween(float $ux, float $uy, float $vx, float $vy): float
399    {
400        $sign = $ux * $vy - $uy * $vx < 0.0 ? -1.0 : 1.0;
401        $dot  = $ux * $vx + $uy * $vy;
402        $lenU = sqrt($ux * $ux + $uy * $uy);
403        $lenV = sqrt($vx * $vx + $vy * $vy);
404        $cos  = $lenU * $lenV > 0.0 ? max(-1.0, min(1.0, $dot / ($lenU * $lenV))) : 1.0;
405
406        return $sign * acos($cos);
407    }

409    /** @return array{'c', float, float, float, float, float, float} */
410    private static function quadToCubic(float $x0, float $y0, float $qx, float $qy, float $x1, float $y1): array
411    {
412        return [
413            'c',
414            $x0 + 2.0 / 3.0 * ($qx - $x0),
415            $y0 + 2.0 / 3.0 * ($qy - $y0),
416            $x1 + 2.0 / 3.0 * ($qx - $x1),
417            $y1 + 2.0 / 3.0 * ($qy - $y1),
418            $x1,
419            $y1,
420        ];
421    }

423    /**
424     * @param list<string> $tokens
425     * @return array{float, float, int}
426     */
427    private static function readPoint(array $tokens, int $pos, bool $isRelative, float $currentX, float $currentY): array
428    {
429        $x = self::readNumber($tokens, $pos);
430        $y = self::readNumber($tokens, $pos + 1);
431
432        if ($isRelative) {
433            $x += $currentX;
434            $y += $currentY;
435        }
436
437        return [$x, $y, $pos + 2];
438    }

440    /**
441     * Every repeatable path command (M/L/H/V/C/S/Q/T/A) requires at
442     * least one argument set — the `while` loops that read repeats
443     * happily accept zero, so this guards the case the loop itself
444     * can't catch: a bare command letter with nothing following it.
445     *
446     * @param list<string> $tokens
447     */
448    private static function requireArgument(array $tokens, int $pos, int $count, string $command): void
449    {
450        if ($pos >= $count || self::isCommandLetter($tokens[$pos])) {
451            throw new InvalidArgumentException("SVG path command \"{$command}\" is missing its required argument(s).");
452        }
453    }

455    /** @param list<string> $tokens */
456    private static function readNumber(array $tokens, int $pos): float
457    {
458        if (!isset($tokens[$pos]) || !is_numeric($tokens[$pos])) {
459            throw new InvalidArgumentException('Malformed SVG path data: expected a number.');
460        }
461
462        return (float) $tokens[$pos];
463    }

465    private static function isCommandLetter(string $token): bool
466    {
467        return strlen($token) === 1 && isset(self::COMMAND_LETTERS[strtoupper($token)]);
468    }

470    /** @var array<string, true> */
471    private const array COMMAND_LETTERS = [
472        'M' => true, 'L' => true, 'H' => true, 'V' => true,
473        'C' => true, 'S' => true, 'Q' => true, 'T' => true,
474        'A' => true, 'Z' => true,
475    ];

477    /**
478     * Splits path data into command letters and numbers. SVG's loose
479     * separator rules (commas, whitespace, or nothing at all between a
480     * sign-prefixed number and its predecessor) are all accepted; the
481     * one documented gap is packed arc flags with no separator at all
482     * (e.g. `A5,5,0,0110,10` for large-arc=1,sweep=1) — real-world tool
483     * output almost always separates them, and unpacking that specific
484     * ambiguity isn't attempted here.
485     *
486     * @return list<string>
487     */
488    private static function tokenize(string $d): array
489    {
490        preg_match_all('/[MmLlHhVvCcSsQqTtAaZz]|[-+]?(?:\d*\.\d+|\d+\.?\d*)(?:[eE][-+]?\d+)?/', $d, $matches);
491
492        return $matches[0];
493    }
