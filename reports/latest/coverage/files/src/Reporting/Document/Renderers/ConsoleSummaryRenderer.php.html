35/**
36 * Walks a `Document` to Console's plain-text form. Only the blocks
37 * `ConsoleReporter` actually uses for its Passed section
38 * (Heading/FoldingTree/TileGrid, D-091's addendum — Problems/Untested
39 * stay hand-written, ANSI-colored, unconverted) are implemented; any
40 * other block type is a genuine bug, not silently ignored.
41 *
42 * The tree is rendered as a table: hierarchy in the first column,
43 * every number in a column of its own. Figures written into prose
44 * (`· 44 tests · 0.646s`) land at a different offset on every row,
45 * so no two of them can be compared without being read.
46 */
47final readonly class ConsoleSummaryRenderer
48{
49    /** Columns for the tree's own column, connectors included. */
50    private const int LABEL = 34;
51
52    /** Cells in the share bar. */
53    private const int BAR = 12;
54
55    /** Columns a packed group of names may fill. */
56    private const int PACKED = 58;
57
58    /** Share of the run under which a row is a candidate for grouping. */
59    private const float FLOOR = 0.01;
60
61    /**
62     * Rows worth collapsing into one.
63     *
64     * ⚠ Below this the group costs what it saves: a heading plus a
65     * name list to stand in for two rows is two rows.
66     */
67    private const int GROUP = 3;
68
69    public function render(Document $document): string
70    {
71        $output = '';
72
73        foreach ($document->blocks as $block) {
74            $output .= $this->block($block);
75        }
76
77        return $output;
78    }
79
80    private function block(Block $block): string
81    {
82        return match (true) {
83            $block instanceof Heading     => $this->heading($block),
84            $block instanceof FoldingTree => $this->foldingTree($block),
85            $block instanceof TileGrid    => $this->tileGrid($block),
86            default                       => throw new UnexpectedValueException('Unknown block type: ' . $block::class),
87        };
88    }
89
90    private function heading(Heading $heading): string
91    {
92        return $this->runText($heading->runs) . "\n";
93    }
94
95    private function foldingTree(FoldingTree $tree): string
96    {
97        // Scaled to the slowest row rather than to the run: one
98        // directory holding most of the time leaves every other bar
99        // empty, and twenty identical empty bars say nothing.
100        $peak  = 0.0;
101        $total = 0.0;
102
103        foreach ($tree->visible as $node) {
104            $peak = max($peak, $this->peak($node));
105            $total += $node->time;
106        }
107
108        $out = sprintf(
109            "%s%s%s   share\n",
110            str_repeat(' ', self::LABEL),
111            Str::padLeft('tests', 7),
112            Str::padLeft('time', 10),
113        );
114
115        $nodes = $tree->visible;
116        $last  = count($nodes);
117        $index = 0;
118
119        foreach ($nodes as $label => $node) {
120            $index++;
121            $out .= $this->node($label, $node, '', $index === $last, true, $peak, $total * self::FLOOR);
122        }
123
124        if ($tree->foldedDirectoryCount > 0) {
125            $out .= $this->row(
126                sprintf('%d more %s', $tree->foldedDirectoryCount, $tree->foldedDirectoryCount === 1 ? 'directory' : 'directories'),
127                $tree->foldedTestCount,
128                $tree->foldedDuration,
129                $peak,
130            );
131        }
132
133        return $out;
134    }
135
136    /**
137     * One directory, then whatever hangs under it.
138     *
139     * Children and files are one set here, because a reader scanning
140     * for where the time went does not care which of the two a row is.
141     */
142    private function node(string $label, FoldingNode $node, string $prefix, bool $last, bool $root, float $peak, float $floor): string
143    {
144        $out = $this->row(
145            $prefix . ($root ? '' : ($last ? '└─ ' : '├─ ')) . $label,
146            $node->subtreePassedCount(),
147            $node->time,
148            $peak,
149        );
150
151        $inner  = $prefix . ($root ? '' : ($last ? '   ' : '│  '));
152        $rows   = $this->entries($node);
153        $slow   = [];
154        $packed = [];
155
156        foreach ($rows as $row) {
157            if ($row['time'] < $floor && !$row['flagged']) {
158                $packed[] = $row;
159
160                continue;
161            }
162
163            $slow[] = $row;
164        }
165
166        // A group of two is a group that saves nothing.
167        if (count($packed) < self::GROUP) {
168            $slow   = $rows;
169            $packed = [];
170        }
171
172        $tail = count($slow);
173
174        foreach ($slow as $index => $row) {
175            $child = $row['node'];
176            $final = $index === $tail - 1 && $packed === [];
177
178            $out .= $child instanceof FoldingNode
179                ? $this->node($row['label'], $child, $inner, $final, false, $peak, $floor)
180                : $this->row($inner . ($final ? '└─ ' : '├─ ') . $row['label'], $row['tests'], $row['time'], $peak);
181        }
182
183        // A group that covers every entry restates the row above it,
184        // number for number; the names alone are the whole content.
185        return $out . $this->packed($packed, $inner, $peak, $slow !== []);
186    }
187
188    /**
189     * The fast tail, named rather than counted.
190     *
191     * ⚠ Grouped, not cut: the rows collapse, the names do not. A line
192     * reading "+ 24 more" is the one thing nobody can act on.
193     *
194     * @param list<array{label: string, tests: int, time: float, flagged: bool, node: FoldingNode|null}> $rows
195     */
196    private function packed(array $rows, string $prefix, float $peak, bool $summarise): string
197    {
198        if ($rows === []) {
199            return '';
200        }
201
202        $tests = 0;
203        $time  = 0.0;
204        $names = [];
205
206        usort($rows, static fn(array $a, array $b): int => $b['tests'] <=> $a['tests']);
207
208        foreach ($rows as $row) {
209            $tests += $row['tests'];
210            $time += $row['time'];
211            $names[] = sprintf('%s %d', $row['label'], $row['tests']);
212        }
213
214        // Named for WHY they are grouped, not for their total: the
215        // total is already the row's own time column.
216        $out = $summarise
217            ? $this->row(
218                sprintf('%s└─ %d under %d%%', $prefix, count($rows), (int) round(self::FLOOR * 100)),
219                $tests,
220                $time,
221                $peak,
222            )
223            : '';
224
225        $line = '';
226
227        foreach ($names as $name) {
228            $piece = $line === '' ? $name : '  ·  ' . $name;
229
230            if (Str::width($line) + Str::width($piece) > self::PACKED) {
231                $out .= $prefix . '      ' . $line . "\n";
232                $line = $name;
233
234                continue;
235            }
236
237            $line .= $piece;
238        }
239
240        // Never empty: the early return guarantees a row, and every
241        // branch of the loop leaves a name in hand.
242        return $out . $prefix . '      ' . $line . "\n";
243    }
244
245    /**
246     * A node's children and its own files, slowest first.
247     *
248     * @return list<array{label: string, tests: int, time: float, flagged: bool, node: FoldingNode|null}>
249     */
250    private function entries(FoldingNode $node): array
251    {
252        $rows = [];
253
254        foreach ($node->children as $label => $child) {
255            $rows[] = [
256                'label'   => $label,
257                'tests'   => $child->subtreePassedCount(),
258                'time'    => $child->time,
259                'flagged' => $child->holdsFlagged(),
260                'node'    => $child,
261            ];
262        }
263
264        foreach (TileGrid::build($node->files)->tiles as $tile) {
265            $rows[] = [
266                'label'   => PrettyName::ofFile($tile->file),
267                'tests'   => $tile->passedCount,
268                'time'    => $tile->duration,
269                'flagged' => $tile->rank > 0,
270                'node'    => null,
271            ];
272        }
273
274        usort($rows, static fn(array $a, array $b): int => $b['time'] <=> $a['time']);
275
276        return $rows;
277    }
278
279    private function row(string $label, int $tests, float $time, float $peak): string
280    {
281        // ⚠ Width, not bytes: every connector here is multi-byte, so
282        // str_pad() would spend three columns of padding on each one
283        // and the numbers would land wherever the depth left them.
284        return sprintf(
285            "%s%s%s   %s\n",
286            Str::padRight(Str::truncate($label, self::LABEL - 1), self::LABEL),
287            Str::padLeft((string) $tests, 7),
288            Str::padLeft(sprintf('%.3fs', $time), 10),
289            $this->bar($time, $peak),
290        );
291    }
292
293    private function bar(float $time, float $peak): string
294    {
295        $filled = $peak <= 0.0 ? 0 : min(self::BAR, (int) round($time / $peak * self::BAR));
296
297        return str_repeat('█', $filled) . str_repeat('·', self::BAR - $filled);
298    }
299
300    /** The slowest row anywhere under a node, which sets the bar's scale. */
301    private function peak(FoldingNode $node): float
302    {
303        $peak = 0.0;
304
305        foreach ($this->entries($node) as $row) {
306            $peak = max($peak, $row['time']);
307
308            if ($row['node'] instanceof FoldingNode) {
309                $peak = max($peak, $this->peak($row['node']));
310            }
311        }
312
313        return $peak;
314    }
315
316    private function tileGrid(TileGrid $grid, int $depth = 1): string
317    {
318        $indent = str_repeat('  ', $depth);
319        $out    = '';
320
321        foreach ($grid->tiles as $tile) {
322            $out .= sprintf("%s%s — %d %s\n", $indent, PrettyName::ofFile($tile->file), $tile->passedCount, $tile->passedCount === 1 ? 'test' : 'tests');
323        }
324
325        return $out;
326    }
327
328    /**
329     * @param list<Run> $runs
330     */
331    private function runText(array $runs): string
332    {
333        return implode('', array_map(static fn(Run $run): string => $run instanceof Text ? $run->content : throw new UnexpectedValueException('Unknown run type: ' . $run::class), $runs));
334    }
