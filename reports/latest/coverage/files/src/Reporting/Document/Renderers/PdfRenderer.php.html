73    public function render(
74        Document $document,
75        string $title,
76        string $author,
77        string $producer,
78        ?DateTimeImmutable $createdAt,
79        float $runtime,
80    ): string {
81        $this->runtime   = $runtime;
82        $this->spotlight = [];
83
84        foreach ($document->blocks as $block) {
85            if ($block instanceof RankedList) {
86                foreach ($block->entries as $entry) {
87                    $this->spotlight[$entry->file] = true;
88                }
89            }
90        }
91
92        $hasToc       = ($document->blocks[0] ?? null) instanceof Toc;
93        $tocEntries   = [];
94        $tocPageCount = 0;
95
96        if ($hasToc) {
97            // Pass 1 (dry run): render the body alone — its own bookmark()
98            // calls collect (level, title, pageIndex) for every heading,
99            // as if there were no TOC in front. Output is discarded.
100            $this->pdf = new PdfPrimitives();
101
102            $this->body($document->blocks);
103
104            $tocEntries = $this->pdf->bookmarkEntries();
105
106            // Self-size the TOC by really laying it out once on a
107            // throwaway primitives instance — reuses the exact same
108            // line-wrap/page-break logic pass 2 will use for real,
109            // rather than a separate estimate that could drift from it.
110            // The trailing forced break gives the TOC its own dedicated
111            // page(s), so chapter one always starts fresh — the same
112            // ensure(HEIGHT) a PageBreak block itself uses.
113            $sizer     = new PdfPrimitives();
114            $this->pdf = $sizer;
115            $this->renderTocPages($tocEntries, 0);
116            $this->pdf->ensure(PdfPrimitives::HEIGHT);
117            $tocPageCount = count($sizer->pages);
118        }
119
120        // Pass 2 (real render): TOC page(s) first, with page numbers
121        // shifted by $tocPageCount, then the body — its true page
122        // numbers and bookmark /Dest targets come out correct since the
123        // TOC now genuinely precedes it.
124        $this->pdf = new PdfPrimitives();
125
126        if ($hasToc) {
127            $this->renderTocPages($tocEntries, $tocPageCount);
128            $this->pdf->ensure(PdfPrimitives::HEIGHT);
129        }
130
131        $this->body($document->blocks);
132
133        return $this->pdf->document($title, $author, $producer, $createdAt);
134    }

136    /**
137     * @param list<array{level: int, title: string, pageIndex: int}> $entries
138     */
139    private function renderTocPages(array $entries, int $pageShift): void
140    {
141        $this->pdf->ensure(15.0 * 1.4);
142        $this->pdf->text(PdfPrimitives::MARGIN, $this->pdf->y - 15.0, PdfPrimitives::BOLD, 15.0, $this->pdf->encode('Table of Contents'));
143        $this->pdf->y -= 15.0 * 1.4 + 8.0;
144
145        $columns = $this->WIDTH() - 2 * PdfPrimitives::MARGIN;
146
147        foreach ($entries as $entry) {
148            $indent = ($entry['level'] - 1) * 14.0;
149            $font   = $entry['level'] === 1 ? PdfPrimitives::BOLD : PdfPrimitives::HELVETICA;
150            $size   = $entry['level'] === 1 ? 11.0 : 10.0;
151
152            $this->pdf->ensure($size * 1.4);
153            $baseline = $this->pdf->y - $size;
154            $title    = $this->pdf->encode($entry['title']);
155            $pageText = $this->pdf->encode((string) ($pageShift + $entry['pageIndex'] + 1));
156            $pageX    = PdfPrimitives::MARGIN + $columns - $this->pdf->measure($pageText, $font, $size);
157
158            $this->pdf->text(PdfPrimitives::MARGIN + $indent, $baseline, $font, $size, $title);
159            $this->pdf->text($pageX, $baseline, $font, $size, $pageText);
160            $this->pdf->leader(PdfPrimitives::MARGIN + $indent + $this->pdf->measure($title, $font, $size), $pageX, $baseline);
161
162            $this->pdf->y -= $size * 1.4;
163        }
164    }

166    /** @phpcpd-ignore-clone Dispatch table — one arm per block type, one table per output format by design. */
167    private function block(Block $block): void
168    {
169        match (true) {
170            $block instanceof Heading       => $this->heading($block),
171            $block instanceof Image         => $this->image($block),
172            $block instanceof Svg           => $this->svg($block),
173            $block instanceof PageBreak     => $this->pdf->ensure(PdfPrimitives::HEIGHT),
174            $block instanceof Paragraph     => $this->paragraph($block),
175            $block instanceof BulletList    => $this->bulletList($block),
176            $block instanceof Badge         => $this->badge($block),
177            $block instanceof ProblemList   => $this->problemList($block),
178            $block instanceof ProportionBar => $this->proportionBar($block),
179            $block instanceof RankedList    => $this->rankedList($block),
180            $block instanceof FoldingTree   => $this->foldingTree($block),
181            $block instanceof TileGrid      => $this->tileGrid($block, $this->WIDTH() - 2 * PdfPrimitives::MARGIN),

191    /**
192     * Every block but the TOC, in order. A title followed by its badge is
193     * one line — `[logo] Test report [OK]` — the verdict read where the
194     * report is named.
195     *
196     * @param list<Block> $blocks
197     */
198    private function body(array $blocks): void
199    {
200        for ($i = 0, $count = count($blocks); $i < $count; $i++) {
201            $block = $blocks[$i];
202            $next  = $blocks[$i + 1] ?? null;
203
204            if ($block instanceof Heading && $block->level === 1 && $next instanceof Badge) {
205                $this->titleLine($block, $next);
206                $i++;
207            } elseif (!$block instanceof Toc) {
208                $this->block($block);
209            }
210        }
211    }

330    {
331        $total = 0;
332

335        }

452    private function node(string $label, FoldingNode $node, int $depth): void
453    {
454        $indent = $depth * 14.0;
455
456        if ($node->children === [] && count($node->files) === 1) {
457            foreach ($node->files as $file => $tests) {
458                $this->inline($label, $file, $tests, $indent);
459            }
460
461            return;
462        }
463
464        if ($node->files === [] && FoldingNode::onlyInlinable($node->children)) {
465            foreach ($node->children as $childLabel => $child) {
466                foreach ($child->files as $file => $tests) {
467                    $this->inline($label . '/' . $childLabel, $file, $tests, $indent);
468                }
469            }
470
471            return;
472        }
473
474        $own  = $node->ownPassedCount();
475        $meta = $own > 0
476            ? sprintf('· %d %s · %.3fs', $own, $own === 1 ? 'test' : 'tests', $node->time)
477            : sprintf('· %.3fs', $node->time);
478
479        $this->pdf->ensure(10.0 * 1.4 + 13.0);
480        $bold = $this->pdf->encode($label);
481        $this->pdf->text(PdfPrimitives::MARGIN + $indent, $this->pdf->y - 10.0, PdfPrimitives::BOLD, 10.0, $bold);
482        $this->pdf->text(PdfPrimitives::MARGIN + $indent + $this->pdf->measure($bold, PdfPrimitives::BOLD, 10.0) + 5.0, $this->pdf->y - 10.0, PdfPrimitives::HELVETICA, 8.5, $this->pdf->encode($meta), self::GRAY);
483        $this->pdf->y -= 10.0 * 1.4 + 2.0;
484
485        if ($node->files !== []) {
486            $this->tileGrid(TileGrid::build($node->files), $this->WIDTH() - 2 * PdfPrimitives::MARGIN - $indent, $indent);
487        }
488
489        foreach ($node->children as $childLabel => $child) {
490            $this->node($childLabel, $child, $depth + 1);
491        }
492    }

495     * @param non-empty-string   $file
496     * @param list<TestFinished> $tests
497     */
498    private function inline(string $label, string $file, array $tests, float $indent): void
499    {
500        $this->pdf->ensure(13.0);
501
502        $baseline = $this->pdf->y - 9.5;
503        $bold     = $this->pdf->encode($label);
504        $name     = $this->pdf->encode('· ' . PrettyName::ofFile($file));
505        $passed   = FoldingNode::passedCountOf($tests);
506        $meta     = $this->pdf->encode(sprintf(
507            '· %d %s · %.3fs',
508            $passed,
509            $passed === 1 ? 'test' : 'tests',
510            FoldingNode::timeOf($tests),
511        ));
512        $nameX = PdfPrimitives::MARGIN + $indent + $this->pdf->measure($bold, PdfPrimitives::BOLD, 10.0) + 5.0;
513        $rank  = FoldingNode::rankOf($tests);
514
515        $this->pdf->text(PdfPrimitives::MARGIN + $indent, $baseline, PdfPrimitives::BOLD, 10.0, $bold);
516
517        if ($rank > 0) {
518            $this->pdf->disc($nameX + 8.5, $baseline + 3.0, 2.2, $rank >= 3 ? self::RED : self::AMBER);
519            $nameX += 6.0;
520        }
521
522        $this->pdf->text($nameX, $baseline, PdfPrimitives::HELVETICA, 9.0, $name);
523        $this->pdf->text($nameX + $this->pdf->measure($name, PdfPrimitives::HELVETICA, 9.0) + 5.0, $baseline, PdfPrimitives::HELVETICA, 8.5, $meta, self::GRAY);
524
525        $this->pdf->y -= 13.0 + 2.0;
526    }

528    private function tileGrid(TileGrid $grid, float $columns, float $indent = 0.0): void
529    {
530        $gutter = 12.0;
531        $cell   = ($columns - 2.0 * $gutter) / 3.0;
532
533        /** @var list<array{label: string, stat: string, span: int, rank: int}> $queue */
534        $queue = [];
535
536        foreach ($grid->tiles as $tile) {
537            $stat = (string) $tile->passedCount;
538
539            if ($this->runtime > 0.0 && $tile->duration > 0.02 * $this->runtime && !isset($this->spotlight[$tile->file])) {
540                $stat = sprintf('%s · %.3fs', $stat, $tile->duration);
541            }
542
543            $stat   = $this->pdf->encode($stat);
544            $label  = $this->pdf->encode(PrettyName::ofFile($tile->file));
545            $needed = 9.0 + $this->pdf->measure($label, PdfPrimitives::HELVETICA, 9.0) + 8.0 + $this->pdf->measure($stat, PdfPrimitives::HELVETICA, 8.0);
546
547            $queue[] = [
548                'label' => $label,
549                'stat'  => $stat,
550                'span'  => match (true) {
551                    $needed <= $cell                 => 1,
552                    $needed <= 2.0 * $cell + $gutter => 2,
553                    default                          => 3,
554                },
555                'rank' => $tile->rank,
556            ];
557        }
558
559        $slot = 0;
560
561        while ($queue !== []) {
562            $room = 3 - $slot % 3;
563            $pick = null;
564
565            foreach ($queue as $index => $tile) {
566                if ($tile['span'] <= $room) {
567                    $pick = $index;
568
569                    break;
570                }
571            }
572
573            if ($pick === null) {
574                $this->pdf->y -= 13.0;
575                $slot += $room;
576
577                continue;
578            }
579
580            $tile = $queue[$pick];
581            unset($queue[$pick]);
582            $queue = array_values($queue);
583
584            if ($slot % 3 === 0) {
585                $this->pdf->ensure(13.0);
586            }
587
588            $x        = PdfPrimitives::MARGIN + $indent + ($slot % 3) * ($cell + $gutter);
589            $width    = $tile['span'] * $cell + ($tile['span'] - 1) * $gutter;
590            $baseline = $this->pdf->y - 9.5;
591
592            if ($tile['rank'] > 0) {
593                $this->pdf->disc($x + 2.5, $baseline + 3.0, 2.2, $tile['rank'] >= 3 ? self::RED : self::AMBER);
594            }
595
596            $this->pdf->text($x + 9.0, $baseline, PdfPrimitives::HELVETICA, 9.0, $tile['label']);
597            $this->pdf->text($x + $width - $this->pdf->measure($tile['stat'], PdfPrimitives::HELVETICA, 8.0), $baseline, PdfPrimitives::HELVETICA, 8.0, $tile['stat'], self::GRAY);
598
599            $slot += $tile['span'];
600
601            if ($slot % 3 === 0) {
602                $this->pdf->y -= 13.0;
603            }
604        }
605
606        if ($slot % 3 !== 0) {
607            $this->pdf->y -= 13.0;
608        }
609
610        $this->pdf->y -= 6.0;
611    }

613    /**
614     * @return array{float, float, float}
615     */
616    private function tone(Tone $tone): array
617    {
618        return match ($tone) {
619            Tone::Success => self::GREEN,
620            Tone::Caution => self::AMBER,
621            Tone::Danger  => self::RED,
622            Tone::Muted   => self::GRAY,
623        };
624    }

626    /**
627     * @param list<Run> $runs
628     */
629    private function runText(array $runs): string
630    {
631        return implode('', array_map(static fn(Run $run): string => match (true) {
632            $run instanceof Code, $run instanceof Strong, $run instanceof Text => $run->content,
633            default                                                            => throw new UnexpectedValueException('Unknown run type: ' . $run::class),
634        }, $runs));
635    }
