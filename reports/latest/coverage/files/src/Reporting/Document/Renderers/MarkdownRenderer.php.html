44/**
45 * Walks a `Document` to Markdown text — the same
46 * `match (true) { $block instanceof X => ... }` dispatch as the lrv
47 * pattern this is modeled on (D-091's addendum), one method per
48 * block/run type.
49 */
50final readonly class MarkdownRenderer
51{
52    public function render(Document $document): string
53    {
54        $output = '';
55
56        foreach ($document->blocks as $block) {
57            $output .= $this->block($block);
58        }
59
60        return $output;
61    }
62
63    /** @phpcpd-ignore-clone Dispatch table — one arm per block type, one table per output format by design. */
64    private function block(Block $block): string
65    {
66        return match (true) {
67            $block instanceof Heading       => $this->heading($block),
68            $block instanceof Paragraph     => $this->paragraph($block),
69            $block instanceof BulletList    => $this->bulletList($block),
70            $block instanceof Badge         => $this->badge($block),
71            $block instanceof ProblemList   => $this->problemList($block),
72            $block instanceof ProportionBar => $this->proportionBar($block),
73            $block instanceof RankedList    => $this->rankedList($block),
74            $block instanceof FoldingTree   => $this->foldingTree($block),
75            $block instanceof TileGrid      => $this->tileGrid($block),
76            $block instanceof Toc           => '<!-- toc -->' . "\n\n",
77            $block instanceof Image, $block instanceof Svg
78                                            => "_An image is included here (not embeddable in Markdown)._\n\n",
79            $block instanceof PageBreak     => "---\n\n",
80            default                         => throw new UnexpectedValueException('Unknown block type: ' . $block::class),
81        };
82    }
83
84    private function heading(Heading $heading): string
85    {
86        return str_repeat('#', $heading->level) . ' ' . $this->runs($heading->runs) . "\n\n";
87    }
88
89    private function paragraph(Paragraph $paragraph): string
90    {
91        return $this->runs($paragraph->runs) . "\n\n";
92    }
93
94    private function bulletList(BulletList $list): string
95    {
96        $out = '';
97
98        foreach ($list->items as $item) {
99            $out .= '- ' . $this->runs($item) . "\n";
100        }
101
102        return $out . "\n";
103    }
104
105    private function badge(Badge $badge): string
106    {
107        return '**' . $badge->text . "**\n\n";
108    }
109
110    private function problemList(ProblemList $list): string
111    {
112        $out = '';
113
114        foreach ($list->entries as $entry) {
115            $out .= sprintf(
116                "### `%s` — %s\n\n",
117                $entry->testName,
118                $entry->outcome,
119            );
120
121            if ($entry->detail !== null) {
122                $out .= "```\n" . $entry->detail . "\n```\n\n";
123            }
124        }
125
126        return $out;
127    }
128
129    /**
130     * The passed/skipped/flagged/failed share strip — a plain
131     * percentage line, since Markdown has no visual bar to draw.
132     */
133    private function proportionBar(ProportionBar $bar): string
134    {
135        $total = 0;
136
137        foreach ($bar->shares as $share) {
138            $total += $share->count;
139        }
140
141        if ($total === 0) {
142            return '';
143        }
144
145        $parts = [];
146
147        foreach ($bar->shares as $share) {
148            if ($share->count === 0) {
149                continue;
150            }
151
152            $parts[] = sprintf('%s %.0f%%', $this->toneLabel($share->tone), $share->count / $total * 100);
153        }
154
155        return implode(' · ', $parts) . "\n\n";
156    }
157
158    private function toneLabel(Tone $tone): string
159    {
160        return match ($tone) {
161            Tone::Success => 'Passed',
162            Tone::Muted   => 'Skipped',
163            Tone::Caution => 'Flagged',
164            Tone::Danger  => 'Failed',
165        };
166    }
167
168    private function rankedList(RankedList $list): string
169    {
170        if ($list->entries === []) {
171            return '';
172        }
173
174        $out = '';
175
176        foreach ($list->entries as $index => $entry) {
177            $out .= sprintf(
178                "%d. %s · %s — %.3fs\n",
179                $index + 1,
180                $this->cell($entry->testName),
181                $this->cell(PrettyName::ofFile($entry->file)),
182                $entry->duration,
183            );
184        }
185
186        return $out . "\n";
187    }
188
189    private function foldingTree(FoldingTree $tree): string
190    {
191        $out = '';
192
193        foreach ($tree->visible as $label => $node) {
194            $out .= $this->node($label, $node);
195        }
196
197        if ($tree->foldedDirectoryCount > 0) {
198            $out .= sprintf(
199                "+ %d more %s · %d %s · %.3fs\n\n",
200                $tree->foldedDirectoryCount,
201                $tree->foldedDirectoryCount === 1 ? 'directory' : 'directories',
202                $tree->foldedTestCount,
203                $tree->foldedTestCount === 1 ? 'test' : 'tests',
204                $tree->foldedDuration,
205            );
206        }
207
208        return $out;
209    }
210
211    private function node(string $label, FoldingNode $node): string
212    {
213        $own = $node->ownPassedCount();
214        $out = sprintf(
215            "#### %s %s\n\n",
216            $label,
217            $own > 0 ? sprintf('(%d %s, %.3fs)', $own, $own === 1 ? 'test' : 'tests', $node->time) : sprintf('(%.3fs)', $node->time),
218        );
219
220        // Every individual test, not just the file's aggregate — the
221        // same per-test table the old flat "## Results" section
222        // showed, now organized under the directory tree instead of
223        // losing that detail to match Pdf's compact tile grid.
224        foreach ($node->files as $file => $tests) {
225            $out .= sprintf("##### %s\n\n", PrettyName::ofFile($file));
226            $out .= "| Test | Outcome | Time |\n|------|---------|-----:|\n";
227
228            foreach ($tests as $test) {
229                $out .= sprintf(
230                    "| %s | %s %s | %.3fs |\n",
231                    $this->cell(PrettyName::ofTest($test->test)),
232                    $this->mark($test->outcome),
233                    $test->outcome->value,
234                    $test->duration,
235                );
236            }
237
238            $out .= "\n";
239        }
240
241        foreach ($node->children as $childLabel => $child) {
242            $out .= $this->node($childLabel, $child);
243        }
244
245        return $out;
246    }
247
248    /**
249     * @return non-empty-string
250     */
251    private function mark(Outcome $outcome): string
252    {
253        return match ($outcome) {
254            Outcome::Passed     => '✅',
255            Outcome::Failed     => '❌',
256            Outcome::Errored    => '💥',
257            Outcome::Skipped    => '⏭️',
258            Outcome::Incomplete => '🚧',
259            Outcome::Risky      => '⚠️',
260        };
261    }
262
263    private function tileGrid(TileGrid $grid): string
264    {
265        $out = '';
266
267        foreach ($grid->tiles as $tile) {
268            $out .= sprintf(
269                "- %s — %d %s (%.3fs)\n",
270                PrettyName::ofFile($tile->file),
271                $tile->passedCount,
272                $tile->passedCount === 1 ? 'test' : 'tests',
273                $tile->duration,
274            );
275        }
276
277        return $out . "\n";
278    }
279
280    /**
281     * @param list<Run> $runs
282     */
283    private function runs(array $runs): string
284    {
285        return implode('', array_map(fn(Run $run): string => match (true) {
286            $run instanceof Code   => '`' . $this->cell($run->content) . '`',
287            $run instanceof Strong => '**' . $this->cell($run->content) . '**',
288            $run instanceof Text   => $this->cell($run->content),
289            default                => throw new UnexpectedValueException('Unknown run type: ' . $run::class),
290        }, $runs));
291    }
292
293    /**
294     * Table cells and inline text cannot contain unescaped pipes or
295     * newlines.
296     */
297    private function cell(string $value): string
298    {
299        return str_replace(['|', "\n"], ['\\|', ' '], $value);
300    }
