113    {

159     * Writes flowing text at the cursor and advances it: one line when it
160     * fits, wrapped at word boundaries by flow() when it does not, so a
161     * long paragraph stays inside the right margin instead of running
162     * off the page.

164     * @param ?array{float, float, float} $rgb
165     */
166    public function line(string $font, float $size, string $text, ?array $rgb = null, float $indent = 0.0): void
167    {
168        foreach ($this->flow($this->encode($text), self::WIDTH - 2 * self::MARGIN - $indent, $font, $size) as $row) {
169            $this->ensure($size * 1.4);
170            $this->text(self::MARGIN + $indent, $this->y - $size, $font, $size, $row, $rgb);
171            $this->y -= $size * 1.4;
172        }
173    }

306    /**
307     * Width, height, and PDF color space straight from the JPEG's own
308     * headers — no hand-rolled marker walk needed, `getimagesizefromstring`
309     * already reports the component count. The one thing it doesn't
310     * expose is Adobe's inverted-CMYK convention (`APP14` present on a
311     * 4-component JPEG), which every CMYK JPEG out of Adobe tooling
312     * uses — so `/Decode [1 0 1 0 1 0 1 0]` is applied whenever that
313     * marker is present, matching FPDF/mPDF's own long-standing rule.
314     *
315     * @return array{int, int, string, bool}
316     */
317    private function jpegInfo(string $jpegBytes): array
318    {
319        $info = @getimagesizefromstring($jpegBytes);
320
321        if ($info === false || $info[2] !== IMAGETYPE_JPEG) {
322            throw new InvalidArgumentException('Image data is not a JPEG.');
323        }
324
325        $components = $info['channels'] ?? 3;
326        $colorSpace = match ($components) {
327            1       => 'DeviceGray',
328            4       => 'DeviceCMYK',
329            default => 'DeviceRGB',
330        };
331
332        $invert = $components === 4 && str_contains($jpegBytes, 'Adobe');
333
334        return [$info[0], $info[1], $colorSpace, $invert];
335    }

337    /**
338     * The PDF `/Outlines` bookmark tree, built from the flat `$bookmarks`
339     * list by walking it once with a level-keyed stack: an entry's parent
340     * is the nearest earlier entry with a strictly lower level (or the
341     * synthetic root when none exists), the same convention every
342     * heading-outline in Markdown/AsciiDoc/etc. already uses. A second
343     * pass groups entries by parent to derive `/Prev`/`/Next`/`/First`/
344     * `/Last`/`/Count`, since siblings aren't contiguous once a parent
345     * has grandchildren interleaved in document order.
346     *
347     * @return array<int, string> object id => object body, including the
348     *                            root `/Type /Outlines` object at $rootId
349     */
350    private function outlineObjects(int $rootId): array
351    {
352        if ($this->bookmarks === []) {
353            return [];
354        }
355
356        $ids = [];
357
358        foreach (array_keys($this->bookmarks) as $index) {
359            $ids[$index] = $rootId + 1 + $index;
360        }
361
362        /** @var array<int, int> $parentOf entry index => parent object id (rootId or another entry's id) */
363        $parentOf = [];
364        /** @var list<array{int, int}> $stack open ancestors as [level, objectId] */
365        $stack = [];
366
367        foreach ($this->bookmarks as $index => $entry) {
368            while ($stack !== [] && end($stack)[0] >= $entry['level']) {
369                array_pop($stack);
370            }
371
372            $parentOf[$index] = $stack === [] ? $rootId : end($stack)[1];
373            $stack[]          = [$entry['level'], $ids[$index]];
374        }
375
376        /** @var array<int, list<int>> $childrenOf parent object id => list of entry indices, in document order */
377        $childrenOf = [];
378        /** @var array<int, int> $positionOf entry index => its position among its siblings */
379        $positionOf = [];
380
381        foreach (array_keys($this->bookmarks) as $index) {
382            $parent                = $parentOf[$index];
383            $childrenOf[$parent][] = $index;
384            $positionOf[$index]    = count($childrenOf[$parent]) - 1;
385        }
386
387        $rootChildren = $childrenOf[$rootId] ?? [];
388        $objects      = [
389            $rootId => sprintf(
390                '<< /Type /Outlines /First %d 0 R /Last %d 0 R /Count %d >>',
391                $ids[$rootChildren[0]],
392                $ids[$rootChildren[count($rootChildren) - 1]],
393                count($this->bookmarks),
394            ),
395        ];
396
397        foreach ($this->bookmarks as $index => $entry) {
398            $siblings = $childrenOf[$parentOf[$index]];
399            $position = $positionOf[$index];
400            $kids     = $childrenOf[$ids[$index]] ?? [];
401
402            $object = sprintf(
403                '<< /Title (%s) /Parent %d 0 R',
404                $this->escape($this->encode($entry['title'])),
405                $parentOf[$index],
406            );
407
408            if ($position > 0) {
409                $object .= sprintf(' /Prev %d 0 R', $ids[$siblings[$position - 1]]);
410            }
411
412            if ($position < count($siblings) - 1) {
413                $object .= sprintf(' /Next %d 0 R', $ids[$siblings[$position + 1]]);
414            }
415
416            if ($kids !== []) {
417                $object .= sprintf(
418                    ' /First %d 0 R /Last %d 0 R /Count %d',
419                    $ids[$kids[0]],
420                    $ids[$kids[count($kids) - 1]],
421                    count($kids),
422                );
423            }
424
425            $object .= sprintf(' /Dest [%d 0 R /XYZ null null null] >>', 7 + 2 * $entry['pageIndex']);
426
427            $objects[$ids[$index]] = $object;
428        }
429
430        return $objects;
431    }

548    /**
549     * UTF-8 to the fonts' WinAnsi encoding: Latin-1 passes through,
550     * the common typographic points map to their Windows-1252 slots,
551     * anything else degrades to '?'. Decoded by hand — no ext-iconv,
552     * no ext-mbstring.
553     */
554    public function encode(string $utf8): string
555    {
556        $out    = '';
557        $length = strlen($utf8);
558
559        for ($i = 0; $i < $length; $i++) {
560            $byte = ord($utf8[$i]);
561
562            if ($byte < 0x80) {
563                $code = $byte;
564            } elseif (($byte & 0xE0) === 0xC0) {
565                $code = (($byte & 0x1F) << 6) | (ord(($utf8[++$i] ?? "\0")[0]) & 0x3F);
566            } elseif (($byte & 0xF0) === 0xE0) {
567                $code = (($byte & 0x0F) << 12) | ((ord(($utf8[++$i] ?? "\0")[0]) & 0x3F) << 6) | (ord(($utf8[++$i] ?? "\0")[0]) & 0x3F);
568            } else {
569                $code = 0x10000;
570                $i += 3;
571            }
572
573            $out .= match (true) {
574                $code === 0x2018 => chr(0x91),
575                $code === 0x2019 => chr(0x92),
576                $code === 0x201C => chr(0x93),
577                $code === 0x201D => chr(0x94),
578                $code === 0x2013 => chr(0x96),
579                $code === 0x2014 => chr(0x97),
580                $code === 0x2026 => chr(0x85),
581                $code < 0x20     => ' ',
582                $code > 0xFF     => '?',
583                default          => chr($code),
584            };
585        }
586
587        return $out;
588    }

590    /**
591     * The rendered width of already-encoded text, in points.
592     */
593    public function measure(string $latin1, string $font, float $size): float
594    {
595        if ($font === self::COURIER) {
596            return strlen($latin1) * 600 * $size / 1000.0;
597        }
598
599        $widths  = $font === self::BOLD ? self::BOLD_WIDTHS : self::HELVETICA_WIDTHS;
600        $default = $font === self::BOLD ? 611 : 556;
601        $units   = 0;
602
603        foreach (str_split($latin1) as $character) {
604            $code = ord($character[0]);
605            $units += $widths[$code - 32] ?? $default;
606        }
607
608        return $units * $size / 1000.0;
609    }

611    /**
612     * Splits already-encoded text into lines that fit — the report
613     * never abbreviates with an ellipsis; names wrap instead. An
614     * overlong single word hard-splits.
615     *
616     * @return non-empty-list<string>
617     */
618    public function flow(string $latin1, float $width, string $font, float $size): array
619    {
620        $lines   = [];
621        $current = '';
622
623        foreach (explode(' ', $latin1) as $word) {
624            $candidate = $current === '' ? $word : $current . ' ' . $word;
625
626            if ($this->measure($candidate, $font, $size) <= $width) {
627                $current = $candidate;
628
629                continue;
630            }
631
632            if ($current !== '') {
633                $lines[] = $current;
634            }
635
636            while ($this->measure($word, $font, $size) > $width) {
637                $kept = '';
638
639                foreach (str_split($word) as $character) {
640                    if ($kept !== '' && $this->measure($kept . $character, $font, $size) > $width) {
641                        break;
642                    }
643
644                    $kept .= $character;
645                }
646
647                $lines[] = $kept;
648                $word    = substr($word, strlen($kept));
649            }
650
651            $current = $word;
652        }
653
654        $lines[] = $current;
655
656        return $lines;
657    }

659    /**
660     * Failure text as encoded Courier lines: newlines respected, long
661     * lines chunked at the fixed-pitch column budget.
662     *
663     * @return list<string>
664     */
665    public function wrap(string $utf8, float $size, float $width): array
666    {
667        $columns = max(1, (int) floor($width / (600 * $size / 1000.0)));
668        $rows    = [];
669
670        foreach (explode("\n", str_replace("\r\n", "\n", $utf8)) as $line) {
671            $encoded = $this->encode($line);
672
673            foreach ($encoded === '' ? [''] : str_split($encoded, $columns) as $chunk) {
674                $rows[] = $chunk;
675            }
676        }
677
678        return $rows;
679    }

681    // -- The document ----------------------------------------------
682
683    /**
684     * The PDF file: header, objects, cross-reference table, trailer.
685     * Objects 1–6 are fixed (catalog, page tree, three fonts, info);
686     * each page then contributes a page object and a content stream.
687     * Finalizes the last in-progress page itself, so a caller never
688     * has to touch `$pages`/`$ops` directly.
689     */
690    public function document(string $title, string $author, string $producer, ?DateTimeImmutable $createdAt): string
691    {
692        $this->pages[]      = $this->ops;
693        $this->pageImages[] = $this->currentPageImages;
694
695        $count = count($this->pages);
696        $kids  = '';
697
698        for ($i = 0; $i < $count; $i++) {
699            $kids .= sprintf('%d 0 R ', 7 + 2 * $i);
700        }
701
702        // Image XObjects follow every page/content object pair, in
703        // registration order — their ids only depend on $count, known
704        // up front, so pages can reference them within the same loop
705        // that builds the page objects below.
706        $imageBaseId   = 7 + 2 * $count;
707        $imageIdByName = [];
708
709        foreach (array_keys($this->images) as $index => $name) {
710            $imageIdByName[$name] = $imageBaseId + $index;
711        }
712
713        // The outline (bookmark) tree follows every image XObject, its
714        // own id block sized by $bookmarks alone — known up front, same
715        // reasoning as $imageBaseId above.
716        $outlineRootId  = $imageBaseId + count($this->images);
717        $outlineObjects = $this->outlineObjects($outlineRootId);
718
719        // Archival metadata, all derived from the configuration and
720        // the event stream — never from the wall clock, so the same
721        // run still produces byte-identical output.
722        $info = sprintf(
723            '<< /Title (%s) /Author (%s) /Producer (%s)',
724            $this->escape($this->encode($title)),
725            $this->escape($this->encode($author)),
726            $producer,
727        );
728
729        if ($createdAt instanceof DateTimeImmutable) {
730            $info .= sprintf(
731                " /CreationDate (D:%s%s')",
732                $createdAt->format('YmdHis'),
733                str_replace(':', "'", $createdAt->format('P')),
734            );
735        }
736
737        $info .= ' >>';
738
739        $objects = [
740            1 => sprintf(
741                '<< /Type /Catalog /Pages 2 0 R%s >>',
742                $outlineObjects === [] ? '' : sprintf(' /Outlines %d 0 R', $outlineRootId),
743            ),
744            2 => sprintf('<< /Type /Pages /Kids [ %s] /Count %d >>', $kids, $count),
745            3 => '<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica /Encoding /WinAnsiEncoding >>',
746            4 => '<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica-Bold /Encoding /WinAnsiEncoding >>',
747            5 => '<< /Type /Font /Subtype /Type1 /BaseFont /Courier /Encoding /WinAnsiEncoding >>',
748            6 => $info,
749        ];
750
751        $brand = $this->encode($producer);
752
753        foreach ($this->pages as $i => $content) {
754            // The footer sits under a hairline rule, clear of the
755            // content area ensure() protects.
756            $footer = $this->encode(sprintf('Page %d of %d', $i + 1, $count));
757            $stream = $content . sprintf(
758                "0.5 w 0.75 G %s 44.00 m %s 44.00 l S\n",
759                $this->number(self::MARGIN),
760                $this->number(self::WIDTH - self::MARGIN),
761            ) . sprintf(
762                "BT 0.46 0.46 0.46 rg /F1 8.00 Tf %s 30.00 Td (%s) Tj ET\n",
763                $this->number(self::MARGIN),
764                $brand,
765            ) . sprintf(
766                "BT 0.46 0.46 0.46 rg /F1 8.00 Tf %s 30.00 Td (%s) Tj ET\n",
767                $this->number(self::WIDTH - self::MARGIN - $this->measure($footer, self::HELVETICA, 8.0)),
768                $footer,
769            );
770
771            $xobject = '';
772
773            if ($this->pageImages[$i] !== []) {
774                $entries = '';
775
776                foreach (array_keys($this->pageImages[$i]) as $name) {
777                    $entries .= sprintf('/%s %d 0 R ', $name, $imageIdByName[$name]);
778                }
779
780                $xobject = sprintf(' /XObject << %s>>', $entries);
781            }
782
783            $objects[7 + 2 * $i] = sprintf(
784                '<< /Type /Page /Parent 2 0 R /MediaBox [0 0 %s %s] /Resources << /Font << /F1 3 0 R /F2 4 0 R /F3 5 0 R >>%s >> /Contents %d 0 R >>',
785                $this->number(self::WIDTH),
786                $this->number(self::HEIGHT),
787                $xobject,
788                8 + 2 * $i,
789            );
790            $objects[8 + 2 * $i] = sprintf(
791                "<< /Length %d >>\nstream\n%sendstream",
792                strlen($stream),
793                $stream,
794            );
795        }
796
797        foreach ($this->images as $name => $image) {
798            $decode = $image['invert'] ? ' /Decode [1 0 1 0 1 0 1 0]' : '';
799
800            $objects[$imageIdByName[$name]] = sprintf(
801                "<< /Type /XObject /Subtype /Image /Width %d /Height %d /ColorSpace /%s /BitsPerComponent 8 /Filter /DCTDecode%s /Length %d >>\nstream\n%sendstream",
802                $image['width'],
803                $image['height'],
804                $image['colorSpace'],
805                $decode,
806                strlen($image['data']),
807                $image['data'],
808            );
809        }
810
811        foreach ($outlineObjects as $id => $object) {
812            $objects[$id] = $object;
813        }
814
815        $body    = "%PDF-1.4\n%\xE2\xE3\xCF\xD3\n";
816        $offsets = [];
817
818        foreach ($objects as $id => $object) {
819            $offsets[$id] = strlen($body);
820            $body .= sprintf("%d 0 obj\n%s\nendobj\n", $id, $object);
821        }
822
823        $start = strlen($body);
824        $body .= sprintf("xref\n0 %d\n0000000000 65535 f \n", count($objects) + 1);
825
826        foreach ($offsets as $offset) {
827            $body .= sprintf("%010d 00000 n \n", $offset);
828        }
829
830        return $body . sprintf(
831            "trailer\n<< /Size %d /Root 1 0 R /Info 6 0 R >>\nstartxref\n%d\n%%%%EOF\n",
832            count($objects) + 1,
833            $start,
834        );
835    }
