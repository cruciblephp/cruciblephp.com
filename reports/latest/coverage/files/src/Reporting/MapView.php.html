61/**
62 * The suite as a map: one cell per test, filling as the run goes.
63 *
64 * Borrowed from the disk utilities of the late eighties, which solved this
65 * on 80x25 and 4.77 MHz. Three things made those readable, and all three
66 * are here:
67 *
68 * - **The space is reserved before the work starts.** Every cell in the
69 *   window is drawn pending at the first frame, so the size of the job is
70 *   visible immediately rather than inferred from a number that grows.
71 * - **It never exceeds its part of the screen.** A population too big for
72 *   the window is shown a window at a time — `tests 1–780 of 1450` — and
73 *   the frame is rewritten when the run passes the end of one. A cell is
74 *   always exactly one test: nothing is averaged away, nothing reflows,
75 *   nothing scrolls.
76 * - **Elapsed and remaining are always on screen**, beside one progress
77 *   bar that is always on screen too.
78 *
79 * ⚠ Only where a screen can be redrawn. {@see Capabilities::animation()}
80 * excludes pipes, CI and `TERM=dumb`; there this degrades to one summary
81 * line, because a map is a thing you watch and a log is a thing you read.
82 */
83#[ProgressView(
84    key: 'map',
85    description: 'The suite as a map: a cell per test, a progress bar, elapsed time and a remaining estimate.',
86    params: [
87        ['name' => 'fullscreen', 'type' => 'bool', 'required' => false, 'default' => false, 'description' => 'Draw on the alternate screen instead of inline, and restore the scrollback on exit'],
88        ['name' => 'colors', 'type' => 'bool', 'required' => false, 'default' => false, 'description' => 'Enable ANSI color output'],
89        ['name' => 'columns', 'type' => 'int', 'required' => false, 'default' => 64, 'description' => 'Progress characters per line, for the report drawn after the map'],
90        ['name' => 'results', 'type' => 'bool', 'required' => false, 'default' => true, 'description' => 'Print the problem list and the passed tree once the map is done'],
91        ['name' => 'reverseList', 'type' => 'bool', 'required' => false, 'default' => false, 'description' => 'List the problems last-first'],
92        ['name' => 'compact', 'type' => 'bool', 'required' => false, 'default' => false, 'description' => 'Tally only, nothing between the map and it'],
93        ['name' => 'configuration', 'type' => 'string', 'required' => false, 'default' => '', 'description' => 'The configuration file to name on the frame, instead of printing it above one that covers it'],

281    public function handle(Envelope $envelope): void
282    {
283        $event = $envelope->event;
284
285        if ($event instanceof RunStarted) {
286            $this->begin($event->tests);
287        } elseif ($event instanceof TestStarted) {
288            $this->current = $event->test->toString();
289        } elseif ($event instanceof TestFinished) {
290            $this->record($event->outcome);
291        } elseif ($event instanceof RunFinished || $event instanceof RunInterrupted) {
292            // The frame comes down FIRST. On the alternate screen the
293            // report would otherwise be written onto a surface that is
294            // about to be thrown away, and inline it would land inside
295            // the region the presenter is still reconciling.
296            $this->end();
297        }
298
299        $this->report->handle($envelope);
300    }

302    /**
303     * Reserve the window, then draw it empty.
304     *
305     * ⚠ A run whose total is unknown (`tests: 0`) gets no map. A map that
306     * grows while you watch it is the thing this exists to replace, so it
307     * falls back rather than inventing a size.
308     */
309    /**
310     * Open the frame before the run, with a note in place of the brand.
311     *
312     * ⚠ Only ever called when the same conditions {@see begin()} checks
313     * already hold; a frame opened here that `begin()` then declines to
314     * keep is torn down there rather than left on the screen.
315     */
316    public function opening(string $note): void
317    {
318        if (!Capabilities::animation($this->terminal) || $this->terminal->columns() < self::MIN_WIDTH) {
319            return;
320        }
321
322        $this->status  = $note;
323        $this->started = microtime(true);
324        $this->columns = max(1, $this->terminal->columns() - 2 * self::GUTTER);
325        $this->live    = true;
326
327        // ⚠ The map's full height from the start, before its population
328        // is known. Opening at nought rows and growing when the count
329        // arrives moves every readout below it down the screen, and the
330        // one thing this frame promises is that nothing moves.
331        $this->rows = $this->fullscreen ? $this->screenRows() : self::INLINE_ROWS;
332
333        if ($this->fullscreen) {
334            Runtime::setPresenter(new ModalPresenter(Capabilities::color($this->terminal)));
335        }
336
337        $this->presenter->begin($this->terminal);
338        $this->draw();
339
340        $sweep = new ForkedAnimation(
341            $this->terminal,
342            function (int $tick): string {
343                $this->phase = $tick;
344                $this->draw();
345
346                return '';
347            },
348            self::OPENING_INTERVAL,
349            owns: true,
350        );
351
352        // No fork, no sweep: the clock's own tick still moves the frame
353        // once a second, which is worse to watch and better than still.
354        if ($sweep->start()) {
355            $this->sweep = $sweep;
356
357            return;
358        }
359
360        $this->startClock();
361    }
362
363    /**
364     * Take the surface back from the opening sweep.
365     *
366     * ⚠ The reconciler is reset, not merely stopped. The child changed
367     * cells this process's own model still believes it wrote, so the
368     * next diff would skip exactly the cells the child left behind.
369     */
370    private function settleOpening(): void
371    {
372        if (!$this->sweep instanceof ForkedAnimation) {
373            return;
374        }
375
376        $this->sweep->stop();
377        $this->sweep = null;
378        $this->phase = null;
379
380        $this->presenter->interrupt($this->terminal, '');
381    }

383    private function begin(int $tests): void
384    {
385        $this->settleOpening();
386
387        $this->started = $this->live ? $this->started : microtime(true);
388        $this->status  = '';
389        $this->total   = $tests;
390
391        if ($tests <= 0 || !Capabilities::animation($this->terminal) || $this->terminal->columns() < self::MIN_WIDTH) {
392            // An opening frame with no run behind it is closed here, not
393            // left holding a screen nothing will ever draw into.
394            $this->close();
395
396            return;
397        }
398
399        $this->columns = max(1, $this->terminal->columns() - 2 * self::GUTTER);
400
401        $budget = $this->fullscreen ? $this->screenRows() : self::INLINE_ROWS;
402
403        // Only as tall as the population needs — unless a frame is
404        // already open at the budget, in which case shrinking it to fit
405        // would move everything below it, which is the same defect as
406        // growing it.
407        $this->rows   = $this->live ? $this->rows : max(1, min($budget, (int) ceil($tests / $this->columns)));
408        $this->window = $this->rows * $this->columns;
409        $this->live   = true;
410
411        // ⚠ While the map owns the screen, a question asked by anything
412        // else must not be drawn into it. Routing prompts through a modal
413        // gives them a surface of their own and hands this one back
414        // untouched; without it a prompt lands in the middle of the frame
415        // and the frame never recovers.
416        if ($this->fullscreen) {
417            Runtime::setPresenter(new ModalPresenter(Capabilities::color($this->terminal)));
418        }
419
420        $this->presenter->begin($this->terminal);
421        $this->startClock();
422        $this->draw();
423    }

425    /**
426     * Whether a run told only this much will draw its own framed banner.
427     *
428     * Asked by the CLI before the configuration is loaded, so it reads
429     * the command line alone: the banner has to be decided before there

435    {
436        if ($view !== null) {

439
440        return !$testdox && !$teamcity && Capabilities::animation($terminal);

443    /** The map's row budget on a full screen: a reasonable part of it, not all. */
444    private function screenRows(): int
445    {
446        $usable = max(1, $this->terminal->lines() - self::CHROME);
447
448        return max(1, min($usable, max(self::MIN_ROWS, (int) ($usable * self::SCREEN_SHARE))));
449    }

451    private function record(Outcome $outcome): void
452    {
453        $this->tally[$outcome->value] = ($this->tally[$outcome->value] ?? 0) + 1;
454
455        if (!$this->live) {
456            $this->done++;
457
458            return;
459        }
460
461        $this->cells[$this->done] = $outcome;
462        $this->done++;
463
464        $page = intdiv($this->done - 1, $this->window);
465        $now  = microtime(true);
466
467        // A page turn is not a frame worth throttling: the whole map has
468        // changed, and holding it back leaves a window of the wrong tests
469        // on screen.
470        if ($page === $this->page && $now - $this->lastDraw < self::REDRAW_INTERVAL && $this->done < $this->total) {
471            return;
472        }
473
474        $this->page     = $page;
475        $this->lastDraw = $now;
476
477        $this->draw();
478    }

505    /**
506     * The frame: a title strip, the map, and the instrument panel.
507     *
508     * Three boxes side by side with the bar bare underneath: the disk
509     * utilities' Time and Sector readouts and their key, but without the
510     * fourth box they put around the bar. Every row not spent on a
511     * border is a row of map, and nothing moves between frames, so the
512     * eye learns where each number lives once and stops looking for it.
513     */
514    private function draw(): void
515    {
516        $width = max(1, $this->terminal->columns());
517        $inner = max(3, $width - 10);
518
519        // Every cell keeps at least a column, whatever the legend asks
520        // for: a negative width is not a narrow box, it is a crash.
521        $legend = min($this->legendWidth($inner), max(1, $inner - 2));
522        $time   = max(1, intdiv($inner - $legend, 2));
523        $tests  = max(1, $inner - $legend - $time);
524
525        $buffer = new Buffer($width, $this->rows + self::CHROME);
526        $y      = 0;
527
528        $this->boxCap($width)->drawInto($buffer, 0, $y++);
529        $this->titleStrip($width)->drawInto($buffer, 0, $y++);
530
531        // Joined to the strip by a titled rule rather than closed and
532        // reopened: two adjacent borders spend a row saying nothing.
533        $top = new Line();
534        $this->boxTop($top, 'Suite', $width, '├', '┤');
535        $top->drawInto($buffer, 0, $y++);
536
537        for ($row = 0; $row < $this->rows; $row++) {
538            $this->mapRow($row, $width)->drawInto($buffer, 0, $y++);
539        }
540
541        // The running test belongs to the map, under a rule rather than
542        // in a box of its own: it names the cell being filled directly
543        // above it, and a border between them would deny that.
544        $this->boxDivider($width)->drawInto($buffer, 0, $y++);
545        $this->currentRow($width)->drawInto($buffer, 0, $y++);
546
547        $bottom = new Line();
548        $this->boxBottom($bottom, $width);
549        $bottom->drawInto($buffer, 0, $y++);
550
551        for ($row = 0; $row < self::BOX_ROWS; $row++) {
552            $this->panelRow($row, $time, $tests, $legend)->drawInto($buffer, 0, $y++);
553        }
554
555        $this->barRow($width)->drawInto($buffer, 0, $y++);
556
557        $close = new Line();
558        $this->boxBottom($close, $width);
559        $close->drawInto($buffer, 0, $y);
560
561        $this->presenter->present($this->terminal, $buffer);
562    }

564    /**
565     * The test being run, on one line, truncated to the screen.
566     *
567     * ⚠ Truncated, never wrapped. A long test name that took a second
568     * row would move every row below it, and the whole point of the
569     * panel is that nothing under the map ever moves.
570     */
571    private function currentRow(int $width): Line
572    {
573        $line = new Line();
574
575        $this->wrap($line, $width, function (Line $into) use ($width): Line {
576            if ($this->current === '') {
577                return $into;
578            }
579
580            return $into
581                ->add('▸ ', $this->fg(Theme::accent()))
582                ->add(Str::truncate($this->current, max(1, $width - 6)), Style::none()->dim());
583        });
584
585        return $line;
586    }

588    /** `├───┤`: a rule across a box without closing it. */

591        return (new Line())->add('├' . str_repeat('─', max(0, $width - 2)) . '┤', $this->edge());

594    /** Room for the widest legend entry, and never more than a third. */
595    private function legendWidth(int $width): int
596    {
597        return max(self::MIN_BOX, min(self::LEGEND_WIDTH, (int) ($width * 0.40)));
598    }

600    /**
601     * One row of the panel band, drawn across all three cells at once.
602     *
603     * One box with shared walls rather than three boxes side by side:
604     * abutting boxes put `││` down both seams and spend two columns
605     * saying what one says, and there is nothing under three separate
606     * boxes for the bar's own border to hang from.
607     */
608    private function panelRow(int $row, int $time, int $tests, int $legend): Line
609    {
610        $line = new Line();
611
612        if ($row === 0) {
613            return $this->bandTop($time, $tests, $legend);
614        }
615
616        if ($row === self::BOX_ROWS - 1) {
617            return $this->bandJoin($time, $tests, $legend);
618        }
619
620        $elapsed   = microtime(true) - $this->started;
621        $remaining = $this->done === 0 ? 0.0 : ($elapsed / $this->done) * ($this->total - $this->done);
622
623        [$timeLabel, $timeValue] = match ($row) {
624            1       => ['Estimated', $this->clock($elapsed + $remaining)],
625            2       => ['Elapsed', $this->clock($elapsed)],
626            default => ['Remaining', $this->clock($remaining)],
627        };
628
629        [$testLabel, $testValue] = match ($row) {
630            1       => ['Current', number_format($this->done)],
631            2       => ['Total', number_format($this->total)],
632            default => ['Failed', number_format($this->failures())],
633        };
634
635        $this->bandCell($line, $time, fn(Line $into): Line => $this->readout($into, $timeLabel, $timeValue, $time));
636        $this->bandCell($line, $tests, fn(Line $into): Line => $this->readout($into, $testLabel, $testValue, $tests));
637        $this->bandCell($line, $legend, fn(Line $into): Line => $this->legendCells($into, $row, $legend));
638
639        return $line->add('│', $this->edge());
640    }

642    /** `┌─ Time ─┬─ Tests ─┬─ Legend ─┐`: the band opened, each cell named. */
643    private function bandTop(int ...$cells): Line
644    {
645        $line  = new Line();
646        $glyph = '┌';
647
648        foreach (['Time', 'Tests', 'Legend'] as $index => $title) {
649            $label = ' ' . $title . ' ';
650
651            $line
652                ->add($glyph . '─', $this->edge())
653                ->add($label, Style::none()->bold())
654                ->add(str_repeat('─', max(0, $cells[$index] + 1 - Str::width($label))), $this->edge());
655
656            $glyph = '┬';
657        }
658
659        return $line->add('┐', $this->edge());
660    }

662    /** `├─┴─┴─┤`: the band closed and the bar's own box opened beneath it. */
663    private function bandJoin(int ...$cells): Line
664    {
665        $line  = new Line();
666        $glyph = '├';
667
668        foreach ($cells as $cell) {
669            $line->add($glyph . str_repeat('─', max(0, $cell + 2)), $this->edge());
670
671            $glyph = '┴';
672        }
673
674        return $line->add('┤', $this->edge());
675    }

711    /**
712     * The key, two glyphs to a row.
713     *
714     * Six rows of key to read a map is more chrome than map, and the
715     * glyph is the part being looked up, not the word beside it.
716     */
717    private function legendCells(Line $line, int $row, int $box): Line
718    {
719        $entries = match ($row) {
720            1       => [[Outcome::Passed, 'passed'], [Outcome::Failed, 'failed']],
721            2       => [[Outcome::Risky, 'risky'], [Outcome::Incomplete, 'incomplete']],
722            default => [[Outcome::Skipped, 'skipped'], [null, 'pending']],
723        };
724
725        // ⚠ The column is measured, not assumed. A fixed label width
726        // overran a narrow box and the padding went to zero, which took
727        // the box's own right border off the screen with it.
728        $cell = max(3, intdiv(max(1, $box) - 1, 2));
729
730        foreach ($entries as $index => [$outcome, $label]) {
731            if ($index > 0) {
732                $line->add(' ');
733            }
734
735            $line
736                ->add($outcome instanceof Outcome ? $this->glyph($outcome) : '░', $outcome instanceof Outcome ? $this->fg($this->colour($outcome)) : Style::none()->dim())
737                ->add(' ' . Str::padRight(Str::truncate($label, $cell - 2), $cell - 2), Style::none()->dim());
738        }
739
740        return $line;
741    }

743    /**
744     * The one bar, with the window it belongs to and the percentage.
745     *
746     * ⚠ The bar takes what is left rather than the whole row. A bar the
747     * width of the terminal is a bar whose ends nobody can compare, and
748     * the columns it gives back say which tests are on screen.
749     */
750    private function barRow(int $width): Line
751    {
752        $line  = new Line();
753        $inner = max(1, $width - 4);
754
755        $this->wrap($line, $width, function (Line $into) use ($inner): Line {
756            $ratio   = $this->total <= 0 ? 0.0 : min(1.0, $this->done / $this->total);
757            $percent = sprintf('%3d%%', (int) round($ratio * 100));
758
759            // Before a run there is no population to describe, and "0
760            // tests" beside an empty bar reads as a finished run of
761            // nothing rather than one that has not started.
762            $window = match (true) {
763                $this->total <= 0             => '',
764                $this->window >= $this->total => sprintf('%s tests', number_format($this->total)),
765                default                       => sprintf('tests %s–%s of %s', number_format($this->first() + 1), number_format($this->last()), number_format($this->total)),
766            };
767
768            $bar    = max(4, $inner - Str::width($window) - Str::width($percent) - 3);
769            $filled = (int) round($ratio * $bar);
770
771            return $into
772                ->add($window, Style::none()->dim())
773                ->add('  ')
774                ->add(str_repeat('█', $filled), $this->fg(Theme::accent()))
775                ->add(str_repeat('░', $bar - $filled), Style::none()->dim())
776                ->add(' ' . $percent, Style::none()->bold());
777        });
778
779        return $line;
780    }

805    /**
806     * Put a box's side borders around whatever the callback appends.
807     *
808     * The callback writes into the same line rather than building its
809     * own, because a {@see Line} cannot be concatenated into another.
810     * The padding is measured from what it actually wrote, so a row that
811     * overruns is clipped by the buffer instead of pushing the border
812     * off the screen.
813     *
814     * @param Closure(Line): Line $content
815     */
816    private function wrap(Line $line, int $width, Closure $content): void
817    {
818        $inner = max(1, $width - 4);
819
820        $line->add('│ ', $this->edge());
821
822        $before = $line->width();
823        $content($line);
824        $used = $line->width() - $before;
825
826        $line
827            ->add(str_repeat(' ', max(0, $inner - $used)))
828            ->add(' │', $this->edge());
829    }

831    /**
832     * Tick once a second, so the clock moves when nothing else does.
833     *
834     * ⚠ A frame is otherwise only drawn when a test finishes, and the
835     * runs worth watching are the ones where that stops happening — a
836     * twelve-second browser test would freeze the clock exactly when
837     * someone is looking at it to decide whether anything is alive.
838     *
839     * Safe against the runner's IPC: the supervisor polls with
840     * `@stream_select(..., 0.05s)`, so a signal that interrupts it
841     * returns false into a suppressed call and the next poll follows.
842     */
843    private function startClock(): void
844    {
845        if (!$this->canTick()) {
846            return;
847        }
848
849        pcntl_async_signals(true);
850        pcntl_signal(SIGALRM, $this->tick(...));
851        pcntl_alarm(1);
852    }
853
854    private function tick(): void
855    {
856        if (!$this->live) {
857            return;
858        }
859
860        $this->lastDraw = microtime(true);
861        $this->draw();
862
863        pcntl_alarm(1);
864    }
865
866    /** ⚠ Disarmed before the surface goes, never after: a tick into a torn-down frame draws onto the restored screen. */
867    private function stopClock(): void
868    {
869        if (!$this->canTick()) {
870            return;
871        }
872
873        pcntl_alarm(0);
874        pcntl_signal(SIGALRM, SIG_DFL);
875    }
876
877    /** Without pcntl the clock still shows the time; it just moves on events. */
878    private function canTick(): bool
879    {
880        return function_exists('pcntl_alarm')
881            && function_exists('pcntl_signal')
882            && function_exists('pcntl_async_signals');
883    }

885    /**
886     * `14:07`, its colon blinking a second on, a second off.
887     *
888     * Hours and minutes only: the seconds are the one field nobody
889     * reads off a wall clock, and the blink already says the run is
890     * alive — which is the whole job the digits were doing.
891     *
892     * The blink is derived from the clock rather than counted, so a
893     * frame drawn by a test finishing and one drawn by the tick agree
894     * on what the colon is doing.
895     */

897    {

907    /** The wordmark and the byline, centred inside the frame's own border. */
908    private function titleStrip(int $width): Line
909    {
910        $line  = new Line();
911        $inner = max(1, $width - 4);
912        $now   = time();
913        $clock = $this->wallClock();
914
915        // The configuration is named HERE rather than printed above the
916        // frame, where the frame immediately covers it and the run ends
917        // with it sitting under the report as though it were output.
918        $file = $this->configuration === ''
919            ? ''
920            : ' ' . Str::truncate($this->configuration, max(4, intdiv($inner, 3)));
921
922        $this->wrap($line, $width, function (Line $into) use ($inner, $now, $clock, $file): Line {
923            $brand = $this->status === ''
924                ? sprintf('v%s · %s', $this->release(), Version::DOMAIN)
925                : $this->status;
926            $room = max(0, $inner - 1 - Str::width($file) - Str::width($clock));
927
928            if (Str::width($brand) > $room) {
929                $brand = $this->status === '' ? Version::DOMAIN : Str::truncate($this->status, max(1, $room));
930            }
931
932            // Centred on the FRAME, not on what the file and the clock
933            // leave over: centring the remainder puts the brand as far
934            // right as the file is wide, which is visibly off.
935            $lead = max(0, intdiv($inner - Str::width($brand), 2) - 1 - Str::width($file));
936
937            // Centred on what the file and the clock leave, both of which
938            // are pinned to an edge: a brand that drifted with either
939            // would be a third thing moving on every frame.
940            return $into
941                ->add($this->markAt($this->phase ?? $now), $this->fg($this->markColour($this->phase ?? $now)))
942                ->add($file, Style::none()->dim())
943                ->add(str_repeat(' ', $lead))
944                ->add($brand, Style::none()->bold())
945                ->add(str_repeat(' ', max(0, $room - $lead - Str::width($brand))))
946                ->add($clock, Style::none()->dim());
947        });
948
949        return $line;
950    }

978    private function mapRow(int $row, int $width): Line
979    {
980        $line  = new Line();
981        $start = $this->first() + $row * $this->columns;
982
983        $this->wrap($line, $width, function (Line $into) use ($start): Line {
984            for ($column = 0; $column < $this->columns; $column++) {
985                $index = $start + $column;
986
987                if ($index >= $this->total) {
988                    $into->add(' ');
989
990                    continue;
991                }
992
993                $outcome = $this->cells[$index] ?? null;
994
995                $into->add(
996                    $outcome instanceof Outcome ? $this->glyph($outcome) : '░',
997                    $outcome instanceof Outcome ? $this->fg($this->colour($outcome)) : Style::none()->dim(),
998                );
999            }
1000
1001            return $into;
1002        });
1003
1004        return $line;
1005    }

1048    /** Seconds as mm:ss, or h:mm:ss once a run has earned the hour. */

1050    {

1053        $minutes = intdiv($whole % 3600, 60);
1054

1056            ? sprintf('%d:%02d:%02d', $hours, $minutes, $whole % 60)
1057            : sprintf('%02d:%02d', $minutes, $whole % 60);
